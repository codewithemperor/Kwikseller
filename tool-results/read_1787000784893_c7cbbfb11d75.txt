   101→function formatDateTime(value?: string) {
   102→  if (!value) return "—";
   103→  return new Intl.DateTimeFormat("en-NG", { dateStyle: "medium", timeStyle: "short" }).format(new Date(value));
   104→}

   106→function formatNGN(n: number) {

   108→    style: "currency",
   109→    currency: "NGN",
   110→    maximumFractionDigits: 0,
   111→  }).format(n);
   112→}

   178→  const [filter, setFilter] = React.useState<"ALL" | "CREDIT" | "DEBIT">("ALL");
   179→  const [dateRange, setDateRange] = React.useState<"7d" | "30d" | "90d" | "all">("all");
   180→  const [categoryFilter, setCategoryFilter] = React.useState<string>("ALL");
   181→  const [searchTerm, setSearchTerm] = React.useState("");
   182→  const [showAdvanced, setShowAdvanced] = React.useState(false);

   184→  // ── Redemption modal state ──

   188→  function openRedeemModal(type: WalletRedemptionType) {

   193→  // ── Derived state (declared BEFORE any early return so the Rules of
   194→  // Hooks aren't violated). All hooks gracefully handle `wallet` being
   195→  // undefined by returning empty defaults.
   196→  const transactions = wallet?.transactions ?? [];
   197→
   198→  const txs = React.useMemo(() => {
   199→    let list = transactions.slice();
   200→    // Type filter
   201→    if (filter !== "ALL") list = list.filter((t) => t.type === filter);
   202→    // Category filter
   203→    if (categoryFilter !== "ALL") list = list.filter((t) => t.category === categoryFilter);
   204→    // Date range filter
   205→    if (dateRange !== "all") {
   206→      const days = dateRange === "7d" ? 7 : dateRange === "30d" ? 30 : 90;
   207→      const cutoff = Date.now() - days * 86400000;
   208→      list = list.filter((t) => new Date(t.createdAt).getTime() >= cutoff);
   209→    }
   210→    // Search filter (case-insensitive on description)
   211→    const term = searchTerm.trim().toLowerCase();
   212→    if (term) {
   213→      list = list.filter(
   214→        (t) =>
   215→          t.description.toLowerCase().includes(term) ||
   216→          t.category.replace(/_/g, " ").toLowerCase().includes(term),
   217→      );
   218→    }
   219→    return list;
   220→  }, [transactions, filter, categoryFilter, dateRange, searchTerm]);
   221→
   222→  // Available categories derived from the wallet's own transactions
   223→  const availableCategories = React.useMemo(() => {
   224→    const seen = new Set<string>();
   225→    for (const t of transactions) seen.add(t.category);
   226→    return Array.from(seen);

   229→  const hasActiveFilters =
   230→    filter !== "ALL" || categoryFilter !== "ALL" || dateRange !== "all" || searchTerm.trim().length > 0;

   232→  function clearAllFilters() {
   233→    setFilter("ALL");
   234→    setCategoryFilter("ALL");
   235→    setDateRange("all");
   236→    setSearchTerm("");

   238→
   239→  // Export the currently-filtered transactions to CSV. Respects all active
   240→  // filters (type, category, date range, search) so the buyer gets exactly
   241→  // what they see on screen. Reuses the shared toCSV/downloadCSV utilities
   242→  // (same as the orders page CSV export).
   243→  //
   244→  // The dummy wallet API doesn't return per-transaction running balances,
   245→  // so we derive them locally from the FILTERED list (using only the
   246→  // transactions the buyer is exporting — otherwise the numbers wouldn't
   247→  // reconcile to `wallet.balance`). Algorithm:
   248→  //   1. Sort the filtered txs oldest-first by `createdAt`.
   249→  //   2. Compute totalDelta = sum(credits) - sum(debits).
   250→  //   3. startingBalance = wallet.balance - totalDelta (the balance before
   251→  //      the oldest filtered tx).
   252→  //   4. Walk forward, applying each tx's delta to a running balance and
   253→  //      recording it in a Map<txId, coinsAfter>.
   254→  //   5. Emit CSV rows in the original `txs` order (matches what's on
   255→  //      screen), looking up each row's "Coins After" from the Map.
   256→  // After the most-recent filtered tx, the running balance equals
   257→  // wallet.balance exactly.
   258→  function handleExportCSV() {
   259→    if (txs.length === 0) {
   260→      kwikToast.error("No transactions to export", "Adjust your filters and try again.");
   261→      return;
   262→    }
   263→    const currentBalance = wallet?.balance ?? 0;
   264→    const totalDelta = txs.reduce(
   265→      (sum, t) => sum + (t.type === "CREDIT" ? t.amount : -t.amount),
   266→      0,
   267→    );
   268→    const coinsAfterById = new Map<string, number>();
   269→    let running = currentBalance - totalDelta;
   270→    // Walk oldest → newest by `createdAt` (the API list order isn't
   271→    // guaranteed to be strictly chronological in dummy mode).
   272→    const chronological = [...txs].sort(
   273→      (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   274→    );
   275→    for (const t of chronological) {
   276→      running += t.type === "CREDIT" ? t.amount : -t.amount;
   277→      coinsAfterById.set(t.id, running);
   278→    }
   279→    const rows = txs.map((t) => ({
   280→      Date: new Date(t.createdAt).toLocaleString("en-NG", { dateStyle: "medium", timeStyle: "short" }),
   281→      Type: t.type === "CREDIT" ? "Earned" : "Spent",
   282→      Category: t.category.replace(/_/g, " ").toLowerCase().replace(/\b\w/g, (c) => c.toUpperCase()),
   283→      Description: t.description,
   284→      Amount: t.type === "CREDIT" ? `+${t.amount}` : `-${t.amount}`,
   285→      "Coins After": coinsAfterById.get(t.id) ?? "",
   286→    }));
   287→    const csv = toCSV(rows);
   288→    const date = new Date().toISOString().slice(0, 10);
   289→    const suffix = hasActiveFilters ? "-filtered" : "";
   290→    downloadCSV(`kwikseller-wallet${suffix}-${date}`, csv);
   291→    kwikToast.success("CSV downloaded", `${txs.length} transaction${txs.length === 1 ? "" : "s"} exported.`);
   292→  }
   293→
   294→  if (isLoading) return <PageLoading label="Loading wallet…" />;
   295→
   296→  if (!wallet) {
   297→    return (
   298→      <EmptyState
   299→        variant="default"
   300→        icon={<WalletIcon className="h-12 w-12" />}
   301→        title="Wallet unavailable"
   302→        description="We couldn't load your KwikCoins wallet right now."
   303→      />
   304→    );
   305→  }
