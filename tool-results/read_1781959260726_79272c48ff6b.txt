    61→// ==================== Local types ====================

    63→type SubscriptionStatus = "ACTIVE" | "CANCELLED" | "EXPIRED" | "NONE";
    64→
    65→type CurrentSubscription = {
    66→  planId: string;
    67→  status: SubscriptionStatus;
    68→  startDate?: string;
    69→  endDate?: string;
    70→  renewalDate?: string;

    75→type BillingEntry = {
    76→  id: string;
    77→  date: string;
    78→  plan: string;

    84→// ==================== LocalStorage helpers ====================
    85→
    86→const SUBSCRIPTION_KEY = "kwikseller_vendor_subscription";
    87→const BILLING_KEY = "kwikseller_vendor_billing_history";
    88→
    89→function loadSubscription(): CurrentSubscription {
    90→  if (typeof window === "undefined")
    91→    return { planId: "STARTER", status: "NONE" };
    92→  try {
    93→    const raw = localStorage.getItem(SUBSCRIPTION_KEY);
    94→    return raw ? JSON.parse(raw) : { planId: "STARTER", status: "NONE" };
    95→  } catch {
    96→    return { planId: "STARTER", status: "NONE" };
    97→  }
    98→}
    99→
   100→function saveSubscription(sub: CurrentSubscription) {
   101→  if (typeof window === "undefined") return;
   102→  localStorage.setItem(SUBSCRIPTION_KEY, JSON.stringify(sub));
   103→}
   104→
   105→function loadBilling(): BillingEntry[] {
   106→  if (typeof window === "undefined") return [];
   107→  try {
   108→    const raw = localStorage.getItem(BILLING_KEY);
   109→    return raw ? JSON.parse(raw) : [];
   110→  } catch {
   111→    return [];
   112→  }
   113→}
   114→
   115→function addBillingEntry(plan: Plan, status: "Paid" | "Pending" | "Failed") {
   116→  const entries = loadBilling();
   117→  const entry: BillingEntry = {
   118→    id: `inv_${Date.now()}`,
   119→    date: new Date().toISOString(),
   120→    plan: plan.name,
   121→    amount: plan.price,
   122→    status,
   123→  };
   124→  entries.unshift(entry);
   125→  localStorage.setItem(BILLING_KEY, JSON.stringify(entries));
   126→}
   127→
   128→// ==================== Helpers ====================
   129→
   130→function getStatusBadge(status: SubscriptionStatus) {
   131→  switch (status) {
   132→    case "ACTIVE":
   133→      return (
   134→        <span className="text-xs font-medium text-green-600">Active</span>
   135→      );
   136→    case "CANCELLED":
   137→      return (
   138→        <span className="text-xs font-medium text-gray-500">Cancelled</span>
   139→      );
   140→    case "EXPIRED":
   141→      return (
   142→        <span className="text-xs font-medium text-red-600">Expired</span>
   143→      );
   144→    default:
   145→      return null;
   146→  }
   147→}
   148→
   149→function formatLimit(value: number) {
   150→  return value === -1 ? "Unlimited" : String(value);
   151→}
   152→
   153→function getPlanById(id: string): Plan | undefined {
   154→  return PLANS.find((p) => p.id === id);
   155→}
