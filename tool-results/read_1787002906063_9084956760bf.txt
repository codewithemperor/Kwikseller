   166→   166→// ─── Timeline stage computation ────────────────────────────────────────────
   167→   167→//
   168→   168→// 9 stages:
   169→   169→//   1. Order Placed
   170→   170→//   2. Vendor Quote
   171→   171→//   3. Quote Agreed
   172→   172→//   4. Payment
   173→   173→//   5. Kwikscrow Holding
   174→   174→//   6. Processing
   175→   175→//   7. Pickup / Delivery
   176→   176→//   8. Confirmed
   177→   177→//   9. Completed
   178→   178→//
   179→   179→// `current` = the stage the order is currently AT (highlighted).
   180→   180→// `reached` = stages the order has already passed (filled green/orange).
   181→   181→// Cancelled orders show all stages greyed with a "Cancelled" label.
   182→   182→
   183→   183→interface TimelineStage {
   184→   184→  key: string;
   185→   185→  label: string;
   186→   186→  hint: string;
   187→   187→  reached: boolean;
   188→   188→  current: boolean;
   189→   189→}
   190→   190→
   191→   191→function computeTimelineStages(order: MarketplaceOrder): TimelineStage[] {
   192→   192→  const isCancelled = order.status === "CANCELLED" || order.status === "REFUNDED";
   193→   193→  const isPickup = order.deliveryMethod === "PICKUP";
   194→   194→  const paymentPaid = order.paymentStatus === "PAID";
   195→   195→  const quoteAgreed =
   196→   196→    order.quoteStatus === "AGREED" ||
   197→   197→    // PICKUP auto-agrees at checkout, and any paid order implies the quote was agreed.
   198→   198→    (isPickup && order.status !== "PENDING") ||
   199→   199→    paymentPaid;
   200→   200→  const quoteQuoted =
   201→   201→    order.quoteStatus !== "PENDING_VENDOR_QUOTE" || quoteAgreed;
   202→   202→  const escrowHeld =
   203→   203→    order.escrow?.status === "HELD" ||
   204→   204→    order.escrow?.status === "PENDING_RELEASE" ||
   205→   205→    order.escrow?.status === "RELEASED" ||
   206→   206→    order.escrow?.status === "PARTIAL";
   207→   207→  const isProcessing =
   208→   208→    order.status === "PROCESSING" ||
   209→   209→    (order.fulfillments?.some((f) => f.status === "PROCESSING") ?? false);
   210→   210→  const deliveryReady =
   211→   211→    order.delivery?.status === "READY_FOR_PICKUP" ||
   212→   212→    order.delivery?.status === "PICKED_UP" ||
   213→   213→    order.delivery?.status === "IN_TRANSIT" ||
   214→   214→    order.delivery?.status === "ARRIVED" ||
   215→   215→    order.delivery?.status === "DELIVERED" ||
   216→   216→    order.status === "FULFILLED" ||
   217→   217→    order.status === "SHIPPED" ||
   218→   218→    order.status === "DELIVERED";
   219→   219→  const confirmed =
   220→   220→    order.delivery?.customerConfirmed === true ||
   221→   221→    order.status === "DELIVERED" ||
   222→   222→    order.status === "COMPLETED";
   223→   223→  const completed =
   224→   224→    order.status === "COMPLETED" ||
   225→   225→    order.escrow?.status === "RELEASED";
   226→   226→
   227→   227→  // Determine the "current" stage.
   228→   228→  let currentKey = "placed";
   229→   229→  if (completed) currentKey = "completed";
   230→   230→  else if (confirmed) currentKey = "confirmed";
   231→   231→  else if (deliveryReady) currentKey = "delivery";
   232→   232→  else if (isProcessing) currentKey = "processing";
   233→   233→  else if (escrowHeld) currentKey = "escrow";
   234→   234→  else if (paymentPaid) currentKey = "payment";
   235→   235→  else if (quoteAgreed) currentKey = "agreed";
   236→   236→  else if (quoteQuoted) currentKey = "quote";
   237→   237→
   238→   238→  const stages: Array<[string, string, string, boolean]> = [
   239→   239→    ["placed", "Order Placed", "Your order was placed and sent to the vendor.", true],
   240→   240→    ["quote", "Vendor Quote", "The vendor is preparing a delivery quote.", quoteQuoted],
   241→   241→    ["agreed", "Quote Agreed", "Both parties agreed on the delivery fee.", quoteAgreed],
   242→   242→    ["payment", "Payment", "You paid via Paystack.", paymentPaid],
   243→   243→    ["escrow", `${KwisCrow.NAME} Holding`, "Funds are held safely in escrow.", escrowHeld],
   244→   244→    ["processing", "Processing", "The vendor is preparing your order.", isProcessing],
   245→   245→    ["delivery", isPickup ? "Ready for Pickup" : "Out for Delivery", isPickup ? "Your order is ready for pickup." : "Your order is on its way.", deliveryReady],
   246→   246→    ["confirmed", "Confirmed", "You confirmed receipt of the order.", confirmed],
   247→   247→    ["completed", "Completed", "Escrow released to the vendor. Order complete.", completed],
   248→   248→  ];
   249→   249→
   250→   250→  return stages.map(([key, label, hint, reached]) => ({
   251→   251→    key,
   252→   252→    label,
   253→   253→    hint,
   254→   254→    reached: isCancelled ? false : reached,
   255→   255→    current: isCancelled ? false : currentKey === key,
   256→   256→  }));
   257→   257→}

   452→   452→function QuoteSection({
   453→   453→  order,
   454→   454→  quote,
   455→   455→}: {
   456→   456→  order: MarketplaceOrder;
   457→   457→  quote: OrderQuote | null | undefined;

   459→   459→  // Only render for STANDARD_DELIVERY (PICKUP auto-agrees at checkout).
   460→   460→  if (order.deliveryMethod !== "STANDARD_DELIVERY") return null;
   461→   461→
   462→   462→  // Sync quoteStatus: prefer the Quote entity's status when present
   463→   463→  // (it may be ahead of the denormalized Order.quoteStatus by a few ms).
   464→   464→  const quoteStatus: QuoteStatus = quote?.status ?? order.quoteStatus;

   474→   474→  const agreedAmount = quote?.agreedAmount ?? (quoteStatus === "AGREED" ? order.agreedDeliveryFee : null);

   478→   478→      orderId={order.id}
   479→   479→      quoteStatus={quoteStatus}

   481→   481→      agreedAmount={agreedAmount ?? null}
   482→   482→      latestVendorNote={latestVendorRevision?.note}
   483→   483→      latestCustomerReductionAmount={latestCustomerReduction?.amount}
   484→   484→      latestCustomerReductionNote={latestCustomerReduction?.note}
   485→   485→      revisionCount={revisions.length}
   486→   486→      createdAt={quote?.createdAt}
