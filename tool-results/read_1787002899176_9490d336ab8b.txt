   166→// ─── Timeline stage computation ────────────────────────────────────────────
   167→//
   168→// 9 stages:
   169→//   1. Order Placed
   170→//   2. Vendor Quote
   171→//   3. Quote Agreed
   172→//   4. Payment
   173→//   5. Kwikscrow Holding
   174→//   6. Processing
   175→//   7. Pickup / Delivery
   176→//   8. Confirmed
   177→//   9. Completed
   178→//
   179→// `current` = the stage the order is currently AT (highlighted).
   180→// `reached` = stages the order has already passed (filled green/orange).
   181→// Cancelled orders show all stages greyed with a "Cancelled" label.
   182→
   183→interface TimelineStage {
   184→  key: string;
   185→  label: string;
   186→  hint: string;
   187→  reached: boolean;
   188→  current: boolean;
   189→}
   190→
   191→function computeTimelineStages(order: MarketplaceOrder): TimelineStage[] {
   192→  const isCancelled = order.status === "CANCELLED" || order.status === "REFUNDED";
   193→  const isPickup = order.deliveryMethod === "PICKUP";
   194→  const paymentPaid = order.paymentStatus === "PAID";
   195→  const quoteAgreed =
   196→    order.quoteStatus === "AGREED" ||
   197→    // PICKUP auto-agrees at checkout, and any paid order implies the quote was agreed.
   198→    (isPickup && order.status !== "PENDING") ||
   199→    paymentPaid;
   200→  const quoteQuoted =
   201→    order.quoteStatus !== "PENDING_VENDOR_QUOTE" || quoteAgreed;
   202→  const escrowHeld =
   203→    order.escrow?.status === "HELD" ||
   204→    order.escrow?.status === "PENDING_RELEASE" ||
   205→    order.escrow?.status === "RELEASED" ||
   206→    order.escrow?.status === "PARTIAL";
   207→  const isProcessing =
   208→    order.status === "PROCESSING" ||
   209→    (order.fulfillments?.some((f) => f.status === "PROCESSING") ?? false);
   210→  const deliveryReady =
   211→    order.delivery?.status === "READY_FOR_PICKUP" ||
   212→    order.delivery?.status === "PICKED_UP" ||
   213→    order.delivery?.status === "IN_TRANSIT" ||
   214→    order.delivery?.status === "ARRIVED" ||
   215→    order.delivery?.status === "DELIVERED" ||
   216→    order.status === "FULFILLED" ||
   217→    order.status === "SHIPPED" ||
   218→    order.status === "DELIVERED";
   219→  const confirmed =
   220→    order.delivery?.customerConfirmed === true ||
   221→    order.status === "DELIVERED" ||
   222→    order.status === "COMPLETED";
   223→  const completed =
   224→    order.status === "COMPLETED" ||
   225→    order.escrow?.status === "RELEASED";
   226→
   227→  // Determine the "current" stage.
   228→  let currentKey = "placed";
   229→  if (completed) currentKey = "completed";
   230→  else if (confirmed) currentKey = "confirmed";
   231→  else if (deliveryReady) currentKey = "delivery";
   232→  else if (isProcessing) currentKey = "processing";
   233→  else if (escrowHeld) currentKey = "escrow";
   234→  else if (paymentPaid) currentKey = "payment";
   235→  else if (quoteAgreed) currentKey = "agreed";
   236→  else if (quoteQuoted) currentKey = "quote";
   237→
   238→  const stages: Array<[string, string, string, boolean]> = [
   239→    ["placed", "Order Placed", "Your order was placed and sent to the vendor.", true],
   240→    ["quote", "Vendor Quote", "The vendor is preparing a delivery quote.", quoteQuoted],
   241→    ["agreed", "Quote Agreed", "Both parties agreed on the delivery fee.", quoteAgreed],
   242→    ["payment", "Payment", "You paid via Paystack.", paymentPaid],
   243→    ["escrow", `${KwisCrow.NAME} Holding`, "Funds are held safely in escrow.", escrowHeld],
   244→    ["processing", "Processing", "The vendor is preparing your order.", isProcessing],
   245→    ["delivery", isPickup ? "Ready for Pickup" : "Out for Delivery", isPickup ? "Your order is ready for pickup." : "Your order is on its way.", deliveryReady],
   246→    ["confirmed", "Confirmed", "You confirmed receipt of the order.", confirmed],
   247→    ["completed", "Completed", "Escrow released to the vendor. Order complete.", completed],
   248→  ];
   249→
   250→  return stages.map(([key, label, hint, reached]) => ({
   251→    key,
   252→    label,
   253→    hint,
   254→    reached: isCancelled ? false : reached,
   255→    current: isCancelled ? false : currentKey === key,
   256→  }));
   257→}

   450→// ─── Quote negotiation section ─────────────────────────────────────────────
   451→
   452→function QuoteSection({
   453→  order,
   454→  quote,
   455→}: {
   456→  order: MarketplaceOrder;
   457→  quote: OrderQuote | null | undefined;
   458→}) {
   459→  // Only render for STANDARD_DELIVERY (PICKUP auto-agrees at checkout).
   460→  if (order.deliveryMethod !== "STANDARD_DELIVERY") return null;
   461→
   462→  // Sync quoteStatus: prefer the Quote entity's status when present
   463→  // (it may be ahead of the denormalized Order.quoteStatus by a few ms).
   464→  const quoteStatus: QuoteStatus = quote?.status ?? order.quoteStatus;
   465→
   466→  const revisions: QuoteRevision[] = quote?.revisions ?? [];
   467→  const latestVendorRevision = [...revisions]
   468→    .reverse()
   469→    .find((r) => r.type === "VENDOR_QUOTE" || r.type === "VENDOR_REVISE");
   470→  const latestCustomerReduction = [...revisions]
   471→    .reverse()
   472→    .find((r) => r.type === "CUSTOMER_REQUEST_REDUCTION");
   473→  const currentAmount = quote?.currentAmount ?? order.agreedDeliveryFee ?? 0;
   474→  const agreedAmount = quote?.agreedAmount ?? (quoteStatus === "AGREED" ? order.agreedDeliveryFee : null);
   475→
   476→  return (
   477→    <QuoteNegotiationCard
   478→      orderId={order.id}
   479→      quoteStatus={quoteStatus}
   480→      currentAmount={currentAmount}
   481→      agreedAmount={agreedAmount ?? null}
   482→      latestVendorNote={latestVendorRevision?.note}
   483→      latestCustomerReductionAmount={latestCustomerReduction?.amount}
   484→      latestCustomerReductionNote={latestCustomerReduction?.note}
   485→      revisionCount={revisions.length}
   486→      createdAt={quote?.createdAt}
   487→    />
   488→  );
   489→}

   491→function QuoteNegotiationCard({
   492→  orderId,
   493→  quoteStatus,
   494→  currentAmount,
   495→  agreedAmount,
   496→  latestVendorNote,
   497→  latestCustomerReductionAmount,
   498→  latestCustomerReductionNote,
   499→  revisionCount,
   500→  createdAt,
   501→}: {
   502→  orderId: string;
   503→  quoteStatus: QuoteStatus;
   504→  currentAmount: number;
   505→  agreedAmount: number | null;
   506→  latestVendorNote?: string | null;
   507→  latestCustomerReductionAmount?: number;
   508→  latestCustomerReductionNote?: string | null;
   509→  revisionCount: number;
   510→  createdAt?: string;
   511→}) {
   512→  const acceptQuote = useAcceptQuote(orderId);
   513→  const requestReduction = useRequestReduction(orderId);
   514→  const rejectQuote = useRejectQuote(orderId);
   515→  const initializePayment = useInitializePayment(orderId);
   516→  const router = useRouter();
   517→
   518→  const [reductionOpen, setReductionOpen] = React.useState(false);
   519→  const [reductionAmount, setReductionAmount] = React.useState("");
   520→  const [reductionNote, setReductionNote] = React.useState("");
   521→
   522→  const canAcceptOrReduce =
   523→    quoteStatus === "QUOTED" || quoteStatus === "VENDOR_REVISED";
   524→
   525→  const handleAccept = async () => {
   526→    try {
   527→      await acceptQuote.mutateAsync({});
   528→      kwikToast.success(
   529→        "Quote accepted",
   530→        "You can now proceed to payment.",
   531→      );
   532→    } catch (err) {
   533→      const message = err instanceof Error ? err.message : "Failed to accept quote.";
   534→      kwikToast.error("Couldn't accept quote", message);
   535→    }
   536→  };
   537→
   538→  const handleRequestReduction = async () => {
   539→    const amount = Number(reductionAmount);
   540→    if (!Number.isFinite(amount) || amount <= 0) {
   541→      kwikToast.warning("Enter a valid amount", "Please enter a positive delivery fee.");
   542→      return;
   543→    }
   544→    if (amount >= currentAmount) {
   545→      kwikToast.warning(
   546→        "Amount too high",
   547→        "Your proposed amount must be lower than the vendor's current quote.",
   548→      );
   549→      return;
   550→    }
   551→    try {
   552→      await requestReduction.mutateAsync({
   553→        amount,
   554→        note: reductionNote.trim() || undefined,
   555→      });
   556→      kwikToast.success(
   557→        "Reduction requested",
   558→        "The vendor will review your proposed amount.",
   559→      );
   560→      setReductionOpen(false);
   561→      setReductionAmount("");
   562→      setReductionNote("");
   563→    } catch (err) {
   564→      const message = err instanceof Error ? err.message : "Failed to request reduction.";
   565→      kwikToast.error("Couldn't request reduction", message);
   566→    }
   567→  };
   568→
   569→  const handleReject = async () => {
   570→    if (
   571→      !window.confirm(
   572→        "Reject this quote? The order will be cancelled and cannot be reopened.",
   573→      )
   574→    ) {
   575→      return;
   576→    }
   577→    try {
   578→      await rejectQuote.mutateAsync({});
   579→      kwikToast.info("Quote rejected", "The order has been cancelled.");
   580→    } catch (err) {
   581→      const message = err instanceof Error ? err.message : "Failed to reject quote.";
   582→      kwikToast.error("Couldn't reject quote", message);
   583→    }
   584→  };
   585→
   586→  const handleProceedToPayment = async () => {
   587→    try {
   588→      const result = await initializePayment.mutateAsync();
   589→      if (result.authorizationUrl) {
   590→        kwikToast.info(
   591→          "Redirecting to Paystack",
   592→          "Complete your payment to confirm the order.",
   593→        );
   594→        // Use a full-page redirect so the Paystack callback can return here.
   595→        window.location.href = result.authorizationUrl;
   596→      } else {
   597→        kwikToast.error(
   598→          "Payment initialization failed",
   599→          "No authorization URL was returned by the server.",
   600→        );
   601→      }
   602→    } catch (err) {
   603→      const message =
   604→        err instanceof Error ? err.message : "Failed to initialize payment.";
   605→      kwikToast.error("Couldn't initialize payment", message);
   606→    }
   607→  };
   608→
   609→  return (
   610→    <section
   611→      aria-label="Vendor quote negotiation"
   612→      className="overflow-hidden rounded-2xl border border-kwik-border bg-kwik-bg-surface shadow-sm"
   613→    >
   614→      <div className="kwik-gradient px-5 py-4 sm:px-6">
   615→        <div className="flex flex-wrap items-start justify-between gap-3">
   616→          <div>
   617→            <p className="text-[11px] font-bold uppercase tracking-[0.18em] text-white/80">
   618→              Delivery quote
   619→            </p>
   620→            <h2 className="mt-1 font-heading text-lg font-bold text-white sm:text-xl">
   621→              Vendor quotation
   622→            </h2>
   623→            {createdAt && (
   624→              <p className="mt-0.5 text-xs text-white/80">
   625→                Opened {formatDate(createdAt)}
   626→                {revisionCount > 0 && ` · ${revisionCount} revision${revisionCount === 1 ? "" : "s"}`}
   627→              </p>
   628→            )}
   629→          </div>
   630→          <div className="flex items-center gap-2 rounded-full bg-white/15 px-3 py-1.5 text-xs font-semibold text-white backdrop-blur">
   631→            <StatusBadgeTransparent status={quoteStatus} />
   632→          </div>
   633→        </div>
   634→      </div>
   635→
   636→      <div className="px-5 py-5 sm:px-6">
   637→        {/* Status-dependent message + actions */}
   638→        {quoteStatus === "PENDING_VENDOR_QUOTE" && (
   639→          <div className="flex items-center gap-3 rounded-xl bg-kwik-orange/5 p-4 text-sm text-kwik-orange">
   640→            <RefreshCw className="h-4 w-4 animate-spin" />
   641→            <span>
   642→              Waiting for the vendor to provide a delivery quote. This page
   643→              updates automatically.
   644→            </span>
   645→          </div>
   646→        )}
   647→
   648→        {canAcceptOrReduce && (
   649→          <div className="space-y-4">
   650→            <div className="rounded-xl border border-kwik-border-light bg-gray-50 p-4">
   651→              <p className="text-xs font-semibold uppercase tracking-wide text-gray-500">
   652→                Vendor&apos;s quoted delivery fee
   653→              </p>
   654→              <p className="mt-1 text-2xl font-bold text-foreground tabular-nums">
   655→                {formatCurrency(currentAmount)}
   656→              </p>

   668→            {!reductionOpen ? (
   669→              <div className="flex flex-col gap-2 sm:flex-row-reverse sm:items-center sm:justify-start">
   670→                <button
   671→                  type="button"
   672→                  onClick={handleAccept}
   673→                  disabled={acceptQuote.isPending}

   678→                      <span className="h-4 w-4 animate-spin rounded-full border-2 border-white/40 border-t-white" />

   690→                  onClick={() => setReductionOpen(true)}

   718→                    Your proposed amount (₦)

   721→                    id="reduction-amount"

   949→function DeliverySection({ order }: { order: MarketplaceOrder }) {
   950→  const isPickup = order.deliveryMethod === "PICKUP";
   951→  const delivery = order.delivery;
   952→  const address = order.address;
   953→
   954→  let addressLine = "No delivery address on this order";
   955→  if (delivery?.deliveryAddress) {
   956→    addressLine = delivery.deliveryAddress;
   957→  } else if (address) {
   958→    addressLine = [
   959→      address.fullName,
   960→      address.line1,
   961→      address.line2,
   962→      address.city,
   963→      address.state,
   964→      address.country,
   965→    ]
   966→      .filter(Boolean)
   967→      .join(", ");
   968→  } else if (order.deliveryLocalGovernment && order.deliveryState) {
   969→    addressLine = `${order.deliveryLocalGovernment}, ${order.deliveryState}, Nigeria`;
   970→  }
   971→
   972→  return (
   973→    <section className="rounded-2xl border border-kwik-border bg-kwik-bg-surface p-5">
   974→      <header className="flex items-center gap-2">
   975→        <Truck className="h-5 w-5 text-kwik-orange" />
   976→        <h2 className="font-semibold text-foreground">Delivery</h2>
   977→      </header>
   978→
   979→      <div className="mt-3 space-y-2 text-sm">
   980→        <div className="flex items-center justify-between">
   981→          <span className="text-kwik-muted">Method</span>
   982→          <span className="font-semibold text-foreground">
   983→            {isPickup ? "Pickup" : "Standard delivery"}
   984→          </span>
   985→        </div>
   986→        {delivery && (
   987→          <div className="flex items-center justify-between">
   988→            <span className="text-kwik-muted">Status</span>
   989→            <StatusBadge status={delivery.status} />
   990→          </div>
   991→        )}
   992→        {delivery?.currentLocation && (
   993→          <div className="flex items-center justify-between text-xs">
   994→            <span className="text-kwik-muted">Current location</span>
   995→            <span className="text-foreground">{delivery.currentLocation}</span>
   996→          </div>
   997→        )}
   998→        {delivery?.estimatedMinutes != null && (
   999→          <div className="flex items-center justify-between text-xs">
  1000→            <span className="text-kwik-muted">ETA</span>
  1001→            <span className="text-foreground">{delivery.estimatedMinutes} min</span>
  1002→          </div>
  1003→        )}
  1004→      </div>
  1005→
  1006→      <div className="mt-4 flex items-start gap-3 rounded-xl bg-gray-50 px-3 py-3 text-sm">
  1007→        <MapPin className="mt-0.5 h-4 w-4 shrink-0 text-kwik-orange" />
  1008→        <div>
  1009→          <p className="font-semibold text-foreground">
  1010→            {isPickup ? "Pickup address" : "Delivery address"}
  1011→          </p>
  1012→          <p className="mt-0.5 text-xs leading-5 text-gray-600">{addressLine}</p>
  1013→          {delivery?.deliveryContactName && (
  1014→            <p className="mt-1 text-xs text-kwik-muted">

  1016→              {delivery.deliveryContactPhone ? ` · ${delivery.deliveryContactPhone}` : ""}
  1017→            </p>
  1018→          )}
  1019→          {delivery?.deliveredAt && (
  1020→            <p className="mt-2 text-xs text-kwik-green">
  1021→              Delivered {formatDate(delivery.deliveredAt)}
  1022→            </p>
  1023→          )}
  1024→        </div>
  1025→      </div>
  1026→    </section>
  1027→  );
  1028→}

  1032→function OrderSummary({ order }: { order: MarketplaceOrder }) {
  1033→  const subtotal = Number(order.subtotal ?? 0);
  1034→  const processingFee = Number(order.processingFeeAmount ?? 0);
  1035→  const deliveryFee = Number(order.agreedDeliveryFee ?? order.shippingFee ?? 0);
  1036→  const discount = Number(order.discount ?? 0);
  1037→  const total = Number(order.totalAmount ?? subtotal + processingFee + deliveryFee - discount);
  1038→
  1039→  return (
  1040→    <section className="rounded-2xl border border-kwik-border bg-kwik-bg-surface p-5">
  1041→      <h2 className="font-semibold text-foreground">Order summary</h2>
  1042→      <dl className="mt-3 space-y-2 text-sm">
  1043→        <div className="flex items-center justify-between">
  1044→          <dt className="text-kwik-muted">Subtotal</dt>
  1045→          <dd className="font-medium text-foreground tabular-nums">
  1046→            {formatCurrency(subtotal)}
  1047→          </dd>
  1048→        </div>
  1049→        {processingFee > 0 && (
  1050→          <div className="flex items-center justify-between">
  1051→            <dt className="text-kwik-muted">
  1052→              Processing fee
  1053→              {order.processingFeePercent
  1054→                ? ` (${Number(order.processingFeePercent)}%)`
  1055→                : ""}
  1056→            </dt>
  1057→            <dd className="font-medium text-foreground tabular-nums">
  1058→              {formatCurrency(processingFee)}
  1059→            </dd>
  1060→          </div>
  1061→        )}
  1062→        <div className="flex items-center justify-between">
  1063→          <dt className="text-kwik-muted">Delivery fee</dt>
  1064→          <dd className="font-medium text-foreground tabular-nums">
  1065→            {order.deliveryMethod === "STANDARD_DELIVERY" &&
  1066→            order.quoteStatus !== "AGREED"
  1067→              ? "To be agreed"
  1068→              : formatCurrency(deliveryFee)}
  1069→          </dd>
  1070→        </div>
  1071→        {discount > 0 && (
  1072→          <div className="flex items-center justify-between text-kwik-green">
  1073→            <dt>Discount</dt>
  1074→            <dd className="font-medium tabular-nums">−{formatCurrency(discount)}</dd>

  1078→          <dt className="font-bold text-foreground">Total</dt>
  1079→          <dd className="font-bold text-kwik-orange tabular-nums">
  1080→            {formatCurrency(total)}
  1081→          </dd>
  1082→        </div>
  1083→      </dl>
  1084→    </section>
  1085→  );
  1086→}

  1090→function CustomerActions({ order }: { order: MarketplaceOrder }) {
  1091→  const router = useRouter();
  1092→  const confirmReceipt = useConfirmReceipt(order.id);
  1093→  const cancelOrder = useCancelOrder(order.id);
  1094→  const initializePayment = useInitializePayment(order.id);
  1095→
  1096→  const quoteAgreed = order.quoteStatus === "AGREED";
  1097→  const paymentPending = order.paymentStatus === "PENDING";
  1098→  const paymentPaid = order.paymentStatus === "PAID";
  1099→  const deliveryReady =
  1100→    order.delivery?.status === "DELIVERED" ||
  1101→    order.delivery?.status === "READY_FOR_PICKUP" ||
  1102→    order.delivery?.status === "ARRIVED";
  1103→  const alreadyConfirmed = order.delivery?.customerConfirmed === true;
  1104→  const isCancelled = order.status === "CANCELLED" || order.status === "REFUNDED";
  1105→
  1106→  // 1. Proceed to Payment — quote agreed, not yet paid
  1107→  const showProceedToPayment = quoteAgreed && paymentPending && !isCancelled;
  1108→
  1109→  // 2. Confirm receipt — paid AND (delivered OR ready for pickup)
  1110→  const showConfirmReceipt =
  1111→    paymentPaid && deliveryReady && !alreadyConfirmed && !isCancelled;
  1112→
  1113→  // 3. Cancel — payment pending AND quote not yet agreed (and not cancelled)
  1114→  const showCancel =
  1115→    paymentPending && !quoteAgreed && !isCancelled;
  1116→
  1117→  // 4. Track order link — show once paid+processing or beyond
  1118→  const showTrack =
  1119→    (paymentPaid || order.status === "PROCESSING" || order.status === "FULFILLED") &&
  1120→    !isCancelled;
  1121→
  1122→  if (!showProceedToPayment && !showConfirmReceipt && !showCancel && !showTrack) {
  1123→    return null;
  1124→  }
  1125→
  1126→  const handleProceedToPayment = async () => {
  1127→    try {
  1128→      const result = await initializePayment.mutateAsync();
  1129→      if (result.authorizationUrl) {
  1130→        kwikToast.info(
  1131→          "Redirecting to Paystack",
  1132→          "Complete your payment to confirm the order.",
  1133→        );
  1134→        window.location.href = result.authorizationUrl;
  1135→      } else {
  1136→        kwikToast.error(
  1137→          "Payment initialization failed",
  1138→          "No authorization URL was returned by the server.",
  1139→        );
  1140→      }
  1141→    } catch (err) {
  1142→      const message =
  1143→        err instanceof Error ? err.message : "Failed to initialize payment.";
  1144→      kwikToast.error("Couldn't initialize payment", message);
  1145→    }
  1146→  };
  1147→
  1148→  const handleConfirmReceipt = async () => {
  1149→    if (
  1150→      !window.confirm(
  1151→        `Confirm receipt of this order? ${KwisCrow.NAME} will release the funds to the vendor.`,
  1152→      )
  1153→    ) {
  1154→      return;
  1155→    }
  1156→    try {
  1157→      await confirmReceipt.mutateAsync();
  1158→      kwikToast.success(
  1159→        "Receipt confirmed",
  1160→        `${KwisCrow.NAME} has released the funds to the vendor. Thank you!`,
  1161→      );
  1162→    } catch (err) {
  1163→      const message =
  1164→        err instanceof Error ? err.message : "Failed to confirm receipt.";
  1165→      kwikToast.error("Couldn't confirm receipt", message);
  1166→    }
  1167→  };
  1168→
  1169→  const handleCancel = async () => {
  1170→    const reason = window.prompt("Reason for cancelling this order? (optional)");
  1171→    try {
  1172→      await cancelOrder.mutateAsync({ reason: reason || undefined });
  1173→      kwikToast.info("Order cancelled", "No payment was taken.");
  1174→      router.push("/orders");
  1175→    } catch (err) {
  1176→      const message =
  1177→        err instanceof Error ? err.message : "Failed to cancel order.";
  1178→      kwikToast.error("Couldn't cancel order", message);
  1179→    }
  1180→  };
  1181→
  1182→  return (
  1183→    <section className="rounded-2xl border border-kwik-border bg-kwik-bg-surface p-5">
  1184→      <h2 className="font-semibold text-foreground">Actions</h2>
  1185→      <div className="mt-3 flex flex-col gap-2 sm:flex-row sm:flex-wrap">
  1186→        {showProceedToPayment && (
  1187→          <button
  1188→            type="button"
  1189→            onClick={handleProceedToPayment}
  1190→            disabled={initializePayment.isPending}
  1191→            className="inline-flex h-11 items-center justify-center gap-2 rounded-md bg-secondary-500 px-5 text-sm font-bold text-white shadow-sm transition hover:bg-secondary-600 disabled:cursor-not-allowed disabled:opacity-70"
  1192→          >
  1193→            {initializePayment.isPending ? (
  1194→              <>
  1195→                <span className="h-4 w-4 animate-spin rounded-full border-2 border-white/40 border-t-white" />
  1196→                Initializing…
  1197→              </>
  1198→            ) : (
  1199→              <>
  1200→                <CreditCard className="h-4 w-4" />
  1201→                Proceed to payment
  1202→              </>
  1203→            )}
  1204→          </button>
  1205→        )}
  1206→        {showConfirmReceipt && (
  1207→          <button
  1208→            type="button"
  1209→            onClick={handleConfirmReceipt}
  1210→            disabled={confirmReceipt.isPending}
  1211→            className="inline-flex h-11 items-center justify-center gap-2 rounded-md bg-kwik-green px-5 text-sm font-bold text-white shadow-sm transition hover:brightness-105 disabled:cursor-not-allowed disabled:opacity-70"
  1212→          >
  1213→            {confirmReceipt.isPending ? (
  1214→              <>
  1215→                <span className="h-4 w-4 animate-spin rounded-full border-2 border-white/40 border-t-white" />
  1216→                Confirming…
  1217→              </>
  1218→            ) : (
  1219→              <>
  1220→                <CheckCircle2 className="h-4 w-4" />
  1221→                Confirm receipt
  1222→              </>
  1223→            )}
  1224→          </button>
  1225→        )}
  1226→        {showTrack && (
  1227→          <Link
  1228→            href={`/orders/${encodeURIComponent(order.id)}/track`}
  1229→            className="inline-flex h-11 items-center justify-center gap-2 rounded-md border border-kwik-border bg-kwik-bg-surface px-5 text-sm font-semibold text-foreground transition hover:border-kwik-orange hover:text-kwik-orange"
  1230→          >
  1231→            <Truck className="h-4 w-4" />
  1232→            Track order
  1233→          </Link>
  1234→        )}
  1235→        {showCancel && (
  1236→          <button
  1237→            type="button"
  1238→            onClick={handleCancel}
  1239→            disabled={cancelOrder.isPending}
  1240→            className="inline-flex h-11 items-center justify-center gap-2 rounded-md border border-kwik-red/30 px-5 text-sm font-semibold text-kwik-red transition hover:bg-kwik-red/5 disabled:cursor-not-allowed disabled:opacity-70"
  1241→          >
  1242→            {cancelOrder.isPending ? "Cancelling…" : "Cancel order"}
  1243→          </button>
  1244→        )}
  1245→      </div>
  1246→    </section>
  1247→  );
  1248→}
