   106→function formatCurrency(amount: number) {
   107→  return new Intl.NumberFormat("en-NG", {
   108→    style: "currency",
   109→    currency: "NGN",
   110→    maximumFractionDigits: 0,

   114→function unwrapApiData<T>(value: unknown): T {
   115→  if (value && typeof value === "object" && "data" in value) {
   116→    return (value as { data: T }).data;
   117→  }
   118→  return value as T;
   119→}

   121→function getSearchHistory() {
   122→  if (typeof window === "undefined") return [];
   123→  try {
   124→    const raw = localStorage.getItem("kwikseller-search-history");
   125→    return raw ? JSON.parse(raw) : [];
   126→  } catch {
   127→    return [];
   128→  }
   129→}

   131→function campaignHref(campaign: PoolCampaign) {
   132→  const slug = campaign.title
   133→    .toLowerCase()
   134→    .replace(/[^a-z0-9]+/g, "-")
   135→    .replace(/(^-|-$)/g, "")
   136→    .slice(0, 72);
   137→  return `/group-buy/${slug || "campaign"}-${campaign.id}`;
   138→}

   211→function PoolOfferCard({ offer }: { offer: PoolOffer }) {
   212→  const addItem = useCartStore((state) => state.addItem);
   213→  const isInCart = useCartStore((state) =>
   214→    state.items.some((item) => item.productId === offer.product?.id || item.poolOfferId === offer.id),
   215→  );
   216→  const productId = offer.product?.id;
   217→  const image = offer.product?.images?.[0]?.url ?? null;
   218→  const name = offer.product?.name ?? offer.poolProduct?.name ?? "Partner-fulfilled offer";
   219→
   220→  const addToCart = () => {
   221→    if (!productId) {
   222→      kwikToast.error("This Pool offer is not ready for checkout yet.");
   223→      return;
   224→    }
   225→
   226→    addItem({
   227→      productId,
   228→      poolOfferId: offer.id,
   229→      name,
   230→      price: offer.retailPrice,
   231→      comparePrice: offer.poolProduct?.suggestedRetailPrice,
   232→      image: image ?? "",
   233→      store: offer.store?.name,
   234→      storeId: offer.store?.id,
   235→      storeSlug: offer.store?.slug,
   236→      storeName: offer.store?.name,
   237→      productType: "PHYSICAL",
   238→      productSource: "POOL_RESALE",
   239→      requiresShipping: true,
   240→    });
   241→    kwikToast.success("Offer added to cart");
   242→  };
   243→
   244→  return (
   245→    <article className="group border-b border-neutral-200 pb-4 dark:border-white/10">
   246→      <div className="relative aspect-[3/4] min-h-[148px] overflow-hidden bg-neutral-100 sm:min-h-[190px] md:aspect-[4/5]">
   247→        <AppImage src={image} alt={name} className="h-full w-full object-cover transition duration-500 group-hover:scale-105" fallbackVariant="product" />
   248→      </div>
   249→      <div className="mt-3 space-y-3">
   250→        <div>
   251→          <h3 className="line-clamp-2 text-sm font-semibold text-kwik-dark dark:text-white">{name}</h3>
   252→          <p className="mt-1 line-clamp-1 text-xs text-kwik-muted dark:text-white/55">{offer.store?.name ?? "Verified vendor"}</p>
   253→        </div>
   254→        <div className="flex items-end justify-between gap-3">
   255→          <div>
   256→            <p className="text-base font-bold text-kwik-dark dark:text-white">{formatCurrency(offer.retailPrice)}</p>
   257→          </div>
   258→          <button
   259→            type="button"
   260→            onClick={addToCart}
   261→            className={`inline-flex h-8 w-8 items-center justify-center rounded-md text-white transition ${
   262→              isInCart ? "bg-emerald-600 hover:bg-emerald-700" : "bg-kwik-orange hover:bg-kwik-orange-hover"
   263→            }`}
   264→            aria-label={isInCart ? "Added to cart" : "Add to cart"}
   265→          >
   266→            {isInCart ? <Check className="h-3.5 w-3.5" /> : <ShoppingCart className="h-3.5 w-3.5" />}
   267→          </button>
   268→        </div>
   269→      </div>
   270→    </article>
   271→  );
   272→}

   274→function CampaignRow({ campaign }: { campaign: PoolCampaign }) {
   275→  const progress =
   276→    campaign.targetQuantity > 0
   277→      ? Math.min(100, Math.round((campaign.committedQuantity / campaign.targetQuantity) * 100))
   278→      : 0;
   279→
   280→  return (
   281→    <div className="border-b border-white/15 py-4 transition hover:border-white/35">
   282→      <div className="flex items-start justify-between gap-4">
   283→        <Link href={campaignHref(campaign)} className="min-w-0 flex-1">
   284→          <p className="text-xs font-semibold uppercase text-cyan-100">{campaign.status}</p>
   285→          <h3 className="mt-1 text-sm font-semibold text-white">{campaign.title}</h3>
   286→          <p className="mt-1 text-xs text-white/65">{campaign.poolProduct?.name ?? "Pool campaign"}</p>
   287→        </Link>
   288→        <p className="text-sm font-bold text-white">{formatCurrency(campaign.unitPrice)}</p>
   289→      </div>
   290→      <div className="mt-4 h-1.5 overflow-hidden rounded-full bg-white/15">
   291→        <div className="h-full rounded-full bg-white" style={{ width: `${progress}%` }} />
   292→      </div>
   293→      <div className="mt-2 flex justify-between text-xs text-white/65">
   294→        <span>{campaign.committedQuantity} committed</span>
   295→        <span>{campaign.targetQuantity} target</span>
   296→      </div>
   297→      <div className="mt-3 flex gap-2">
   298→        <Link
   299→          href={campaignHref(campaign)}
   300→          className="inline-flex h-8 flex-1 items-center justify-center rounded-md border border-white/20 text-xs font-semibold text-white transition hover:bg-white/10"
   301→        >
   302→          View details
   303→        </Link>
   304→        <button
   305→          type="button"
   306→          onClick={() => kwikToast.success(`Subscribed to ${campaign.title}`)}
   307→          className="inline-flex h-8 flex-1 items-center justify-center rounded-md bg-white text-xs font-semibold text-kwik-blue"
   308→        >
   309→          Subscribe
   310→        </button>
   311→      </div>
   312→    </div>
   313→  );
   314→}

   334→export function MarketplaceHomeFeedPage() {
   335→  const [feed, setFeed] = React.useState<HomeFeedResponse | null>(null);
   336→  const [poolOffers, setPoolOffers] = React.useState<PoolOffer[]>([]);
   337→  const [campaigns, setCampaigns] = React.useState<PoolCampaign[]>([]);
   338→  const [homeSellers, setHomeSellers] = React.useState<HomeSeller[]>([]);
   339→  const [isLoading, setIsLoading] = React.useState(true);
   340→  const [error, setError] = React.useState<string | null>(null);
   341→  const [activeBanner, setActiveBanner] = React.useState(0);
   342→  const [quickViewProduct, setQuickViewProduct] = React.useState<MarketplaceProduct | null>(null);
   343→  const setCachedHomeFeed = useHomeFeedStore((state) => state.setHomeFeed);
   344→  const cartItems = useCartStore((state) => state.items);
   345→  const wishlistItems = useWishlistStore((state) => state.items);
   346→  const recentlyViewedItems = useRecentlyViewedStore((state) => state.items);
   347→
   348→  React.useEffect(() => {
   349→    let isMounted = true;
   350→
   351→    const loadFeed = async () => {
   352→      const navEntry = performance.getEntriesByType("navigation")[0] as PerformanceNavigationTiming | undefined;
   353→      const isPageReload = navEntry?.type === "reload";
   354→
   355→      const cachedHome = useHomeFeedStore.getState();
   356→      if (!isPageReload && cachedHome.isFresh() && cachedHome.feed) {
   357→        setFeed(cachedHome.feed);
   358→        setPoolOffers(cachedHome.poolOffers);
   359→        setCampaigns(cachedHome.campaigns);
   360→        setIsLoading(false);
   361→        return;
   362→      }
   363→
   364→      setIsLoading(true);
   365→      setError(null);
   366→      try {
   367→        const [homeResponse, poolResponse, campaignsResponse] = await Promise.allSettled([
   368→          marketplaceApi.getHomeFeed(),
   369→          marketplaceApi.getPoolOffers({ limit: 6 }),
   370→          marketplaceApi.getPoolCampaigns({ limit: 4 }),
   371→        ]);
   372→
   373→        if (!isMounted) return;
   374→        if (homeResponse.status === "fulfilled") {
   375→          const nextFeed = unwrapApiData<HomeFeedResponse>(homeResponse.value.data);
   376→          const nextPoolOffers = poolResponse.status === "fulfilled"
   377→            ? unwrapApiData<PoolOffer[]>(poolResponse.value.data)
   378→            : [];
   379→          const nextCampaigns = campaignsResponse.status === "fulfilled"
   380→            ? unwrapApiData<PoolCampaign[]>(campaignsResponse.value.data)
   381→            : [];
   382→
   383→          setFeed(nextFeed);
   384→          setPoolOffers(nextPoolOffers);
   385→          setCampaigns(nextCampaigns);
   386→          setCachedHomeFeed({
   387→            feed: nextFeed,
   388→            poolOffers: nextPoolOffers,
   389→            campaigns: nextCampaigns,
   390→          });
   391→        } else {
   392→          throw homeResponse.reason;
   393→        }
   394→      } catch (err) {
   395→        if (isMounted) {
   396→          // Graceful fallback: render the curated mock feed so the homepage
   397→          // stays rich and explorable when the live API is unreachable
   398→          // (e.g. preview/staging without a backend, or a network blip).
   399→          setFeed(mockHomeFeed);
   400→          setPoolOffers(mockPoolOffers);
   401→          setCampaigns(mockPoolCampaigns);
   402→          setError(null);
   403→        }
   404→      } finally {
   405→        if (isMounted) setIsLoading(false);
   406→      }
   407→    };
   408→
   409→    loadFeed();
   410→    return () => {
   411→      isMounted = false;
   412→    };
   413→  }, [setCachedHomeFeed]);
   414→
   415→  React.useEffect(() => {
   416→    let isMounted = true;
   417→    marketplaceApi.getSellers({ limit: 4 })
   418→      .then((response) => {
   419→        if (!isMounted) return;
   420→        const data = unwrapApiData<any>(response.data);
   421→        const list = Array.isArray(data) ? data : data?.sellers || [];
   422→        setHomeSellers(list.map((seller: any, index: number) => {
   423→          const name = seller.name || seller.storeName || "Kwikseller vendor";
   424→          const slug = seller.username || seller.slug || seller.storeUsername || name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "") || String(seller.id ?? index);
   425→          return {
   426→            id: String(seller.id ?? slug),
   427→            name,
   428→            slug,
   429→            image: seller.banner || seller.bannerUrl || seller.coverUrl || seller.image || null,
   430→            logo: seller.logo || seller.logoUrl || null,
   431→            productCount: Number(seller.productCount ?? seller.products ?? 0),
   432→            isVerified: Boolean(seller.isVerified ?? seller.verified ?? true),
   433→          };
   434→        }));
   435→      })
   436→      .catch(() => {
   437→        if (isMounted) setHomeSellers([]);
   438→      });
   439→    return () => {
   440→      isMounted = false;
   441→    };
   442→  }, []);
   443→
   444→  React.useEffect(() => {
   445→    const count = feed?.heroBanners.length ?? 0;
   446→    if (count < 2) return undefined;
   447→
   448→    const interval = window.setInterval(() => {
   449→      setActiveBanner((current) => (current + 1) % count);
   450→    }, 4500);
   451→
   452→    return () => window.clearInterval(interval);
   453→  }, [feed?.heroBanners.length]);
   454→
   455→  if (isLoading) return <MarketplaceHomeSkeleton />;
   456→
   457→  if (error || !feed) {
   458→    return (
   459→      <div className="bg-background px-4 py-20">
   460→        <div className="container mx-auto max-w-2xl text-center">
   461→          <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-full border border-neutral-200">
   462→            <PackageOpen className="h-7 w-7 text-kwik-orange" />
   463→          </div>
   464→          <h1 className="mt-5 text-2xl font-semibold text-kwik-dark dark:text-white">Marketplace could not load</h1>
   465→          <p className="mx-auto mt-2 max-w-md text-sm leading-6 text-kwik-muted dark:text-white/60">
   466→            {error || "We could not reach the live catalog right now. Please try again in a moment."}
   467→          </p>
   468→          <button
   469→            type="button"
   470→            onClick={() => window.location.reload()}
   471→            className="mt-6 inline-flex h-11 items-center justify-center rounded-md bg-kwik-dark px-5 text-sm font-semibold text-white"
   472→          >
   473→            Retry marketplace
   474→          </button>
   475→        </div>
   476→      </div>
   477→    );
   478→  }
   479→
   480→  const banners = feed.heroBanners.length
   481→    ? feed.heroBanners
   482→    : [
   483→        {
   484→          id: "fallback",
   485→          title: "Pool-ready marketplace",
   486→          subtitle: "Shop vendor stock, resale offers, group buys, and digital products in one cart.",
   487→          image: null,
   488→          href: "/search",
   489→          badge: "Kwikseller",
   490→        },
   491→      ];
   492→  const banner = banners[activeBanner % banners.length];
   493→  const memberSignals = {
   494→    cartProductIds: cartItems.map((item) => item.productId),
   495→    wishlistProductIds: wishlistItems.map((item) => item.id),
   496→    recentlyViewedIds: recentlyViewedItems.map((item) => item.id),
   497→    searchHistory: getSearchHistory(),
   498→  };
   499→  const rankedFeatured = rankProductsForMember(feed.featuredProducts, memberSignals);
   500→  const rankedTrending = rankProductsForMember(feed.trendingProducts, memberSignals);
   501→  const stockProducts = rankedFeatured.filter((item) => item.productSource !== "POOL_RESALE");
   502→  const digitalProducts = [...rankedFeatured, ...rankedTrending].filter(
   503→    (item) => item.productType === "DIGITAL",
   504→  );
   505→
   506→  return (
   507→    <div className="bg-background pb-12">
   508→      <section className="border-b border-border bg-background">
   509→        <div className="container mx-auto px-4 py-5 md:py-7">
   510→          <div className="lg:hidden">
   511→            <Link href={banner.href || "/search"} className="block">
   512→              <div className="aspect-[16/10] overflow-hidden bg-neutral-100">
   513→                <AppImage src={banner.image} alt={banner.title} className="h-full w-full object-cover" fallbackVariant="product" />
   514→              </div>
   515→            </Link>
   516→            {banners.length > 1 && (
   517→              <div className="mt-3 flex justify-center gap-2" aria-label="Marketplace banners">
   518→                {banners.map((item, index) => (
   519→                  <button
   520→                    key={`${item.id}-${index}`}
   521→                    type="button"
   522→                    onClick={() => setActiveBanner(index)}
   523→                    className={`h-1.5 rounded-full transition-all ${index === activeBanner ? "w-6 bg-kwik-dark" : "w-1.5 bg-neutral-300"}`}
   524→                    aria-label={`Show banner ${index + 1}`}
   525→                  />
   526→                ))}
   527→              </div>
   528→            )}
   529→          </div>
   530→
   531→          <div className="hidden gap-5 lg:grid lg:grid-cols-[minmax(0,1.45fr)_380px] xl:grid-cols-[minmax(0,1.55fr)_430px]">
   532→            <Link href={banner.href || "/search"} className="block h-[390px] overflow-hidden bg-neutral-100 xl:h-[430px]">
   533→              <AppImage src={banner.image} alt={banner.title} className="h-full w-full object-cover transition duration-700 hover:scale-[1.02]" fallbackVariant="product" />
   534→            </Link>
   535→
   536→            <aside className="flex h-[390px] flex-col justify-between border border-neutral-200 p-6 dark:border-white/10 xl:h-[430px]">
   537→              <div>
   538→                <p className="text-xs font-semibold uppercase tracking-wide text-kwik-orange">{banner.badge || "Kwikseller Pool"}</p>
   539→                <h1 className="mt-4 text-3xl font-semibold leading-tight text-kwik-dark dark:text-white xl:text-4xl">{banner.title}</h1>
   540→                <p className="mt-3 text-sm leading-6 text-kwik-muted dark:text-white/60">{banner.subtitle}</p>
   541→              </div>
