    53→  delay = 0,

    64→      initial={{ opacity: 0, y: 30 }}

    67→      className={className}
    68→    >
    69→      {children}
    70→    </motion.div>
    71→  );

   403→function HowItWorksStep({
   404→  step,
   405→  index,
   406→  totalSteps,
   407→}: {
   408→  step: (typeof howItWorksSteps)[number];
   409→  index: number;
   410→  totalSteps: number;
   411→}) {
   412→  const ref = useRef<HTMLDivElement>(null);
   413→  const isInView = useInView(ref, { once: true, margin: "-80px" });
   414→  const isOdd = step.number % 2 === 1;
   415→
   416→  return (
   417→    <motion.div
   418→      ref={ref}
   419→      initial={{ opacity: 0, y: 50 }}
   420→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 50 }}
   421→      transition={{ duration: 0.7, delay: index * 0.05, ease: "easeOut" }}
   422→    >
   423→      {/* Connecting line (except last) */}
   424→      {index < totalSteps - 1 && (
   425→        <div className="hidden md:flex justify-center">
   426→          <div className="w-px border-l-2 border-dashed border-default-200 my-4" />
   427→        </div>
   428→      )}
   429→
   430→      <div
   431→        className={cn(
   432→          "flex flex-col gap-8 md:gap-12 lg:gap-16",
   433→          isOdd ? "md:flex-row" : "md:flex-row-reverse",
   434→        )}
   435→      >
   436→        {/* Image side */}
   437→        <div className="w-full md:w-1/2 max-w-lg mx-auto">
   438→          <StepImage step={step} isInView={isInView} />
   439→        </div>
   440→
   441→        {/* Text side */}
   442→        <motion.div
   443→          initial={{ opacity: 0, x: isOdd ? 30 : -30 }}
   444→          animate={
   445→            isInView
   446→              ? { opacity: 1, x: 0 }
   447→              : { opacity: 0, x: isOdd ? 30 : -30 }
   448→          }
   449→          transition={{ duration: 0.6, delay: 0.25, ease: "easeOut" }}
   450→          className="w-full md:w-1/2 flex flex-col justify-center"
   451→        >
   452→          {/* Step number */}
   453→          <div className="flex items-center gap-3 mb-5">
   454→            <div
   455→              className={cn(
   456→                "w-12 h-12 rounded-full flex items-center justify-center text-white font-bold text-lg shadow-md",
   457→                step.color,
   458→              )}
   459→            >
   460→              {step.number}
   461→            </div>
   462→            <div className={cn("flex-1 h-0.5 bg-accent/30 rounded-full")} />
   463→          </div>
   464→
   465→          <h3 className="text-2xl md:text-3xl font-bold mb-4">{step.title}</h3>
   466→          <p className="text-default-500 text-base md:text-lg leading-relaxed max-w-md">
   467→            {step.description}
   468→          </p>
   469→
   470→          {/* Step progress */}
   471→          <div className="flex items-center gap-2 mt-6">
   472→            {howItWorksSteps.map((s) => (
   473→              <div
   474→                key={s.number}
   475→                className={cn(
   476→                  "h-1.5 rounded-full transition-all duration-500",

   484→              Step {step.number} of {totalSteps}
   485→            </span>
   486→          </div>
   487→        </motion.div>
   488→      </div>
   489→    </motion.div>
   490→  );
