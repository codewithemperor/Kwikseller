    37→    37→function formatCount(n: number): string {
    38→    38→  if (n >= 1000) return `${(n / 1000).toFixed(1).replace(/\.0$/, "")}K`;
    39→    39→  return String(n);
    40→    40→}

    42→    42→function timeAgo(ts: number): string {
    43→    43→  const diff = Date.now() - ts;
    44→    44→  const mins = Math.floor(diff / 60000);
    45→    45→  if (mins < 1) return "just now";
    46→    46→  if (mins < 60) return `${mins}m ago`;
    47→    47→  const hours = Math.floor(mins / 60);
    48→    48→  if (hours < 24) return `${hours}h ago`;
    49→    49→  const days = Math.floor(hours / 24);
    50→    50→  if (days < 7) return `${days}d ago`;
    51→    51→  return new Intl.DateTimeFormat("en-NG", { dateStyle: "medium" }).format(new Date(ts));
    52→    52→}

    60→    60→function isFiltersEmpty(f: SearchFiltersState): boolean {
    61→    61→  return (
    62→    62→    f.minPrice === undefined &&
    63→    63→    f.maxPrice === undefined &&
    64→    64→    f.rating === undefined &&
    65→    65→    !f.category &&
    66→    66→    !f.brandId &&
    67→    67→    !f.storeId &&
    68→    68→    !f.state
    69→    69→  );
    70→    70→}

    72→    72→function parseFiltersFromParams(
    73→    73→  params: URLSearchParams,
    74→    74→): { query: string; filters: SearchFiltersState; sort: SortValue } {
    75→    75→  const query = (params.get("q") ?? "").trim();
    76→    76→  const filters: SearchFiltersState = {
    77→    77→    minPrice: params.get("minPrice") ? Number(params.get("minPrice")) : undefined,
    78→    78→    maxPrice: params.get("maxPrice") ? Number(params.get("maxPrice")) : undefined,
    79→    79→    rating: params.get("rating") ? Number(params.get("rating")) : undefined,
    80→    80→    category: params.get("category") ?? undefined,
    81→    81→    brandId: params.get("brandId") ?? undefined,
    82→    82→    storeId: params.get("storeId") ?? undefined,
    83→    83→    state: params.get("state") ?? undefined,
    84→    84→  };
    85→    85→  // Strip NaN / undefined
    86→    86→  for (const k of Object.keys(filters) as (keyof SearchFiltersState)[]) {
    87→    87→    const v = filters[k];
    88→    88→    if (v === undefined || v === null || (typeof v === "number" && Number.isNaN(v))) {
    89→    89→      delete filters[k];
    90→    90→    }
    91→    91→  }
    92→    92→  const sort = (params.get("sort") as SortValue) || "relevance";
    93→    93→  return { query, filters, sort };
    94→    94→}
    95→    95→
    96→    96→function buildSearchParams(
    97→    97→  query: string,
    98→    98→  filters: SearchFiltersState,
    99→    99→  sort: SortValue,
   100→   100→): URLSearchParams {
   101→   101→  const params = new URLSearchParams();
   102→   102→  if (query) params.set("q", query);
   103→   103→  if (filters.minPrice !== undefined && !Number.isNaN(filters.minPrice)) {
   104→   104→    params.set("minPrice", String(filters.minPrice));
   105→   105→  }
   106→   106→  if (filters.maxPrice !== undefined && !Number.isNaN(filters.maxPrice)) {
   107→   107→    params.set("maxPrice", String(filters.maxPrice));
   108→   108→  }
   109→   109→  if (filters.rating !== undefined && !Number.isNaN(filters.rating)) {
   110→   110→    params.set("rating", String(filters.rating));
   111→   111→  }
   112→   112→  if (filters.category) params.set("category", filters.category);
   113→   113→  if (filters.brandId) params.set("brandId", filters.brandId);
   114→   114→  if (filters.storeId) params.set("storeId", filters.storeId);
   115→   115→  if (filters.state) params.set("state", filters.state);
   116→   116→  if (sort !== "relevance") params.set("sort", sort);
   117→   117→  return params;
   118→   118→}
