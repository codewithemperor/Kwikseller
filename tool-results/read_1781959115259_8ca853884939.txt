   115→function formatRelativeTime(dateStr: string): string {
   116→  if (!dateStr) return "";
   117→  const now = Date.now();
   118→  const then = new Date(dateStr).getTime();
   119→  const diffMs = now - then;
   120→  const diffSec = Math.floor(diffMs / 1000);
   121→  const diffMin = Math.floor(diffSec / 60);
   122→  const diffHr = Math.floor(diffMin / 60);
   123→  const diffDay = Math.floor(diffHr / 24);
   124→
   125→  if (diffSec < 60) return "just now";
   126→  if (diffMin < 60) return `${diffMin}m ago`;
   127→  if (diffHr < 24) return `${diffHr}h ago`;
   128→  if (diffDay < 7) return `${diffDay}d ago`;
   129→  return formatDate(dateStr);
   130→}

   160→// ==================== Countdown Timer Component ====================

   162→function CountdownTimer({ targetDate }: { targetDate: string }) {
   163→  const [timeLeft, setTimeLeft] = React.useState<{
   164→    hours: number;
   165→    minutes: number;
   166→    seconds: number;
   167→    expired: boolean;
   168→  } | null>(null);
   169→
   170→  React.useEffect(() => {
   171→    const calculate = () => {
   172→      const now = Date.now();
   173→      const target = new Date(targetDate).getTime();
   174→      const diff = target - now;
   175→
   176→      if (diff <= 0) {
   177→        setTimeLeft({ hours: 0, minutes: 0, seconds: 0, expired: true });
   178→        return;
   179→      }
   180→
   181→      setTimeLeft({
   182→        hours: Math.floor(diff / (1000 * 60 * 60)),
   183→        minutes: Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)),
   184→        seconds: Math.floor((diff % (1000 * 60)) / 1000),
   185→        expired: false,
   186→      });
   187→    };
   188→
   189→    calculate();
   190→    const interval = setInterval(calculate, 1000);
   191→    return () => clearInterval(interval);
   192→  }, [targetDate]);
   193→
   194→  if (!timeLeft) {
   195→    return <span className="text-xs text-gray-400">...</span>;
   196→  }
   197→
   198→  if (timeLeft.expired) {
   199→    return (
   200→      <span className="text-xs font-medium text-amber-600">
   201→        Releasing soon...
   202→      </span>
   203→    );
   204→  }
   205→
   206→  const pad = (n: number) => String(n).padStart(2, "0");
   207→
   208→  return (
   209→    <span className="font-mono text-xs tabular-nums text-amber-600">
   210→      Release in {pad(timeLeft.hours)}h {pad(timeLeft.minutes)}m {pad(timeLeft.seconds)}s
   211→    </span>
   212→  );
   213→}
