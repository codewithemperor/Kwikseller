    53→  children,
    54→  className = "",
    55→  delay = 0,
    56→}: {

    70→    >
    71→      {children}
    72→    </motion.div>
    73→  );

   104→function CounterAnimation({
   105→  target,
   106→  suffix = "",
   107→  prefix = "",
   108→  display,
   109→}: {
   110→  target: number;
   111→  suffix?: string;
   112→  prefix?: string;
   113→  display?: string;
   114→}) {
   115→  const [count, setCount] = useState(0);
   116→  const [done, setDone] = useState(false);
   117→  const ref = useRef<HTMLSpanElement>(null);
   118→  const isInView = useInView(ref, { once: true });
   119→  React.useEffect(() => {
   120→    if (!isInView) return;
   121→    let start = 0;
   122→    const duration = 2000;
   123→    const increment = target / (duration / 16);
   124→    const timer = setInterval(() => {
   125→      start += increment;
   126→      if (start >= target) {
   127→        setCount(target);
   128→        setDone(true);
   129→        clearInterval(timer);
   130→      } else {
   131→        setCount(Math.floor(start));
   132→      }
   133→    }, 16);
   134→    return () => clearInterval(timer);
   135→  }, [isInView, target]);
   136→  return (
   137→    <span ref={ref}>
   138→      {done && display
   139→        ? display
   140→        : `${prefix}${count.toLocaleString()}${suffix}`}
   141→    </span>
   142→  );
   143→}

   145→function AnimatedNumber({
   146→  value,
   147→  decimals = 1,
   148→}: {
   149→  value: number;
   150→  decimals?: number;
   151→}) {
   152→  const ref = useRef<HTMLSpanElement>(null);
   153→  const isInView = useInView(ref, { once: true });
   154→  const motionVal = useMotionValue(0);
   155→  const rounded = useTransform(motionVal, (v) => v.toFixed(decimals));
   156→
   157→  React.useEffect(() => {
   158→    if (isInView) {
   159→      const controls = animate(motionVal, value, {
   160→        duration: 1.2,
   161→        ease: "easeOut",
   162→      });
   163→      return controls.stop;
   164→    }
   165→  }, [isInView, motionVal, value]);
   166→
   167→  return <motion.span ref={ref}>{rounded}</motion.span>;
   168→}

   264→function inferCategory(haystack: string): VendorCategory {
   265→  const lc = haystack.toLowerCase();
   266→  for (const { match, category } of CATEGORY_KEYWORDS) {
   267→    if (lc.includes(match)) return category;
   268→  }
   269→  return "Fashion";
   270→}

   272→function initialsFor(name: string) {
   273→  return (
   274→    name
   275→      .split(/\s+/)
   276→      .filter(Boolean)
   277→      .slice(0, 2)
   278→      .map((part) => part[0]?.toUpperCase())
   279→      .join("") || "KS"
   280→  );
   281→}

   283→function formatCount(n: number): string {
   284→  if (n >= 1000) return `${(n / 1000).toFixed(1).replace(/\.0$/, "")}K`;
   285→  return String(n);
   286→}

   288→function toVendorData(
   289→  store: {
   290→    id: string;
   291→    name: string;
   292→    slug?: string;
   293→    description?: string | null;
   294→    logoUrl?: string | null;
   295→    bannerUrl?: string | null;
   296→    rating?: number;
   297→    reviewCount?: number;
   298→    productCount?: number;
   299→    location?: string | null;
   300→    isVerified?: boolean;
   301→    category?: string | null;
   302→  },
   303→  index: number,
   304→): VendorData {
   305→  const category = inferCategory(
   306→    `${store.name} ${store.description ?? ""} ${store.category ?? ""}`,
   307→  );
   308→  const rating = Number(store.rating ?? 0);
   309→  const reviewCount = Number(store.reviewCount ?? 0);
   310→  const productCount = Number(store.productCount ?? 0);
   311→  const badge: VendorData["badge"] =
   312→    rating >= 4.8 ? "Top Rated" : productCount >= 100 ? "Featured" : "Verified";
   313→  const badgeColor =
   314→    badge === "Featured"
   315→      ? "bg-amber-500"
   316→      : badge === "Top Rated"
   317→        ? "bg-emerald-500"
   318→        : "bg-gray-500";
   319→  return {
   320→    id: String(store.id),
   321→    slug: store.slug,
   322→    storeName: store.name,
   323→    initials: initialsFor(store.name),
   324→    category,
   325→    location: store.location || "Africa",
   326→    description:
   327→      store.description ||
   328→      "Verified vendor on Kwikseller — quality products, escrow protected.",
   329→    products: formatCount(productCount),
   330→    rating,
   331→    sold: formatCount(reviewCount),
   332→    badge,
   333→    badgeColor,
   334→    coverColor: COVER_COLORS[index % COVER_COLORS.length],
   335→    tags: [category],
   336→    isVerified: Boolean(store.isVerified ?? true),
   337→  };
   338→}

   642→  const [searchQuery, setSearchQuery] = useState("");
   643→  const [sortBy, setSortBy] = useState<VendorSortValue>("rating");
   644→  const [isSortOpen, setIsSortOpen] = useState(false);

   647→  // the dummy API when NEXT_PUBLIC_USE_DUMMY_DATA=true, the real backend
   648→  // otherwise — no hardcoded fallback list).
   649→  const storesQuery = useStores();
   650→  const isLoadingVendors = storesQuery.isLoading;
   651→  const apiVendors = useMemo<VendorData[]>(
   652→    () =>
   653→      ((storesQuery.data ?? []) as Array<Record<string, unknown>>).map(
   654→        (store, index) =>
   655→          toVendorData(
   656→            {
   657→              id: String(store.id ?? index),
   658→              name: String(store.name ?? store.storeName ?? "Kwikseller vendor"),
   659→              slug: store.slug ? String(store.slug) : store.username ? String(store.username) : undefined,
   660→              description: (store.description ?? store.tagline) as string | null,
   661→              logoUrl: (store.logoUrl ?? store.logo) as string | null,
   662→              bannerUrl: store.bannerUrl as string | null,
   663→              rating: Number(store.rating ?? store.averageRating ?? 0),
   664→              reviewCount: Number(store.reviewCount ?? 0),
   665→              productCount: Number(store.productCount ?? 0),
   666→              location: (store.location ?? (store.address as { state?: string } | undefined)?.state) as string | null,
   667→              isVerified: Boolean(store.isVerified ?? store.verified ?? true),
   668→              category: store.category as string | null,
   669→            },
   670→            index,
   671→          ),
   672→      ),
   673→    [storesQuery.data],
   674→  );
   675→
   676→  const filteredVendors = useMemo(() => {
   677→    const result = apiVendors.filter((v) => {
   678→      const matchesCategory =
   679→        activeCategory === "All" || v.category === activeCategory;
   680→      const matchesSearch =
   681→        searchQuery.trim() === "" ||
   682→        v.storeName.toLowerCase().includes(searchQuery.toLowerCase()) ||
   683→        v.category.toLowerCase().includes(searchQuery.toLowerCase()) ||
   684→        v.location.toLowerCase().includes(searchQuery.toLowerCase()) ||
   685→        v.tags.some((t) => t.toLowerCase().includes(searchQuery.toLowerCase()));
   686→      return matchesCategory && matchesSearch;
   687→    });
   688→
   689→    switch (sortBy) {
   690→      case "az":
   691→        result.sort((a, b) => a.storeName.localeCompare(b.storeName));
   692→        break;
   693→      case "za":
   694→        result.sort((a, b) => b.storeName.localeCompare(a.storeName));
   695→        break;
   696→      case "rating":
   697→        result.sort((a, b) => b.rating - a.rating);
   698→        break;
   699→    }
   700→
   701→    return result;
   702→  }, [activeCategory, searchQuery, sortBy, apiVendors]);
   703→

   705→    <>
   706→      <section className="relative z-[95] border-b border-divider bg-white dark:bg-background">
   707→        <div className="container mx-auto px-4 py-4">

   710→              <h1 className="text-xl font-bold text-kwik-dark dark:text-white">Vendors</h1>

   712→                {isLoadingVendors ? "Loading verified sellers" : "Browse verified sellers"}
   713→              </p>
   714→            </div>
   715→            <div className="relative shrink-0">
   716→              <button
   717→                type="button"
   718→                onClick={() => setIsSortOpen((value) => !value)}
   719→                className="inline-flex h-9 items-center gap-2 rounded-md border border-kwik-border bg-white px-3 text-xs font-semibold text-kwik-dark transition hover:border-kwik-orange dark:border-white/10 dark:bg-white/5 dark:text-white"
   720→                aria-expanded={isSortOpen}
   721→              >
   722→                <SlidersHorizontal className="h-3.5 w-3.5 text-kwik-orange" />
   723→                Sort
   724→              </button>
   725→              {isSortOpen && (
   726→                <div className="absolute right-0 top-11 z-[110] w-40 overflow-hidden rounded-md border border-kwik-border bg-background shadow-xl">
   727→                  {VENDOR_SORT_OPTIONS.map((opt) => (
   728→                <button
   729→                  key={opt.value}
   730→                  type="button"
   731→                  onClick={() => {
   732→                    setSortBy(opt.value);
   733→                    setIsSortOpen(false);
   734→                  }}
   735→                  className={`block w-full px-3 py-2 text-left text-xs font-medium transition ${
   736→                    sortBy === opt.value
   737→                      ? "bg-kwik-orange text-white"
   738→                      : "text-kwik-gray-light hover:bg-neutral-50 hover:text-kwik-dark dark:text-white/65 dark:hover:bg-white/10 dark:hover:text-white"
   739→                  }`}
   740→                >
   741→                  {opt.label}
   742→                </button>
   743→                  ))}
   744→                </div>
   745→              )}
   746→            </div>
   747→          </div>
   748→        </div>
   749→      </section>
   750→      {/* Hero section is intentionally omitted in this refactor — the

   752→          affordance. Keeping the layout compact for a focused browse flow. */}
   753→
   754→      {/* ─── 2. Vendor Categories Filter ─────────────────────── */}

   756→        <div className="container mx-auto px-0 md:px-4">
