    45→function AnimatedSection({
    46→  children,
    47→  className = "",
    48→  delay = 0,
    49→}: {
    50→  children: React.ReactNode;
    51→  className?: string;
    52→  delay?: number;
    53→}) {
    54→  const ref = useRef<HTMLDivElement>(null);
    55→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    56→  return (
    57→    <motion.div
    58→      ref={ref}
    59→      initial={{ opacity: 0, y: 30 }}
    60→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    61→      transition={{ duration: 0.5, delay, ease: "easeOut" as const }}
    62→      className={className}
    63→    >
    64→      {children}
    65→    </motion.div>
    66→  );
    67→}
