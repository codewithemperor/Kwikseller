    84→function formatNGN(n: number) {
    85→  return new Intl.NumberFormat("en-NG", {
    86→    style: "currency",
    87→    currency: "NGN",

    92→function formatDateTime(value?: string | null) {
    93→  if (!value) return "—";
    94→  return new Intl.DateTimeFormat("en-NG", { dateStyle: "medium", timeStyle: "short" }).format(new Date(value));

    97→// Map order status → step index in the fulfilment journey.
    98→const STEPS = [
    99→  { key: "PENDING", label: "Order Placed", icon: Package, description: "Vendor received your order" },
   100→  { key: "CONFIRMED", label: "Confirmed", icon: CheckCircle2, description: "Vendor confirmed & quoted delivery" },
   101→  { key: "READY", label: "Ready for Dispatch", icon: PackageCheck, description: "Packed and ready to ship" },
   102→  { key: "SHIPPED", label: "Shipped", icon: Truck, description: "Out for delivery" },
   103→  { key: "DELIVERED", label: "Delivered", icon: Home, description: "Package delivered" },
   104→];
   105→
   106→function statusToStepIndex(status: string): number {
   107→  const idx = STEPS.findIndex((s) => s.key === status);
   108→  if (status === "OUT_FOR_DELIVERY") return 3;
   109→  if (status === "CANCELLED" || status === "REJECTED") return -1;
   110→  return idx === -1 ? 0 : idx;
   111→}
