    52→function AnimatedSection({
    53→  children,
    54→  className = "",
    55→  delay = 0,
    56→}: {
    57→  children: React.ReactNode;
    58→  className?: string;
    59→  delay?: number;
    60→}) {
    61→  const ref = useRef<HTMLDivElement>(null);
    62→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    63→  return (
    64→    <motion.div
    65→      ref={ref}
    66→      initial={{ opacity: 0, y: 30 }}
    67→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    68→      transition={{ duration: 0.5, delay, ease: "easeOut" as const }}
    69→      className={className}
    70→    >
    71→      {children}
    72→    </motion.div>
    73→  );

   104→function CounterAnimation({
   105→  target,
   106→  suffix = "",
   107→  prefix = "",
   108→  display,
   109→}: {
   110→  target: number;
   111→  suffix?: string;
   112→  prefix?: string;
   113→  display?: string;
   114→}) {
   115→  const [count, setCount] = useState(0);
   116→  const [done, setDone] = useState(false);
   117→  const ref = useRef<HTMLSpanElement>(null);
   118→  const isInView = useInView(ref, { once: true });
   119→  React.useEffect(() => {
   120→    if (!isInView) return;
   121→    let start = 0;
   122→    const duration = 2000;
   123→    const increment = target / (duration / 16);
   124→    const timer = setInterval(() => {
   125→      start += increment;
   126→      if (start >= target) {
   127→        setCount(target);
   128→        setDone(true);
   129→        clearInterval(timer);
   130→      } else {
   131→        setCount(Math.floor(start));
   132→      }
   133→    }, 16);
   134→    return () => clearInterval(timer);
   135→  }, [isInView, target]);
   136→  return (
   137→    <span ref={ref}>
   138→      {done && display
   139→        ? display
   140→        : `${prefix}${count.toLocaleString()}${suffix}`}
   141→    </span>
   142→  );
   143→}

   145→function AnimatedNumber({
   146→  value,
   147→  decimals = 1,
   148→}: {
   149→  value: number;
   150→  decimals?: number;
   151→}) {
   152→  const ref = useRef<HTMLSpanElement>(null);
   153→  const isInView = useInView(ref, { once: true });
   154→  const motionVal = useMotionValue(0);
   155→  const rounded = useTransform(motionVal, (v) => v.toFixed(decimals));
   156→
   157→  React.useEffect(() => {
   158→    if (isInView) {
   159→      const controls = animate(motionVal, value, {
   160→        duration: 1.2,
   161→        ease: "easeOut",
   162→      });
   163→      return controls.stop;
   164→    }
   165→  }, [isInView, motionVal, value]);
   166→
   167→  return <motion.span ref={ref}>{rounded}</motion.span>;
   168→}
