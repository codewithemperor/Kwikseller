   101→function formatDateTime(value?: string) {
   102→  if (!value) return "—";
   103→  return new Intl.DateTimeFormat("en-NG", { dateStyle: "medium", timeStyle: "short" }).format(new Date(value));
   104→}
   105→
   106→function formatNGN(n: number) {
   107→  return new Intl.NumberFormat("en-NG", {
   108→    style: "currency",
   109→    currency: "NGN",
   110→    maximumFractionDigits: 0,
   111→  }).format(n);
   112→}

   160→function WalletContent() {

   171→    queryFn: async () => {
   172→      const res = await api.get<Tier[]>("wallet/tiers");
   173→      return res.data || [];
   174→    },
   175→    staleTime: 5 * 60 * 1000,
   176→  });
   177→
   178→  const [filter, setFilter] = React.useState<"ALL" | "CREDIT" | "DEBIT">("ALL");
   179→  const [dateRange, setDateRange] = React.useState<"7d" | "30d" | "90d" | "all">("all");
   180→  const [categoryFilter, setCategoryFilter] = React.useState<string>("ALL");
   181→  const [searchTerm, setSearchTerm] = React.useState("");
   182→  const [showAdvanced, setShowAdvanced] = React.useState(false);
   183→
   184→  // ── Redemption modal state ──
   185→  const [redeemOpen, setRedeemOpen] = React.useState(false);
   186→  const [redeemType, setRedeemType] = React.useState<WalletRedemptionType>("CASH");
   187→
   188→  function openRedeemModal(type: WalletRedemptionType) {
   189→    setRedeemType(type);
   190→    setRedeemOpen(true);
   191→  }
   192→
   193→  // ── Derived state (declared BEFORE any early return so the Rules of
   194→  // Hooks aren't violated). All hooks gracefully handle `wallet` being
   195→  // undefined by returning empty defaults.
   196→  const transactions = wallet?.transactions ?? [];
   197→
   198→  const txs = React.useMemo(() => {
   199→    let list = transactions.slice();
   200→    // Type filter
   201→    if (filter !== "ALL") list = list.filter((t) => t.type === filter);
   202→    // Category filter
   203→    if (categoryFilter !== "ALL") list = list.filter((t) => t.category === categoryFilter);
   204→    // Date range filter
   205→    if (dateRange !== "all") {
   206→      const days = dateRange === "7d" ? 7 : dateRange === "30d" ? 30 : 90;
   207→      const cutoff = Date.now() - days * 86400000;
   208→      list = list.filter((t) => new Date(t.createdAt).getTime() >= cutoff);
   209→    }
   210→    // Search filter (case-insensitive on description)
   211→    const term = searchTerm.trim().toLowerCase();
   212→    if (term) {
   213→      list = list.filter(
   214→        (t) =>
   215→          t.description.toLowerCase().includes(term) ||
   216→          t.category.replace(/_/g, " ").toLowerCase().includes(term),
   217→      );
   218→    }
   219→    return list;
   220→  }, [transactions, filter, categoryFilter, dateRange, searchTerm]);
   221→
   222→  // Available categories derived from the wallet's own transactions
   223→  const availableCategories = React.useMemo(() => {
   224→    const seen = new Set<string>();
   225→    for (const t of transactions) seen.add(t.category);
   226→    return Array.from(seen);
   227→  }, [transactions]);
   228→
   229→  const hasActiveFilters =
   230→    filter !== "ALL" || categoryFilter !== "ALL" || dateRange !== "all" || searchTerm.trim().length > 0;
   231→
   232→  function clearAllFilters() {
   233→    setFilter("ALL");
   234→    setCategoryFilter("ALL");
   235→    setDateRange("all");
   236→    setSearchTerm("");
   237→  }
   238→
   239→  // Export the currently-filtered transactions to CSV. Respects all active
   240→  // filters (type, category, date range, search) so the buyer gets exactly
   241→  // what they see on screen. Reuses the shared toCSV/downloadCSV utilities
   242→  // (same as the orders page CSV export).
   243→  function handleExportCSV() {
   244→    if (txs.length === 0) {
   245→      kwikToast.error("No transactions to export", {
   246→        description: "Adjust your filters and try again.",
   247→      });
   248→      return;
   249→    }
   250→    const rows = txs.map((t) => ({
   251→      Date: new Date(t.createdAt).toLocaleString("en-NG", { dateStyle: "medium", timeStyle: "short" }),
   252→      Type: t.type === "CREDIT" ? "Earned" : "Spent",
   253→      Category: t.category.replace(/_/g, " ").toLowerCase().replace(/\b\w/g, (c) => c.toUpperCase()),
   254→      Description: t.description,
   255→      Amount: t.type === "CREDIT" ? `+${t.amount}` : `-${t.amount}`,
   256→      "Coins After": "",
   257→    }));
   258→    const csv = toCSV(rows);
   259→    const date = new Date().toISOString().slice(0, 10);
   260→    const suffix = hasActiveFilters ? "-filtered" : "";
   261→    downloadCSV(`kwikseller-wallet${suffix}-${date}`, csv);
   262→    kwikToast.success("CSV downloaded", {
   263→      description: `${txs.length} transaction${txs.length === 1 ? "" : "s"} exported.`,
   264→    });
   265→  }
   266→
   267→  if (isLoading) return <PageLoading label="Loading wallet…" />;
   268→
   269→  if (!wallet) {
   270→    return (
   271→      <EmptyState
   272→        variant="default"
   273→        icon={<WalletIcon className="h-12 w-12" />}
   274→        title="Wallet unavailable"
   275→        description="We couldn't load your KwikCoins wallet right now."
   276→      />
   277→    );
   278→  }
   279→
   280→  const CurrentTierIcon = TIER_ICONS[wallet.tier] ?? Coins;
   281→
   282→  return (
   283→    <div className="mx-auto max-w-4xl">
   284→      <Link
   285→        href="/profile"
   286→        className="inline-flex items-center gap-2 text-sm font-semibold text-kwik-muted transition hover:text-foreground"
   287→      >
   288→        <ArrowLeft className="h-4 w-4" /> Profile
   289→      </Link>
   290→
   291→      <h1 className="mt-3 font-heading text-2xl font-bold text-foreground">KwikCoins Wallet</h1>
   292→      <p className="mt-1 text-sm text-kwik-muted">
   293→        Earn coins on every order and redeem them for discounts, ad credit, and rewards.
   294→      </p>
   295→
   296→      {/* Balance card */}
   297→      <motion.div
   298→        initial={{ opacity: 0, y: 12 }}
   299→        animate={{ opacity: 1, y: 0 }}
   300→        className="mt-6 overflow-hidden rounded-3xl bg-kwik-gradient p-6 text-white shadow-lg sm:p-8"
   301→      >
   302→        <div className="flex flex-wrap items-start justify-between gap-4">
   303→          <div>
   304→            <div className="flex items-center gap-2 text-white/80">
   305→              <Coins className="h-5 w-5" />
   306→              <span className="text-sm font-medium uppercase tracking-wider">Current balance</span>
   307→            </div>
   308→            <p className="mt-2 font-heading text-4xl font-bold sm:text-5xl">
   309→              {wallet.balance.toLocaleString()}{" "}
   310→              <span className="text-lg font-medium text-white/80">KwikCoins</span>
   311→            </p>
   312→            <p className="mt-2 text-sm text-white/70">
   313→              ≈ ₦{wallet.nairaEquivalent.toLocaleString()} value · {wallet.earningRate}× earning rate
   314→            </p>
   315→          </div>
   316→          <div className="flex items-center gap-2 rounded-full bg-white/15 px-3 py-1.5 backdrop-blur-sm">
   317→            <CurrentTierIcon className="h-4 w-4" />
   318→            <span className="text-sm font-semibold">{wallet.tier} Tier</span>
   319→          </div>
   320→        </div>
   321→
   322→        {/* Tier progress */}
   323→        <div className="mt-6">
   324→          <div className="flex items-center justify-between text-xs text-white/80">
   325→            <span>Progress to {wallet.nextTier}</span>
   326→            <span>{wallet.balance} / {wallet.nextTierThreshold}</span>
   327→          </div>
   328→          <div className="mt-1.5 h-2.5 w-full overflow-hidden rounded-full bg-white/20">
   329→            <motion.div
   330→              initial={{ width: 0 }}
   331→              animate={{ width: `${wallet.tierProgress}%` }}
   332→              transition={{ duration: 0.8, ease: "easeOut" }}
   333→              className="h-full rounded-full bg-white"
   334→            />
   335→          </div>
   336→        </div>
   337→
   338→        {/* Quick actions — wired to open the redemption modal */}
   339→        <div className="mt-6 grid grid-cols-3 gap-2">
   340→          <button
   341→            type="button"
   342→            onClick={() => openRedeemModal("CASH")}
   343→            className="flex flex-col items-center gap-1 rounded-xl bg-white/10 px-2 py-3 text-center backdrop-blur-sm transition hover:bg-white/20 focus:outline-none focus-visible:ring-2 focus-visible:ring-white/60"
   344→          >
   345→            <Gift className="h-5 w-5" />
   346→            <span className="text-xs font-semibold">Redeem</span>
   347→          </button>
   348→          <button
   349→            type="button"
   350→            onClick={() => openRedeemModal("AD_CREDIT")}
   351→            className="flex flex-col items-center gap-1 rounded-xl bg-white/10 px-2 py-3 text-center backdrop-blur-sm transition hover:bg-white/20 focus:outline-none focus-visible:ring-2 focus-visible:ring-white/60"
   352→          >
   353→            <Zap className="h-5 w-5" />
   354→            <span className="text-xs font-semibold">Ad Credit</span>
   355→          </button>
   356→          <button

   361→            <Sparkles className="h-5 w-5" />

   371→

   373→      <div className="mt-6 grid grid-cols-2 gap-4 sm:grid-cols-3">
   374→        <div className="rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-4">
   375→          <div className="flex items-center gap-2 text-kwik-green">
   376→            <TrendingUp className="h-4 w-4" />
   377→            <span className="text-xs font-semibold uppercase tracking-wide">Lifetime earned</span>
   378→          </div>
   379→          <p className="mt-2 font-heading text-xl font-bold text-foreground">
   380→            {wallet.lifetimeEarned.toLocaleString()}
   381→          </p>
   382→          <p className="text-xs text-kwik-muted">KwikCoins</p>
   383→        </div>
   384→        <div className="rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-4">
   385→          <div className="flex items-center gap-2 text-kwik-red">
   386→            <TrendingDown className="h-4 w-4" />
   387→            <span className="text-xs font-semibold uppercase tracking-wide">Lifetime spent</span>

   390→            {wallet.lifetimeSpent.toLocaleString()}

   392→          <p className="text-xs text-kwik-muted">KwikCoins</p>
   393→        </div>
   394→        <div className="col-span-2 rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-4 sm:col-span-1">
   395→          <div className="flex items-center gap-2 text-kwik-orange">

   409→          <h2 className="font-heading text-lg font-bold text-foreground">Membership Tiers</h2>

   420→                    "rounded-2xl border p-4",

   498→          {(["ALL", "CREDIT", "DEBIT"] as const).map((f) => (
