   291→/* ─── Create Product Wizard Modal ─── */
   292→
   293→function CreateProductModal({
   294→  isOpen,
   295→  onClose,
   296→  onCreated,
   297→}: {
   298→  isOpen: boolean;
   299→  onClose: () => void;
   300→  onCreated: () => void;
   301→}) {
   302→  const [form, setForm] = React.useState(blankForm);
   303→  const [step, setStep] = React.useState(0);
   304→  const [isSaving, setIsSaving] = React.useState(false);
   305→  const [isUploading, setIsUploading] = React.useState(false);
   306→
   307→  const resetAndClose = () => {
   308→    setForm(blankForm);
   309→    setStep(0);
   310→    onClose();
   311→  };
   312→
   313→  const validateStep = (nextStep: number) => {
   314→    if (step === 0 && (!form.name.trim() || !form.productType)) {
   315→      kwikToast.error("Add product name and type first");
   316→      return;
   317→    }
   318→    if (step === 1 && Number(form.price) <= 0) {
   319→      kwikToast.error("Add a valid product price");
   320→      return;
   321→    }
   322→    setStep(nextStep);
   323→  };
   324→
   325→  const createProduct = async () => {
   326→    if (!form.name.trim()) { kwikToast.error("Product name is required"); setStep(0); return; }
   327→    if (Number(form.price) <= 0) { kwikToast.error("Price must be greater than zero"); setStep(1); return; }
   328→
   329→    setIsSaving(true);
   330→    try {
   331→      await vendorCommerceApi.createProduct({
   332→        name: form.name,
   333→        description: form.description || undefined,
   334→        price: Number(form.price),
   335→        comparePrice: form.comparePrice ? Number(form.comparePrice) : undefined,
   336→        sku: form.sku || undefined,
   337→        productType: form.productType,
   338→        requiresShipping: form.productType === "PHYSICAL",
   339→        trackInventory: form.productType === "PHYSICAL",
   340→        initialStock: form.productType === "PHYSICAL" ? Number(form.initialStock) : undefined,
   341→        lowStock: form.productType === "PHYSICAL" ? Number(form.lowStock) : undefined,
   342→        images: form.images,
   343→        poolEnabled: form.productType === "PHYSICAL" ? form.poolEnabled : false,
   344→        poolBasePrice: form.poolEnabled ? Number(form.poolBasePrice || form.price) : undefined,
   345→        poolMinSalePrice: form.poolEnabled ? Number(form.poolMinSalePrice || form.poolBasePrice || form.price) : undefined,
   346→        poolMaxSelectableQuantity: form.poolEnabled && form.poolMaxSelectableQuantity ? Number(form.poolMaxSelectableQuantity) : undefined,
   347→      });
   348→      kwikToast.success("Product created");
   349→      resetAndClose();
   350→      onCreated();
   351→    } catch (error) {
   352→      kwikToast.error(error instanceof Error ? error.message : "Product creation failed");
   353→    } finally {
   354→      setIsSaving(false);
   355→    }
   356→  };
   357→
   358→  const uploadImages = async (files: FileList | null) => {
   359→    if (!files?.length) return;
   360→    setIsUploading(true);
   361→    try {
   362→      const uploads = await Promise.all(Array.from(files).slice(0, 5).map((f) => uploadApi.productImage(f)));
   363→      const urls = uploads.map(uploadedUrl).filter(Boolean);
   364→      setForm((c) => ({ ...c, images: [...c.images, ...urls].slice(0, 5) }));
   365→      kwikToast.success(`${urls.length} image${urls.length === 1 ? "" : "s"} uploaded`);
   366→    } catch (error) {
   367→      kwikToast.error(error instanceof Error ? error.message : "Image upload failed");
   368→    } finally {
   369→      setIsUploading(false);
   370→    }
   371→  };
   372→
   373→  return (
   374→    <AppModal isOpen={isOpen} onClose={resetAndClose} title="Add Product" description="Complete the required steps, then confirm." className="sm:max-w-3xl">
   375→      <div className="space-y-6">
   376→        {/* Step indicator */}
   377→        <div className="grid grid-cols-4 gap-2">
   378→          {wizardSteps.map((label, i) => {
   379→            const done = i < step;
   380→            const active = i === step;
   381→            return (
   382→              <button key={label} type="button" onClick={() => i <= step && setStep(i)} className="flex flex-col items-center gap-2 text-center">
   383→                <span className={`flex h-10 w-10 items-center justify-center rounded-full border text-sm font-semibold ${done ? "border-gray-900 bg-gray-900 text-white" : active ? "border-gray-900 bg-white text-gray-900" : "border-gray-300 bg-white text-gray-400"}`}>
   384→                  {done ? <Check className="h-4 w-4" /> : i + 1}
   385→                </span>
   386→                <span className="hidden text-xs font-medium text-gray-500 sm:block">{label}</span>
   387→              </button>
   388→            );
   389→          })}
   390→        </div>
   391→
   392→        {/* Step 0 — Basic Info */}
   393→        {step === 0 && (
   394→          <div className="space-y-4">
   395→            <FieldInput required label="Product name" value={form.name} onChange={(e) => setForm((v) => ({ ...v, name: e.target.value }))} />
   396→            <FieldSelect label="Product type" value={form.productType} onChange={(e) => setForm((v) => ({ ...v, productType: e.target.value as ProductType }))}>
   397→              <option value="PHYSICAL">Physical</option>
   398→              <option value="DIGITAL">Digital</option>
   399→            </FieldSelect>
   400→            <FieldTextarea label="Description" value={form.description} onChange={(e) => setForm((v) => ({ ...v, description: e.target.value }))} />
   401→          </div>
   402→        )}
   403→
   404→        {/* Step 1 — Pricing & Stock */}
   405→        {step === 1 && (
   406→          <div className="space-y-4">
   407→            <div className="grid gap-3 sm:grid-cols-2">
   408→              <FieldInput required type="number" min={0} label="Price" value={form.price} onChange={(e) => setForm((v) => ({ ...v, price: Number(e.target.value) }))} />
   409→              <FieldInput type="number" min={0} label="Compare price" value={form.comparePrice} onChange={(e) => setForm((v) => ({ ...v, comparePrice: Number(e.target.value) }))} />
   410→            </div>
   411→            {form.productType === "PHYSICAL" ? (
   412→              <div className="grid gap-3 sm:grid-cols-2">
   413→                <FieldInput type="number" min={0} label="Initial stock" value={form.initialStock} onChange={(e) => setForm((v) => ({ ...v, initialStock: Number(e.target.value) }))} />
   414→                <FieldInput type="number" min={0} label="Low stock alert" value={form.lowStock} onChange={(e) => setForm((v) => ({ ...v, lowStock: Number(e.target.value) }))} />
   415→              </div>
   416→            ) : (
   417→              <p className="text-sm text-gray-500">Digital products default to unlimited delivery.</p>
   418→            )}
   419→          </div>
   420→        )}
   421→
   422→        {/* Step 2 — Details */}
   423→        {step === 2 && (
   424→          <div className="space-y-4">
   425→            <FieldInput label="SKU" value={form.sku} onChange={(e) => setForm((v) => ({ ...v, sku: e.target.value }))} />
   426→            <label className="block">
   427→              <span className="text-xs font-semibold text-gray-500">Product images</span>
   428→              <div className="mt-2 flex min-h-44 cursor-pointer flex-col items-center justify-center rounded-lg border border-dashed border-gray-300 bg-gray-50 p-6 text-center">
   429→                <Upload className="h-8 w-8 text-gray-400" />
   430→                <p className="mt-2 text-sm font-medium text-gray-700">
   431→                  <span className="text-blue-600 underline">Click to upload</span> or drag and drop
   432→                </p>
   433→                <p className="mt-1 text-xs text-gray-400">{`${form.images.length}/5 images added`}</p>
   434→                <input type="file" accept="image/*" multiple onChange={(e) => uploadImages(e.target.files)} className="sr-only" />
   435→              </div>
   436→            </label>
   437→            {form.images.length > 0 && (
   438→              <div className="grid grid-cols-5 gap-2">
   439→                {form.images.map((url) => (
   440→                  <button key={url} type="button" onClick={() => setForm((c) => ({ ...c, images: c.images.filter((img) => img !== url) }))} className="aspect-square overflow-hidden rounded-lg border border-gray-200 bg-gray-50" title="Remove image">
   441→                    <img src={url} alt="" className="h-full w-full object-cover" />
   442→                  </button>
   443→                ))}
   444→              </div>
   445→            )}
   446→            {form.productType === "PHYSICAL" && (
   447→              <div className="border border-gray-200 p-4">
   448→                <AppSwitch
   449→                  isSelected={form.poolEnabled}
   450→                  onChange={(sel) => setForm((v) => ({
   451→                    ...v,
   452→                    poolEnabled: sel,
   453→                    poolBasePrice: sel && !v.poolBasePrice ? Number(v.price) : v.poolBasePrice,
   454→                    poolMinSalePrice: sel && !v.poolMinSalePrice ? Number(v.price) : v.poolMinSalePrice,
   455→                  }))}
   456→                  label="Make available in Pool"
   457→                  description="Other vendors can select and sell this product."
   458→                />
   459→                {form.poolEnabled && (
   460→                  <div className="mt-4 grid gap-3 sm:grid-cols-3">
   461→                    <FieldInput type="number" min={0} label="Source price" value={form.poolBasePrice || form.price} onChange={(e) => setForm((v) => ({ ...v, poolBasePrice: Number(e.target.value) }))} />
   462→                    <FieldInput type="number" min={form.poolBasePrice || form.price} label="Min sale price" value={form.poolMinSalePrice || form.poolBasePrice || form.price} onChange={(e) => setForm((v) => ({ ...v, poolMinSalePrice: Number(e.target.value) }))} />
   463→                    <FieldInput type="number" min={0} label="Pool quantity" placeholder="Use stock" value={form.poolMaxSelectableQuantity || ""} onChange={(e) => setForm((v) => ({ ...v, poolMaxSelectableQuantity: Number(e.target.value) }))} />
   464→                  </div>
   465→                )}
   466→              </div>
   467→            )}
   468→          </div>
   469→        )}
   470→
   471→        {/* Step 3 — Review */}
   472→        {step === 3 && (
   473→          <div className="space-y-4">
   474→            <div className="border border-gray-200 p-5">
   475→              <p className="text-xs font-semibold uppercase tracking-wide text-gray-400">Review</p>
   476→              <h3 className="mt-2 text-xl font-semibold text-gray-900">{form.name || "Untitled product"}</h3>
   477→              <p className="mt-2 text-sm leading-6 text-gray-500">{form.description || "No description added."}</p>
   478→              <div className="mt-4 grid gap-3 sm:grid-cols-3">
   479→                <div><p className="text-xs text-gray-400">Type</p><p className="font-semibold text-gray-900">{form.productType}</p></div>
   480→                <div><p className="text-xs text-gray-400">Price</p><p className="font-semibold text-gray-900">{formatCurrency(form.price)}</p></div>
   481→                <div><p className="text-xs text-gray-400">Images</p><p className="font-semibold text-gray-900">{form.images.length}</p></div>
   482→              </div>
   483→            </div>
   484→            {form.poolEnabled && (
   485→              <p className="text-sm font-medium text-emerald-600">
   486→                This product will be available in Pool at source price {formatCurrency(form.poolBasePrice || form.price)}.
   487→              </p>
   488→            )}
   489→          </div>
   490→        )}
   491→
   492→        {/* Footer */}
   493→        <div className="flex items-center justify-between gap-3 border-t border-gray-200 pt-4">
   494→          <AppButton type="button" variant="secondary" disabled={step === 0 || isSaving} onClick={() => setStep((c) => Math.max(0, c - 1))}>
   495→            <ChevronLeft className="h-4 w-4" />
   496→            Back
   497→          </AppButton>
   498→          {step < wizardSteps.length - 1 ? (
   499→            <AppButton type="button" onClick={() => validateStep(step + 1)}>
   500→              Next
   501→              <ChevronRight className="h-4 w-4" />
   502→            </AppButton>
   503→          ) : (
   504→            <AppButton type="button" disabled={isSaving || isUploading} onClick={createProduct}>
   505→              <Plus className="h-4 w-4" />
   506→              Create product
   507→            </AppButton>
   508→          )}
   509→        </div>
   510→      </div>
   511→    </AppModal>
   512→  );
   513→}

   515→/* ─── Pagination ─── */
   516→
   517→function Pagination({
   518→  total,
   519→  page,
   520→  onPageChange,
   521→}: {
   522→  total: number;
   523→  page: number;
   524→  onPageChange: (p: number) => void;
   525→}) {
   526→  const totalPages = Math.max(1, Math.ceil(total / ITEMS_PER_PAGE));
   527→  const start = (page - 1) * ITEMS_PER_PAGE + 1;
   528→  const end = Math.min(page * ITEMS_PER_PAGE, total);
   529→
   530→  const pages: (number | "...")[] = [];
   531→  if (totalPages <= 7) {
   532→    for (let i = 1; i <= totalPages; i++) pages.push(i);
   533→  } else {
   534→    pages.push(1);
   535→    if (page > 3) pages.push("...");
   536→    for (let i = Math.max(2, page - 1); i <= Math.min(totalPages - 1, page + 1); i++) pages.push(i);
   537→    if (page < totalPages - 2) pages.push("...");
   538→    pages.push(totalPages);
   539→  }
   540→
   541→  return (
   542→    <div className="flex flex-col items-center justify-between gap-4 sm:flex-row">
   543→      <p className="text-sm text-gray-500">
   544→        Showing {total > 0 ? start : 0}–{end} of {total} products
   545→      </p>
   546→      <div className="flex items-center gap-1">
   547→        <button
   548→          type="button"
   549→          disabled={page <= 1}
   550→          onClick={() => onPageChange(page - 1)}
   551→          className="inline-flex h-8 items-center gap-1 rounded px-2 text-sm font-medium text-gray-600 transition hover:bg-gray-100 disabled:cursor-not-allowed disabled:opacity-40"
   552→        >
   553→          <ChevronLeft className="h-3.5 w-3.5" />
   554→          Prev
   555→        </button>
   556→        {pages.map((p, i) =>
   557→          p === "..." ? (
   558→            <span key={`dots-${i}`} className="px-1 text-sm text-gray-400">...</span>
   559→          ) : (
   560→            <button
   561→              key={p}
   562→              type="button"
   563→              onClick={() => onPageChange(p as number)}
   564→              className={`inline-flex h-8 w-8 items-center justify-center rounded text-sm font-medium transition ${p === page ? "bg-gray-900 text-white" : "text-gray-600 hover:bg-gray-100"}`}
   565→            >
   566→              {p}
   567→            </button>
   568→          )
   569→        )}
   570→        <button
   571→          type="button"
   572→          disabled={page >= totalPages}
   573→          onClick={() => onPageChange(page + 1)}
   574→          className="inline-flex h-8 items-center gap-1 rounded px-2 text-sm font-medium text-gray-600 transition hover:bg-gray-100 disabled:cursor-not-allowed disabled:opacity-40"
   575→        >
   576→          Next
   577→          <ChevronRight className="h-3.5 w-3.5" />
   578→        </button>
   579→      </div>
   580→    </div>
   581→  );
   582→}
