    67→// ==================== Configuration ====================

    69→const getBaseURL = (): string => {
    70→  const configuredUrl = process.env.NEXT_PUBLIC_API_URL
    71→
    72→  if (!configuredUrl) {
    73→    return typeof window !== 'undefined' ? '/api/v1' : 'http://localhost:4000/api/v1'
    74→  }
    75→
    76→  const withProtocol = /^https?:\/\//i.test(configuredUrl)
    77→    ? configuredUrl
    78→    : `https://${configuredUrl}`
    79→  const withoutTrailingSlash = withProtocol.replace(/\/+$/, '')
    80→
    81→  return withoutTrailingSlash.endsWith('/api/v1')
    82→    ? withoutTrailingSlash
    83→    : `${withoutTrailingSlash}/api/v1`
    84→}

    86→// Storage keys
    87→const STORAGE_KEYS = {
    88→  ACCESS_TOKEN: 'kwikseller_access_token',
    89→  REFRESH_TOKEN: 'kwikseller_refresh_token',
    90→  AUTH_STORE: 'kwikseller_auth',
    91→}
    92→
    93→// ==================== Token Refresh Mutex ====================
    94→// Prevents concurrent 401 responses from triggering multiple refresh calls.
    95→// Only one refresh request is in-flight at a time; other requests queue and
    96→// resolve once the refresh completes.
    97→
    98→let isRefreshing = false
    99→let refreshQueue: Array<{
   100→  resolve: (token: string) => void
   101→  reject: (error: unknown) => void
   102→}> = []
   103→
   104→function processRefreshQueue(error: unknown, token: string | null = null): void {
   105→  refreshQueue.forEach(({ resolve, reject }) => {
   106→    if (error) reject(error)
   107→    else if (token) resolve(token)
   108→  })
   109→  refreshQueue = []
   110→}

   112→// ==================== Token Management ====================
   113→
   114→export const tokenManager = {
   115→  getAccessToken: (): string | null => {
   116→    if (typeof window === 'undefined') return null
   117→    return localStorage.getItem(STORAGE_KEYS.ACCESS_TOKEN)
   118→  },
   119→
   120→  getRefreshToken: (): string | null => {
   121→    if (typeof window === 'undefined') return null
   122→    return localStorage.getItem(STORAGE_KEYS.REFRESH_TOKEN)
   123→  },
   124→
   125→  setTokens: (accessToken: string, refreshToken?: string): void => {
   126→    if (typeof window === 'undefined') return
   127→    localStorage.setItem(STORAGE_KEYS.ACCESS_TOKEN, accessToken)
   128→    if (refreshToken) {
   129→      localStorage.setItem(STORAGE_KEYS.REFRESH_TOKEN, refreshToken)
   130→    }
   131→  },
   132→
   133→  clearTokens: (): void => {
   134→    if (typeof window === 'undefined') return
   135→    localStorage.removeItem(STORAGE_KEYS.ACCESS_TOKEN)
   136→    localStorage.removeItem(STORAGE_KEYS.REFRESH_TOKEN)
   137→    localStorage.removeItem(STORAGE_KEYS.AUTH_STORE)
   138→  },
   139→
   140→  isAuthenticated: (): boolean => {
   141→    return !!tokenManager.getAccessToken()
   142→  },
   143→}

   145→// ==================== Create Axios Instance ====================
   146→
   147→const createApiClient = (): AxiosInstance => {
   148→  const instance = axios.create({
   149→    baseURL: getBaseURL(),
   150→    timeout: 30000,
   151→    headers: {
   152→      'Content-Type': 'application/json',
   153→    },
   154→  })
   155→
   156→  const redirectToLogin = () => {
   157→    if (typeof window === 'undefined') return
   158→    tokenManager.clearTokens()
   159→    const currentPath = window.location.pathname
   160→    const currentHost = window.location.hostname
   161→    const isAdminApp = currentHost.includes('admin') || currentPath.startsWith('/admin')
   162→    const isRiderApp = currentHost.includes('rider') || currentPath.startsWith('/rider')
   163→    const isVendorApp = currentHost.includes('vendor') || currentPath.startsWith('/dashboard')
   164→    let loginPath = '/login'
   165→    if (isAdminApp) loginPath = '/admin/login'
   166→    else if (isRiderApp) loginPath = '/rider/login'
   167→    // Vendor app login is at /login (same as default), but we preserve
   168→    // the dashboard redirect so the user lands back where they were
   169→    window.location.href = `${loginPath}?redirect=${encodeURIComponent(currentPath)}`
   170→  }
   171→
   172→  // Request interceptor - Add auth token
   173→  instance.interceptors.request.use(
   174→    (config: InternalAxiosRequestConfig) => {
   175→      // Skip auth if explicitly requested
   176→      if ((config as RequestConfig).skipAuth) {
   177→        return config
   178→      }
   179→
   180→      const token = tokenManager.getAccessToken()
   181→      if (token && config.headers) {
   182→        config.headers.Authorization = `Bearer ${token}`
   183→      }
   184→      return config
   185→    },
   186→    (error) => Promise.reject(error)
   187→  )
   188→
   189→  // Response interceptor - Handle errors and token refresh with mutex
   190→  instance.interceptors.response.use(
   191→    (response: AxiosResponse) => response,
   192→    async (error: AxiosError<ApiError>) => {
   193→      const originalRequest = error.config as InternalAxiosRequestConfig & {
   194→        _retry?: boolean
   195→      }
   196→      const isAuthEndpoint = originalRequest?.url?.includes('/auth/')
   197→
   198→      // Handle 401 Unauthorized - Try to refresh token (with mutex to prevent race conditions)
   199→      if (error.response?.status === 401 && !originalRequest._retry && !isAuthEndpoint) {
   200→        if (isRefreshing) {
   201→          // Another refresh is already in-flight — queue this request
   202→          return new Promise((resolve, reject) => {
   203→            refreshQueue.push({
   204→              resolve: (newToken: string) => {
   205→                if (originalRequest.headers) {
   206→                  originalRequest.headers.Authorization = `Bearer ${newToken}`
   207→                }
   208→                resolve(instance(originalRequest))
   209→              },
   210→              reject: (err: unknown) => reject(err),
   211→            })
   212→          })
   213→        }
   214→
   215→        originalRequest._retry = true
   216→        isRefreshing = true
   217→
   218→        try {
   219→          const refreshToken = tokenManager.getRefreshToken()
   220→          if (refreshToken) {
   221→            const response = await axios.post(`${getBaseURL()}/auth/refresh`, {
   222→              refreshToken,
   223→            })
   224→
   225→            const { accessToken, refreshToken: newRefreshToken } = response.data.data || response.data
   226→            tokenManager.setTokens(accessToken, newRefreshToken)
   227→
   228→            // Notify queued requests
   229→            processRefreshQueue(null, accessToken)
   230→
   231→            // Retry original request with new token
   232→            if (originalRequest.headers) {
   233→              originalRequest.headers.Authorization = `Bearer ${accessToken}`
   234→            }
   235→            return instance(originalRequest)
   236→          }
   237→          // No refresh token available
   238→          processRefreshQueue(new Error('No refresh token available'), null)
   239→          tokenManager.clearTokens()
   240→          redirectToLogin()
   241→        } catch (refreshError) {
   242→          // Refresh failed
   243→          processRefreshQueue(refreshError, null)
   244→          tokenManager.clearTokens()
   245→          redirectToLogin()
   246→        } finally {
   247→          isRefreshing = false
   248→        }
   249→      }
   250→
   251→      // Handle other errors
   252→      const apiError: ApiError = {
   253→        statusCode: error.response?.status || 500,
   254→        message: error.response?.data?.message || error.message || 'An error occurred',
   255→        error: error.response?.data?.error,
   256→        details: error.response?.data?.details,
   257→      }
   258→
   259→      return Promise.reject(apiError)
   260→    }
   261→  )
   262→
   263→  return instance
   264→}
   265→
   266→// Create singleton instance
   267→const apiClient = createApiClient()

   271→export const api = {
   272→  get: <T = unknown>(
   273→    url: string,
   274→    config?: RequestConfig
   275→  ): Promise<ApiResponse<T>> => apiClient.get(url, config).then((res) => res.data),
   276→
   277→  post: <T = unknown>(
   278→    url: string,
   279→    data?: unknown,
   280→    config?: RequestConfig
   281→  ): Promise<ApiResponse<T>> => apiClient.post(url, data, config).then((res) => res.data),
   282→
   283→  put: <T = unknown>(
   284→    url: string,
   285→    data?: unknown,
   286→    config?: RequestConfig
   287→  ): Promise<ApiResponse<T>> => apiClient.put(url, data, config).then((res) => res.data),
   288→
   289→  patch: <T = unknown>(
   290→    url: string,
   291→    data?: unknown,
   292→    config?: RequestConfig
   293→  ): Promise<ApiResponse<T>> => apiClient.patch(url, data, config).then((res) => res.data),
   294→
   295→  delete: <T = unknown>(
   296→    url: string,
   297→    config?: RequestConfig
   298→  ): Promise<ApiResponse<T>> => apiClient.delete(url, config).then((res) => res.data),
   299→}
