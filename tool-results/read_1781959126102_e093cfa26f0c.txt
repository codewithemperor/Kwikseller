   115→   115→function formatRelativeTime(dateStr: string): string {
   116→   116→  if (!dateStr) return "";
   117→   117→  const now = Date.now();
   118→   118→  const then = new Date(dateStr).getTime();
   119→   119→  const diffMs = now - then;
   120→   120→  const diffSec = Math.floor(diffMs / 1000);
   121→   121→  const diffMin = Math.floor(diffSec / 60);
   122→   122→  const diffHr = Math.floor(diffMin / 60);
   123→   123→  const diffDay = Math.floor(diffHr / 24);
   124→   124→
   125→   125→  if (diffSec < 60) return "just now";
   126→   126→  if (diffMin < 60) return `${diffMin}m ago`;
   127→   127→  if (diffHr < 24) return `${diffHr}h ago`;
   128→   128→  if (diffDay < 7) return `${diffDay}d ago`;
   129→   129→  return formatDate(dateStr);
   130→   130→}

   162→   162→function CountdownTimer({ targetDate }: { targetDate: string }) {
   163→   163→  const [timeLeft, setTimeLeft] = React.useState<{
   164→   164→    hours: number;
   165→   165→    minutes: number;
   166→   166→    seconds: number;
   167→   167→    expired: boolean;
   168→   168→  } | null>(null);
   169→   169→
   170→   170→  React.useEffect(() => {
   171→   171→    const calculate = () => {
   172→   172→      const now = Date.now();
   173→   173→      const target = new Date(targetDate).getTime();
   174→   174→      const diff = target - now;
   175→   175→
   176→   176→      if (diff <= 0) {
   177→   177→        setTimeLeft({ hours: 0, minutes: 0, seconds: 0, expired: true });
   178→   178→        return;
   179→   179→      }
   180→   180→
   181→   181→      setTimeLeft({
   182→   182→        hours: Math.floor(diff / (1000 * 60 * 60)),
   183→   183→        minutes: Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)),
   184→   184→        seconds: Math.floor((diff % (1000 * 60)) / 1000),
   185→   185→        expired: false,
   186→   186→      });
   187→   187→    };
   188→   188→
   189→   189→    calculate();
   190→   190→    const interval = setInterval(calculate, 1000);
   191→   191→    return () => clearInterval(interval);
   192→   192→  }, [targetDate]);
   193→   193→
   194→   194→  if (!timeLeft) {
   195→   195→    return <span className="text-xs text-gray-400">...</span>;
   196→   196→  }
   197→   197→
   198→   198→  if (timeLeft.expired) {
   199→   199→    return (
   200→   200→      <span className="text-xs font-medium text-amber-600">
   201→   201→        Releasing soon...
   202→   202→      </span>
   203→   203→    );
   204→   204→  }
   205→   205→
   206→   206→  const pad = (n: number) => String(n).padStart(2, "0");
   207→   207→
   208→   208→  return (
   209→   209→    <span className="font-mono text-xs tabular-nums text-amber-600">
   210→   210→      Release in {pad(timeLeft.hours)}h {pad(timeLeft.minutes)}m {pad(timeLeft.seconds)}s
   211→   211→    </span>
   212→   212→  );
   213→   213→}
