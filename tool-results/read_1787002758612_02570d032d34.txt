    42→    42→/**
    43→    43→ * Generates a reasonably-unique idempotency key per checkout attempt.
    44→    44→ * The backend uses it to dedupe a ParentCheckout (and its per-vendor
    45→    45→ * Orders) so a network retry returns the original result instead of
    46→    46→ * double-creating orders. We regenerate it whenever the user starts a
    47→    47→ * new checkout submission (not on every render).
    48→    48→ */
    49→    49→function newIdempotencyKey(): string {
    50→    50→  if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
    51→    51→    return `chk_${crypto.randomUUID()}`;
    52→    52→  }
    53→    53→  return `chk_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    54→    54→}
