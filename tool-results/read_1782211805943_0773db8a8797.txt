   124→function getTimelineStageIndex(status: DeliveryStatus): number {
   125→  const stageKeys = TIMELINE_STAGES.map((s) => s.key);
   126→  const statusIdx = stageKeys.indexOf(status as (typeof TIMELINE_STAGES)[number]["key"]);
   127→  if (statusIdx >= 0) return statusIdx;
   128→  // Statuses before ACCEPTED map to -1 (all dots empty)
   129→  if (STATUS_ORDER.indexOf(status) < STATUS_ORDER.indexOf("ACCEPTED")) return -1;
   130→  // Statuses after DELIVERED map to last index (all dots filled)
   131→  return TIMELINE_STAGES.length - 1;
   132→}

   134→function getRelativeTime(dateStr: string): string {
   135→  const now = new Date();
   136→  const date = new Date(dateStr);
   137→  const diffMs = now.getTime() - date.getTime();
   138→  const diffMins = Math.floor(diffMs / 60000);
   139→  const diffHours = Math.floor(diffMs / 3600000);
   140→  const diffDays = Math.floor(diffMs / 86400000);
   141→
   142→  if (diffMins < 1) return "Just now";
   143→  if (diffMins < 60) return `${diffMins}m ago`;
   144→  if (diffHours < 24) return `${diffHours}h ago`;
   145→  if (diffDays < 7) return `${diffDays}d ago`;
   146→  return formatDate(dateStr);
   147→}

   150→  return ["DELIVERED", "COMPLETED", "CANCELLED", "RETURNED"].includes(status);

   331→function loadDeliveries(): DeliveryItem[] | null {
   332→  if (typeof window === "undefined") return null;
   333→  try {
   334→    const raw = localStorage.getItem(DELIVERIES_KEY);

   392→function DetailTimeline({ delivery }: { delivery: DeliveryItem }) {
   393→  const entries: Array<{ key: string; label: string; time?: string; color: string }> = [];
   394→
   395→  const push = (key: string, label: string, time?: string, color = "text-muted-foreground") => {
   396→    if (time) entries.push({ key, label, time, color });
   397→  };
   398→
   399→  push("created", "Order created", delivery.createdAt, "text-muted-foreground");
   400→  push("accepted", "Accepted by vendor", delivery.acceptedAt, "text-muted-foreground");
   401→  push("preparing", "Preparing order", delivery.preparingAt, "text-amber-600");
   402→  push("ready", "Ready for pickup", delivery.readyAt, "text-orange-600");
   403→  push("picked_up", "Picked up by rider", delivery.pickedUpAt, "text-blue-600");
   404→  push("in_transit", "In transit", delivery.inTransitAt, "text-blue-600");
   405→  push("arrived", "Arrived at destination", delivery.arrivedAt, "text-purple-600");
   406→  push("delivered", "Delivered", delivery.deliveredAt, "text-green-600");
   407→  push("completed", "Completed", delivery.completedAt, "text-green-700");
   408→  if (delivery.cancelledAt) push("cancelled", "Cancelled", delivery.cancelledAt, "text-red-500");
   409→  if (delivery.returnedAt) push("returned", "Returned", delivery.returnedAt, "text-red-400");
   410→
   411→  if (entries.length === 0) return null;
   412→
   413→  return (
   414→    <div className="relative pl-6">
   415→      <div className="absolute left-[7px] top-2 bottom-2 w-px bg-default-100" />
   416→      <div className="space-y-4">
   417→        {entries.map((entry) => (
   418→          <div key={entry.key} className="relative flex items-start gap-3">
   419→            <div className="absolute -left-6 top-1.5 h-3.5 w-3.5 rounded-full border-2 border-kwik-border bg-surface" />
   420→            <div className="min-w-0">
   421→              <p className="text-sm text-foreground">{entry.label}</p>
   422→              <p className="text-xs text-muted-foreground mt-0.5">{getRelativeTime(entry.time!)}</p>
   423→            </div>
   424→          </div>
   425→        ))}
   426→      </div>
   427→    </div>
   428→  );
   429→}

   433→export default function DeliveriesPage() {
   434→  const [deliveries, setDeliveries] = React.useState<DeliveryItem[]>([]);
   435→  const [isLoading, setIsLoading] = React.useState(true);
   436→  const [dateRange, setDateRange] = React.useState<DateRange>("month");
   437→  const [statusFilter, setStatusFilter] = React.useState<StatusFilter>("ALL");
   438→  const [searchQuery, setSearchQuery] = React.useState("");
   439→  const [expandedId, setExpandedId] = React.useState<string | null>(null);
   440→  const [error, setError] = React.useState<string | null>(null);
   441→
   442→  // ==================== Data Loading ====================
   443→
   444→  React.useEffect(() => {
   445→    const load = async () => {
   446→      setIsLoading(true);
   447→      setError(null);
   448→      try {
   449→        // Try deliveriesApi first (may not exist yet)
   450→        const apiClient = await import("@kwikseller/api-client");
   451→        const deliveriesApi = (apiClient as Record<string, unknown>).deliveriesApi as {
   452→          list: () => Promise<{ data: unknown }>;
   453→        } | undefined;
   454→
   455→        if (deliveriesApi?.list) {
   456→          const response = await deliveriesApi.list();
   457→          const data = unwrapApiData<DeliveryItem[]>(response.data);
   458→          if (Array.isArray(data) && data.length > 0) {
   459→            setDeliveries(data);
   460→            setIsLoading(false);
   461→            return;
   462→          }
   463→        }
   464→
   465→        // Fallback to localStorage
   466→        const saved = loadDeliveries();
   467→        if (saved && saved.length > 0) {
   468→          setDeliveries(saved);
   469→        } else {
   470→          // Fallback to demo data
   471→          setDeliveries(createDemoDeliveries());
   472→        }
   473→      } catch {
   474→        // Fallback to localStorage or demo data
   475→        const saved = loadDeliveries();
   476→        setDeliveries(saved || createDemoDeliveries());
   477→      } finally {
   478→        setIsLoading(false);
   479→      }
   480→    };
   481→    load();
   482→  }, []);
   483→
   484→  // Persist deliveries when they change
   485→  React.useEffect(() => {
   486→    if (deliveries.length > 0 && !isLoading) {
   487→      saveDeliveries(deliveries);
   488→    }
   489→  }, [deliveries, isLoading]);
   490→
   491→  // ==================== Derived State ====================
   492→
   493→  const activeDeliveries = React.useMemo(() => {
   494→    return deliveries.filter((d) => !isTerminalStatus(d.status));
   495→  }, [deliveries]);
   496→
   497→  const completedDeliveries = React.useMemo(() => {
   498→    return deliveries.filter((d) => isTerminalStatus(d.status));
   499→  }, [deliveries]);
   500→
   501→  const displayedDeliveries = React.useMemo(() => {
   502→    const isFilteringTerminal =
   503→      statusFilter === "DELIVERED" || statusFilter === "COMPLETED";
   504→
   505→    let list = isFilteringTerminal ? completedDeliveries : activeDeliveries;
   506→
   507→    // Filter by status
   508→    if (statusFilter !== "ALL") {
   509→      if (statusFilter === "DELIVERED") {
   510→        list = list.filter((d) => d.status === "DELIVERED");
   511→      } else if (statusFilter === "COMPLETED") {
   512→        list = list.filter((d) => d.status === "COMPLETED");
   513→      } else {
   514→        list = list.filter((d) => d.status === statusFilter);
   515→      }
   516→    }
   517→
   518→    // Search filter
   519→    if (searchQuery) {
   520→      const q = searchQuery.toLowerCase();
   521→      list = list.filter(
   522→        (d) =>
   523→          d.orderId.toLowerCase().includes(q) ||
   524→          d.customerName.toLowerCase().includes(q) ||
   525→          d.deliveryAddress.toLowerCase().includes(q) ||
   526→          (d.riderName && d.riderName.toLowerCase().includes(q))
   527→      );
   528→    }
   529→
   530→    return list;
   531→  }, [activeDeliveries, completedDeliveries, statusFilter, searchQuery]);
   532→
   533→  // Stats
   534→  const todayDeliveredCount = React.useMemo(() => {
   535→    const todayStart = new Date();
   536→    todayStart.setHours(0, 0, 0, 0);
   537→    return deliveries.filter(
   538→      (d) =>
   539→        (d.status === "DELIVERED" || d.status === "COMPLETED") &&
   540→        d.deliveredAt &&
   541→        new Date(d.deliveredAt) >= todayStart
   542→    ).length;
   543→  }, [deliveries]);
   544→
   545→  const avgDeliveryMinutes = React.useMemo(() => {
   546→    const completed = deliveries.filter(
   547→      (d) => d.acceptedAt && d.deliveredAt
   548→    );
   549→    if (completed.length === 0) return null;
   550→    const totalMinutes = completed.reduce((sum, d) => {
   551→      return sum + (new Date(d.deliveredAt!).getTime() - new Date(d.acceptedAt!).getTime()) / 60000;
   552→    }, 0);
   553→    const avg = Math.round(totalMinutes / completed.length);
   554→    if (avg < 60) return `${avg}m`;
   555→    if (avg < 1440) return `${Math.round(avg / 60)}h`;
   556→    return `${Math.round(avg / 1440)}d`;
   557→  }, [deliveries]);
   558→
   559→  // ==================== Action Handlers ====================
   560→

   700→            <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">
   701→              Active Deliveries
   702→            </p>
   703→            <p className="mt-1 text-xl font-semibold text-foreground">

   762→
   763→      {/* ==================== Section 4: Delivery List ==================== */}

   794→              </p>
   795→            </div>
   796→          ) : (
   797→            <div className="flex flex-col items-center justify-center py-16">
   798→              <Truck className="h-10 w-10 text-muted-foreground/50" strokeWidth={1.5} />
   799→              <p className="mt-3 text-sm font-medium text-muted-foreground">
   800→                No active deliveries
   801→              </p>
   802→              <p className="mt-1 text-xs text-muted-foreground">
   803→                {searchQuery || statusFilter !== "ALL"
   804→                  ? "No results match your current filters."
   805→                  : "Active deliveries will appear here when orders are placed."}
   806→              </p>
   807→            </div>
   808→          )}
   809→        </section>
   810→      ) : (
   811→        <section>
   812→          <div className="border-b border-kwik-border">
   813→            {displayedDeliveries.map((d) => (
   814→              <DeliveryRow
   815→                key={d.id}
   816→                delivery={d}
   817→                isExpanded={expandedId === d.id}
   818→                onToggle={() => toggleExpanded(d.id)}
   819→                onMarkReady={handleMarkReady}
   820→                onConfirmPickup={handleConfirmPickup}
   821→              />
   822→            ))}
   823→          </div>
   824→        </section>
   825→      )}
   826→
   827→      {/* Error state */}
   828→      {error && (
   829→        <section className="flex flex-col items-center justify-center py-12">
   830→          <AlertCircle className="h-10 w-10 text-red-400 dark:text-red-500" strokeWidth={1.5} />
   831→          <p className="mt-3 text-sm text-red-600 dark:text-red-400">{error}</p>
   832→          <AppButton
   833→            variant="secondary"
   834→            size="sm"
   835→            className="mt-3"
   836→            onClick={() => window.location.reload()}
   837→          >
   838→            Try Again
   839→          </AppButton>
   840→        </section>
   841→      )}
   842→    </motion.div>
   843→  );
   844→}
