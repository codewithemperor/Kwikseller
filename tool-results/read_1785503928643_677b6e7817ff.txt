    40→function getCategoryStyle(name: string, slug: string, index: number): CategoryStyle {
    41→  const key = (slug || "").toLowerCase();
    42→  if (CATEGORY_STYLES[key]) return CATEGORY_STYLES[key];
    43→  for (const [k, v] of Object.entries(CATEGORY_STYLES)) {
    44→    if (name.toLowerCase().includes(k) || key.includes(k)) return v;
    45→  }
    46→  const colorIdx = index % CARD_ACCENT_COLORS.length;
    47→  return {
    48→    color: CARD_ACCENT_COLORS[colorIdx],
    49→    textColor: CARD_TEXT_COLORS[colorIdx],
    50→    Icon: DEFAULT_STYLE.Icon,
    51→  };
    52→}

    65→function StaggerWrap({ children }: { children: React.ReactNode }) {
    66→  const ref = useRef(null);

    68→  return (
    69→    <motion.div ref={ref} initial="hidden" animate={inView ? "visible" : "hidden"} variants={staggerVariants}>
    70→      {children}
    71→    </motion.div>
    72→  );

    86→const withTimeout = async <T,>(promise: Promise<T>, ms = 8000): Promise<T> => {
    87→  let timer: ReturnType<typeof setTimeout> | undefined;
    88→  try {
    89→    return await Promise.race([
    90→      promise,
    91→      new Promise<never>((_, reject) => {
    92→        timer = setTimeout(() => reject(new Error("Request timed out")), ms);
    93→      }),
    94→    ]);
    95→  } finally {
    96→    if (timer) clearTimeout(timer);
    97→  }
    98→};

   102→function getProductImage(p: SearchableProduct & { images?: Array<{ url?: string } | string>; featuredImage?: { url?: string } | string }): string {
   103→  const firstImage = Array.isArray(p.images) ? p.images[0] : null;
   104→  if (typeof p.image === "string") return p.image;
   105→  if (typeof firstImage === "string") return firstImage;
   106→  return firstImage?.url || (typeof p.featuredImage === "string" ? p.featuredImage : p.featuredImage?.url) || "";
   107→}
   108→
   109→function getProductTags(value: unknown): string[] {
   110→  if (!Array.isArray(value)) return [];
   111→  return value
   112→    .map((tag) => {
   113→      if (typeof tag === "string") return tag;
   114→      if (tag && typeof tag === "object") {
   115→        const record = tag as { tag?: { name?: string }; name?: string };
   116→        return record.tag?.name || record.name || "";
   117→      }
   118→      return "";
   119→    })
   120→    .filter(Boolean);
   121→}
   122→
   123→function getEntityName(value: unknown): string {
   124→  if (!value) return "";
   125→  if (typeof value === "string") return value;
   126→  if (typeof value === "object") return (value as { name?: string; slug?: string }).name || (value as { slug?: string }).slug || "";
   127→  return "";
   128→}

   130→function toMarketplaceProduct(p: SearchableProduct & { images?: Array<{ url?: string } | string>; store?: unknown; category?: unknown; storeName?: string; categoryName?: string; tags?: Array<string | { tag?: { name?: string }; name?: string }> }): MarketplaceProduct {
   131→  const image = getProductImage(p);
   132→  const tags = getProductTags(p.tags);
   133→  const store = getEntityName(p.store) || p.storeName || "";
   134→  const category = p.categorySlug || p.categoryName || getEntityName(p.category);
   135→  return {
   136→    id: p.id,
   137→    slug: p.slug,
   138→    name: p.name,
   139→    price: p.price,
   140→    comparePrice: p.comparePrice,
   141→    image,
   142→    rating: p.rating || 0,
   143→    reviewCount: p.reviewCount || 0,
   144→    store,
   145→    storeId: p.storeId,
   146→    storeSlug: p.storeSlug,
   147→    category,
   148→    isNew: p.isNew,
   149→    tag: p.categoryName || getEntityName(p.category),
   150→    description: p.description,
   151→    images: image ? [image] : [],
   152→    features: tags.slice(0, 4),
   153→    specifications: [],
   154→    reviews: [],
   155→  };
   156→}

   555→                <button

   557→                  onClick={() => setSearchQuery("")}
   558→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-kwik-muted hover:text-kwik-dark-medium transition-colors"
   559→                >
   560→                  <span className="text-xs font-medium">✕</span>
   561→                </button>
   562→              )}
   563→            </div>
   564→
   565→            {/* Sort options */}
   566→            <div className="flex items-center gap-2">
   567→              <span className="text-xs font-semibold text-kwik-gray-light uppercase tracking-wider whitespace-nowrap hidden sm:inline">
   568→                Sort:
   569→              </span>
   570→              <div className="flex items-center gap-1 rounded-md border border-kwik-border bg-white p-1 dark:border-white/10 dark:bg-white/5">
   571→                {CATEGORY_SORT_OPTIONS.map((opt) => (
   572→                  <button
   573→                    key={opt.value}
   574→                    type="button"
   575→                    onClick={() => setSortBy(opt.value)}
   576→                    className={`rounded-lg px-3 py-1.5 text-xs font-medium transition-all duration-200 ${
   577→                      sortBy === opt.value
   578→                        ? "bg-kwik-orange text-white shadow-sm"
   579→                        : "text-kwik-gray-light hover:bg-neutral-50 hover:text-kwik-dark dark:text-white/65 dark:hover:bg-white/10 dark:hover:text-white"
   580→                    }`}
   581→                  >
   582→                    {opt.label}
   583→                  </button>
   584→                ))}
   585→              </div>
   586→            </div>
   587→          </div>
   588→        </div>
   589→      </div>
   590→      </>
   591→      )}
   592→
   593→      {/* Category grid */}
   594→      <div className="container mx-auto px-4 py-6">
   595→        {isLoading ? (
   596→          <div className="flex items-center justify-center py-20">
   597→            <Loader2 className="h-8 w-8 animate-spin text-kwik-orange" />
   598→          </div>
   599→        ) : filteredCategories.length === 0 ? (
   600→          <motion.div
   601→            initial={{ opacity: 0, y: 20 }}
   602→            animate={{ opacity: 1, y: 0 }}
   603→            transition={{ duration: 0.4 }}
   604→            className="flex flex-col items-center justify-center py-20"
   605→          >
   606→            <div className="h-28 w-28 rounded-3xl bg-gradient-to-br from-kwik-orange/10 to-kwik-orange/5 flex items-center justify-center mb-5">
   607→              <PackageOpen className="h-12 w-12 text-kwik-orange/60" />
   608→            </div>
   609→            <h3 className="text-lg font-semibold text-kwik-dark mb-2">
   610→              {searchQuery ? "No matching categories" : "No categories yet"}
   611→            </h3>
   612→            <p className="text-sm text-kwik-gray-light text-center max-w-[320px] mb-4">
   613→              {searchQuery
   614→                ? `No categories match "${searchQuery}". Try a different search term.`
   615→                : "Categories will appear here once sellers start listing products."}
   616→            </p>
   617→            {searchQuery && (
   618→              <button
   619→                type="button"
   620→                onClick={() => setSearchQuery("")}
   621→                className="rounded-xl px-5 py-2.5 text-sm font-medium text-kwik-orange border border-kwik-orange hover:bg-kwik-orange-tint transition-colors"
   622→              >
   623→                Clear search
   624→              </button>
   625→            )}
   626→          </motion.div>
   627→        ) : (
   628→          <>
   629→            {/* Results info */}
   630→            {searchQuery && (
   631→              <motion.p
   632→                initial={{ opacity: 0 }}
   633→                animate={{ opacity: 1 }}
   634→                className="text-sm text-kwik-gray-light mb-4"
   635→              >
   636→                <span className="font-semibold text-kwik-dark">{filteredCategories.length}</span>{" "}
   637→                {filteredCategories.length === 1 ? "category" : "categories"} found
   638→                for "<span className="text-kwik-orange">{searchQuery}</span>"
   639→              </motion.p>
   640→            )}
   641→
   642→            <StaggerWrap>
   643→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-4">
   644→                {filteredCategories.map((category, index) => {
   645→                  const style = getCategoryStyle(
   646→                    category.name,
   647→                    category.slug || category.id,
   648→                    index,
   649→                  );
   650→                  const { Icon } = style;
   651→
   652→                  return (
   653→                    <StaggerChild key={`${category.id}-${index}`} index={index} className="w-full">
   654→                      <motion.button
   655→                        type="button"
   656→                        onClick={() => router.push(`/categories?name=${category.slug || category.id}`)}
   657→                        whileHover={{ y: -4 }}
   658→                        whileTap={{ scale: 0.98 }}
   659→                        className="group h-full w-full cursor-pointer border border-neutral-200 bg-white p-4 text-left transition-all duration-300 hover:border-kwik-dark dark:border-white/10 dark:bg-white/5 dark:hover:border-white/50"
   660→                      >
   661→                        <div className="flex items-center gap-4">
   662→                          {/* Colored icon box */}
   663→                          <div
   664→                            className={cn(
   665→                              "w-12 h-12 rounded-xl flex items-center justify-center shrink-0 text-white shadow-md group-hover:scale-110 transition-transform",
   666→                              style.color,
   667→                            )}
   668→                          >
   669→                            <Icon className="w-6 h-6" />
   670→                          </div>
   671→
   672→                          {/* Category info */}
   673→                          <div className="flex-1 min-w-0">
   674→                            <h3 className="font-semibold text-base text-kwik-dark dark:text-white">
   675→                              {category.name}
   676→                            </h3>
   677→                          </div>
   678→
   679→                          {/* Chevron */}
   680→                          <ChevronRight className="w-4 h-4 text-kwik-muted group-hover:text-kwik-orange group-hover:translate-x-1 transition-all shrink-0 mt-1" />
   681→                        </div>
   682→                      </motion.button>
   683→                    </StaggerChild>
   684→                  );
   685→                })}
   686→              </div>
   687→            </StaggerWrap>
   688→          </>
   689→        )}
   690→      </div>
   691→    </div>
   692→  );
   693→}

   696→
   697→function CategoriesPageContent() {
   698→  const searchParams = useSearchParams();
   699→  // URL format: /categories?name=electronics
   700→  const slug = searchParams.get('name') || '';
   701→
   702→  if (slug) {
   703→    return <CategoryDetailView slug={slug} />;
   704→  }
   705→
   706→  return <AllCategoriesView />;
   707→}
