    68→function unwrapApiData<T>(value: unknown): T {
    69→  if (value && typeof value === "object" && "data" in value) {
    70→    return (value as { data: T }).data;
    71→  }
    72→  return value as T;
    73→}
    74→
    75→function unwrapNestedApiData<T>(value: unknown): T {
    76→  const payload = value as { data?: unknown };
    77→  const nested = payload?.data as { data?: unknown } | undefined;
    78→  return (nested?.data ?? payload?.data ?? value) as T;
    79→}

    99→function extractValidationIssues(error: unknown): CartValidationIssue[] {
   100→  const responseData = (error as { response?: { data?: unknown } })?.response?.data;
   101→  const candidates = [
   102→    responseData,
   103→    responseData && typeof responseData === "object" && "data" in responseData
   104→      ? (responseData as { data?: unknown }).data
   105→      : null,
   106→    responseData && typeof responseData === "object" && "message" in responseData
   107→      ? (responseData as { message?: unknown }).message
   108→      : null,
   109→  ];
   110→
   111→  for (const candidate of candidates) {
   112→    if (candidate && typeof candidate === "object" && "errors" in candidate) {
   113→      const errors = (candidate as { errors?: unknown }).errors;
   114→      if (Array.isArray(errors)) return errors as CartValidationIssue[];
   115→    }
   116→  }
   117→
   118→  return [];
   119→}

   132→function formatDeliveryWindow(rate?: DeliveryRate | null) {
   133→  if (!rate?.estimatedDeliveryStart || !rate?.estimatedDeliveryEnd) return null;
   134→  const formatter = new Intl.DateTimeFormat("en-NG", { month: "short", day: "numeric" });
   135→  return `${formatter.format(new Date(rate.estimatedDeliveryStart))} - ${formatter.format(new Date(rate.estimatedDeliveryEnd))}`;
   136→}

   138→function groupCartItems(items: ReturnType<typeof useCartStore.getState>["items"]) {
   139→  const groups = new Map<
   140→    string,
   141→    {
   142→      storeId: string;
   143→      storeSlug?: string;
   144→      storeName: string;
   145→      items: typeof items;
   146→      subtotal: number;
   147→      requiresShipping: boolean;
   148→      hasDigitalDelivery: boolean;
   149→      hasPoolResale: boolean;
   150→    }
   151→  >();
   152→
   153→  for (const item of items) {
   154→    const storeId = item.storeId ?? item.storeSlug ?? item.store ?? "kwikseller-store";
   155→    if (!groups.has(storeId)) {
   156→      groups.set(storeId, {
   157→        storeId,
   158→        storeSlug: item.storeSlug,
   159→        storeName: item.storeName ?? item.store ?? "Kwikseller vendor",
   160→        items: [],
   161→        subtotal: 0,
   162→        requiresShipping: false,
   163→        hasDigitalDelivery: false,
   164→        hasPoolResale: false,
   165→      });
   166→    }
   167→
   168→    const group = groups.get(storeId)!;
   169→    group.items.push(item);
   170→    group.subtotal += item.price * item.quantity;
   171→    group.requiresShipping = group.requiresShipping || (item.requiresShipping !== false && item.productType !== "DIGITAL");
   172→    group.hasDigitalDelivery = group.hasDigitalDelivery || item.productType === "DIGITAL" || item.requiresShipping === false;
   173→    group.hasPoolResale = group.hasPoolResale || item.productSource === "POOL_RESALE" || Boolean(item.poolOfferId);
   174→  }
   175→
   176→  return [...groups.values()];
   177→}
