   299→/* ─── Create Product Wizard Modal ─── */
   300→
   301→function CreateProductModal({
   302→  isOpen,
   303→  onClose,
   304→  onCreated,
   305→}: {
   306→  isOpen: boolean;
   307→  onClose: () => void;
   308→  onCreated: () => void;
   309→}) {
   310→  const [form, setForm] = React.useState(blankForm);
   311→  const [step, setStep] = React.useState(0);
   312→  const [isSaving, setIsSaving] = React.useState(false);
   313→  const [isUploading, setIsUploading] = React.useState(false);
   314→
   315→  const resetAndClose = () => {
   316→    setForm(blankForm);
   317→    setStep(0);
   318→    onClose();
   319→  };
   320→
   321→  const validateStep = (nextStep: number) => {
   322→    if (step === 0 && (!form.name.trim() || !form.productType)) {
   323→      kwikToast.error("Add product name and type first");
   324→      return;
   325→    }
   326→    if (step === 1 && Number(form.price) <= 0) {
   327→      kwikToast.error("Add a valid product price");
   328→      return;
   329→    }
   330→    setStep(nextStep);
   331→  };
   332→
   333→  const createProduct = async () => {
   334→    if (!form.name.trim()) { kwikToast.error("Product name is required"); setStep(0); return; }
   335→    if (Number(form.price) <= 0) { kwikToast.error("Price must be greater than zero"); setStep(1); return; }
   336→
   337→    setIsSaving(true);
   338→    try {
   339→      await vendorCommerceApi.createProduct({
   340→        name: form.name,
   341→        description: form.description || undefined,
   342→        price: Number(form.price),
   343→        comparePrice: form.comparePrice ? Number(form.comparePrice) : undefined,
   344→        sku: form.sku || undefined,
   345→        productType: form.productType,
   346→        requiresShipping: form.productType === "PHYSICAL",
   347→        trackInventory: form.productType === "PHYSICAL",
   348→        initialStock: form.productType === "PHYSICAL" ? Number(form.initialStock) : undefined,
   349→        lowStock: form.productType === "PHYSICAL" ? Number(form.lowStock) : undefined,
   350→        images: form.images,
   351→        poolEnabled: form.productType === "PHYSICAL" ? form.poolEnabled : false,
   352→        poolBasePrice: form.poolEnabled ? Number(form.poolBasePrice || form.price) : undefined,
   353→        poolMinSalePrice: form.poolEnabled ? Number(form.poolMinSalePrice || form.poolBasePrice || form.price) : undefined,
   354→        poolMaxSelectableQuantity: form.poolEnabled && form.poolMaxSelectableQuantity ? Number(form.poolMaxSelectableQuantity) : undefined,
   355→      });
   356→      kwikToast.success("Product created");
   357→      resetAndClose();
   358→      onCreated();
   359→    } catch (error) {
   360→      kwikToast.error(error instanceof Error ? error.message : "Product creation failed");
   361→    } finally {
   362→      setIsSaving(false);
   363→    }
   364→  };
   365→
   366→  const uploadImages = async (files: FileList | null) => {
   367→    if (!files?.length) return;
   368→    setIsUploading(true);
   369→    try {
   370→      const uploads = await Promise.all(Array.from(files).slice(0, 5).map((f) => uploadApi.productImage(f)));
   371→      const urls = uploads.map(uploadedUrl).filter(Boolean);
   372→      setForm((c) => ({ ...c, images: [...c.images, ...urls].slice(0, 5) }));
   373→      kwikToast.success(`${urls.length} image${urls.length === 1 ? "" : "s"} uploaded`);
   374→    } catch (error) {
   375→      kwikToast.error(error instanceof Error ? error.message : "Image upload failed");
   376→    } finally {
   377→      setIsUploading(false);
   378→    }
   379→  };
   380→
   381→  return (
   382→    <AppModal isOpen={isOpen} onClose={resetAndClose} title="Add Product" description="Complete the required steps, then confirm." className="sm:max-w-3xl">
   383→      <div className="space-y-6">
   384→        {/* Step indicator */}
   385→        <div className="grid grid-cols-4 gap-2">
   386→          {wizardSteps.map((label, i) => {
   387→            const done = i < step;
   388→            const active = i === step;
   389→            return (
   390→              <button key={label} type="button" onClick={() => i <= step && setStep(i)} className="flex flex-col items-center gap-2 text-center">
   391→                <span className={`flex h-10 w-10 items-center justify-center rounded-full border text-sm font-semibold ${done ? "border-foreground bg-foreground text-background" : active ? "border-foreground bg-surface text-foreground" : "border-kwik-border bg-surface text-muted-foreground"}`}>
   392→                  {done ? <Check className="h-4 w-4" /> : i + 1}
   393→                </span>
   394→                <span className="hidden text-xs font-medium text-muted-foreground sm:block">{label}</span>
   395→              </button>
   396→            );
   397→          })}
   398→        </div>
   399→
   400→        {/* Step 0 — Basic Info */}
   401→        {step === 0 && (
   402→          <div className="space-y-4">
   403→            <FieldInput required label="Product name" value={form.name} onChange={(e) => setForm((v) => ({ ...v, name: e.target.value }))} />
   404→            <FieldSelect label="Product type" value={form.productType} onChange={(e) => setForm((v) => ({ ...v, productType: e.target.value as ProductType }))}>
   405→              <option value="PHYSICAL">Physical</option>
   406→              <option value="DIGITAL">Digital</option>
   407→            </FieldSelect>
   408→            <FieldTextarea label="Description" value={form.description} onChange={(e) => setForm((v) => ({ ...v, description: e.target.value }))} />
   409→          </div>
   410→        )}
   411→
   412→        {/* Step 1 — Pricing & Stock */}
   413→        {step === 1 && (
   414→          <div className="space-y-4">
   415→            <div className="grid gap-3 sm:grid-cols-2">
   416→              <FieldInput required type="number" min={0} label="Price" value={form.price} onChange={(e) => setForm((v) => ({ ...v, price: Number(e.target.value) }))} />
   417→              <FieldInput type="number" min={0} label="Compare price" value={form.comparePrice} onChange={(e) => setForm((v) => ({ ...v, comparePrice: Number(e.target.value) }))} />
   418→            </div>
   419→            {form.productType === "PHYSICAL" ? (
   420→              <div className="grid gap-3 sm:grid-cols-2">
   421→                <FieldInput type="number" min={0} label="Initial stock" value={form.initialStock} onChange={(e) => setForm((v) => ({ ...v, initialStock: Number(e.target.value) }))} />
   422→                <FieldInput type="number" min={0} label="Low stock alert" value={form.lowStock} onChange={(e) => setForm((v) => ({ ...v, lowStock: Number(e.target.value) }))} />
   423→              </div>
   424→            ) : (
   425→              <p className="text-sm text-muted-foreground">Digital products default to unlimited delivery.</p>
   426→            )}
   427→          </div>
   428→        )}
   429→
   430→        {/* Step 2 — Details */}
   431→        {step === 2 && (
   432→          <div className="space-y-4">
   433→            <FieldInput label="SKU" value={form.sku} onChange={(e) => setForm((v) => ({ ...v, sku: e.target.value }))} />
   434→            <label className="block">
   435→              <span className="text-xs font-semibold text-muted-foreground">Product images</span>
   436→              <div className="mt-2 flex min-h-44 cursor-pointer flex-col items-center justify-center rounded-lg border border-dashed border-kwik-border bg-default-100 p-6 text-center">
   437→                <Upload className="h-8 w-8 text-muted-foreground" />
   438→                <p className="mt-2 text-sm font-medium text-foreground">
   439→                  <span className="text-accent underline">Click to upload</span> or drag and drop
   440→                </p>
   441→                <p className="mt-1 text-xs text-muted-foreground">{`${form.images.length}/5 images added`}</p>
   442→                <input type="file" accept="image/*" multiple onChange={(e) => uploadImages(e.target.files)} className="sr-only" />
   443→              </div>
   444→            </label>
   445→            {form.images.length > 0 && (
   446→              <div className="grid grid-cols-5 gap-2">
   447→                {form.images.map((url) => (
   448→                  <button key={url} type="button" onClick={() => setForm((c) => ({ ...c, images: c.images.filter((img) => img !== url) }))} className="aspect-square overflow-hidden rounded-lg border border-kwik-border bg-default-100" title="Remove image">
   449→                    <img src={url} alt="" className="h-full w-full object-cover" />
   450→                  </button>
   451→                ))}
   452→              </div>
   453→            )}
   454→            {form.productType === "PHYSICAL" && (
   455→              <div className="border border-kwik-border p-4">
   456→                <AppSwitch
   457→                  isSelected={form.poolEnabled}
   458→                  onChange={(sel) => setForm((v) => ({
   459→                    ...v,
   460→                    poolEnabled: sel,
   461→                    poolBasePrice: sel && !v.poolBasePrice ? Number(v.price) : v.poolBasePrice,
   462→                    poolMinSalePrice: sel && !v.poolMinSalePrice ? Number(v.price) : v.poolMinSalePrice,
   463→                  }))}
   464→                  label="Make available in Pool"
   465→                  description="Other vendors can select and sell this product."
   466→                />
   467→                {form.poolEnabled && (
   468→                  <div className="mt-4 grid gap-3 sm:grid-cols-3">
   469→                    <FieldInput type="number" min={0} label="Source price" value={form.poolBasePrice || form.price} onChange={(e) => setForm((v) => ({ ...v, poolBasePrice: Number(e.target.value) }))} />
   470→                    <FieldInput type="number" min={form.poolBasePrice || form.price} label="Min sale price" value={form.poolMinSalePrice || form.poolBasePrice || form.price} onChange={(e) => setForm((v) => ({ ...v, poolMinSalePrice: Number(e.target.value) }))} />
   471→                    <FieldInput type="number" min={0} label="Pool quantity" placeholder="Use stock" value={form.poolMaxSelectableQuantity || ""} onChange={(e) => setForm((v) => ({ ...v, poolMaxSelectableQuantity: Number(e.target.value) }))} />
   472→                  </div>
   473→                )}
   474→              </div>
   475→            )}
   476→          </div>
   477→        )}
   478→
   479→        {/* Step 3 — Review */}
   480→        {step === 3 && (
   481→          <div className="space-y-4">
   482→            <div className="border border-kwik-border p-5">
   483→              <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Review</p>
   484→              <h3 className="mt-2 text-xl font-semibold text-foreground">{form.name || "Untitled product"}</h3>
   485→              <p className="mt-2 text-sm leading-6 text-muted-foreground">{form.description || "No description added."}</p>
   486→              <div className="mt-4 grid gap-3 sm:grid-cols-3">
   487→                <div><p className="text-xs text-muted-foreground">Type</p><p className="font-semibold text-foreground">{form.productType}</p></div>
   488→                <div><p className="text-xs text-muted-foreground">Price</p><p className="font-semibold text-foreground">{formatCurrency(form.price)}</p></div>
   489→                <div><p className="text-xs text-muted-foreground">Images</p><p className="font-semibold text-foreground">{form.images.length}</p></div>
   490→              </div>
   491→            </div>
   492→            {form.poolEnabled && (
   493→              <p className="text-sm font-medium text-emerald-600 dark:text-emerald-400">
   494→                This product will be available in Pool at source price {formatCurrency(form.poolBasePrice || form.price)}.
   495→              </p>
   496→            )}
   497→          </div>
   498→        )}
   499→
   500→        {/* Footer */}
   501→        <div className="flex items-center justify-between gap-3 border-t border-kwik-border pt-4">
   502→          <AppButton type="button" variant="secondary" disabled={step === 0 || isSaving} onClick={() => setStep((c) => Math.max(0, c - 1))}>
   503→            <ChevronLeft className="h-4 w-4" />
   504→            Back
   505→          </AppButton>
   506→          {step < wizardSteps.length - 1 ? (
   507→            <AppButton type="button" onClick={() => validateStep(step + 1)}>
   508→              Next
   509→              <ChevronRight className="h-4 w-4" />
   510→            </AppButton>
   511→          ) : (
   512→            <AppButton type="button" disabled={isSaving || isUploading} onClick={createProduct}>
   513→              <Plus className="h-4 w-4" />
   514→              Create product
   515→            </AppButton>
   516→          )}
   517→        </div>
   518→      </div>
   519→    </AppModal>
   520→  );
   521→}

   523→/* ─── Pagination ─── */
   524→
   525→function Pagination({
   526→  total,
   527→  page,
   528→  onPageChange,
   529→}: {
   530→  total: number;
   531→  page: number;
   532→  onPageChange: (p: number) => void;
   533→}) {
   534→  const totalPages = Math.max(1, Math.ceil(total / ITEMS_PER_PAGE));
   535→  const start = (page - 1) * ITEMS_PER_PAGE + 1;
   536→  const end = Math.min(page * ITEMS_PER_PAGE, total);
   537→
   538→  const pages: (number | "...")[] = [];
   539→  if (totalPages <= 7) {
   540→    for (let i = 1; i <= totalPages; i++) pages.push(i);
   541→  } else {
   542→    pages.push(1);
   543→    if (page > 3) pages.push("...");
   544→    for (let i = Math.max(2, page - 1); i <= Math.min(totalPages - 1, page + 1); i++) pages.push(i);
   545→    if (page < totalPages - 2) pages.push("...");
   546→    pages.push(totalPages);
   547→  }
   548→
   549→  return (
   550→    <div className="flex flex-col items-center justify-between gap-4 sm:flex-row">
   551→      <p className="text-sm text-muted-foreground">
   552→        Showing {total > 0 ? start : 0}–{end} of {total} products
   553→      </p>
   554→      <div className="flex items-center gap-1">
   555→        <button
   556→          type="button"
   557→          disabled={page <= 1}
   558→          onClick={() => onPageChange(page - 1)}
   559→          className="inline-flex h-8 items-center gap-1 rounded px-2 text-sm font-medium text-muted-foreground transition hover:bg-default-100 disabled:cursor-not-allowed disabled:opacity-40"
   560→        >
   561→          <ChevronLeft className="h-3.5 w-3.5" />
   562→          Prev
   563→        </button>
   564→        {pages.map((p, i) =>
   565→          p === "..." ? (
   566→            <span key={`dots-${i}`} className="px-1 text-sm text-muted-foreground">...</span>
   567→          ) : (
   568→            <button
   569→              key={p}
   570→              type="button"
   571→              onClick={() => onPageChange(p as number)}
   572→              className={`inline-flex h-8 w-8 items-center justify-center rounded text-sm font-medium transition ${p === page ? "bg-foreground text-background" : "text-muted-foreground hover:bg-default-100"}`}
   573→            >
   574→              {p}
   575→            </button>
   576→          )
   577→        )}
   578→        <button
   579→          type="button"
   580→          disabled={page >= totalPages}
   581→          onClick={() => onPageChange(page + 1)}
   582→          className="inline-flex h-8 items-center gap-1 rounded px-2 text-sm font-medium text-muted-foreground transition hover:bg-default-100 disabled:cursor-not-allowed disabled:opacity-40"
   583→        >
   584→          Next
   585→          <ChevronRight className="h-3.5 w-3.5" />
   586→        </button>
   587→      </div>
   588→    </div>
   589→  );
   590→}
