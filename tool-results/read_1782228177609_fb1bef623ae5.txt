   132→  private async resolveStoreId(user: AuthContext) {
   133→    if (user.storeId) {
   134→      return user.storeId;
   135→    }
   136→
   137→    const userId = this.getUserId(user);
   138→    const store = await this.db()
   139→      .store?.findFirst({
   140→        where: { vendorId: userId },
   141→        select: { id: true },
   142→      })
   143→      .catch(() => null);
   144→
   145→    if (store?.id) {
   146→      return store.id;
   147→    }
   148→
   149→    const vendor = await this.db().user?.findUnique({
   150→      where: { id: userId },
   151→      include: { profile: true },
   152→    });
   153→    if (!vendor || `${vendor.role}`.toUpperCase() !== 'VENDOR') {
   154→      throw new ForbiddenException('Vendor store is required for this action');
   155→    }
   156→
   157→    const baseName =
   158→      [vendor.profile?.firstName, vendor.profile?.lastName].filter(Boolean).join(' ').trim() ||
   159→      vendor.email?.split('@')[0] ||
   160→      'Vendor Store';
   161→    const created = await this.db().store?.create({
   162→      data: {
   163→        vendorId: userId,
   164→        name: baseName,
   165→        slug: `${this.slugify(baseName)}-${randomUUID().slice(0, 6)}`,
   166→        category: 'other',
   167→      },
   168→      select: { id: true },
   169→    });
   170→
   171→    if (!created?.id) {
   172→      throw new ForbiddenException('Vendor store is required for this action');
   173→    }
   174→
   175→    await Promise.allSettled([
   176→      this.db().storefrontDesign?.create({
   177→        data: {
   178→          storeId: created.id,
   179→          headingFont: 'SORA',
   180→          bodyFont: 'FIGTREE',
   181→        },
   182→      }),
   183→      this.db().storeDeliverySetting?.create({
   184→        data: {
   185→          storeId: created.id,
   186→          manualDeliveryEnabled: true,
   187→          kwiksellerDeliveryEnabled: false,
   188→          processingDays: 1,
   189→        },
   190→      }),
   191→    ]);
   192→
   193→    return created.id;
   194→  }

   252→  private assertTemplateKey<T extends keyof typeof STOREFRONT_TEMPLATE_OPTIONS>(field: T, value?: string) {
   253→    if (!value) return undefined;
   254→    const options = STOREFRONT_TEMPLATE_OPTIONS[field] as readonly string[];
   255→    if (!options.includes(value)) {
   256→      throw new BadRequestException(`${field} must be one of: ${options.join(', ')}`);

   261→  private assertProductCanPublish(product: any) {
   262→    if (!product.categoryId) {
   263→      throw new BadRequestException('Publishing requires a category');
   264→    }
   265→    const images = Array.isArray(product.images) ? product.images : [];
   266→    if (!images.length) {
   267→      throw new BadRequestException('Publishing requires at least one product image');
   268→    }
   269→    if (Number(product.price ?? 0) <= 0) {
   270→      throw new BadRequestException('Publishing requires a valid price');
   271→    }
   272→    const isTrackedPhysical =
   273→      product.productType === 'PHYSICAL' &&
   274→      (product.trackInventory ?? product.inventoryPolicy === 'TRACKED');
   275→    if (isTrackedPhysical && Number(product.stock ?? 0) < 1) {
   276→      throw new BadRequestException('Publishing tracked physical products requires available stock');
   277→    }
   278→  }

   344→  async addCartItem(user: AuthContext, dto: AddCartItemDto) {
   345→    this.getUserId(user);
   346→    const db = this.db();
   347→    const cart = await this.getCart(user);
   348→    const product = await db.product?.findUnique({
   349→      where: { id: dto.productId },
   350→      include: {
   351→        inventoryItems: true,
   352→        digitalAssets: true,
   353→        store: true,
   354→        poolProduct: { include: { inventoryItems: true } },
   355→        vendorPoolOffers: {
   356→          where: { isActive: true, status: 'ACTIVE' },
   357→          include: {
   358→            poolProduct: { include: { inventoryItems: true } },
   359→            sourceStore: true,
   360→            sourceProduct: { include: { inventoryItems: true, store: true } },
   361→          },
   362→        },
   363→      },
   364→    });
   365→
   366→    if (!product) {
   367→      throw new NotFoundException('Product not found');
   368→    }
   369→
   370→    let poolOffer: any = null;
   371→    if (dto.poolOfferId) {
   372→      poolOffer = await db.vendorPoolOffer?.findUnique({
   373→        where: { id: dto.poolOfferId },
   374→        include: {
   375→          poolProduct: { include: { inventoryItems: true } },
   376→          store: true,
   377→          sourceStore: true,
   378→          sourceProduct: { include: { inventoryItems: true, store: true } },
   379→        },
   380→      });
   381→      if (!poolOffer || !poolOffer.isActive) {
   382→        throw new BadRequestException('Pool offer is not available');
   383→      }
   384→    }
   385→
   386→    const existing = await db.cartItem?.findFirst({
   387→      where: {
   388→        cartId: cart.id,
   389→        productId: dto.productId,
   390→        variantId: dto.variantId ?? null,
   391→        poolOfferId: dto.poolOfferId ?? null,
   392→      },
   393→    });
   394→    const requestedQuantity = Number(existing?.quantity ?? 0) + Number(dto.quantity ?? 0);
   395→    await this.assertCartLineCanBeAdded({
   396→      product,
   397→      quantity: requestedQuantity,
   398→      variantId: dto.variantId,
   399→      poolOfferId: dto.poolOfferId,
   400→      poolOffer,
   401→    });
   402→
   403→    if (existing) {
   404→      await db.cartItem.update({
   405→        where: { id: existing.id },
   406→        data: { quantity: existing.quantity + dto.quantity },
   407→      });
   408→    } else {
   409→      await db.cartItem?.create({
   410→        data: {
   411→          cartId: cart.id,
   412→          productId: dto.productId,
   413→          variantId: dto.variantId,
   414→          poolOfferId: dto.poolOfferId,
   415→          quantity: dto.quantity,
   416→          price: product.salePrice ?? product.price,
   417→          productType: product.productType ?? 'PHYSICAL',
   418→          productSource: product.productSource ?? 'VENDOR_STOCK',
   419→          requiresShipping: product.requiresShipping ?? product.productType !== 'DIGITAL',
   420→        },
   421→      });
   422→    }
   423→
   424→    return this.getCart(user);
   425→  }

   513→  async clearStoreCart(user: AuthContext, storeSlug: string) {
   514→    const userId = this.getUserId(user);
   515→    const cart = await this.db().cart?.findFirst({
   516→      where: { userId },
   517→      include: this.cartValidationInclude(),
   518→    });

   520→
   521→    const itemIds = (cart.items ?? [])
   522→      .filter((item: any) => this.cartItemMatchesStoreSlug(item, storeSlug))
   523→      .map((item: any) => item.id)
   524→      .filter(Boolean);
   525→
   526→    if (itemIds.length) {
   527→      await this.db().cartItem?.deleteMany({
   528→        where: { cartId: cart.id, id: { in: itemIds } },
   529→      });
   530→    }
   531→
   532→    return this.getCart(user);
   533→  }

   558→  async lookupDeliveryRate(state: string, localGovernment: string) {
   559→    const normalizedState = this.normalizeLocationPart(state ?? '');
   560→    const normalizedLga = this.normalizeLocationPart(localGovernment ?? '');
   561→
   562→    if (!normalizedState || !normalizedLga) {
   563→      throw new BadRequestException('State and local government are required');
   564→    }
   565→
   566→    const rate = await (this.db() as any).deliveryRate?.findFirst({
   567→      where: {
   568→        state: { equals: normalizedState, mode: 'insensitive' },
   569→        localGovernment: { equals: normalizedLga, mode: 'insensitive' },
   570→        isActive: true,
   571→      },
   572→    });
   573→
   574→    if (!rate) {
   575→      throw new NotFoundException('Delivery is not active for this state and local government yet');
   576→    }
   577→
   578→    return {
   579→      ...rate,
   580→      ...this.deliveryEstimate(rate),
   581→      dispatchNote: 'Kwikseller operations will assign manual dispatch after payment.',
   582→    };
   583→  }

   593→    const db = this.db();

   595→    const cartWhere = dto.cartId ? { id: dto.cartId, userId } : { userId };
   596→
   597→    const result = await db.$transaction(async (tx: any) => {
   598→      if (dto.idempotencyKey) {
   599→        const existingCheckout = await tx.parentCheckout.findFirst({
   600→          where: { buyerId: userId, idempotencyKey: dto.idempotencyKey },
   601→          include: {
   602→            payment: true,
   603→            orders: {
   604→              include: { items: true, store: true },
   605→              orderBy: { createdAt: 'asc' },
   606→            },
   607→          },
   608→        });
   609→
   610→        if (existingCheckout?.payment) {
   611→          return {
   612→            parentCheckout: existingCheckout,
   613→            orders: existingCheckout.orders,
   614→            order: existingCheckout.orders[0],
   615→            payment: existingCheckout.payment,
   616→            authorizationUrl: existingCheckout.payment.authorizationUrl,
   617→            reference: existingCheckout.payment.reference,
   618→            requiresShipping: existingCheckout.orders.some((order: any) => Number(order.shippingFee ?? 0) > 0),
   619→          };
   620→        }
   621→
   622→        const existingOrder = await tx.order.findFirst({
   623→          where: { buyerId: userId, idempotencyKey: dto.idempotencyKey },
   624→          include: { payment: true, items: true },
   625→        });
   626→
   627→        if (existingOrder?.payment) {
   628→          return {
   629→            order: existingOrder,
   630→            payment: existingOrder.payment,
   631→            authorizationUrl: existingOrder.payment.authorizationUrl,
   632→            reference: existingOrder.payment.reference,
   633→            requiresShipping: this.withCartTotals({ items: existingOrder.items }).requiresShipping,

   638→      const cart = await tx.cart.findFirst({

   640→        include: this.cartValidationInclude(),

   644→      if (dto.storeSlug) {

   646→          where: { slug: dto.storeSlug },
   647→          select: { id: true },
   648→        });

   650→          throw new BadRequestException('Vendor store not found');
   651→        }
   652→      }
   653→
   654→      const checkoutCart = dto.storeSlug ? this.scopeCartToStore(cart, dto.storeSlug) : cart;
   655→      const checkoutCartItemIds = (checkoutCart?.items ?? [])
   656→        .map((item: any) => item.id)

   660→        throw new BadRequestException(dto.storeSlug ? 'Vendor cart is empty' : 'Cart is empty');

   662→
   663→      const validation = this.buildCartValidation(checkoutCart);
   664→      if (!validation.valid) {
   665→        throw new BadRequestException({
   666→          message: 'Cart validation failed',
   667→          errors: validation.errors,
   668→          warnings: validation.warnings,
   669→        });

   672→      const groupedItems = this.groupCartItemsByStore(checkoutCart.items);
   673→      const groups = [...groupedItems.values()];
   674→      const hasPhysicalItems = groups.some((group) => group.requiresShipping);
   675→      if (hasPhysicalItems && !dto.shippingAddress) {
   676→        throw new BadRequestException('Shipping address is required for physical products');
   677→      }
   678→
   679→      const deliveryQuotes = new Map<string, any>();
   680→      if (hasPhysicalItems) {
   681→        for (const group of groups) {
   682→          if (group.requiresShipping) {
   683→            deliveryQuotes.set(group.storeId, await this.resolveDeliveryQuote(tx, dto.shippingAddress, group));

   686→      }

   692→              line1: dto.shippingAddress.addressLine1,
   693→              line2: dto.shippingAddress.addressLine2,

   736→        const groupDiscount = groupDiscounts.get(group.storeId) ?? 0;

   745→            shippingFee: groupShippingFee,
   746→            totalAmount: groupTotal,
   747→            discount: groupDiscount,
   748→            couponId: coupon?.couponId,
   749→            deliveryRateId: group.requiresShipping ? deliveryQuote?.rate.id : null,

   751→            deliveryLocalGovernment: group.requiresShipping ? deliveryQuote?.location?.lgaName : null,
   752→            estimatedDeliveryStart: group.requiresShipping ? deliveryQuote?.estimatedDeliveryStart : null,

   755→            status: 'PENDING_PAYMENT',
   756→            paymentStatus: 'PENDING',
   757→            checkoutReference: `${reference}-${index + 1}`,

   759→            items: {
   760→              create: group.items.map((item: any) => ({
   761→              productId: item.productId,
   762→              variantId: item.variantId,
   763→              poolOfferId: item.poolOfferId,
   764→              quantity: item.quantity,
   765→              unitPrice: item.price,
   766→              totalPrice: Number(item.price ?? 0) * Number(item.quantity ?? 0),
   767→              isPoolItem: Boolean(item.poolOfferId),
   768→              productType: item.productType ?? item.product?.productType ?? 'PHYSICAL',
   769→                productSource: item.productSource ?? item.product?.productSource ?? 'VENDOR_STOCK',
   770→                sellerStoreId: item.product?.storeId ?? item.poolOffer?.storeId,
   771→                sourceStoreId: item.poolOffer?.sourceStoreId ?? item.product?.poolSourceStoreId ?? group.storeId,
   772→                sourceProductId: item.poolOffer?.sourceProductId ?? item.product?.poolSourceProductId,
   773→                sourceBasePrice: item.poolOffer?.sourceBasePrice ?? item.product?.poolSourceBasePrice,
   774→                resellerMargin: item.poolOfferId
   775→                  ? Math.max(0, Number(item.price ?? 0) - Number(item.poolOffer?.sourceBasePrice ?? item.product?.poolSourceBasePrice ?? 0))
   776→                  : 0,
   777→                platformFeeAmount: 0,
   778→                fulfillmentStatus: 'PENDING',
   779→              })),
   780→            },
   781→          },
   782→          include: { items: true, store: true },
   783→        });
   784→
   785→        for (const orderItem of order.items) {
   786→          const cartItem = group.items.find((item: any) => this.lineItemKey(item) === this.lineItemKey(orderItem));
   787→          await this.reserveInventoryForOrderItem(tx, orderItem, cartItem);
   788→          await this.createPoolSettlementForOrderItem(tx, order, orderItem, cartItem);
   789→        }
   790→
   791→        orders.push(order);
   792→      }
   793→
   794→      const payment = await tx.payment.create({
   795→        data: {
   796→          parentCheckoutId: parentCheckout.id,
   797→          entityType: 'CHECKOUT',
   798→          entityId: parentCheckout.id,
   799→          amount: totalAmount,
   800→          gateway: 'PAYSTACK',
   801→          status: 'PENDING',

   807→        where: dto.storeSlug
   808→          ? { cartId: cart.id, id: { in: checkoutCartItemIds } }
   809→          : { cartId: cart.id },

   811→      await this.writeAudit(tx, {
   812→        userId,
   813→        action: 'checkout.created',
   814→        entity: 'ParentCheckout',
   815→        entityId: parentCheckout.id,
   816→        changes: {
   817→          reference,
   818→          total: totalAmount,
   819→          shippingFee,
   820→          discount,
   821→          couponCode: coupon?.code,
   822→            deliveryRateId: [...deliveryQuotes.values()][0]?.rate.id,
   823→          vendorOrders: orders.map((order: any) => order.id),
   824→          reservedItems: orders.reduce((sum: number, order: any) => sum + order.items.length, 0),
   825→        },
   826→      });
   827→
   828→      return {
   829→        parentCheckout,
   830→        orders,
   831→        order: orders[0],
   832→        payment,
   833→        authorizationUrl: payment.authorizationUrl,
   834→        reference,
   835→        requiresShipping: totals.requiresShipping,
   836→      };
   837→    });
   838→
   839→    try {
   840→      const callbackUrl = this.defaultPaymentCallbackUrl(result.reference);
   841→      const initialized = await this.paystack.initializeTransaction({
   842→        email: buyerEmail,
   843→        amount: result.parentCheckout?.totalAmount ?? result.order.totalAmount,
   844→        reference: result.reference,
   845→        callbackUrl,
   846→        metadata: {
   847→          parentCheckoutId: result.parentCheckout?.id,
   848→          orderIds: result.orders?.map((order: any) => order.id) ?? [result.order.id],
   849→          buyerId: userId,
   850→          source: 'kwikseller_checkout',
   851→        },
   852→      });
   853→
   854→      const payment = await db.payment.update({
   855→        where: { id: result.payment.id },
   856→        data: {
   857→          authorizationUrl: initialized.authorizationUrl,
   858→          gatewayResponse: JSON.stringify({
   859→            initialize: initialized.raw,
   860→            accessCode: initialized.accessCode,
   861→          }),
   862→        },
   863→      });
   864→
   865→      return {
   866→        ...result,
   867→        payment,
   868→        authorizationUrl: initialized.authorizationUrl,
   869→        requiresShipping:
   870→          result.requiresShipping ??
   871→          this.withCartTotals({ items: result.orders?.flatMap((order: any) => order.items) ?? result.order.items })

   937→  async verifyPayment(reference: string) {
   938→    const db = this.db();
   939→    const payment = await db.payment?.findUnique({
   940→      where: { reference },
   941→      include: {
   942→        order: { include: { items: true } },
   943→        parentCheckout: { include: { orders: { include: { items: true, store: true } } } },
   944→      },
   945→    });
   946→
   947→    if (!payment) {
   948→      throw new NotFoundException('Payment reference not found');
   949→    }
   950→
   951→    if (payment.status === 'PENDING') {
   952→      const verification = await this.paystack.verifyTransaction(reference);
   953→      const gatewayStatus = verification?.data?.status;
   954→
   955→      if (gatewayStatus === 'success') {
   956→        return this.processSuccessfulPayment(reference, {
   957→          source: 'payment.verify',
   958→          payload: verification.data,
   959→        });
   960→      }
   961→
   962→      if (['failed', 'abandoned'].includes(gatewayStatus)) {
   963→        await this.failPaymentReference(reference, verification.data);
   964→      }
   965→    }
   966→
   967→    const refreshed = await db.payment?.findUnique({
   968→      where: { reference },
   969→      include: {
   970→        order: { include: { items: true, fulfillments: true } },
   971→        parentCheckout: {
   972→          include: { orders: { include: { items: true, fulfillments: true, store: true } } },
   973→        },
   974→      },
   975→    });
   976→
   977→    return {
   978→      reference,
   979→      status: refreshed?.status ?? payment.status,
   980→      order: refreshed?.order ?? payment.order,
   981→      parentCheckout: refreshed?.parentCheckout ?? payment.parentCheckout,
   982→      orders: refreshed?.parentCheckout?.orders ?? (refreshed?.order ? [refreshed.order] : undefined),
   983→      verifiedAt: refreshed?.verifiedAt ?? payment.verifiedAt,
   984→    };
   985→  }

   987→  async handlePaystackWebhook(signature: string | undefined, body: unknown) {
   988→    const secret = this.config.get<string>('payment.paystackSecret');
   989→    if (!secret) {
   990→      throw new ForbiddenException('Paystack secret is not configured');
   991→    }
   992→
   993→    const payload = JSON.stringify(body);
   994→    const expected = createHmac('sha512', secret).update(payload).digest('hex');
   995→    if (!signature || !this.safeEqual(signature, expected)) {
   996→      throw new ForbiddenException('Invalid Paystack signature');
   997→    }
   998→
   999→    const event = body as any;
  1000→    if (event.event !== 'charge.success') {
  1001→      return { received: true, ignored: true };
  1002→    }
  1003→
  1004→    const reference = event.data?.reference;
  1005→    if (!reference) {
  1006→      throw new BadRequestException('Webhook reference is missing');
  1007→    }
  1008→
  1009→    const idempotencyKey = this.paystackWebhookIdempotencyKey(event);
  1010→    const webhookEvent = await this.recordWebhookEvent(idempotencyKey, event.event, reference, event);
  1011→    if (webhookEvent.idempotent) {
  1012→      return { received: true, reference, idempotent: true };
  1013→    }
  1014→
  1015→    try {
  1016→      const result = await this.processSuccessfulPayment(reference, {
  1017→        source: 'payment.webhook.paystack.charge_success',
  1018→        payload: event.data ?? {},
  1019→      });
  1020→      await this.markWebhookEventProcessed(webhookEvent.id);
  1021→      return { received: true, reference, idempotent: result.idempotent };
  1022→    } catch (error) {
  1023→      await this.markWebhookEventFailed(webhookEvent.id, error);
  1024→      throw error;
  1025→    }
  1026→  }

  1062→  /**
  1063→   * Buyer confirms they received the order. Sets delivery.customerConfirmed,
  1064→   * initiates escrow release (escrow → PENDING_RELEASE), and notifies the
  1065→   * vendor. The actual fund credit happens via the escrow auto-release cron.

  1067→  async confirmDelivery(user: AuthContext, orderId: string) {
  1068→    const userId = this.getUserId(user);
  1069→    const db = this.db();
  1070→
  1071→    const order = await db.order?.findFirst({
  1072→      where: { id: orderId, buyerId: userId },
  1073→      include: { delivery: true, store: { select: { vendorId: true, name: true } } },
  1074→    });
  1075→
  1076→    if (!order) {
  1077→      throw new NotFoundException('Order not found');
  1078→    }
  1079→
  1080→    if (!['SHIPPED', 'DELIVERED', 'FULFILLED'].includes(order.status)) {
  1081→      throw new BadRequestException(
  1082→        `Cannot confirm delivery for an order in ${order.status} status. Order must be shipped, fulfilled, or delivered.`,
  1083→      );
  1084→    }
  1085→
  1086→    if (!order.delivery) {
  1087→      throw new NotFoundException('No delivery record found for this order');
  1088→    }
  1089→
  1090→    if (order.delivery.customerConfirmed) {
  1091→      throw new BadRequestException('Delivery has already been confirmed');
  1092→    }
  1093→
  1094→    // Mark delivery as customer-confirmed
  1095→    await db.delivery?.update({
  1096→      where: { id: order.delivery.id },
  1097→      data: {
  1098→        customerConfirmed: true,
  1099→        customerConfirmedAt: new Date(),
  1100→        status: 'DELIVERED',
  1101→        deliveredAt: order.delivery.deliveredAt ?? new Date(),
  1102→      },
  1103→    });
  1104→
  1105→    // Initiate escrow release (escrow → PENDING_RELEASE)
  1106→    await this.escrowService
  1107→      .initiateRelease(order.delivery.id)
  1108→      .catch((err) => {
  1109→        this.logger.warn(
  1110→          `EscrowService.initiateRelease failed for delivery ${order.delivery!.id}: ${
  1111→            err instanceof Error ? err.message : String(err)
  1112→          }`,
  1113→        );
  1114→      });
  1115→
  1116→    // Notify vendor that the buyer confirmed delivery
  1117→    const vendorId = order.store?.vendorId ?? order.storeId;
  1118→    const orderRef = order.checkoutReference ?? orderId.slice(-8);
  1119→    await this.notificationService
  1120→      .create({
  1121→        userId: vendorId,
  1122→        type: 'DELIVERY_CONFIRMED',
  1123→        title: 'Delivery Confirmed by Buyer',
  1124→        message: `Buyer confirmed delivery for order #${orderRef}. Escrow funds will be released shortly.`,
  1125→        data: { orderId, deliveryId: order.delivery.id },
  1126→      })
  1127→      .catch(() => undefined);
  1128→
  1129→    // Email the buyer a thank-you / confirmation
  1130→    const buyer = await db.user?.findUnique({
  1131→      where: { id: userId },
  1132→      include: { profile: true },
  1133→    });
  1134→    if (buyer?.email) {
  1135→      await this.emailService
  1136→        .sendEmail(
  1137→          buyer.email,
  1138→          `Delivery confirmed — order #${orderRef}`,
  1139→          'order-delivered',
  1140→          {
  1141→            customerName: buyer.profile?.firstName || buyer.email,
  1142→            orderRef,
  1143→            storeName: order.store?.name ?? 'the vendor',
  1144→          },
  1145→        )
  1146→        .catch(() => undefined);
  1147→    }
  1148→
  1149→    return db.order?.findUnique({
  1150→      where: { id: orderId },
  1151→      include: {
  1152→        items: { include: { product: true } },
  1153→        delivery: true,
  1154→        escrow: true,
  1155→      },
  1156→    });
  1157→  }

  1159→  /**
  1160→   * Buyer opens a dispute on an order. Freezes the escrow (escrow → DISPUTED),
  1161→   * records the dispute reason/evidence, and notifies the vendor + admin.
  1162→   */
  1163→  async openDispute(
  1164→    user: AuthContext,
  1165→    orderId: string,
  1166→    reason: string,
  1167→    evidence?: string,
  1168→  ) {
  1169→    const userId = this.getUserId(user);
  1170→    const db = this.db();
  1171→
  1172→    const order = await db.order?.findFirst({
  1173→      where: { id: orderId, buyerId: userId },
  1174→      include: { escrow: true, store: { select: { vendorId: true, name: true } } },
  1175→    });
  1176→
  1177→    if (!order) {
  1178→      throw new NotFoundException('Order not found');
  1179→    }
  1180→
  1181→    if (['CANCELLED', 'REFUNDED'].includes(order.status)) {
  1182→      throw new BadRequestException(
  1183→        `Cannot dispute an order in ${order.status} status`,
  1184→      );
  1185→    }
  1186→
  1187→    if (order.disputeStatus === 'OPENED') {
  1188→      throw new BadRequestException('A dispute is already open for this order');
  1189→    }
  1190→
  1191→    // Freeze escrow (escrow → DISPUTED) and update order dispute fields
  1192→    await this.escrowService.freezeForDispute(orderId, reason).catch((err) => {
  1193→      this.logger.warn(
  1194→        `EscrowService.freezeForDispute failed for order ${orderId}: ${
  1195→          err instanceof Error ? err.message : String(err)
  1196→        }`,
  1197→      );
  1198→      // Still record the dispute on the order even if escrow freeze fails
  1199→    });
  1200→
  1201→    // Ensure order dispute fields are set (freezeForDispute also does this, but
  1202→    // we repeat here in case the escrow call failed)
  1203→    await db.order?.update({
  1204→      where: { id: orderId },
  1205→      data: {
  1206→        disputeStatus: 'OPENED',
  1207→        disputeReason: reason,
  1208→        disputeEvidence: evidence ?? null,
  1209→      },
  1210→    });
  1211→
  1212→    const orderRef = order.checkoutReference ?? orderId.slice(-8);
  1213→    const vendorId = order.store?.vendorId ?? order.storeId;
  1214→
  1215→    // Notify vendor
  1216→    await this.notificationService
  1217→      .create({
  1218→        userId: vendorId,
  1219→        type: 'DISPUTE_OPENED',
  1220→        title: 'A Dispute Has Been Opened',
  1221→        message: `Buyer opened a dispute on order #${orderRef}: ${reason}`,
  1222→        data: { orderId, reason },
  1223→      })
  1224→      .catch(() => undefined);
  1225→
  1226→    // Notify admins (users with ADMIN role) — best-effort
  1227→    await (async () => {
  1228→      try {
  1229→        const admins = await db.user?.findMany({
  1230→          where: { role: 'ADMIN' },
  1231→          select: { id: true, email: true },
  1232→        });
  1233→        if (admins && admins.length > 0) {
  1234→          await Promise.allSettled(
  1235→            admins.map((admin: any) =>
  1236→              this.notificationService.create({
  1237→                userId: admin.id,
  1238→                type: 'DISPUTE_OPENED',
  1239→                title: 'New Dispute Requires Review',
  1240→                message: `Order #${orderRef} dispute: ${reason}`,
  1241→                data: { orderId, reason, vendorId },
  1242→              }),
  1243→            ),
  1244→          );
  1245→        }
  1246→      } catch {
  1247→        // best-effort
  1248→      }
  1249→    })();
  1250→
  1251→    return db.order?.findUnique({
  1252→      where: { id: orderId },
  1253→      include: { escrow: true, delivery: true, store: { select: { name: true } } },
  1254→    });
  1255→  }

  1553→  async updateVendorProduct(user: AuthContext, productId: string, dto: UpdateVendorProductDto) {

  1609→        throw new BadRequestException('Only tracked physical products can be made available in Pool');

  1612→        throw new BadRequestException('Pool base price must be greater than zero');
  1613→      }
  1614→      if (Number(nextPoolMinSalePrice ?? 0) < Number(nextPoolBasePrice ?? 0)) {
  1615→        throw new BadRequestException('Pool minimum sale price cannot be lower than the base price');
  1616→      }
  1617→    }
  1618→    if (dto.status === 'ACTIVE') {
  1619→      this.assertProductCanPublish({
  1620→        ...next,
