    66→function formatDateTime(value?: string) {
    67→  if (!value) return "—";
    68→  return new Intl.DateTimeFormat("en-NG", { dateStyle: "medium", timeStyle: "short" }).format(new Date(value));

    71→function formatNGN(n: number) {
    72→  return new Intl.NumberFormat("en-NG", {
    73→    style: "currency",
    74→    currency: "NGN",
    75→    maximumFractionDigits: 0,
    76→  }).format(n);

   418→// ── RedeemModal ───────────────────────────────────────────────────────────

   420→function RedeemModal({
   421→  open,
   422→  onClose,
   423→  wallet,
   424→  initialType,
   425→}: {
   426→  open: boolean;
   427→  onClose: () => void;
   428→  wallet: WalletData;
   429→  initialType: WalletRedemptionType;
   430→}) {
   431→  const redeem = useRedeemWallet();
   432→  const [redemptionType, setRedemptionType] = React.useState<WalletRedemptionType>(initialType);
   433→  const [amountInput, setAmountInput] = React.useState<string>("");
   434→
   435→  // Sync the selected type when the parent passes a new initial type
   436→  // (e.g. when the user clicks "Ad Credit" vs "Transfer" on the wallet card).
   437→  React.useEffect(() => {
   438→    if (open) setRedemptionType(initialType);
   439→  }, [open, initialType]);
   440→
   441→  // Reset the amount input whenever the modal closes so the next open is clean.
   442→  React.useEffect(() => {
   443→    if (!open) {
   444→      const t = setTimeout(() => setAmountInput(""), 180);
   445→      return () => clearTimeout(t);
   446→    }
   447→    return undefined;
   448→  }, [open]);
   449→
   450→  const amount = Number(amountInput) || 0;
   451→  const nairaValue = amount * 10; // 1 KwikCoin = ₦10
   452→  const exceedsBalance = amount > wallet.balance;
   453→  const canSubmit =
   454→    !redeem.isPending && amount > 0 && !exceedsBalance && amountInput !== "";
   455→
   456→  function setQuick(value: number) {
   457→    setAmountInput(String(value));
   458→  }
   459→
   460→  function setMax() {
   461→    setAmountInput(String(wallet.balance));
   462→  }
   463→
   464→  async function handleConfirm() {
   465→    if (!canSubmit) return;
   466→    try {
   467→      const result = await redeem.mutateAsync({ amount, redemptionType });
   468→      kwikToast.success(
   469→        "Redemption successful",
   470→        `${amount.toLocaleString()} KwikCoins converted. New balance: ${result.newBalance.toLocaleString()} coins.`,
   471→      );
   472→      onClose();
   473→    } catch (e) {
   474→      kwikToast.error(
   475→        "Redemption failed",
   476→        e instanceof Error ? e.message : "Please try again.",
   477→      );
   478→    }
   479→  }
   480→
   481→  return (
   482→    <AnimatePresence>
   483→      {open ? (
   484→        <motion.div
   485→          initial={{ opacity: 0 }}
   486→          animate={{ opacity: 1 }}
   487→          exit={{ opacity: 0 }}
   488→          className="fixed inset-0 z-50 flex items-end justify-center p-0 sm:items-center sm:p-4"
   489→          role="dialog"
   490→          aria-modal="true"
   491→          aria-label="Redeem KwikCoins"
   492→        >
   493→          {/* Overlay */}
   494→          <div
   495→            className="absolute inset-0 bg-black/50 backdrop-blur-sm"
   496→            onClick={() => {
   497→              if (!redeem.isPending) onClose();
   498→            }}
   499→          />
   500→
   501→          {/* Modal card */}
   502→          <motion.div
   503→            initial={{ scale: 0.96, opacity: 0, y: 16 }}
   504→            animate={{ scale: 1, opacity: 1, y: 0 }}
   505→            exit={{ scale: 0.96, opacity: 0, y: 16 }}
   506→            transition={{ type: "spring", duration: 0.32, bounce: 0.18 }}
   507→            className="relative z-10 flex max-h-[92vh] w-full flex-col overflow-hidden rounded-t-3xl border border-kwik-border-light bg-kwik-bg-surface shadow-2xl sm:max-w-lg sm:rounded-3xl"
   508→          >
   509→            {/* Header */}
   510→            <div className="flex items-center justify-between border-b border-kwik-border-light bg-kwik-bg-page px-5 py-4">
   511→              <div className="flex items-center gap-2.5">
   512→                <div className="flex h-9 w-9 items-center justify-center rounded-xl bg-kwik-orange/10">
   513→                  <Coins className="h-5 w-5 text-kwik-orange" />
   514→                </div>
   515→                <div>
   516→                  <h2 className="font-heading text-base font-bold text-foreground">Redeem KwikCoins</h2>
   517→                  <p className="text-xs text-kwik-muted">Convert your coins to cash, ad credit, or transfer.</p>
   518→                </div>
   519→              </div>
   520→              <button
   521→                type="button"
   522→                onClick={() => {
   523→                  if (!redeem.isPending) onClose();
   524→                }}
   525→                aria-label="Close"
   526→                disabled={redeem.isPending}
   527→                className="rounded-lg p-1.5 text-kwik-muted transition hover:bg-kwik-bg-light hover:text-foreground disabled:opacity-50"
   528→              >
   529→                <X className="h-5 w-5" />
   530→              </button>
   531→            </div>
   532→
   533→            {/* Body */}
   534→            <div className="flex-1 overflow-y-auto px-5 py-4">
   535→              {/* Current balance */}
   536→              <div className="flex items-center justify-between rounded-2xl border border-kwik-border-light bg-kwik-bg-page px-4 py-3">
   537→                <div>
   538→                  <p className="text-xs font-semibold uppercase tracking-wide text-kwik-muted">Available balance</p>

   540→                    {wallet.balance.toLocaleString()}{" "}
   541→                    <span className="text-sm font-normal text-kwik-muted">KwikCoins</span>
   542→                  </p>
   543→                </div>
   544→                <div className="text-right">
   545→                  <p className="text-xs font-semibold uppercase tracking-wide text-kwik-muted">Value</p>

   548→                  </p>

   550→              </div>

   571→                      )}
   572→                    >
   573→                      <Icon
   574→                        className={cn(
   575→                          "h-5 w-5",
   576→                          selected ? "text-kwik-orange" : "text-kwik-muted",

   578→                      />

   584→                        </span>
   585→                      ) : null}
   586→                    </button>
   587→                  );
   588→                })}
   589→              </div>
   590→
   591→              {/* Amount input */}
   592→              <div className="mt-4">
   593→                <label
   594→                  htmlFor="redeem-amount"

   600→                  id="redeem-amount"
   601→                  type="number"
   602→                  inputMode="numeric"

   604→                  max={wallet.balance}
   605→                  value={amountInput}
   606→                  onChange={(e) => setAmountInput(e.target.value)}
   607→                  placeholder="Enter amount"
   608→                  className={cn(
   609→                    "h-11 w-full rounded-xl border bg-kwik-bg-page px-3 text-sm font-semibold text-foreground outline-none transition placeholder:font-normal placeholder:text-kwik-muted focus:ring-2",
   610→                    exceedsBalance
   611→                      ? "border-kwik-red focus:ring-kwik-red/20"
   612→                      : "border-kwik-border-light focus:border-kwik-orange focus:ring-kwik-orange/20",
   613→                  )}
   614→                />

   621→                      onClick={() => setQuick(q)}
   622→                      disabled={q > wallet.balance}

   626→                          ? "border-kwik-orange bg-kwik-orange/10 text-kwik-orange"

   630→                      {q}

   634→                    type="button"

   638→                    Max
   639→                  </button>
   640→                </div>
   641→                {exceedsBalance ? (
   642→                  <p className="mt-1.5 text-xs text-kwik-red">
   643→                    Amount exceeds your available balance.
   644→                  </p>
   645→                ) : null}
   646→              </div>
   647→
   648→              {/* Live preview */}
   649→              <div className="mt-4 rounded-2xl bg-kwik-orange/5 p-4">
   650→                <div className="flex items-center justify-between">
   651→                  <div className="flex items-center gap-2">
   652→                    <Sparkles className="h-4 w-4 text-kwik-orange" />
   653→                    <span className="text-sm font-semibold text-foreground">You receive</span>
   654→                  </div>
   655→                  <p className="font-heading text-lg font-bold text-kwik-orange">
   656→                    {formatNGN(nairaValue)}
   657→                  </p>
   658→                </div>
   659→                <p className="mt-1 text-xs text-kwik-muted">
   660→                  1 KwikCoin = ₦10 · Rate locked at {wallet.tier} tier ({wallet.earningRate}× earning).
   661→                </p>
   662→              </div>
   663→            </div>
   664→
   665→            {/* Footer */}
   666→            <div className="flex items-center gap-3 border-t border-kwik-border-light bg-kwik-bg-page px-5 py-4">
   667→              <button
   668→                type="button"
   669→                onClick={() => {
   670→                  if (!redeem.isPending) onClose();
   671→                }}
   672→                disabled={redeem.isPending}
   673→                className="inline-flex h-11 flex-1 items-center justify-center rounded-xl border border-kwik-border-light bg-kwik-bg-surface px-4 text-sm font-semibold text-kwik-dark transition hover:bg-kwik-bg-light disabled:opacity-50 sm:flex-none sm:px-5"

   679→                onClick={handleConfirm}

   682→              >
   683→                {redeem.isPending ? (
   684→                  <>
   685→                    <Loader2 className="h-4 w-4 animate-spin" /> Redeeming…
   686→                  </>
   687→                ) : (
   688→                  <>
   689→                    <Check className="h-4 w-4" /> Confirm redemption
   690→                  </>
   691→                )}
   692→              </button>
   693→            </div>
   694→          </motion.div>
   695→        </motion.div>
   696→      ) : null}
   697→    </AnimatePresence>
   698→  );
   699→}
