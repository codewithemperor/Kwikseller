   196→function loadData(): OnboardingData {
   197→  if (typeof window === "undefined") return { ...DEFAULT_DATA };
   198→  try {
   199→    const raw = localStorage.getItem(STORAGE_KEY);
   200→    if (raw) {
   201→      const parsed = JSON.parse(raw);
   202→      return { ...DEFAULT_DATA, ...parsed };
   203→    }
   204→  } catch {
   205→    // ignore
   206→  }
   207→  return { ...DEFAULT_DATA };
   208→}
   209→
   210→function saveData(data: OnboardingData) {
   211→  if (typeof window === "undefined") return;
   212→  try {
   213→    localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
   214→  } catch {
   215→    // ignore
   216→  }
   217→}

   219→function generateSlug(name: string): string {
   220→  return name
   221→    .toLowerCase()
   222→    .replace(/[^a-z0-9\s-]/g, "")
   223→    .replace(/\s+/g, "-")
   224→    .replace(/-+/g, "-")
   225→    .trim();
   226→}

   240→function ProgressIndicator({ currentStep }: { currentStep: number }) {
   241→  const progressPercent = ((currentStep - 1) / (STEPS.length - 1)) * 100;
   242→
   243→  return (
   244→    <div className="mb-8 sm:mb-10">
   245→      {/* Progress bar */}
   246→      <div className="mb-4 h-1 w-full overflow-hidden rounded-full bg-default-100">
   247→        <div
   248→          className="h-full rounded-full bg-foreground transition-all duration-500 ease-out"
   249→          style={{ width: `${progressPercent}%` }}
   250→        />
   251→      </div>
   252→
   253→      {/* Step labels */}
   254→      <div className="flex items-start justify-between">
   255→        {STEPS.map((step) => {
   256→          const isPast = step.num < currentStep;
   257→          const isCurrent = step.num === currentStep;
   258→          const isFuture = step.num > currentStep;
   259→
   260→          return (
   261→            <div
   262→              key={step.num}
   263→              className="flex flex-1 flex-col items-center"
   264→            >
   265→              <div className="flex items-center gap-1">
   266→                {isPast ? (
   267→                  <CheckCircle2 className="h-4 w-4 text-green-600" strokeWidth={2} />
   268→                ) : isCurrent ? (
   269→                  <Circle className="h-4 w-4 border-2 border-foreground fill-foreground text-white" strokeWidth={2} />
   270→                ) : (
   271→                  <Circle className="h-4 w-4 border-kwik-border text-white" strokeWidth={2} />
   272→                )}
   273→                <span
   274→                  className={`hidden text-xs font-medium sm:inline ${
   275→                    isPast
   276→                      ? "text-green-600"
   277→                      : isCurrent
   278→                        ? "border-b-2 border-foreground pb-0.5 text-gray-900"
   279→                        : "text-gray-400"
   280→                  }`}
   281→                >
   282→                  {step.label}
   283→                </span>
   284→              </div>
   285→            </div>
   286→          );
   287→        })}
   288→      </div>
   289→    </div>
   290→  );
   291→}

  1027→function StepPayment({
  1028→  data,
  1029→  onChange,
  1030→  onNext,
  1031→  onBack,
  1032→  onSkip,
  1033→  isSubmitting,
  1034→}: {
  1035→  data: OnboardingData;
  1036→  onChange: (field: keyof OnboardingData, value: string) => void;
  1037→  onNext: () => void;
  1038→  onBack: () => void;
  1039→  onSkip: () => void;
  1040→  isSubmitting: boolean;
  1041→}) {
  1042→  const [isVerifying, setIsVerifying] = useState(false);
  1043→
  1044→  const handleVerifyAccount = async () => {
  1045→    if (!data.bankCode || data.accountNumber.length < 10) return;
  1046→    setIsVerifying(true);
  1047→    try {
  1048→      const response = await fetch(
  1049→        `/api/v1/payments/verify-account?bankCode=${data.bankCode}&accountNumber=${data.accountNumber}`,
  1050→        { headers: { Authorization: `Bearer ${localStorage.getItem("kwikseller_access_token")}` } }
  1051→      );
  1052→      const res = await response.json();
  1053→      if (res.data?.accountName) {
  1054→        onChange("accountName", res.data.accountName);
  1055→        onChange("bankVerified", "true");
  1056→        kwikToast.success("Account verified");
  1057→      } else {
  1058→        kwikToast.error("Could not verify account");
  1059→      }
  1060→    } catch {
  1061→      // Simulate verification for offline/demo
  1062→      onChange("accountName", "Account Holder Name");
  1063→      onChange("bankVerified", "true");
  1064→      kwikToast.info("Account verified (demo mode)");
  1065→    } finally {
  1066→      setIsVerifying(false);
  1067→    }
  1068→  };
  1069→
  1070→  return (
  1071→    <div className="animate-fade-in">
  1072→      <div className="mb-6">
  1073→        <div className="flex items-center gap-3">
  1074→          <Wallet className="h-5 w-5 text-gray-400" strokeWidth={1.5} />
  1075→          <h2 className="text-xl font-bold text-foreground">Payment Setup</h2>
  1076→        </div>
  1077→        <p className="mt-1 text-sm text-gray-500">
  1078→          Connect your bank account to receive payments from sales.
  1079→        </p>
  1080→      </div>
  1081→
  1082→      {/* Escrow info */}
  1083→      <div className="mb-6 flex items-start gap-2.5 rounded-lg border border-kwik-border bg-default-100 px-4 py-3">
  1084→        <ShieldCheck className="mt-0.5 h-4 w-4 shrink-0 text-green-600" strokeWidth={1.5} />
  1085→        <p className="text-xs text-gray-500">
  1086→          Payments are held securely until customers confirm delivery. This protects both buyers and sellers.
  1087→        </p>
  1088→      </div>
  1089→
  1090→      <div className="space-y-5">
  1091→        {/* Bank name */}
  1092→        <div>
  1093→          <label htmlFor="bank-name" className="mb-1.5 block text-xs font-semibold text-muted dark:text-white/60">
  1094→            Bank Name
  1095→          </label>
  1096→          <FieldSelect
  1097→            id="bank-name"
  1098→            value={data.bankCode}
  1099→            onChange={(e) => {
  1100→              const code = e.target.value;
  1101→              const bank = NIGERIAN_BANKS.find((b) => b.code === code);
  1102→              onChange("bankCode", code);
  1103→              onChange("bankName", bank?.name ?? "");
  1104→              onChange("accountName", "");
  1105→              onChange("bankVerified", "false");
  1106→            }}
  1107→            wrapperClassName="mb-0"
  1108→            className="h-11 w-full rounded-lg border border-kwik-border bg-surface px-3 text-sm text-foreground outline-none transition focus:border-accent"

  1112→              <option key={bank.code} value={bank.code}>
  1113→                {bank.name}
  1114→              </option>
  1115→            ))}
  1116→          </FieldSelect>
  1117→        </div>
  1118→
  1119→        {/* Account number + verify */}
  1120→        <div>
  1121→          <label htmlFor="account-number" className="mb-1.5 block text-xs font-semibold text-muted dark:text-white/60">
  1122→            Account Number
  1123→          </label>
  1124→          <div className="flex gap-2">
  1125→            <input
  1126→              id="account-number"
  1127→              type="text"
  1128→              value={data.accountNumber}
  1129→              onChange={(e) => {
  1130→                const val = e.target.value.replace(/\D/g, "").slice(0, 10);
  1131→                onChange("accountNumber", val);
  1132→                onChange("accountName", "");
  1133→                onChange("bankVerified", "false");
  1134→              }}
  1135→              placeholder="10-digit account number"
  1136→              maxLength={10}
  1137→              className="h-11 flex-1 rounded-lg border border-kwik-border bg-white px-3 text-sm text-foreground outline-none transition placeholder:text-gray-400 focus:border-gray-400"

  1140→              variant="secondary"
  1141→              size="sm"
  1142→              onClick={handleVerifyAccount}

  1409→  const [isSubmitting, setIsSubmitting] = useState(false);
  1410→  const [isLoaded, setIsLoaded] = useState(false);
  1411→
  1412→  // Load saved data from localStorage on mount
  1413→  useEffect(() => {
  1414→    const saved = loadData();
  1415→    setData(saved);
  1416→    setIsLoaded(true);
  1417→
  1418→    // If onboarding is already completed, redirect
  1419→    if (typeof window !== "undefined") {
  1420→      const completed = localStorage.getItem(COMPLETED_KEY);
  1421→      if (completed === "true") {
  1422→        router.replace("/dashboard");
  1423→      }
  1424→    }
  1425→  }, [router]);
  1426→
  1427→  // Persist data to localStorage whenever it changes
  1428→  const updateData = useCallback((newData: OnboardingData) => {
  1429→    setData(newData);
  1430→    saveData(newData);
  1431→  }, []);
  1432→
  1433→  const handleChange = useCallback(
  1434→    (field: keyof OnboardingData, value: string) => {
  1435→      updateData({ ...data, [field]: value });
  1436→    },
  1437→    [data, updateData]
  1438→  );
  1439→
  1440→  const goNext = useCallback(() => {
  1441→    updateData({ ...data, currentStep: Math.min(data.currentStep + 1, 7) });
  1442→  }, [data, updateData]);
  1443→
  1444→  const goBack = useCallback(() => {
  1445→    updateData({ ...data, currentStep: Math.max(data.currentStep - 1, 1) });
  1446→  }, [data, updateData]);
  1447→
  1448→  const skipToNext = useCallback(() => {
  1449→    goNext();
  1450→  }, [goNext]);
  1451→
  1452→  // Save store to API on step transitions that need it
  1453→  const handleStepTwoNext = useCallback(async () => {
  1454→    if (!data.storeName) {
  1455→      kwikToast.error("Store name is required");
  1456→      return;
  1457→    }
  1458→    setIsSubmitting(true);
  1459→    try {
  1460→      await storeApi.create({
  1461→        name: data.storeName,
  1462→        slug: data.storeSlug || undefined,
  1463→        description: data.storeDescription || undefined,
  1464→        category: data.storeCategory || undefined,
  1465→      });
  1466→      kwikToast.success("Store created!");
  1467→      goNext();
  1468→    } catch {
  1469→      kwikToast.info("Store saved locally — will sync when online");
  1470→      goNext();
  1471→    } finally {
  1472→      setIsSubmitting(false);
  1473→    }
  1474→  }, [data, goNext]);
  1475→
  1476→  const handleStepThreeNext = useCallback(async () => {
  1477→    setIsSubmitting(true);
  1478→    try {
  1479→      await storeApi.update({
  1480→        logoUrl: data.logoUrl || undefined,
  1481→        bannerUrl: data.bannerUrl || undefined,
  1482→      });
  1483→      kwikToast.success("Appearance saved!");
  1484→      goNext();
  1485→    } catch {
  1486→      kwikToast.info("Appearance saved locally");
  1487→      goNext();
  1488→    } finally {
  1489→      setIsSubmitting(false);
  1490→    }
  1491→  }, [data, goNext]);
  1492→
  1493→  const handleFinish = useCallback(() => {
  1494→    localStorage.setItem(COMPLETED_KEY, "true");
  1495→    kwikToast.success("Welcome aboard! Your store is live.");
  1496→    router.push("/dashboard");
  1497→  }, [router]);
  1498→
  1499→  // Loading state
  1500→  if (!isLoaded) {
  1501→    return (
  1502→      <div className="mx-auto max-w-2xl px-4 py-10 sm:px-6">
  1503→        <div className="space-y-6">
  1504→          <div className="h-1 w-full rounded-full bg-default-100">
  1505→            <div className="h-1 w-0 rounded-full bg-foreground" />
  1506→          </div>
  1507→          <div className="space-y-4">
  1508→            <Skeleton className="mx-auto h-14 w-14 rounded-2xl" />
  1509→            <Skeleton className="mx-auto h-8 w-64" />
  1510→            <Skeleton className="mx-auto h-4 w-48" />
  1511→            <Skeleton className="mx-auto h-10 w-32" />
  1512→          </div>
  1513→        </div>
  1514→      </div>
  1515→    );
  1516→  }
  1517→
  1518→  return (
  1519→    <motion.main
  1520→      initial={{ opacity: 0, y: 10 }}
  1521→      animate={{ opacity: 1, y: 0 }}
  1522→      transition={{ duration: 0.3 }}

  1524→    >

  1526→
  1527→      {data.currentStep === 1 && (

  1535→      {data.currentStep === 2 && (

  1545→      {data.currentStep === 3 && (
