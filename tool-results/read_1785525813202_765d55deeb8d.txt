    75→// ─── Quote form (vendor sets delivery + discount) ──────────────────────────
    76→
    77→function QuoteForm({ order }: { order: ApiOrder }) {
    78→  const quote = useQuoteOrder();
    79→  const action = useVendorOrderAction();
    80→  const [deliveryFee, setDeliveryFee] = useState(String(order.deliveryFee || 1500));
    81→  const [discount, setDiscount] = useState("");
    82→  const [discountType, setDiscountType] = useState<"AMOUNT" | "PERCENT">("AMOUNT");
    83→
    84→  const discNum = Number(discount) || 0;
    85→  const subtotal = order.subtotal;
    86→  const discAmount =
    87→    discountType === "PERCENT" ? Math.round((subtotal * discNum) / 100) : discNum;
    88→  const fee = Number(deliveryFee) || 0;
    89→  const total = subtotal + fee - discAmount + order.platformFee;
    90→
    91→  async function submitQuote() {
    92→    try {
    93→      await quote.mutateAsync({
    94→        orderId: order.id,
    95→        deliveryFee: fee,
    96→        discount: discNum,
    97→        discountType,
    98→      });
    99→      kwikToast.success("Quotation sent", "The buyer has been notified of your delivery fee & discount.");
   100→    } catch (e) {
   101→      kwikToast.error("Failed to quote", e instanceof Error ? e.message : "Try again.");
   102→    }
   103→  }
   104→
   105→  async function reject() {
   106→    try {
   107→      await action.mutateAsync({ orderId: order.id, action: "reject", reason: "Unable to fulfil at this time" });
   108→      kwikToast.info("Order rejected", "The buyer has been notified.");
   109→    } catch (e) {
   110→      kwikToast.error("Failed to reject", e instanceof Error ? e.message : "Try again.");
   111→    }
   112→  }
   113→
   114→  return (
   115→    <div className="mt-4 rounded-xl border border-kwik-border-light bg-kwik-bg-page p-4">
   116→      <p className="text-sm font-semibold text-foreground">Set delivery &amp; discount</p>
   117→      <div className="mt-3 grid gap-3 sm:grid-cols-2">
   118→        <label className="block">
   119→          <span className="flex items-center gap-1.5 text-xs font-medium text-kwik-muted">
   120→            <Truck className="h-3.5 w-3.5" /> Delivery fee (₦)
   121→          </span>
   122→          <input
   123→            type="number"
   124→            value={deliveryFee}
   125→            onChange={(e) => setDeliveryFee(e.target.value)}
   126→            className="mt-1 h-10 w-full rounded-lg border border-kwik-border-light bg-kwik-bg-surface px-3 text-sm text-foreground outline-none focus:border-kwik-orange focus:ring-2 focus:ring-kwik-orange/20"
   127→            min={0}
   128→          />
   129→        </label>
   130→        <label className="block">
   131→          <span className="flex items-center gap-1.5 text-xs font-medium text-kwik-muted">
   132→            <Tag className="h-3.5 w-3.5" /> Discount
   133→          </span>
   134→          <div className="mt-1 flex gap-2">
   135→            <input
   136→              type="number"
   137→              value={discount}
   138→              onChange={(e) => setDiscount(e.target.value)}
   139→              placeholder="0"
   140→              className="h-10 flex-1 rounded-lg border border-kwik-border-light bg-kwik-bg-surface px-3 text-sm text-foreground outline-none focus:border-kwik-orange focus:ring-2 focus:ring-kwik-orange/20"
   141→              min={0}
   142→            />
   143→            <select
   144→              value={discountType}
   145→              onChange={(e) => setDiscountType(e.target.value as "AMOUNT" | "PERCENT")}
   146→              className="h-10 rounded-lg border border-kwik-border-light bg-kwik-bg-surface px-2 text-sm text-foreground outline-none focus:border-kwik-orange"
   147→            >
   148→              <option value="AMOUNT">₦</option>
   149→              <option value="PERCENT">%</option>
   150→            </select>
   151→          </div>
   152→        </label>
   153→      </div>
   154→      <div className="mt-3 flex items-center justify-between rounded-lg bg-kwik-orange/5 px-3 py-2 text-sm">
   155→        <span className="text-kwik-muted">Buyer pays</span>
   156→        <span className="font-bold text-kwik-orange">{formatNGN(total)}</span>
   157→      </div>
   158→      <div className="mt-3 flex flex-wrap gap-2">
   159→        <button
   160→          type="button"
   161→          onClick={submitQuote}
   162→          disabled={quote.isPending}
   163→          className="inline-flex h-10 items-center justify-center gap-1.5 rounded-lg bg-kwik-orange px-4 text-sm font-semibold text-white transition hover:bg-kwik-orange-hover disabled:opacity-60"
   164→        >
   165→          {quote.isPending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Check className="h-4 w-4" />}
   166→          Send quotation
   167→        </button>
   168→        <button
   169→          type="button"
   170→          onClick={reject}
   171→          disabled={action.isPending}
   172→          className="inline-flex h-10 items-center justify-center gap-1.5 rounded-lg border border-kwik-border-light px-4 text-sm font-semibold text-kwik-muted transition hover:bg-kwik-red/5 hover:text-kwik-red disabled:opacity-60"
   173→        >
   174→          <X className="h-4 w-4" /> Reject
   175→        </button>
   176→      </div>
   177→    </div>
   178→  );
   179→}

   183→function VendorOrderCard({ order, index }: { order: ApiOrder; index: number }) {
   184→  const action = useVendorOrderAction();
   185→  const isPending = order.status === "PENDING";
   186→  const isConfirmed = order.status === "CONFIRMED" || order.status === "PROCESSING";
   187→  const isReady = order.status === "READY";
   188→
   189→  async function doAction(act: "accept" | "ready" | "ship", label: string) {
   190→    try {
   191→      await action.mutateAsync({ orderId: order.id, action: act });
   192→      kwikToast.success(label, "Order updated.");
   193→    } catch (e) {
   194→      kwikToast.error("Action failed", e instanceof Error ? e.message : "Try again.");
   195→    }
   196→  }
   197→
   198→  return (
   199→    <motion.article
   200→      initial={{ opacity: 0, y: 12 }}
   201→      animate={{ opacity: 1, y: 0 }}
   202→      transition={{ duration: 0.35, delay: index * 0.04 }}
   203→      className="rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-5"
   204→    >
   205→      <div className="flex flex-wrap items-start justify-between gap-3">
   206→        <div>
   207→          <div className="flex items-center gap-2">
   208→            <h3 className="font-bold text-foreground">{order.orderNumber}</h3>
   209→            <span
   210→              className={cn(
   211→                "inline-flex items-center rounded-full px-2.5 py-0.5 text-xs font-semibold ring-1",
   212→                STATUS_BADGE[order.status] ?? STATUS_BADGE.PENDING,
   213→              )}
   214→            >
   215→              {order.status.replace(/_/g, " ")}
   216→            </span>
   217→          </div>
   218→          <p className="mt-1 text-sm text-kwik-muted">
   219→            From <span className="font-medium text-foreground">{order.buyerName}</span> ·{" "}
   220→            {new Date(order.createdAt).toLocaleDateString("en-NG", { dateStyle: "medium" })}
   221→          </p>
   222→        </div>
   223→        <div className="text-right">
   224→          <p className="text-xs text-kwik-muted">Subtotal</p>
   225→          <p className="font-bold text-foreground">{formatNGN(order.subtotal)}</p>
   226→        </div>
   227→      </div>
   228→
   229→      {/* Items */}
   230→      <div className="mt-3 space-y-2">
   231→        {order.items.map((item) => (
   232→          <div key={item.id} className="flex items-center justify-between gap-3 text-sm">
   233→            <div className="flex items-center gap-2">
   234→              {item.product.image ? (
   235→                // eslint-disable-next-line @next/next/no-img-element
   236→                <img src={item.product.image} alt={item.product.name} className="h-9 w-9 rounded-md object-cover" />
   237→              ) : (
   238→                <Package className="h-9 w-9 text-kwik-muted" />
   239→              )}
   240→              <span className="font-medium text-foreground">{item.product.name}</span>
   241→              <span className="text-kwik-muted">×{item.quantity}</span>
   242→            </div>
   243→            <span className="text-kwik-muted">{formatNGN(item.totalPrice)}</span>
   244→          </div>
   245→        ))}
   246→      </div>
   247→
   248→      {/* Delivery address */}
   249→      <div className="mt-3 flex items-start gap-2 rounded-lg bg-kwik-bg-page p-3 text-sm">
   250→        <MapPin className="mt-0.5 h-4 w-4 shrink-0 text-kwik-orange" />
   251→        <div className="text-kwik-muted">
   252→          <span className="font-medium text-foreground">{order.deliveryAddress.fullName}</span>
   253→          {" · "}
   254→          <span className="inline-flex items-center gap-1">
   255→            <Phone className="h-3 w-3" /> {order.deliveryAddress.phone}
   256→          </span>
   257→          <br />
   258→          {order.deliveryAddress.addressLine1}, {order.deliveryAddress.city}, {order.deliveryAddress.state}
   259→        </div>
   260→      </div>
   261→
   262→      {/* Quotation summary (when quoted) */}
   263→      {!isPending && (
   264→        <div className="mt-3 grid grid-cols-2 gap-2 text-sm sm:grid-cols-4">
   265→          <div className="rounded-lg bg-kwik-bg-page p-2">
   266→            <p className="flex items-center gap-1 text-xs text-kwik-muted"><Truck className="h-3 w-3" /> Delivery</p>
   267→            <p className="font-semibold text-foreground">{formatNGN(order.deliveryFee)}</p>
   268→          </div>
   269→          {order.discount > 0 && (
   270→            <div className="rounded-lg bg-kwik-bg-page p-2">
   271→              <p className="flex items-center gap-1 text-xs text-kwik-muted"><Tag className="h-3 w-3" /> Discount</p>
   272→              <p className="font-semibold text-kwik-green">−{formatNGN(order.discount)}</p>
   273→            </div>
   274→          )}
   275→          <div className="rounded-lg bg-kwik-bg-page p-2">
   276→            <p className="flex items-center gap-1 text-xs text-kwik-muted"><Banknote className="h-3 w-3" /> Platform fee</p>
   277→            <p className="font-semibold text-foreground">{formatNGN(order.platformFee)}</p>
   278→          </div>
   279→          <div className="rounded-lg bg-kwik-orange/5 p-2">
   280→            <p className="text-xs text-kwik-muted">Total</p>
   281→            <p className="font-bold text-kwik-orange">{formatNGN(order.total)}</p>
   282→          </div>
   283→        </div>
   284→      )}
   285→
   286→      {/* Actions */}
   287→      {isPending && <QuoteForm order={order} />}
   288→      {isConfirmed && (
   289→        <button
   290→          type="button"
   291→          onClick={() => doAction("ready", "Marked ready")}
   292→          disabled={action.isPending}
   293→          className="mt-3 inline-flex h-10 items-center gap-1.5 rounded-lg bg-kwik-blue px-4 text-sm font-semibold text-white transition hover:opacity-90 disabled:opacity-60"
   294→        >
   295→          {action.isPending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Package className="h-4 w-4" />}
   296→          Mark as ready
   297→        </button>
   298→      )}
   299→      {isReady && (
   300→        <button
   301→          type="button"
   302→          onClick={() => doAction("ship", "Order shipped")}
   303→          disabled={action.isPending}
   304→          className="mt-3 inline-flex h-10 items-center gap-1.5 rounded-lg bg-kwik-green px-4 text-sm font-semibold text-white transition hover:opacity-90 disabled:opacity-60"
   305→        >
   306→          {action.isPending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Truck className="h-4 w-4" />}
   307→          Mark as shipped
   308→        </button>
   309→      )}
   310→      {(order.status === "SHIPPED" || order.status === "DELIVERED") && order.trackingNumber && (
   311→        <p className="mt-3 text-xs text-kwik-muted">
   312→          Tracking: <span className="font-mono">{order.trackingNumber}</span>
   313→        </p>
   314→      )}
   315→    </motion.article>
   316→  );
   317→}

   470→                    disabled={reply.isPending || !replyText.trim()}

   477→                    )}
   478→                    Post reply
   479→                  </button>
   480→                </div>
   481→              </div>
   482→            </div>
   483→          </motion.div>
   484→        )}
   485→      </AnimatePresence>
   486→
   487→      {/* Reply / Edit button */}

   494→          {review.vendorReply ? (
   495→            <>
   496→              <MessageCircleReply className="h-3.5 w-3.5" /> Edit reply
   497→            </>
   498→          ) : (
   499→            <>
   500→              <MessageCircleReply className="h-3.5 w-3.5" /> Reply to review
   501→            </>
   502→          )}
   503→        </button>
   504→      )}
   505→    </motion.article>

   511→function VendorReviewsContent({ storeId }: { storeId: string }) {

   526→    star: s,

   543→          </p>

   548→        </div>
   549→        <div className="rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-4">

   551→            <MessageCircleReply className="h-3 w-3 text-kwik-orange" /> Reply rate
   552→          </p>
   553→          <p className="mt-1 font-heading text-2xl font-bold text-foreground">{replyRate}%</p>
   554→          <p className="text-xs text-kwik-muted">{repliedCount} of {total} replied</p>
   555→        </div>
   556→      </div>
   557→
   558→      {/* Rating distribution */}
   559→      {total > 0 && (
   560→        <div className="mt-4 rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-4">
   561→          <p className="text-xs font-semibold uppercase tracking-wider text-kwik-muted">
   562→            Rating distribution
   563→          </p>
   564→          <div className="mt-3 space-y-2">

   566→              <button
   567→                key={b.star}
   568→                type="button"
   569→                onClick={() => setRatingFilter(ratingFilter === b.star ? "ALL" : b.star)}
   570→                className={cn(
   571→                  "flex w-full items-center gap-3 rounded-lg px-2 py-1.5 transition",
   572→                  ratingFilter === b.star ? "bg-kwik-orange-tint" : "hover:bg-kwik-bg-page",
   573→                )}
   574→              >
   575→                <div className="flex w-14 items-center gap-1">
   576→                  <span className="text-xs font-medium text-foreground">{b.star}</span>

   580→                  <div
   581→                    className="h-full rounded-full bg-kwik-gradient"
   582→                    style={{ width: `${total > 0 ? (b.count / total) * 100 : 0}%` }}
   583→                  />

   585→                <span className="w-8 text-right text-xs font-medium text-kwik-muted">
   586→                  {b.count}
   587→                </span>
   588→              </button>
   589→            ))}
   590→          </div>
   591→          {ratingFilter !== "ALL" && (
   592→            <button
   593→              onClick={() => setRatingFilter("ALL")}
   594→              className="mt-2 text-xs font-medium text-kwik-orange hover:underline"
   595→            >
   596→              Clear filter ({ratingFilter}★)
   597→            </button>
   598→          )}
   599→        </div>
   600→      )}
   601→
   602→      {/* Reviews list */}
   603→      <div className="mt-6">
   604→        {isLoading ? (
   605→          <PageLoading label="Loading reviews…" />
   606→        ) : filtered.length === 0 ? (
   607→          <EmptyState
   608→            variant="default"
   609→            icon={<MessageSquare className="h-12 w-12" />}

   612→              ratingFilter !== "ALL"

   622→          </div>
   623→        )}
   624→      </div>
