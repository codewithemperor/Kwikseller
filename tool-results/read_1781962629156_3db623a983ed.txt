    54→function generateId() {
    55→  return `msg_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;

    58→function getRelativeTime(dateStr: string): string {
    59→  const now = new Date();
    60→  const date = new Date(dateStr);
    61→  const diffMs = now.getTime() - date.getTime();
    62→  const diffMins = Math.floor(diffMs / 60000);
    63→  const diffHours = Math.floor(diffMs / 3600000);
    64→  const diffDays = Math.floor(diffMs / 86400000);
    65→
    66→  if (diffMins < 1) return "Just now";
    67→  if (diffMins < 60) return `${diffMins}m ago`;
    68→  if (diffHours < 24) return `${diffHours}h ago`;
    69→  if (diffDays < 7) return `${diffDays}d ago`;
    70→  return formatDate(dateStr);
    71→}
    72→
    73→function getDateSeparator(dateStr: string): string {
    74→  const now = new Date();
    75→  const date = new Date(dateStr);
    76→  const diffDays = Math.floor(
    77→    (now.getTime() - date.getTime()) / 86400000
    78→  );
    79→
    80→  if (diffDays === 0) return "Today";
    81→  if (diffDays === 1) return "Yesterday";
    82→  return formatDate(dateStr);
    83→}
    84→
    85→function formatMessageTime(dateStr: string): string {
    86→  const date = new Date(dateStr);
    87→  return date.toLocaleTimeString("en-NG", {
    88→    hour: "numeric",
    89→    minute: "2-digit",
    90→    hour12: true,
    91→  });
    92→}

   319→function loadConversations(): Conversation[] {
   320→  if (typeof window === "undefined") return [];
   321→  try {
   322→    const raw = localStorage.getItem(MESSAGES_KEY);
   323→    return raw ? JSON.parse(raw) : [];
   324→  } catch {
   325→    return [];
   326→  }
   327→}
   328→
   329→function saveConversations(conversations: Conversation[]) {
   330→  if (typeof window === "undefined") return;
   331→  localStorage.setItem(MESSAGES_KEY, JSON.stringify(conversations));
   332→}
