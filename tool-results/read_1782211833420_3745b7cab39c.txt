    35→const deliveryStages = [
    36→  "CONFIRMED",
    37→  "PROCESSING",
    38→  "FULFILLED",
    39→  "SHIPPED",
    40→  "DELIVERED",
    41→] as const;
    42→
    43→const allStages: string[] = ["PENDING", ...deliveryStages, "CANCELLED"];
    44→
    45→function getStageIndex(status: string): number {
    46→  const idx = allStages.indexOf(status);
    47→  return idx >= 0 ? idx : 0;
    48→}
    49→

    59→
    60→function buildActivityLog(order: Order): Array<{ time: string; label: string }> {
    61→  const log: Array<{ time: string; label: string }> = [];
    62→  if (order.createdAt) {
    63→    log.push({ time: order.createdAt, label: "Order placed" });
    64→  }
    65→  if (order.payment?.paidAt) {
    66→    log.push({ time: order.payment.paidAt, label: "Payment confirmed" });
    67→  }
    68→  if (order.payment?.verifiedAt) {
    69→    log.push({ time: order.payment.verifiedAt, label: "Payment verified" });
    70→  }
    71→  if (order.updatedAt && order.updatedAt !== order.createdAt) {
    72→    log.push({ time: order.updatedAt, label: `Status updated to ${order.status}` });
    73→  }
    74→  if (order.estimatedDeliveryEnd) {
    75→    log.push({ time: order.estimatedDeliveryEnd, label: "Estimated delivery date" });
    76→  }
    77→  if (order.fulfillments?.length) {
    78→    order.fulfillments.forEach((f) => {
    79→      if (f.deliveredAt) {
    80→        log.push({ time: f.deliveredAt, label: "Package delivered" });
    81→      } else if (f.createdAt) {
    82→        log.push({ time: f.createdAt, label: `Fulfillment: ${f.status}` });
    83→      }
    84→    });
    85→  }
    86→  return log.sort((a, b) => new Date(a.time).getTime() - new Date(b.time).getTime());
    87→}

   546→              )}
   547→            </div>
   548→
   549→            {/* Rider info */}
   550→            {order.delivery?.rider && (
   551→              <div className="mt-4 border-t border-kwik-border pt-3 space-y-2">
   552→                <p className="text-xs text-muted-foreground">Assigned Rider</p>
   553→                <p className="text-sm font-medium text-foreground">
   554→                  {order.delivery.rider.vehicleType === "BIKE"
   555→                    ? "Bike"
   556→                    : order.delivery.rider.vehicleType === "CAR"
   557→                    ? "Car"
   558→                    : "Truck"}{" "}
   559→                  - {order.delivery.assignedAt
   560→                    ? formatDate(order.delivery.assignedAt)
   561→                    : ""}
   562→                </p>
   563→              </div>
   564→            )}
   565→
   566→            {order.estimatedDeliveryEnd && (
   567→              <div className="mt-3 border-t border-kwik-border pt-3">
   568→                <p className="text-xs text-muted-foreground">Estimated Delivery</p>
   569→                <p className="text-sm font-medium text-foreground">
   570→                  {formatDate(order.estimatedDeliveryEnd)}
   571→                </p>
   572→              </div>
   573→            )}
   574→
   575→            <p className="mt-4 text-xs text-muted-foreground">
   576→              Delivery tracking powered by Kwikseller Logistics
   577→            </p>
   578→          </section>
   579→
   580→          {/* Order Actions */}
   581→          <section className="border border-kwik-border rounded-lg p-4">
   582→            <h2 className="text-sm font-semibold text-foreground mb-4">
   583→              Order Actions
   584→            </h2>
   585→            <div className="space-y-2">
   586→              {(order.status === "PENDING" || order.status === "PAID") && (
   587→                <>
   588→                  <AppButton
   589→                    type="button"
   590→                    variant="primary"
   591→                    fullWidth
   592→                    onClick={handleAccept}
   593→                    isLoading={updatingId === order.id}
   594→                  >
   595→                    Accept Order
   596→                  </AppButton>
   597→                  <AppButton
   598→                    type="button"
   599→                    variant="secondary"
   600→                    fullWidth
   601→                    onClick={() => setRejectModalOpen(true)}
   602→                    disabled={updatingId === order.id}
   603→                  >
   604→                    Reject Order
   605→                  </AppButton>
   606→                </>
   607→              )}
   608→
   609→              {order.status === "CONFIRMED" && (
   610→                <>
   611→                  <AppButton
   612→                    type="button"
   613→                    variant="primary"
   614→                    fullWidth
   615→                    onClick={handleMarkPreparing}
   616→                    isLoading={updatingId === order.id}
   617→                  >
   618→                    Mark as Preparing
   619→                  </AppButton>
   620→                  <AppButton
   621→                    type="button"
   622→                    variant="secondary"
   623→                    fullWidth
   624→                    onClick={handleCancel}
   625→                    disabled={updatingId === order.id}
   626→                  >
   627→                    Cancel Order
   628→                  </AppButton>
   629→                </>
   630→              )}
   631→
   632→              {order.status === "PROCESSING" && (
   633→                <AppButton
   634→                  type="button"
   635→                  variant="primary"
   636→                  fullWidth
   637→                  onClick={handleMarkReady}
   638→                  isLoading={updatingId === order.id}
   639→                >
   640→                  Mark as Ready for Pickup
   641→                </AppButton>
   642→              )}
   643→
   644→              {order.status === "FULFILLED" && (
   645→                <AppButton
   646→                  type="button"
   647→                  variant="primary"
   648→                  fullWidth
   649→                  onClick={handleConfirmHandoff}
   650→                  isLoading={updatingId === order.id}
   651→                >
   652→                  Confirm Handoff
   653→                </AppButton>
   654→              )}
   655→
   656→              {(order.status === "SHIPPED" ||
   657→                order.status === "DELIVERED" ||
   658→                order.status === "CANCELLED" ||
   659→                order.status === "REFUNDED") && (
   660→                <p className="text-sm text-muted-foreground text-center py-2">
   661→                  No further actions available
   662→                </p>
   663→              )}
   664→            </div>
   665→          </section>
   666→
   667→          {/* Print Invoice */}
   668→          <AppButton
   669→            type="button"
   670→            variant="ghost"
   671→            fullWidth
   672→            onClick={handlePrintInvoice}
   673→          >
   674→            <Printer className="h-4 w-4" />
   675→            Print Invoice
   676→          </AppButton>
   677→        </div>
   678→      </div>
   679→
   680→      {/* Reject Order Modal */}
   681→      <AppModal
   682→        isOpen={rejectModalOpen}
   683→        onClose={() => setRejectModalOpen(false)}
   684→        title="Reject Order"
   685→        description="Please provide a reason for rejecting this order."
   686→        footer={
   687→          <div className="flex justify-end gap-2">
   688→            <AppButton
   689→              type="button"
   690→              variant="secondary"
   691→              onClick={() => setRejectModalOpen(false)}
   692→            >
   693→              Cancel
   694→            </AppButton>
   695→            <AppButton
   696→              type="button"
   697→              variant="danger"
   698→              onClick={handleReject}
   699→              isLoading={updatingId === order.id}
   700→            >
   701→              Reject Order
   702→            </AppButton>
   703→          </div>
   704→        }
   705→      >
   706→        <FieldTextarea
   707→          label="Rejection reason"
   708→          value={rejectReason}
   709→          onChange={(e) => setRejectReason(e.target.value)}
   710→          rows={4}
   711→          placeholder="Explain why you are rejecting this order..."
   712→        />
   713→      </AppModal>
   714→
   715→      {/* Cancel Order Modal */}
   716→      <AppModal
   717→        isOpen={isCancelModalOpen}
   718→        onClose={() => setIsCancelModalOpen(false)}
   719→        title="Cancel Order?"
   720→        description="This action is irreversible. The order will be marked as cancelled and the customer will be notified."
   721→        className="sm:max-w-md"
   722→      >
   723→        <div className="flex items-center justify-end gap-3">
   724→          <AppButton type="button" variant="secondary" onClick={() => setIsCancelModalOpen(false)}>
   725→            Keep order
   726→          </AppButton>
   727→          <AppButton
   728→            type="button"
   729→            variant="danger"
   730→            onClick={confirmCancel}
   731→            isLoading={updatingId === order.id}
   732→          >
   733→            Yes, cancel order
   734→          </AppButton>
   735→        </div>
   736→      </AppModal>
   737→    </motion.div>
   738→  );
   739→}
