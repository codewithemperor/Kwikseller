   150→   150→function getDateParams(range: DateRange, customFrom?: string, customTo?: string) {
   151→   151→  const now = new Date();
   152→   152→  const from = new Date();
   153→   153→  switch (range) {
   154→   154→    case "today":
   155→   155→      from.setHours(0, 0, 0, 0);
   156→   156→      return { from: from.toISOString(), to: now.toISOString() };
   157→   157→    case "7d":
   158→   158→      from.setDate(from.getDate() - 7);
   159→   159→      return { from: from.toISOString(), to: now.toISOString() };
   160→   160→    case "30d":
   161→   161→      from.setDate(from.getDate() - 30);
   162→   162→      return { from: from.toISOString(), to: now.toISOString() };
   163→   163→    case "90d":
   164→   164→      from.setDate(from.getDate() - 90);
   165→   165→      return { from: from.toISOString(), to: now.toISOString() };
   166→   166→    case "year":
   167→   167→      from.setMonth(0, 1);
   168→   168→      from.setHours(0, 0, 0, 0);
   169→   169→      return { from: from.toISOString(), to: now.toISOString() };
   170→   170→    case "custom":
   171→   171→      return {
   172→   172→        from: customFrom ? new Date(customFrom).toISOString() : now.toISOString(),
   173→   173→        to: customTo ? new Date(customTo).toISOString() : now.toISOString(),
   174→   174→      };
   175→   175→  }
   176→   176→}

   219→   219→function mapDashboardToAnalytics(dashboard: VendorDashboardLive): AnalyticsOverview {
   220→   220→  const orders = dashboard.recentOrders ?? [];
   221→   221→  const totalRevenue =
   222→   222→    Number(dashboard.revenue ?? 0) ||
   223→   223→    orders.reduce((sum, order) => sum + Number(order.totalAmount ?? 0), 0);
   224→   224→  const totalOrders = Number(dashboard.ordersCount ?? orders.length);
   225→   225→  const orderBuckets = new Map<string, { revenue: number; orders: number }>();
   226→   226→
   227→   227→  orders.forEach((order) => {
   228→   228→    const key = order.createdAt
   229→   229→      ? new Date(order.createdAt).toISOString().slice(0, 10)
   230→   230→      : new Date().toISOString().slice(0, 10);
   231→   231→    const current = orderBuckets.get(key) ?? { revenue: 0, orders: 0 };
   232→   232→    current.revenue += Number(order.totalAmount ?? 0);
   233→   233→    current.orders += 1;
   234→   234→    orderBuckets.set(key, current);
   235→   235→  });
   236→   236→
   237→   237→  const timeline = [...orderBuckets.entries()]
   238→   238→    .sort(([a], [b]) => a.localeCompare(b))
   239→   239→    .slice(-12);
   240→   240→
   241→   241→  return {
   242→   242→    totalRevenue,
   243→   243→    totalOrders,
   244→   244→    averageOrderValue: totalOrders ? totalRevenue / totalOrders : 0,
   245→   245→    conversionRate: 0,
   246→   246→    returningCustomers: 0,
   247→   247→    topProduct: null,
   248→   248→    revenueOverTime: timeline.map(([date, value]) => ({ date, revenue: value.revenue })),
   249→   249→    ordersOverTime: timeline.map(([date, value]) => ({ date, orders: value.orders })),
   250→   250→    salesByCategory: [],
   251→   251→    trafficSources: [],
   252→   252→    topProducts: [],
   253→   253→    recentOrders: orders.slice(0, 10).map((order) => ({
   254→   254→      id: order.checkoutReference || order.id || "Order",
   255→   255→      customer: order.customer?.name || order.customer?.email || order.user?.name || order.user?.email || "Customer",
   256→   256→      amount: Number(order.totalAmount ?? 0),
   257→   257→      status: order.status || "PENDING",
   258→   258→      date: order.createdAt || new Date().toISOString(),
   259→   259→    })),
   260→   260→  };
   261→   261→}

   726→   726→function BarChart({
   727→   727→  data,
   728→   728→  formatValue,
   729→   729→}: {
   730→   730→  data: { date: string; revenue: number }[];
   731→   731→  formatValue: (n: number) => string;
   732→   732→}) {
   733→   733→  const maxRevenue = React.useMemo(
   734→   734→    () => Math.max(...data.map((d) => d.revenue), 1),
   735→   735→    [data]
   736→   736→  );
   737→   737→
   738→   738→  return (
   739→   739→    <div className="space-y-3">
   740→   740→      {/* Y-axis labels + bars */}
   741→   741→      <div className="flex items-end gap-1" style={{ height: "180px" }}>
   742→   742→        {data.map((item, i) => {
   743→   743→          const height = (item.revenue / maxRevenue) * 100;
   744→   744→          return (
   745→   745→            <div
   746→   746→              key={i}
   747→   747→              className="group relative flex flex-1 flex-col items-center justify-end"
   748→   748→            >
   749→   749→              {/* Tooltip */}

   757→   757→            </div>
