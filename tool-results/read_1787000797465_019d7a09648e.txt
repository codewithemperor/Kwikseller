   209→   209→    }

   239→   239→  // Export the currently-filtered transactions to CSV. Respects all active
   240→   240→  // filters (type, category, date range, search) so the buyer gets exactly
   241→   241→  // what they see on screen. Reuses the shared toCSV/downloadCSV utilities
   242→   242→  // (same as the orders page CSV export).
   243→   243→  //
   244→   244→  // The dummy wallet API doesn't return per-transaction running balances,
   245→   245→  // so we derive them locally from the FILTERED list (using only the
   246→   246→  // transactions the buyer is exporting — otherwise the numbers wouldn't
   247→   247→  // reconcile to `wallet.balance`). Algorithm:
   248→   248→  //   1. Sort the filtered txs oldest-first by `createdAt`.
   249→   249→  //   2. Compute totalDelta = sum(credits) - sum(debits).
   250→   250→  //   3. startingBalance = wallet.balance - totalDelta (the balance before
   251→   251→  //      the oldest filtered tx).
   252→   252→  //   4. Walk forward, applying each tx's delta to a running balance and
   253→   253→  //      recording it in a Map<txId, coinsAfter>.
   254→   254→  //   5. Emit CSV rows in the original `txs` order (matches what's on
   255→   255→  //      screen), looking up each row's "Coins After" from the Map.
   256→   256→  // After the most-recent filtered tx, the running balance equals
   257→   257→  // wallet.balance exactly.
   258→   258→  function handleExportCSV() {
   259→   259→    if (txs.length === 0) {
   260→   260→      kwikToast.error("No transactions to export", "Adjust your filters and try again.");
   261→   261→      return;
   262→   262→    }
   263→   263→    const currentBalance = wallet?.balance ?? 0;
   264→   264→    const totalDelta = txs.reduce(
   265→   265→      (sum, t) => sum + (t.type === "CREDIT" ? t.amount : -t.amount),
   266→   266→      0,
   267→   267→    );
   268→   268→    const coinsAfterById = new Map<string, number>();
   269→   269→    let running = currentBalance - totalDelta;
   270→   270→    // Walk oldest → newest by `createdAt` (the API list order isn't
   271→   271→    // guaranteed to be strictly chronological in dummy mode).
   272→   272→    const chronological = [...txs].sort(
   273→   273→      (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   274→   274→    );
   275→   275→    for (const t of chronological) {
   276→   276→      running += t.type === "CREDIT" ? t.amount : -t.amount;
   277→   277→      coinsAfterById.set(t.id, running);
   278→   278→    }
   279→   279→    const rows = txs.map((t) => ({
   280→   280→      Date: new Date(t.createdAt).toLocaleString("en-NG", { dateStyle: "medium", timeStyle: "short" }),
   281→   281→      Type: t.type === "CREDIT" ? "Earned" : "Spent",
   282→   282→      Category: t.category.replace(/_/g, " ").toLowerCase().replace(/\b\w/g, (c) => c.toUpperCase()),
   283→   283→      Description: t.description,
   284→   284→      Amount: t.type === "CREDIT" ? `+${t.amount}` : `-${t.amount}`,
   285→   285→      "Coins After": coinsAfterById.get(t.id) ?? "",
   286→   286→    }));
   287→   287→    const csv = toCSV(rows);
   288→   288→    const date = new Date().toISOString().slice(0, 10);
   289→   289→    const suffix = hasActiveFilters ? "-filtered" : "";
   290→   290→    downloadCSV(`kwikseller-wallet${suffix}-${date}`, csv);
   291→   291→    kwikToast.success("CSV downloaded", `${txs.length} transaction${txs.length === 1 ? "" : "s"} exported.`);
   292→   292→  }
   293→   293→
   294→   294→  if (isLoading) return <PageLoading label="Loading wallet…" />;
   295→   295→
   296→   296→  if (!wallet) {

   299→   299→        variant="default"

   311→   311→      <Link

   315→   315→        <ArrowLeft className="h-4 w-4" /> Profile
   316→   316→      </Link>
   317→   317→
   318→   318→      <h1 className="mt-3 font-heading text-2xl font-bold text-foreground">KwikCoins Wallet</h1>

   321→   321→      </p>

   395→   395→      {wallet.expiringCoins && wallet.expiringCoins.amount > 0 && (

   846→   846→function ReferralProgramCard({ referral }: { referral: ReferralData }) {

   848→   848→
   849→   849→  function copy(text: string, kind: "code" | "link") {
   850→   850→    if (typeof navigator !== "undefined" && navigator.clipboard) {
   851→   851→      navigator.clipboard.writeText(text).catch(() => {});
   852→   852→    }
   853→   853→    setCopied(kind);
   854→   854→    setTimeout(() => setCopied(null), 2000);
   855→   855→  }
   856→   856→
   857→   857→  const milestoneProgress = Math.min(
   858→   858→    100,
   859→   859→    (referral.successfulReferrals / referral.nextMilestone) * 100,
   860→   860→  );
