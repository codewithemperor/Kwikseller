   190→/* ─── Inline Search Bar (shown in header on /search page) ──── */
   191→
   192→function InlineSearchBar({
   193→  query,
   194→  onSearch,
   195→  onBack,
   196→  showFilters,
   197→  onToggleFilters,
   198→}: {
   199→  query: string;
   200→  onSearch: (q: string) => void;
   201→  onBack: () => void;
   202→  showFilters: boolean;
   203→  onToggleFilters: () => void;
   204→}) {
   205→  const [inputValue, setInputValue] = useState(query);
   206→
   207→  useEffect(() => {
   208→    setInputValue(query);
   209→  }, [query]);
   210→
   211→  const handleSubmit = (e: React.FormEvent) => {
   212→    e.preventDefault();
   213→    const trimmed = inputValue.trim();
   214→    if (trimmed) onSearch(trimmed);
   215→  };
   216→
   217→  return (
   218→    <form onSubmit={handleSubmit} className="flex items-center gap-2 flex-1 min-w-0">
   219→      <button
   220→        type="button"
   221→        onClick={onBack}
   222→        className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl hover:bg-kwik-bg-light transition-colors"
   223→        aria-label="Go back"
   224→      >
   225→        <ArrowLeft className="h-4 w-4 text-kwik-dark-medium" />
   226→      </button>
   227→
   228→      <div className="relative flex-1 min-w-0">
   229→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-kwik-muted" />
   230→        <input
   231→          type="text"
   232→          value={inputValue}
   233→          onChange={(e) => setInputValue(e.target.value)}
   234→          placeholder="Search products, stores, categories..."
   235→          className="w-full h-9 rounded-xl border border-kwik-border bg-kwik-bg-surface pl-9 pr-9 text-sm text-kwik-dark placeholder:text-kwik-muted outline-none focus:border-kwik-orange focus:ring-1 focus:ring-kwik-orange transition-colors"
   236→          autoFocus
   237→        />
   238→        {inputValue && (
   239→          <button
   240→            type="button"
   241→            onClick={() => setInputValue("")}
   242→            className="absolute right-3 top-1/2 -translate-y-1/2 text-kwik-muted hover:text-kwik-dark-medium"
   243→          >
   244→            <X className="h-3.5 w-3.5" />
   245→          </button>
   246→        )}
   247→      </div>
   248→
   249→      <button
   250→        type="button"
   251→        onClick={onToggleFilters}
   252→        className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-xl transition-colors ${
   253→          showFilters ? "bg-kwik-orange-tint text-kwik-orange" : "hover:bg-kwik-bg-light text-kwik-dark-medium"
   254→        }`}
   255→        aria-label="Toggle filters"
   256→      >
   257→        <SlidersHorizontal className="h-4 w-4" />
   258→      </button>
   259→    </form>
   260→  );
   261→}

   307→
   308→export function MarketplaceLayout({ children }: { children: React.ReactNode }) {
   309→  const router = useRouter();
   310→  const pathname = usePathname();
   311→  const searchParams = useSearchParams();
   312→  const { user, isAuthenticated, logout, isLoading: isAuthLoading } = useAuth();
   313→  const [isDrawerOpen, setIsDrawerOpen] = useState(false);
   314→  const [isSearchOpen, setIsSearchOpen] = useState(false);
   315→  const [isAutoSuggestOpen, setIsAutoSuggestOpen] = useState(false);
   316→  const desktopSearchBtnRef = useRef<HTMLButtonElement>(null);
   317→  const [isWishlistOpen, setIsWishlistOpen] = useState(false);
   318→  const [isPageLoading, setIsPageLoading] = useState(true);
   319→  const [isScrolled, setIsScrolled] = useState(false);
   320→  const [showFilters, setShowFilters] = useState(false);
   321→  const isPageLoadingRef = useRef(true);
   322→  const [isClientMounted, setIsClientMounted] = useState(false);
   323→
   324→  const isSearchPage = pathname === "/search";
   325→  const isCartPage = pathname === "/cart";
   326→  const isAuthPage = pathname === "/login" || pathname === "/register" || pathname.startsWith("/forgot-password") || pathname.startsWith("/reset-password");
   327→  const isVendorStorefrontRoute = pathname.startsWith("/vendor/");
   328→  const hideTopNav = isCartPage || isAuthPage || isVendorStorefrontRoute;
   329→  const searchQuery = searchParams.get("q") || "";
   330→  const startNavigationLoading = useCallback(() => {
   331→    isPageLoadingRef.current = true;
   332→    setIsPageLoading(true);
   333→  }, []);
   334→
   335→  useEffect(() => {
   336→    setIsClientMounted(true);
   337→  }, []);
   338→
   339→  useEffect(() => {
   340→    if (!isDrawerOpen) return undefined;
   341→    const previousOverflow = document.body.style.overflow;
   342→    document.body.style.overflow = "hidden";
   343→    return () => {
   344→      document.body.style.overflow = previousOverflow;
   345→    };
   346→  }, [isDrawerOpen]);
   347→
   348→  useEffect(() => {
   349→    const handleScroll = () => {
   350→      setIsScrolled(window.scrollY > 12);
   351→    };
   352→
   353→    window.addEventListener("scroll", handleScroll, { passive: true });
   354→    return () => window.removeEventListener("scroll", handleScroll);
   355→  }, []);
   356→
   357→  useEffect(() => {
   358→    isPageLoadingRef.current = true;
   359→    setIsPageLoading(true);
   360→    const timer = setTimeout(() => {
   361→      isPageLoadingRef.current = false;
   362→      setIsPageLoading(false);
   363→    }, 420);
   364→
   365→    return () => clearTimeout(timer);
   366→  }, [pathname]);
   367→
   368→  useEffect(() => {
   369→    const handleDocumentClick = (event: MouseEvent) => {
   370→      const target = event.target as Element | null;
   371→      const anchor = target?.closest?.("a[href]") as HTMLAnchorElement | null;
   372→      if (!anchor || event.defaultPrevented) return;
   373→      if (anchor.target && anchor.target !== "_self") return;
   374→
   375→      const nextUrl = new URL(anchor.href, window.location.href);
   376→      if (nextUrl.origin !== window.location.origin) return;
   377→      if (nextUrl.pathname === window.location.pathname && nextUrl.search === window.location.search) return;
   378→
   379→      startNavigationLoading();
   380→    };
   381→
   382→    document.addEventListener("click", handleDocumentClick, true);
   383→    return () => document.removeEventListener("click", handleDocumentClick, true);
   384→  }, [startNavigationLoading]);
   385→
   386→  // Reset filters when leaving search page
   387→  useEffect(() => {
   388→    if (!isSearchPage) setShowFilters(false);
   389→  }, [isSearchPage]);
   390→
   391→  const cartItemCount = useCartStore((s) =>
   392→    s.items.length,
   393→  );
   394→  const [prevCartCount, setPrevCartCount] = useState(cartItemCount);
   395→  const [badgeKey, setBadgeKey] = useState(0);
   396→
   397→  // Animate badge when cart count changes
   398→  useEffect(() => {
   399→    if (cartItemCount !== prevCartCount) {
   400→      setBadgeKey((k) => k + 1);
   401→      setPrevCartCount(cartItemCount);
   402→    }
   403→  }, [cartItemCount, prevCartCount]);
   404→  const setCartOpen = useCartStore((s) => s.setCartOpen);
   405→
   406→  const handleLogout = async () => {
   407→    try {
   408→      await logout();
   409→      kwikToast.success("Logged out successfully");
   410→      router.refresh();
   411→    } catch {
   412→      kwikToast.error("Failed to log out");
   413→    }
   414→  };
   415→
   416→  const closeDrawer = useCallback(() => setIsDrawerOpen(false), []);
   417→
   418→  const handleSearchSubmit = useCallback(
   419→    (q: string) => {
   420→      const params = new URLSearchParams();
   421→      params.set("q", q);
   422→      const category = searchParams.get("category");
   423→      if (category) params.set("category", category);
   424→      startNavigationLoading();
   425→      router.push(`/search?${params.toString()}`);
   426→    },
   427→    [router, searchParams, startNavigationLoading],
   428→  );
   429→
   430→  const handleSearchBack = useCallback(() => {
   431→    startNavigationLoading();
   432→    router.push("/");
   433→  }, [router, startNavigationLoading]);
   434→
   435→  const handleToggleFilters = useCallback(() => {
   436→    setShowFilters((prev) => !prev);
   437→    if (isSearchPage) {
   438→      const params = new URLSearchParams(searchParams.toString());
   439→      if (!showFilters) {
   440→        params.set("filters", "true");
   441→      } else {
   442→        params.delete("filters");
   443→      }
   444→      startNavigationLoading();
   445→      router.replace(`/search?${params.toString()}`, { scroll: false });
   446→    }
   447→  }, [isSearchPage, router, searchParams, showFilters, startNavigationLoading]);
   448→
   449→  return (
   450→    <MarketplaceShellProvider
   451→      value={{
   452→        openSearch: () => setIsSearchOpen(true),
   453→        showFilters,
   454→        setShowFilters,
   455→      }}
   456→    >
   457→      <div className="flex min-h-screen flex-col bg-background">
   458→        {!isVendorStorefrontRoute && <PageLoader isLoading={isPageLoading} />}
   459→        <OfflineBanner />
   460→        {!isSearchPage && !isVendorStorefrontRoute && (
   461→          <EnhancedSearchOverlay
   462→            isOpen={isSearchOpen}
   463→            onClose={() => setIsSearchOpen(false)}
   464→          />
   465→        )}
   466→
   467→        {!hideTopNav && (
   468→          <>
   469→            <header
   470→              className={`fixed inset-x-0 top-0 z-[80] bg-background/95 backdrop-blur-md transition-shadow duration-300 ${
   471→                isScrolled
   472→                  ? "shadow-md border-b border-kwik-orange/10"
   473→                  : "border-b border-kwik-border"
   474→              }`}
   475→            >
   476→              <div className="container mx-auto px-0 md:px-4">
   477→                {/* Top row: logo + actions */}
   478→                <div className="flex py-2 md:h-16 items-center justify-between">
   479→                  <div className="flex min-w-0 items-center gap-1 md:gap-2">
   480→                    <Button
   481→                      isIconOnly
   482→                      variant="ghost"
   483→                      className="md:hidden"
   484→                      onPress={() => setIsDrawerOpen(true)}
   485→                      aria-label="Open menu"
   486→                    >
   487→                      <Menu className="h-5 w-5" />
   488→                    </Button>
   489→
   490→                    {/* Logo */}
   491→                    <button
   492→                      type="button"
   493→                      onClick={() => {
   494→                        startNavigationLoading();
   495→                        router.push("/");
   496→                      }}
   497→                      className="flex min-w-0 items-center gap-1.5 transition-opacity hover:opacity-80 md:gap-2"
   498→                    >
   499→                      <div className="y-2">
   500→                        <Image
   501→                          src="/icon.png"
   502→                          alt="KWIKSELLER"
   503→                          width={20}
   504→                          height={20}
   505→                          className="rounded-md md:rounded-lg md:h-8! md:w-8!"
   506→                        />
   507→                      </div>

   509→                        KWIKSELLER
