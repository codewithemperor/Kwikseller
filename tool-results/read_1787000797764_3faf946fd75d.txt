    41→    41→/**
    42→    42→ * Generates a reasonably-unique idempotency key per checkout attempt.
    43→    43→ * The backend uses it to dedupe a ParentCheckout (and its per-vendor
    44→    44→ * Orders) so a network retry returns the original result instead of
    45→    45→ * double-creating orders. We regenerate it whenever the user starts a
    46→    46→ * new checkout submission (not on every render).
    47→    47→ */
    48→    48→function newIdempotencyKey(): string {
    49→    49→  if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
    50→    50→    return `chk_${crypto.randomUUID()}`;
    51→    51→  }
    52→    52→  return `chk_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    53→    53→}
