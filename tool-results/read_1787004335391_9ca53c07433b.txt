     1→"use client";
     2→
     3→import { useState, useEffect, Suspense, useMemo, useCallback } from "react";
     4→import { useSearchParams, useRouter } from "next/navigation";
     5→import dynamic from "next/dynamic";
     6→import { motion, AnimatePresence } from "framer-motion";
     7→import {
     8→  SlidersHorizontal,
     9→  X,
    10→  LayoutGrid,
    11→  List,
    12→  ChevronLeft,
    13→  ChevronRight,
    14→} from "lucide-react";
    15→import { cn } from "@/lib/utils";
    16→import {
    17→  useProducts,
    18→  useCategories,
    19→  useStores,
    20→} from "@/lib/api-hooks";
    21→import { ProductGridSkeleton } from "@/components/ui/loading-state";
    22→import { EmptyState } from "@/components/ui/empty-state";
    23→import { MarketplaceProductCard } from "@/components/landing/shared/marketplace-product-card";
    24→import type { MarketplaceProduct } from "@/data/marketplace-home";
    25→
    26→// Quick-view modal is dynamically imported (client-only) to keep the
    27→// browse bundle small — same pattern as /categories and /search.
    28→const QuickViewModal = dynamic(
    29→  () =>
    30→    import("@/components/landing/quick-view-modal").then(
    31→      (mod) => mod.QuickViewModal,
    32→    ),
    33→  { ssr: false },
    34→);
    35→
    36→function formatNGN(n: number) {
    37→  return new Intl.NumberFormat("en-NG", {
    38→    style: "currency",
    39→    currency: "NGN",
    40→    maximumFractionDigits: 0,
    41→  }).format(n);
    42→}
    43→
    44→type SortOption = "popular" | "newest" | "price-asc" | "price-desc" | "rating";
    45→
    46→const sortOptions: { label: string; value: SortOption }[] = [
    47→  { label: "Most Popular", value: "popular" },
    48→  { label: "Newest First", value: "newest" },
    49→  { label: "Price: Low to High", value: "price-asc" },
    50→  { label: "Price: High to Low", value: "price-desc" },
    51→  { label: "Top Rated", value: "rating" },
    52→];
    53→
    54→const PAGE_SIZE = 12;
    55→
    56→function ProductsBrowseContent() {
    57→  const router = useRouter();
    58→  const searchParams = useSearchParams();
    59→
    60→  const initialQuery = searchParams.get("q") ?? "";
    61→  const initialCategory = searchParams.get("category") ?? "all";
    62→  const initialVendor = searchParams.get("vendor") ?? "all";
    63→  const initialBrandId = searchParams.get("brandId") ?? "";
    64→
    65→  const [query, setQuery] = useState(initialQuery);
    66→  const [category, setCategory] = useState<string>(initialCategory);
    67→  const [vendor, setVendor] = useState<string>(initialVendor);
    68→  const [brandId, setBrandId] = useState<string>(initialBrandId);
    69→  const [sort, setSort] = useState<SortOption>("popular");
    70→  const [view, setView] = useState<"grid" | "list">("grid");
    71→  const [showFilters, setShowFilters] = useState(false);
    72→  const [priceMax, setPriceMax] = useState<number>(50000);
    73→  const [onlyDiscounted, setOnlyDiscounted] = useState(false);
    74→  const [onlyInStock, setOnlyInStock] = useState(false);
    75→  const [page, setPage] = useState(1);
    76→  const [debouncedQuery, setDebouncedQuery] = useState(initialQuery);
    77→
    78→  // Debounce the search query so we don't fire an API call on every keystroke.
    79→  useEffect(() => {
    80→    const t = setTimeout(() => setDebouncedQuery(query), 300);
    81→    return () => clearTimeout(t);
    82→  }, [query]);
    83→
    84→  // Wrap each filter setter so changing any filter resets to page 1. Doing it
    85→  // inline (rather than in a useEffect) avoids the "setState in effect" lint
    86→  // rule and the cascading render it warns about.
    87→  const updateFilter = useCallback(
    88→    <T,>(setter: (v: T) => void) =>
    89→      (v: T) => {
    90→        setter(v);
    91→        setPage(1);
    92→      },
    93→    [],
    94→  );
    95→  // Sync filters to URL (shallow) so the page is shareable/bookmarkable.
    96→  useEffect(() => {
    97→    const params = new URLSearchParams();
    98→    if (query) params.set("q", query);
    99→    if (category !== "all") params.set("category", category);
   100→    if (vendor !== "all") params.set("vendor", vendor);
   101→    if (brandId) params.set("brandId", brandId);
   102→    const qs = params.toString();
   103→    router.replace(qs ? `/products?${qs}` : "/products", { scroll: false });
   104→  }, [query, category, vendor, brandId, router]);
   105→
   106→  const [quickViewProduct, setQuickViewProduct] = useState<MarketplaceProduct | null>(null);
   107→
   108→  // ── Filter option lists from the API ───────────────────────────────────
   109→  const categoriesQuery = useCategories();
   110→  const storesQuery = useStores();
   111→
   112→  const categoryOptions = useMemo(() => {
   113→    const cats = categoriesQuery.data ?? [];
   114→    return [
   115→      { label: "All Categories", value: "all" },
   116→      ...cats.map((c: { name: string; slug: string }) => ({
   117→        label: c.name,
   118→        value: c.slug,
   119→      })),
   120→    ];
   121→  }, [categoriesQuery.data]);
   122→
   123→  const vendorOptions = useMemo(() => {
   124→    const stores = (storesQuery.data ?? []) as Array<{
   125→      name: string;
   126→      slug: string;
   127→    }>;
   128→    return [
   129→      { label: "All Vendors", value: "all" },
   130→      ...stores.map((s) => ({ label: s.name, value: s.slug })),
   131→    ];
   132→  }, [storesQuery.data]);
   133→
   134→  // ── Build API params from local filter state ───────────────────────────
   135→  const sortByParam =
   136→    sort === "popular"
   137→      ? "totalSales"
   138→      : sort === "newest"
   139→        ? "createdAt"
   140→        : sort === "rating"
   141→          ? "rating"
   142→          : "price";
   143→  const sortOrderParam: "asc" | "desc" =
   144→    sort === "price-asc" ? "asc" : "desc";
   145→
   146→  const productsQuery = useProducts({
   147→    search: debouncedQuery.trim() || undefined,
   148→    categoryId: category !== "all" ? category : undefined,
   149→    storeId: vendor !== "all" ? vendor : undefined,
   150→    brandId: brandId || undefined,
   151→    sortBy: sortByParam,
   152→    sortOrder: sortOrderParam,
   153→    page,
   154→    limit: PAGE_SIZE,
   155→  });
   156→
   157→  const apiProducts: MarketplaceProduct[] = productsQuery.data?.products ?? [];
   158→  const meta = productsQuery.data?.meta;
   159→  const totalPages = meta?.totalPages ?? 1;
   160→  const totalCount = meta?.total ?? apiProducts.length;
   161→
   162→  // Apply client-side filters the API doesn't support (price ceiling,
   163→  // "on sale" toggle, "in stock" toggle) to the page's products.
   164→  const filtered = useMemo(() => {
   165→    let list = apiProducts.slice();
   166→    list = list.filter((p) => p.price <= priceMax);
   167→    if (onlyDiscounted) {
   168→      list = list.filter((p) => p.comparePrice && p.comparePrice > p.price);
   169→    }
   170→    if (onlyInStock) {
   171→      list = list.filter((p) => (p.stock ?? 0) > 0);
   172→    }
   173→    return list;
   174→  }, [apiProducts, priceMax, onlyDiscounted, onlyInStock]);
   175→
   176→  const activeFilterCount =
   177→    (category !== "all" ? 1 : 0) +
   178→    (vendor !== "all" ? 1 : 0) +
   179→    (brandId ? 1 : 0) +
   180→    (onlyDiscounted ? 1 : 0) +
   181→    (onlyInStock ? 1 : 0) +
   182→    (priceMax < 50000 ? 1 : 0);
   183→
   184→  function clearFilters() {
   185→    setCategory("all");
   186→    setVendor("all");
   187→    setBrandId("");
   188→    setPriceMax(50000);
   189→    setOnlyDiscounted(false);
   190→    setOnlyInStock(false);
   191→    setPage(1);
   192→  }
   193→
   194→  const handleQuickView = useCallback((p: MarketplaceProduct) => {
   195→    setQuickViewProduct(p);
   196→  }, []);
   197→
   198→  const isLoading = productsQuery.isLoading;
   199→
   200→  return (
   201→    <div className="bg-background min-h-screen">
   202→      {/* ── Page header ── */}
   203→      <section className="kwik-gradient relative overflow-hidden">
   204→        <div className="absolute inset-0 bg-[radial-gradient(circle_at_top_right,rgba(255,255,255,0.18),transparent_55%)]" />
   205→        <div className="container mx-auto max-w-7xl px-4 py-10 md:py-14 relative">
   206→          <motion.div
   207→            initial={{ opacity: 0, y: 12 }}
   208→            animate={{ opacity: 1, y: 0 }}
   209→            transition={{ duration: 0.4 }}
   210→          >
   211→            <p className="text-xs font-semibold uppercase tracking-[0.2em] text-white/75">
   212→              Browse the marketplace
   213→            </p>
   214→            <h1 className="mt-2 font-heading text-3xl font-bold text-white md:text-4xl">
   215→              All Products
   216→            </h1>
   217→            <p className="mt-2 max-w-2xl text-sm leading-6 text-white/85 md:text-base">
   218→              Discover thousands of products from verified vendors across Africa.
   219→              Every purchase is protected by KwisCrow escrow.
   220→            </p>
   221→          </motion.div>
   222→        </div>
   223→      </section>
   224→
   225→      {/* ── Sort + filter bar ── (no in-page search — header search is the
   226→          universal entry point, spec #15. Text filtering still works via
   227→          ?q= URL param; page controls are for sorting & filtering.) */}
   228→      <section className="sticky top-[var(--header-height)] z-30 border-b border-border bg-background/95 backdrop-blur-md">
   229→        <div className="container mx-auto max-w-7xl px-4 py-3">
   230→          <div className="flex flex-col gap-3 md:flex-row md:items-center">
   231→            {/* Result count */}
   232→            <div className="flex-1">
   233→              <p className="text-sm text-kwik-gray-light dark:text-white/70">
   234→                {productsQuery.isLoading ? (
   235→                  <span className="text-kwik-muted">Loading…</span>
   236→                ) : (
   237→                  <>
   238→                    <span className="font-semibold text-kwik-dark dark:text-white">{totalCount}</span>{" "}
   239→                    product{totalCount !== 1 ? "s" : ""}
   240→                    {query ? <> · filtering by &ldquo;<span className="font-semibold text-kwik-orange">{query}</span>&rdquo;</> : null}
   241→                  </>
   242→                )}
   243→              </p>
   244→            </div>
   245→
   246→            {/* Sort */}
   247→            <div className="flex items-center gap-2">
   248→              <div className="relative">
   249→                <select
   250→                  value={sort}
   251→                  onChange={(e) => {
   252→                    setSort(e.target.value as SortOption);
   253→                    setPage(1);
   254→                  }}
   255→                  aria-label="Sort products"
   256→                  className="h-11 appearance-none rounded-xl border border-border bg-surface pl-9 pr-8 text-sm font-medium text-foreground focus:border-kwik-orange focus:outline-none focus:ring-2 focus:ring-kwik-orange/20"
   257→                >
   258→                  {sortOptions.map((opt) => (
   259→                    <option key={opt.value} value={opt.value}>
   260→                      {opt.label}
   261→                    </option>
   262→                  ))}
   263→                </select>
   264→                <SlidersHorizontal className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-kwik-muted" />
   265→              </div>
   266→
   267→              {/* Filter toggle (mobile) */}
   268→              <button
   269→                type="button"
   270→                onClick={() => setShowFilters((v) => !v)}
   271→                className="relative inline-flex h-11 items-center gap-2 rounded-xl border border-border bg-surface px-3 text-sm font-medium text-foreground hover:bg-muted md:hidden"
   272→                aria-label="Toggle filters"
   273→              >
   274→                <SlidersHorizontal className="h-4 w-4" />
   275→                Filters
   276→                {activeFilterCount > 0 ? (
   277→                  <span className="absolute -right-1 -top-1 flex h-5 min-w-5 items-center justify-center rounded-full bg-kwik-orange px-1 text-[10px] font-bold text-white">
   278→                    {activeFilterCount}
   279→                  </span>
   280→                ) : null}
   281→              </button>
   282→
   283→              {/* View toggle */}
   284→              <div className="hidden items-center rounded-xl border border-border bg-surface p-1 md:flex">
   285→                <button
   286→                  type="button"
   287→                  onClick={() => setView("grid")}
   288→                  aria-label="Grid view"
   289→                  aria-pressed={view === "grid"}
   290→                  className={cn(
   291→                    "rounded-lg p-2 transition",
   292→                    view === "grid"
   293→                      ? "bg-kwik-orange text-white"
   294→                      : "text-kwik-muted hover:text-foreground",
   295→                  )}
   296→                >
   297→                  <LayoutGrid className="h-4 w-4" />
   298→                </button>
   299→                <button
   300→                  type="button"
   301→                  onClick={() => setView("list")}
   302→                  aria-label="List view"
   303→                  aria-pressed={view === "list"}
   304→                  className={cn(
   305→                    "rounded-lg p-2 transition",
   306→                    view === "list"
   307→                      ? "bg-kwik-orange text-white"
   308→                      : "text-kwik-muted hover:text-foreground",
   309→                  )}
   310→                >
   311→                  <List className="h-4 w-4" />
   312→                </button>
   313→              </div>
   314→            </div>
   315→          </div>
   316→        </div>
   317→      </section>
   318→
   319→      {/* ── Body: sidebar filters + product grid ── */}
   320→      <section className="container mx-auto max-w-7xl px-4 py-6">
   321→        <div className="flex gap-6">
   322→          {/* Sidebar filters (desktop) */}
   323→          <aside className="hidden w-64 shrink-0 md:block">
   324→            <FilterPanel
   325→              category={category}
   326→              setCategory={updateFilter(setCategory)}
   327→              vendor={vendor}
   328→              setVendor={updateFilter(setVendor)}
   329→              priceMax={priceMax}
   330→              setPriceMax={updateFilter(setPriceMax)}
   331→              onlyDiscounted={onlyDiscounted}
   332→              setOnlyDiscounted={updateFilter(setOnlyDiscounted)}
   333→              onlyInStock={onlyInStock}
   334→              setOnlyInStock={updateFilter(setOnlyInStock)}
   335→              activeFilterCount={activeFilterCount}
   336→              onClear={clearFilters}
   337→              categoryOptions={categoryOptions}
   338→              vendorOptions={vendorOptions}
   339→            />
   340→          </aside>
   341→
   342→          {/* Mobile filter drawer */}
   343→          <AnimatePresence>
   344→            {showFilters ? (
   345→              <motion.div
   346→                initial={{ opacity: 0 }}
   347→                animate={{ opacity: 1 }}
   348→                exit={{ opacity: 0 }}
   349→                className="fixed inset-0 z-50 md:hidden"
   350→              >
   351→                <div
   352→                  className="absolute inset-0 bg-black/40 backdrop-blur-sm"
   353→                  onClick={() => setShowFilters(false)}
   354→                />
   355→                <motion.div
   356→                  initial={{ x: "-100%" }}
   357→                  animate={{ x: 0 }}
   358→                  exit={{ x: "-100%" }}
   359→                  transition={{ type: "spring", damping: 28, stiffness: 280 }}
   360→                  className="absolute left-0 top-0 h-full w-80 max-w-[85vw] overflow-y-auto bg-background p-4 shadow-2xl"
   361→                >
   362→                  <div className="mb-4 flex items-center justify-between">
   363→                    <h2 className="font-heading text-lg font-semibold text-foreground">
   364→                      Filters
   365→                    </h2>
   366→                    <button
   367→                      type="button"
   368→                      onClick={() => setShowFilters(false)}
   369→                      aria-label="Close filters"
   370→                      className="rounded-lg p-1.5 text-kwik-muted hover:bg-muted hover:text-foreground"
   371→                    >
   372→                      <X className="h-5 w-5" />
   373→                    </button>
   374→                  </div>
   375→                  <FilterPanel
   376→                    category={category}
   377→                    setCategory={updateFilter(setCategory)}
   378→                    vendor={vendor}
   379→                    setVendor={updateFilter(setVendor)}
   380→                    priceMax={priceMax}
   381→                    setPriceMax={updateFilter(setPriceMax)}
   382→                    onlyDiscounted={onlyDiscounted}
   383→                    setOnlyDiscounted={updateFilter(setOnlyDiscounted)}
   384→                    onlyInStock={onlyInStock}
   385→                    setOnlyInStock={updateFilter(setOnlyInStock)}
   386→                    activeFilterCount={activeFilterCount}
   387→                    onClear={clearFilters}
   388→                    categoryOptions={categoryOptions}
   389→                    vendorOptions={vendorOptions}
   390→                  />
   391→                </motion.div>
   392→              </motion.div>
   393→            ) : null}
   394→          </AnimatePresence>
   395→
   396→          {/* Results */}
   397→          <div className="min-w-0 flex-1">
   398→            {/* Result count + active chips */}
   399→            <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   400→              <p className="text-sm text-kwik-muted">
   401→                Showing{" "}
   402→                <span className="font-semibold text-foreground">
   403→                  {filtered.length}
   404→                </span>{" "}
   405→                {filtered.length !== totalCount ? (
   406→                  <>
   407→                    of{" "}
   408→                    <span className="font-semibold text-foreground">
   409→                      {totalCount}
   410→                    </span>{" "}
   411→                  </>
   412→                ) : null}
   413→                products
   414→                {totalPages > 1 ? (
   415→                  <>
   416→                    {" "}
   417→                    · Page{" "}
   418→                    <span className="font-semibold text-foreground">{page}</span>{" "}
   419→                    of {totalPages}
   420→                  </>
   421→                ) : null}
   422→              </p>
   423→              {activeFilterCount > 0 ? (
   424→                <button
   425→                  type="button"
   426→                  onClick={clearFilters}
   427→                  className="inline-flex items-center gap-1 text-xs font-medium text-kwik-orange hover:opacity-80"
   428→                >
   429→                  <X className="h-3.5 w-3.5" /> Clear all filters
   430→                </button>
   431→              ) : null}
   432→            </div>
   433→
   434→            {/* Active filter chips */}
   435→            {activeFilterCount > 0 ? (
   436→              <div className="mb-4 flex flex-wrap gap-2">
   437→                {category !== "all" ? (
   438→                  <FilterChip
   439→                    label={
   440→                      categoryOptions.find((c) => c.value === category)?.label ??
   441→                      category
   442→                    }
   443→                    onRemove={() => {
   444→                      setCategory("all");
   445→                      setPage(1);
   446→                    }}
   447→                  />
   448→                ) : null}
   449→                {vendor !== "all" ? (
   450→                  <FilterChip
   451→                    label={
   452→                      vendorOptions.find((v) => v.value === vendor)?.label ??
   453→                      vendor
   454→                    }
   455→                    onRemove={() => {
   456→                      setVendor("all");
   457→                      setPage(1);
   458→                    }}
   459→                  />
   460→                ) : null}
   461→                {brandId ? (
   462→                  <FilterChip
   463→                    label="Brand filter"
   464→                    onRemove={() => {
   465→                      setBrandId("");
   466→                      setPage(1);
   467→                    }}
   468→                  />
   469→                ) : null}
   470→                {onlyDiscounted ? (
   471→                  <FilterChip label="On sale" onRemove={() => {
   472→                    setOnlyDiscounted(false);
   473→                    setPage(1);
   474→                  }} />
   475→                ) : null}
   476→                {onlyInStock ? (
   477→                  <FilterChip label="In stock" onRemove={() => {
   478→                    setOnlyInStock(false);
   479→                    setPage(1);
   480→                  }} />
   481→                ) : null}
   482→                {priceMax < 50000 ? (
   483→                  <FilterChip label={`Under ${formatNGN(priceMax)}`} onRemove={() => {
   484→                    setPriceMax(50000);
   485→                    setPage(1);
   486→                  }} />
   487→                ) : null}
   488→              </div>
   489→            ) : null}
   490→
   491→            {/* Loading skeleton */}
   492→            {isLoading ? (
   493→              <ProductGridSkeleton count={8} columns={view === "list" ? 2 : 4} />
   494→            ) : filtered.length === 0 ? (
   495→              <EmptyState
   496→                variant="search"
   497→                title="No products found"
   498→                description="Try adjusting your search or filters to find what you're looking for."
   499→                action={
   500→                  <button
   501→                    type="button"
   502→                    onClick={() => {
   503→                      setQuery("");
   504→                      clearFilters();
   505→                    }}
   506→                    className="inline-flex h-10 items-center justify-center rounded-xl bg-kwik-orange px-5 text-sm font-semibold text-white hover:bg-kwik-orange-hover"
   507→                  >
   508→                    Reset all
   509→                  </button>
   510→                }
   511→              />
   512→            ) : view === "grid" ? (
   513→              <div className="grid grid-cols-2 gap-4 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 2xl:grid-cols-6">
   514→                {filtered.map((product, idx) => (
   515→                  <motion.div
   516→                    key={product.id}
   517→                    initial={{ opacity: 0, y: 16 }}
   518→                    animate={{ opacity: 1, y: 0 }}
   519→                    transition={{ duration: 0.3, delay: Math.min(idx * 0.03, 0.3) }}
   520→                  >
   521→                    <MarketplaceProductCard
   522→                      product={product}
   523→                      onQuickView={() => handleQuickView(product)}
   524→                    />
   525→                  </motion.div>
   526→                ))}
   527→              </div>
   528→            ) : (
   529→              // List view: same MarketplaceProductCard (no per-page card
   530→              // variants) in a denser 3-column grid so the grid/list toggle
   531→              // still changes the browsing density without inventing a
   532→              // second card component.
   533→              <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   534→                {filtered.map((product, idx) => (
   535→                  <motion.div
   536→                    key={product.id}
   537→                    initial={{ opacity: 0, x: -8 }}
   538→                    animate={{ opacity: 1, x: 0 }}
   539→                    transition={{ duration: 0.25, delay: Math.min(idx * 0.02, 0.2) }}
   540→                  >
   541→                    <MarketplaceProductCard
   542→                      product={product}
   543→                      onQuickView={() => handleQuickView(product)}
   544→                    />
   545→                  </motion.div>
   546→                ))}
   547→              </div>
   548→            )}
   549→
   550→            {/* Pagination */}
   551→            {totalPages > 1 && !isLoading ? (
   552→              <div className="mt-8 flex items-center justify-center gap-2">
   553→                <button
   554→                  type="button"
   555→                  onClick={() => setPage((p) => Math.max(1, p - 1))}
   556→                  disabled={page <= 1}
   557→                  className="inline-flex h-9 items-center gap-1 rounded-lg border border-border bg-surface px-3 text-sm font-medium text-foreground transition hover:bg-muted disabled:cursor-not-allowed disabled:opacity-40"
   558→                  aria-label="Previous page"
   559→                >
   560→                  <ChevronLeft className="h-4 w-4" />
   561→                  Prev
   562→                </button>
   563→                {Array.from({ length: totalPages }).map((_, i) => {
   564→                  const pageNum = i + 1;
   565→                  const isCurrent = pageNum === page;
   566→                  return (
   567→                    <button
   568→                      key={pageNum}
   569→                      type="button"
   570→                      onClick={() => setPage(pageNum)}
   571→                      className={cn(
   572→                        "inline-flex h-9 min-w-9 items-center justify-center rounded-lg border px-3 text-sm font-medium transition",
   573→                        isCurrent
   574→                          ? "border-kwik-orange bg-kwik-orange text-white"
   575→                          : "border-border bg-surface text-foreground hover:bg-muted",
   576→                      )}
   577→                      aria-current={isCurrent ? "page" : undefined}
   578→                    >
   579→                      {pageNum}
   580→                    </button>
   581→                  );
   582→                })}
   583→                <button
   584→                  type="button"
   585→                  onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   586→                  disabled={page >= totalPages}
   587→                  className="inline-flex h-9 items-center gap-1 rounded-lg border border-border bg-surface px-3 text-sm font-medium text-foreground transition hover:bg-muted disabled:cursor-not-allowed disabled:opacity-40"
   588→                  aria-label="Next page"
   589→                >
   590→                  Next
   591→                  <ChevronRight className="h-4 w-4" />
   592→                </button>
   593→              </div>
   594→            ) : null}
   595→          </div>
   596→        </div>
   597→      </section>
   598→
   599→      {/* Quick view modal — opened when a product card is clicked. */}
   600→      <QuickViewModal
   601→        product={quickViewProduct}
   602→        isOpen={!!quickViewProduct}
   603→        onClose={() => setQuickViewProduct(null)}
   604→      />
   605→    </div>
   606→  );
   607→}
