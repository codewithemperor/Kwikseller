   150→function getDateParams(range: DateRange, customFrom?: string, customTo?: string) {
   151→  const now = new Date();
   152→  const from = new Date();
   153→  switch (range) {
   154→    case "today":
   155→      from.setHours(0, 0, 0, 0);
   156→      return { from: from.toISOString(), to: now.toISOString() };
   157→    case "7d":
   158→      from.setDate(from.getDate() - 7);
   159→      return { from: from.toISOString(), to: now.toISOString() };
   160→    case "30d":
   161→      from.setDate(from.getDate() - 30);
   162→      return { from: from.toISOString(), to: now.toISOString() };
   163→    case "90d":
   164→      from.setDate(from.getDate() - 90);
   165→      return { from: from.toISOString(), to: now.toISOString() };
   166→    case "year":
   167→      from.setMonth(0, 1);
   168→      from.setHours(0, 0, 0, 0);
   169→      return { from: from.toISOString(), to: now.toISOString() };
   170→    case "custom":
   171→      return {
   172→        from: customFrom ? new Date(customFrom).toISOString() : now.toISOString(),
   173→        to: customTo ? new Date(customTo).toISOString() : now.toISOString(),
   174→      };
   175→  }
   176→}

   219→function mapDashboardToAnalytics(dashboard: VendorDashboardLive): AnalyticsOverview {
   220→  const orders = dashboard.recentOrders ?? [];
   221→  const totalRevenue =
   222→    Number(dashboard.revenue ?? 0) ||
   223→    orders.reduce((sum, order) => sum + Number(order.totalAmount ?? 0), 0);
   224→  const totalOrders = Number(dashboard.ordersCount ?? orders.length);
   225→  const orderBuckets = new Map<string, { revenue: number; orders: number }>();
   226→
   227→  orders.forEach((order) => {
   228→    const key = order.createdAt
   229→      ? new Date(order.createdAt).toISOString().slice(0, 10)
   230→      : new Date().toISOString().slice(0, 10);
   231→    const current = orderBuckets.get(key) ?? { revenue: 0, orders: 0 };
   232→    current.revenue += Number(order.totalAmount ?? 0);
   233→    current.orders += 1;
   234→    orderBuckets.set(key, current);
   235→  });
   236→
   237→  const timeline = [...orderBuckets.entries()]
   238→    .sort(([a], [b]) => a.localeCompare(b))
   239→    .slice(-12);
   240→
   241→  return {
   242→    totalRevenue,
   243→    totalOrders,
   244→    averageOrderValue: totalOrders ? totalRevenue / totalOrders : 0,
   245→    conversionRate: 0,
   246→    returningCustomers: 0,
   247→    topProduct: null,
   248→    revenueOverTime: timeline.map(([date, value]) => ({ date, revenue: value.revenue })),
   249→    ordersOverTime: timeline.map(([date, value]) => ({ date, orders: value.orders })),
   250→    salesByCategory: [],
   251→    trafficSources: [],
   252→    topProducts: [],
   253→    recentOrders: orders.slice(0, 10).map((order) => ({
   254→      id: order.checkoutReference || order.id || "Order",
   255→      customer: order.customer?.name || order.customer?.email || order.user?.name || order.user?.email || "Customer",
   256→      amount: Number(order.totalAmount ?? 0),
   257→      status: order.status || "PENDING",
   258→      date: order.createdAt || new Date().toISOString(),
   259→    })),
   260→  };
   261→}

   726→function BarChart({
   727→  data,
   728→  formatValue,
   729→}: {
   730→  data: { date: string; revenue: number }[];
   731→  formatValue: (n: number) => string;
   732→}) {
   733→  const maxRevenue = React.useMemo(
   734→    () => Math.max(...data.map((d) => d.revenue), 1),
   735→    [data]
   736→  );
   737→
   738→  return (
   739→    <div className="space-y-3">
   740→      {/* Y-axis labels + bars */}
   741→      <div className="flex items-end gap-1" style={{ height: "180px" }}>
   742→        {data.map((item, i) => {
   743→          const height = (item.revenue / maxRevenue) * 100;
   744→          return (
   745→            <div
   746→              key={i}
   747→              className="group relative flex flex-1 flex-col items-center justify-end"
   748→            >
   749→              {/* Tooltip */}
   750→              <div className="absolute bottom-full mb-2 hidden rounded border border-gray-200 bg-white px-2 py-1 text-xs text-gray-700 group-hover:block">
   751→                {formatShortDate(item.date)}
   752→                <br />
   753→                <span className="font-mono">{formatValue(item.revenue)}</span>
   754→              </div>
   755→              {/* Bar */}
   756→              <div className="w-full max-w-[48px] rounded-t bg-gray-900 transition-all duration-500" style={{ height: `${height}%` }} />
   757→            </div>
   758→          );
   759→        })}
   760→      </div>
   761→      {/* X-axis labels */}
   762→      <div className="flex gap-1">
   763→        {data.map((item, i) => (
   764→          <div key={i} className="flex-1 text-center">
   765→            <span className="text-[10px] text-gray-400">{formatShortDate(item.date)}</span>
   766→          </div>
   767→        ))}
   768→      </div>
   769→    </div>
   770→  );
   771→}
