    65→function resolveCategoryStyle(
    66→  name: string,
    67→  slug: string,
    68→  index: number,
    69→): CategoryStyle {
    70→  const key = (slug || "").toLowerCase();
    71→  if (CATEGORY_STYLES[key]) return CATEGORY_STYLES[key];
    72→  for (const [k, v] of Object.entries(CATEGORY_STYLES)) {
    73→    if (name.toLowerCase().includes(k) || key.includes(k)) return v;
    74→  }
    75→  const colorIdx = index % CATEGORY_CARD_ACCENT_COLORS.length;
    76→  return {
    77→    color: CATEGORY_CARD_ACCENT_COLORS[colorIdx],
    78→    textColor: CATEGORY_CARD_ACCENT_COLORS[colorIdx],
    79→    Icon: DEFAULT_CATEGORY_STYLE.Icon,
    80→  };
    81→}

    91→/**
    92→ * Normalize the category response shape.
    93→ *
    94→ * The dummy route returns a flat `{ ...category, products }` while the real
    95→ * NestJS backend returns `{ category, products }`. This helper accepts either
    96→ * and returns the category object (without products — those come from the
    97→ * dedicated products query).
    98→ */
    99→function normalizeCategory(raw: unknown): Category | null {
   100→  if (!raw || typeof raw !== "object") return null;
   101→  const obj = raw as Record<string, unknown>;
   102→  // Real backend shape: { category: {...}, products: [...] }
   103→  if (obj.category && typeof obj.category === "object") {
   104→    return obj.category as Category;
   105→  }
   106→  // Dummy shape: { ...categoryFields, products: [...] }
   107→  if ("id" in obj || "name" in obj) {
   108→    return obj as unknown as Category;
   109→  }
   110→  return null;
   111→}

   374→// ─── Main page component ────────────────────────────────────────────────────

   376→export default function CategoryDetailPage() {
   377→  const params = useParams<{ id: string }>();
   378→  const router = useRouter();
   379→  const idParam = params?.id ?? "";
   380→
   381→  // Category metadata (param is treated as a slug — CategoryCard links to
   382→  // /categories/[slug], falling back to id when no slug exists).
   383→  const categoryQuery = useCategoryBySlug(idParam);
   384→  const rawCategoryData = categoryQuery.data;
   385→  const category = useMemo(
   386→    () => normalizeCategory(rawCategoryData),
   387→    [rawCategoryData],
   388→  );
   389→
   390→  // Products — fetch a generous batch for the resolved category, then filter,
   391→  // sort and paginate client-side for an instant, consistent UX.
   392→  const productsQuery = useProducts({
   393→    categoryId: category?.id,
   394→    limit: 100,
   395→  });
   396→
   397→  // Brands (for the brand filter).
   398→  const brandsQuery = useBrands();
   399→  const brands = useMemo(
   400→    () =>
   401→      (brandsQuery.data ?? []).map((b: { id: string; name: string }) => ({
   402→        id: b.id,
   403→        name: b.name,
   404→      })),
   405→    [brandsQuery.data],
   406→  );
   407→
   408→  // ── UI state ──
   409→  const [sortBy, setSortBy] = useState<SortValue>("relevance");
   410→  const [sortOpen, setSortOpen] = useState(false);
   411→  const [filters, setFilters] = useState<FilterState>(EMPTY_FILTERS);
   412→  const [activeSubCategoryId, setActiveSubCategoryId] = useState<string | null>(
   413→    null,
   414→  );
   415→  const [visibleCount, setVisibleCount] = useState(PAGE_SIZE);
   416→  const [mobileFiltersOpen, setMobileFiltersOpen] = useState(false);
   417→  const [quickViewProduct, setQuickViewProduct] =
   418→    useState<MarketplaceProduct | null>(null);
   419→
   420→  // Reset visible count when filters/search/sort change. Using the
   421→  // "adjust state during render" pattern (instead of useEffect) avoids
   422→  // cascading renders — see https://react.dev/reference/react/useState#storing-information-from-previous-renders
   423→  const [resetKey, setResetKey] = useState("");
   424→  const currentResetKey = `${sortBy}|${JSON.stringify(filters)}|${activeSubCategoryId}`;
   425→  if (currentResetKey !== resetKey) {
   426→    setResetKey(currentResetKey);
   427→    setVisibleCount(PAGE_SIZE);
   428→  }
   429→
   430→  // All products for this category (mapped to marketplace shape).
   431→  const allProducts: MarketplaceProduct[] = useMemo(() => {
   432→    const raw = productsQuery.data?.products ?? [];
   433→    // Also include products that came embedded in the category response (the
   434→    // dummy route attaches them). We prefer the dedicated products query.
   435→    if (raw.length > 0) return raw;
   436→    if (rawCategoryData && typeof rawCategoryData === "object") {
   437→      const obj = rawCategoryData as unknown as Record<string, unknown>;
   438→      const embedded = obj.products;
   439→      if (Array.isArray(embedded)) {
   440→        return (embedded as unknown[]).map((p) =>
   441→          toMarketplaceProduct(p as Parameters<typeof toMarketplaceProduct>[0]),
   442→        );
   443→      }
   444→    }
   445→    return [];
   446→  }, [productsQuery.data, rawCategoryData]);
   447→
   448→  // Price bounds (for the placeholder hints in the price filter).
   449→  const priceBounds = useMemo(() => {
   450→    if (allProducts.length === 0) return { min: 0, max: 100000 };
   451→    let min = Infinity;
   452→    let max = 0;
   453→    for (const p of allProducts) {
   454→      if (p.price < min) min = p.price;
   455→      if (p.price > max) max = p.price;
   456→    }
   457→    return { min: Math.floor(min), max: Math.ceil(max) };
   458→  }, [allProducts]);
   459→
   460→  // Brand product counts (within this category's product set).
   461→  const brandCounts = useMemo(() => {
   462→    const counts: Record<string, number> = {};
   463→    for (const p of allProducts) {
   464→      if (p.brandId) {
   465→        counts[p.brandId] = (counts[p.brandId] ?? 0) + 1;
   466→      }
   467→    }
   468→    return counts;
   469→  }, [allProducts]);
   470→
   471→  // Only show brands that actually have products in this category.
   472→  const relevantBrands = useMemo(
   473→    () => brands.filter((b) => brandCounts[b.id] > 0),
   474→    [brands, brandCounts],
   475→  );
   476→
   477→  // Apply filters + sort.
   478→  const filteredProducts = useMemo(() => {
   479→    let list = allProducts;
   480→
   481→    // Sub-category filter — narrows to products whose categoryId matches the
   482→    // selected child category.
   483→    if (activeSubCategoryId) {
   484→      list = list.filter((p) => p.categoryId === activeSubCategoryId);
   485→    }
   486→
   487→    // Price range
   488→    const minP = filters.minPrice ? Number(filters.minPrice) : -Infinity;
   489→    const maxP = filters.maxPrice ? Number(filters.maxPrice) : Infinity;
   490→    if (filters.minPrice || filters.maxPrice) {
   491→      list = list.filter((p) => p.price >= minP && p.price <= maxP);
   492→    }
   493→
   494→    // Brands
   495→    if (filters.brandIds.length > 0) {
   496→      list = list.filter(
   497→        (p) => p.brandId != null && filters.brandIds.includes(p.brandId),
   498→      );
   499→    }
   500→
   501→    // Rating
   502→    if (filters.minRating > 0) {
   503→      list = list.filter((p) => p.rating >= filters.minRating);
   504→    }
   505→
   506→    // In stock
   507→    if (filters.inStockOnly) {
   508→      list = list.filter(
   509→        (p) => p.stock === undefined || p.stock > 0 || !p.trackInventory,
   510→      );
   511→    }
   512→
   513→    // Sort
   514→    const sorted = [...list];
   515→    switch (sortBy) {
   516→      case "price-asc":
   517→        sorted.sort((a, b) => a.price - b.price);
   518→        break;
   519→      case "price-desc":
   520→        sorted.sort((a, b) => b.price - a.price);
   521→        break;
   522→      case "rating":
   523→        sorted.sort((a, b) => b.rating - a.rating);
   524→        break;
   525→      case "newest":
   526→        sorted.sort(
   527→          (a, b) => Number(b.isNew ?? false) - Number(a.isNew ?? false),
   528→        );
   529→        break;
   530→      case "best-selling":
   531→        sorted.sort(
   532→          (a, b) => (b.reviewCount ?? 0) - (a.reviewCount ?? 0),
   533→        );
   534→        break;
   535→      default:
   536→        break;
   537→    }
   538→    return sorted;
   539→  }, [allProducts, filters, sortBy, activeSubCategoryId]);
   540→
   541→  const visibleProducts = filteredProducts.slice(0, visibleCount);
   542→  const hasMore = visibleCount < filteredProducts.length;
   543→
   544→  const subCategories = useMemo(
   545→    () => category?.children ?? [],
   546→    [category],
   547→  );
   548→
   549→  const handleClearFilters = useCallback(() => {
   550→    setFilters(EMPTY_FILTERS);
   551→    setActiveSubCategoryId(null);
   552→  }, []);
   553→
   554→  const categoryStyle = category
   555→    ? resolveCategoryStyle(category.name, category.slug, 0)
   556→    : null;
   557→  const CategoryIcon = categoryStyle?.Icon ?? DEFAULT_CATEGORY_STYLE.Icon;
   558→
   559→  const productCount = allProducts.length;
   560→  const filteredCount = filteredProducts.length;
   561→
   562→  // ── Loading ──
   563→  if (categoryQuery.isLoading) {
   564→    return (
   565→      <div className="min-h-screen bg-background">
   566→        <div className="border-b border-border bg-background">
   567→          <div className="container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   568→            <div className="py-3">
   569→              <div className="h-3 w-48 animate-pulse rounded bg-muted" />
   570→            </div>
   571→            <div className="flex items-center gap-4 pb-5">
   572→              <div className="h-14 w-14 animate-pulse rounded-xl bg-muted" />
   573→              <div className="space-y-2">
   574→                <div className="h-6 w-40 animate-pulse rounded bg-muted" />
   575→                <div className="h-3 w-56 animate-pulse rounded bg-muted" />
   576→              </div>
   577→            </div>
   578→          </div>
   579→        </div>
   580→        <div className="container mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
   581→          <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 2xl:grid-cols-6">
   582→            {Array.from({ length: 10 }).map((_, i) => (
   583→              <div
   584→                key={i}
   585→                className="space-y-3 border border-border bg-background p-3"
   586→              >
   587→                <div className="aspect-square animate-pulse bg-muted" />
   588→                <div className="h-3 w-3/4 animate-pulse rounded bg-muted" />
   589→                <div className="h-4 w-1/2 animate-pulse rounded bg-muted" />
   590→              </div>
   591→            ))}
   592→          </div>
   593→        </div>
   594→      </div>
   595→    );
   596→  }

   599→  if (!category) {
   600→    return (
   601→      <div className="min-h-screen bg-background">
   602→        <div className="container mx-auto max-w-7xl px-4 py-16 sm:px-6 lg:px-8">
   603→          <div className="flex flex-col items-center justify-center py-20 text-center">
   604→            <div className="flex h-16 w-16 items-center justify-center rounded-full border border-border bg-muted">
   605→              <PackageOpen className="h-8 w-8 text-muted-foreground" />
   606→            </div>

   608→              Category not found

   610→            <p className="mt-2 max-w-md text-sm text-muted-foreground">
   611→              We couldn&apos;t find the category you&apos;re looking for. It may

   613→            </p>
   614→            <div className="mt-6 flex flex-wrap items-center justify-center gap-3">
   615→              <Link
   616→                href="/categories"
   617→                className="inline-flex h-11 items-center justify-center rounded-md bg-kwik-orange px-5 text-sm font-semibold text-white transition-colors hover:bg-kwik-orange-hover"

   619→                Browse all categories
   620→              </Link>
   621→              <Link
   622→                href="/"
   623→                className="inline-flex h-11 items-center justify-center gap-2 rounded-md border border-border bg-background px-5 text-sm font-semibold text-foreground transition-colors hover:bg-muted"

   628→          </div>
   629→        </div>

   633→
   634→  return (
   635→    <div className="min-h-screen bg-background">

   637→      <div className="border-b border-border bg-background">
   638→        <div className="container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   639→          {/* Breadcrumb */}
   640→          <nav className="flex items-center gap-1.5 py-3 text-xs text-muted-foreground">

   643→              className="transition-colors hover:text-kwik-orange"

   645→              Home

   649→              href="/categories"
   650→              className="transition-colors hover:text-kwik-orange"

   662→                "flex h-14 w-14 shrink-0 items-center justify-center rounded-2xl text-white shadow-md",

   671→              </h1>
   672→              <p className="mt-0.5 text-sm text-muted-foreground">
   673→                {productCount > 0
   674→                  ? `${productCount} product${productCount !== 1 ? "s" : ""} available`

   677→            </div>
   678→          </div>
   679→        </div>
   680→      </div>
   681→
   682→      {/* ── Toolbar ── */}
   683→      <div className="sticky top-0 z-30 border-b border-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   684→        <div className="container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   685→          <div className="flex items-center gap-3 py-3">

   687→            <div className="relative shrink-0">
   688→              <button
   689→                type="button"
   690→                onClick={() => setSortOpen((v) => !v)}

   744→              <Filter className="h-4 w-4 text-kwik-orange" />

   746→            </button>
   747→          </div>
   748→        </div>
   749→      </div>
   750→
   751→      {/* ── Body: sidebar + product grid ── */}

   753→        <div className="flex gap-6">
   754→          {/* Desktop sidebar */}
   755→          <aside className="hidden w-64 shrink-0 lg:block">
   756→            <div className="sticky top-24 max-h-[calc(100vh-7rem)] overflow-y-auto rounded-lg border border-border bg-background p-4">
   757→              <FilterPanel
   758→                filters={filters}
   759→                setFilters={setFilters}
   760→                brands={relevantBrands}
   761→                brandCounts={brandCounts}
   762→                subCategories={subCategories}
   763→                activeSubCategoryId={activeSubCategoryId}
   764→                onSubCategoryChange={setActiveSubCategoryId}

   772→          <div className="min-w-0 flex-1">
   773→            {/* Results meta */}
   774→            <div className="mb-4 flex items-center justify-between gap-3">
   775→              <p className="text-sm text-muted-foreground">
   776→                {productsQuery.isLoading ? (
   777→                  "Loading products…"
   778→                ) : (
   779→                  <>
   780→                    <span className="font-semibold text-foreground">
   781→                      {filteredCount}
   782→                    </span>{" "}
   783→                    product{filteredCount !== 1 ? "s" : ""}
   784→                  </>
   785→                )}
   786→              </p>
   787→            </div>
   788→

   790→            {productsQuery.isLoading && (

   792→                {Array.from({ length: 10 }).map((_, i) => (
   793→                  <div
   794→                    key={i}
   795→                    className="space-y-3 border border-border bg-background p-3"
   796→                  >
   797→                    <div className="aspect-square animate-pulse bg-muted" />

   799→                    <div className="h-4 w-1/2 animate-pulse rounded bg-muted" />
   800→                  </div>
   801→                ))}
   802→              </div>
   803→            )}
   804→

   809→                  <PackageOpen className="h-7 w-7 text-muted-foreground" />

   811→                <h2 className="mt-5 text-lg font-semibold text-foreground">

   822→                  filters.minRating ||
   823→                  filters.inStockOnly ||
   824→                  filters.minPrice ||
   825→                  filters.maxPrice) && (
   826→                  <button
   827→                    type="button"
   828→                    onClick={handleClearFilters}
   829→                    className="mt-5 inline-flex h-10 items-center justify-center rounded-md border border-kwik-orange px-5 text-sm font-semibold text-kwik-orange transition-colors hover:bg-kwik-orange-tint"
   830→                  >
   831→                    Clear filters
   832→                  </button>
   833→                )}
   834→              </div>
   835→            )}
   836→
   837→            {/* Product grid */}
   838→            {!productsQuery.isLoading && filteredCount > 0 && (
   839→              <>
   840→                <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-3 xl:grid-cols-4 2xl:grid-cols-6">
   841→                  {visibleProducts.map((product, index) => (
   842→                    <MarketplaceProductCard
   843→                      key={`${product.id}-${index}`}
   844→                      product={product}
   845→                      onQuickView={() => setQuickViewProduct(product)}

   868→                {!hasMore && filteredCount > PAGE_SIZE && (

   870→                    You&apos;ve seen all {filteredCount} products in{" "}

   873→                )}
   874→              </>
   875→            )}
   876→          </div>

   880→      {/* ── Mobile filter drawer ── */}
   881→      <AnimatePresence>

   885→              initial={{ opacity: 0 }}
   886→              animate={{ opacity: 1 }}

   888→              transition={{ duration: 0.2 }}
   889→              className="fixed inset-0 z-50 bg-black/50 lg:hidden"
   890→              onClick={() => setMobileFiltersOpen(false)}
   891→            />

   893→              initial={{ x: "100%" }}
   894→              animate={{ x: 0 }}
   895→              exit={{ x: "100%" }}
   896→              transition={{ type: "spring", damping: 30, stiffness: 300 }}
   897→              className="fixed bottom-0 right-0 top-0 z-50 flex w-[85%] max-w-sm flex-col bg-background shadow-xl lg:hidden"
   898→            >
   899→              <div className="flex items-center justify-between border-b border-border px-4 py-3">
   900→                <h2 className="text-base font-semibold text-foreground">
   901→                  Filters
   902→                </h2>
   903→                <button
   904→                  type="button"
   905→                  onClick={() => setMobileFiltersOpen(false)}
   906→                  className="rounded-md p-1.5 text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"
   907→                  aria-label="Close filters"
   908→                >
   909→                  <X className="h-5 w-5" />
   910→                </button>
   911→              </div>
   912→              <div className="flex-1 overflow-y-auto p-4">
   913→                <FilterPanel
   914→                  filters={filters}
   915→                  setFilters={setFilters}
   916→                  brands={relevantBrands}
   917→                  brandCounts={brandCounts}
   918→                  subCategories={subCategories}
   919→                  activeSubCategoryId={activeSubCategoryId}
   920→                  onSubCategoryChange={setActiveSubCategoryId}
   921→                  onClear={handleClearFilters}
   922→                  priceBounds={priceBounds}
   923→                />
   924→              </div>
   925→              <div className="border-t border-border p-4">
   926→                <button
   927→                  type="button"
   928→                  onClick={() => setMobileFiltersOpen(false)}
   929→                  className="inline-flex h-11 w-full items-center justify-center rounded-md bg-kwik-orange px-5 text-sm font-semibold text-white transition-colors hover:bg-kwik-orange-hover"
   930→                >
   931→                  Show {filteredCount} result{filteredCount !== 1 ? "s" : ""}
   932→                </button>
   933→              </div>
   934→            </motion.div>
   935→          </>
   936→        )}

   938→
   939→      {/* ── Quick view ── */}
   940→      <QuickViewModal
   941→        product={quickViewProduct}
   942→        isOpen={!!quickViewProduct}

   944→      />
   945→    </div>
   946→  );
