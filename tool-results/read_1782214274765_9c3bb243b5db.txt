    39→    39→
    40→    40→function unwrap<T>(value: unknown): T {
    41→    41→  if (value && typeof value === "object" && "data" in value) {
    42→    42→    return (value as { data: T }).data;
    43→    43→  }
    44→    44→  return value as T;
    45→    45→}
    46→    46→
    47→    47→function shortId(value?: string) {
    48→    48→  return value ? value.slice(0, 12) : "unknown";
    49→    49→}
    50→    50→
    51→    51→function buyerName(order: AdminCommerceOrder): string {
    52→    52→  const profile = order.buyer?.profile;
    53→    53→  if (profile?.firstName || profile?.lastName) {
    54→    54→    return [profile.firstName, profile.lastName].filter(Boolean).join(" ");
    55→    55→  }
    56→    56→  if (order.buyer?.email) return order.buyer.email;
    57→    57→  return "Unknown buyer";
    58→    58→}
    59→    59→
    60→    60→function orderRef(order: AdminCommerceOrder): string {
    61→    61→  return order.checkoutReference ?? shortId(order.id);
    62→    62→}
