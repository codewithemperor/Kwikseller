    69→    69→const getBaseURL = (): string => {
    70→    70→  const configuredUrl = process.env.NEXT_PUBLIC_API_URL
    71→    71→
    72→    72→  if (!configuredUrl) {
    73→    73→    // Client (browser): relative "/api/v1" so requests go through the
    74→    74→    // gateway and hit the in-app route handlers (dummy) or the proxied
    75→    75→    // rewrite (real API).
    76→    76→    // Server (SSR/RSC): absolute localhost:3000 so server-side fetches
    77→    77→    // resolve against the Next.js app's own route handlers. In production
    78→    78→    // set NEXT_PUBLIC_API_URL to the real NestJS backend.
    79→    79→    return typeof window !== 'undefined' ? '/api/v1' : 'http://localhost:3000/api/v1'
    80→    80→  }
    81→    81→
    82→    82→  const withProtocol = /^https?:\/\//i.test(configuredUrl)
    83→    83→    ? configuredUrl
    84→    84→    : `https://${configuredUrl}`
    85→    85→  const withoutTrailingSlash = withProtocol.replace(/\/+$/, '')
    86→    86→
    87→    87→  return withoutTrailingSlash.endsWith('/api/v1')
    88→    88→    ? withoutTrailingSlash
    89→    89→    : `${withoutTrailingSlash}/api/v1`
    90→    90→}

    92→    92→// Storage keys
    93→    93→const STORAGE_KEYS = {
    94→    94→  ACCESS_TOKEN: 'kwikseller_access_token',
    95→    95→  REFRESH_TOKEN: 'kwikseller_refresh_token',
    96→    96→  AUTH_STORE: 'kwikseller_auth',
    97→    97→}
    98→    98→
    99→    99→// ==================== Token Refresh Mutex ====================
   100→   100→// Prevents concurrent 401 responses from triggering multiple refresh calls.
   101→   101→// Only one refresh request is in-flight at a time; other requests queue and
   102→   102→// resolve once the refresh completes.
   103→   103→
   104→   104→let isRefreshing = false
   105→   105→let refreshQueue: Array<{
   106→   106→  resolve: (token: string) => void
   107→   107→  reject: (error: unknown) => void
   108→   108→}> = []
   109→   109→
   110→   110→function processRefreshQueue(error: unknown, token: string | null = null): void {
   111→   111→  refreshQueue.forEach(({ resolve, reject }) => {
   112→   112→    if (error) reject(error)
   113→   113→    else if (token) resolve(token)
   114→   114→  })
   115→   115→  refreshQueue = []
   116→   116→}

   118→   118→// ==================== Token Management ====================
   119→   119→
   120→   120→export const tokenManager = {
   121→   121→  getAccessToken: (): string | null => {
   122→   122→    if (typeof window === 'undefined') return null
   123→   123→    return localStorage.getItem(STORAGE_KEYS.ACCESS_TOKEN)
   124→   124→  },
   125→   125→
   126→   126→  getRefreshToken: (): string | null => {
   127→   127→    if (typeof window === 'undefined') return null
   128→   128→    return localStorage.getItem(STORAGE_KEYS.REFRESH_TOKEN)
   129→   129→  },
   130→   130→
   131→   131→  setTokens: (accessToken: string, refreshToken?: string): void => {
   132→   132→    if (typeof window === 'undefined') return
   133→   133→    localStorage.setItem(STORAGE_KEYS.ACCESS_TOKEN, accessToken)
   134→   134→    if (refreshToken) {
   135→   135→      localStorage.setItem(STORAGE_KEYS.REFRESH_TOKEN, refreshToken)
   136→   136→    }
   137→   137→  },
   138→   138→
   139→   139→  clearTokens: (): void => {
   140→   140→    if (typeof window === 'undefined') return
   141→   141→    localStorage.removeItem(STORAGE_KEYS.ACCESS_TOKEN)
   142→   142→    localStorage.removeItem(STORAGE_KEYS.REFRESH_TOKEN)
   143→   143→    localStorage.removeItem(STORAGE_KEYS.AUTH_STORE)
   144→   144→  },
   145→   145→
   146→   146→  isAuthenticated: (): boolean => {
   147→   147→    return !!tokenManager.getAccessToken()
   148→   148→  },
   149→   149→}
   150→   150→
   151→   151→// ==================== Create Axios Instance ====================
   152→   152→
   153→   153→const createApiClient = (): AxiosInstance => {
   154→   154→  const instance = axios.create({
   155→   155→    baseURL: getBaseURL(),
   156→   156→    timeout: 30000,
   157→   157→    headers: {
   158→   158→      'Content-Type': 'application/json',
   159→   159→    },
   160→   160→  })
   161→   161→
   162→   162→  const redirectToLogin = () => {
   163→   163→    if (typeof window === 'undefined') return
   164→   164→    tokenManager.clearTokens()
   165→   165→    const currentPath = window.location.pathname
   166→   166→    const currentHost = window.location.hostname
   167→   167→    const isAdminApp = currentHost.includes('admin') || currentPath.startsWith('/admin')
   168→   168→    const isRiderApp = currentHost.includes('rider') || currentPath.startsWith('/rider')
   169→   169→    const isVendorApp = currentHost.includes('vendor') || currentPath.startsWith('/dashboard')
   170→   170→    let loginPath = '/login'
   171→   171→    if (isAdminApp) loginPath = '/admin/login'
   172→   172→    else if (isRiderApp) loginPath = '/rider/login'
   173→   173→    // Vendor app login is at /login (same as default), but we preserve
   174→   174→    // the dashboard redirect so the user lands back where they were
   175→   175→    window.location.href = `${loginPath}?redirect=${encodeURIComponent(currentPath)}`
   176→   176→  }
   177→   177→
   178→   178→  // Request interceptor - Add auth token
   179→   179→  instance.interceptors.request.use(
   180→   180→    (config: InternalAxiosRequestConfig) => {
   181→   181→      // Skip auth if explicitly requested
   182→   182→      if ((config as RequestConfig).skipAuth) {
   183→   183→        return config
   184→   184→      }
   185→   185→
   186→   186→      const token = tokenManager.getAccessToken()
   187→   187→      if (token && config.headers) {
   188→   188→        config.headers.Authorization = `Bearer ${token}`
   189→   189→      }
   190→   190→      return config
   191→   191→    },
   192→   192→    (error) => Promise.reject(error)
   193→   193→  )
   194→   194→
   195→   195→  // Response interceptor - Handle errors and token refresh with mutex
   196→   196→  instance.interceptors.response.use(
   197→   197→    (response: AxiosResponse) => response,
   198→   198→    async (error: AxiosError<ApiError>) => {
   199→   199→      const originalRequest = error.config as InternalAxiosRequestConfig & {
   200→   200→        _retry?: boolean
   201→   201→      }
   202→   202→      const isAuthEndpoint = originalRequest?.url?.includes('/auth/')
   203→   203→
   204→   204→      // Handle 401 Unauthorized - Try to refresh token (with mutex to prevent race conditions)
   205→   205→      if (error.response?.status === 401 && !originalRequest._retry && !isAuthEndpoint) {
   206→   206→        if (isRefreshing) {
   207→   207→          // Another refresh is already in-flight — queue this request
   208→   208→          return new Promise((resolve, reject) => {
   209→   209→            refreshQueue.push({
   210→   210→              resolve: (newToken: string) => {
   211→   211→                if (originalRequest.headers) {
   212→   212→                  originalRequest.headers.Authorization = `Bearer ${newToken}`
   213→   213→                }
   214→   214→                resolve(instance(originalRequest))
   215→   215→              },
   216→   216→              reject: (err: unknown) => reject(err),
   217→   217→            })
   218→   218→          })
   219→   219→        }
   220→   220→
   221→   221→        originalRequest._retry = true
   222→   222→        isRefreshing = true
   223→   223→
   224→   224→        try {
   225→   225→          const refreshToken = tokenManager.getRefreshToken()
   226→   226→          if (refreshToken) {
   227→   227→            const response = await axios.post(`${getBaseURL()}/auth/refresh`, {
   228→   228→              refreshToken,
   229→   229→            })
   230→   230→
   231→   231→            const { accessToken, refreshToken: newRefreshToken } = response.data.data || response.data
   232→   232→            tokenManager.setTokens(accessToken, newRefreshToken)
   233→   233→
   234→   234→            // Notify queued requests
   235→   235→            processRefreshQueue(null, accessToken)
   236→   236→
   237→   237→            // Retry original request with new token
   238→   238→            if (originalRequest.headers) {
   239→   239→              originalRequest.headers.Authorization = `Bearer ${accessToken}`
   240→   240→            }
   241→   241→            return instance(originalRequest)
   242→   242→          }
   243→   243→          // No refresh token available
   244→   244→          processRefreshQueue(new Error('No refresh token available'), null)
   245→   245→          tokenManager.clearTokens()
   246→   246→          redirectToLogin()
   247→   247→        } catch (refreshError) {
   248→   248→          // Refresh failed
   249→   249→          processRefreshQueue(refreshError, null)
   250→   250→          tokenManager.clearTokens()
   251→   251→          redirectToLogin()
   252→   252→        } finally {
   253→   253→          isRefreshing = false
   254→   254→        }
   255→   255→      }
   256→   256→
   257→   257→      // Handle other errors
   258→   258→      const apiError: ApiError = {
   259→   259→        statusCode: error.response?.status || 500,
   260→   260→        message: error.response?.data?.message || error.message || 'An error occurred',
   261→   261→        error: error.response?.data?.error,
   262→   262→        details: error.response?.data?.details,
   263→   263→      }
   264→   264→
   265→   265→      return Promise.reject(apiError)
   266→   266→    }
   267→   267→  )
   268→   268→
   269→   269→  return instance
   270→   270→}
   271→   271→
   272→   272→// Create singleton instance
   273→   273→const apiClient = createApiClient()
