   101→function formatDateTime(value?: string) {
   102→  if (!value) return "—";
   103→  return new Intl.DateTimeFormat("en-NG", { dateStyle: "medium", timeStyle: "short" }).format(new Date(value));
   104→}
   105→
   106→function formatNGN(n: number) {
   107→  return new Intl.NumberFormat("en-NG", {
   108→    style: "currency",
   109→    currency: "NGN",
   110→    maximumFractionDigits: 0,
   111→  }).format(n);
   112→}

   131→function WalletContent() {
   132→  const { data: wallet, isLoading } = useQuery<WalletData>({

   145→    },
   146→    staleTime: 5 * 60 * 1000,
   147→  });
   148→
   149→  const [filter, setFilter] = React.useState<"ALL" | "CREDIT" | "DEBIT">("ALL");
   150→  const [dateRange, setDateRange] = React.useState<"7d" | "30d" | "90d" | "all">("all");
   151→  const [categoryFilter, setCategoryFilter] = React.useState<string>("ALL");
   152→  const [searchTerm, setSearchTerm] = React.useState("");
   153→  const [showAdvanced, setShowAdvanced] = React.useState(false);
   154→
   155→  // ── Redemption modal state ──

   157→  const [redeemType, setRedeemType] = React.useState<WalletRedemptionType>("CASH");

   159→  function openRedeemModal(type: WalletRedemptionType) {
   160→    setRedeemType(type);
   161→    setRedeemOpen(true);
   162→  }
   163→
   164→  // ── Derived state (declared BEFORE any early return so the Rules of
   165→  // Hooks aren't violated). All hooks gracefully handle `wallet` being
   166→  // undefined by returning empty defaults.
   167→  const transactions = wallet?.transactions ?? [];
   168→
   169→  const txs = React.useMemo(() => {
   170→    let list = transactions.slice();
   171→    // Type filter
   172→    if (filter !== "ALL") list = list.filter((t) => t.type === filter);
   173→    // Category filter
   174→    if (categoryFilter !== "ALL") list = list.filter((t) => t.category === categoryFilter);
   175→    // Date range filter
   176→    if (dateRange !== "all") {
   177→      const days = dateRange === "7d" ? 7 : dateRange === "30d" ? 30 : 90;
   178→      const cutoff = Date.now() - days * 86400000;
   179→      list = list.filter((t) => new Date(t.createdAt).getTime() >= cutoff);
   180→    }
   181→    // Search filter (case-insensitive on description)
   182→    const term = searchTerm.trim().toLowerCase();
   183→    if (term) {
   184→      list = list.filter(
   185→        (t) =>
   186→          t.description.toLowerCase().includes(term) ||
   187→          t.category.replace(/_/g, " ").toLowerCase().includes(term),
   188→      );
   189→    }
   190→    return list;
   191→  }, [transactions, filter, categoryFilter, dateRange, searchTerm]);
   192→
   193→  // Available categories derived from the wallet's own transactions
   194→  const availableCategories = React.useMemo(() => {
   195→    const seen = new Set<string>();
   196→    for (const t of transactions) seen.add(t.category);
   197→    return Array.from(seen);
   198→  }, [transactions]);
   199→
   200→  const hasActiveFilters =
   201→    filter !== "ALL" || categoryFilter !== "ALL" || dateRange !== "all" || searchTerm.trim().length > 0;
   202→
   203→  function clearAllFilters() {
   204→    setFilter("ALL");
   205→    setCategoryFilter("ALL");
   206→    setDateRange("all");
   207→    setSearchTerm("");
   208→  }
   209→
   210→  // Export the currently-filtered transactions to CSV. Respects all active
   211→  // filters (type, category, date range, search) so the buyer gets exactly
   212→  // what they see on screen. Reuses the shared toCSV/downloadCSV utilities
   213→  // (same as the orders page CSV export).
   214→  //
   215→  // The dummy wallet API doesn't return per-transaction running balances,
   216→  // so we derive them locally from the FILTERED list (using only the
   217→  // transactions the buyer is exporting — otherwise the numbers wouldn't
   218→  // reconcile to `wallet.balance`). Algorithm:
   219→  //   1. Sort the filtered txs oldest-first by `createdAt`.
   220→  //   2. Compute totalDelta = sum(credits) - sum(debits).
   221→  //   3. startingBalance = wallet.balance - totalDelta (the balance before
   222→  //      the oldest filtered tx).
   223→  //   4. Walk forward, applying each tx's delta to a running balance and
   224→  //      recording it in a Map<txId, coinsAfter>.
   225→  //   5. Emit CSV rows in the original `txs` order (matches what's on
   226→  //      screen), looking up each row's "Coins After" from the Map.
   227→  // After the most-recent filtered tx, the running balance equals
   228→  // wallet.balance exactly.
   229→  function handleExportCSV() {
   230→    if (txs.length === 0) {
   231→      kwikToast.error("No transactions to export", "Adjust your filters and try again.");
   232→      return;
   233→    }
   234→    const currentBalance = wallet?.balance ?? 0;
   235→    const totalDelta = txs.reduce(
   236→      (sum, t) => sum + (t.type === "CREDIT" ? t.amount : -t.amount),
   237→      0,
   238→    );
   239→    const coinsAfterById = new Map<string, number>();
   240→    let running = currentBalance - totalDelta;
   241→    // Walk oldest → newest by `createdAt` (the API list order isn't
   242→    // guaranteed to be strictly chronological in dummy mode).
   243→    const chronological = [...txs].sort(
   244→      (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
   245→    );
   246→    for (const t of chronological) {
   247→      running += t.type === "CREDIT" ? t.amount : -t.amount;
   248→      coinsAfterById.set(t.id, running);
   249→    }
   250→    const rows = txs.map((t) => ({
   251→      Date: new Date(t.createdAt).toLocaleString("en-NG", { dateStyle: "medium", timeStyle: "short" }),
   252→      Type: t.type === "CREDIT" ? "Earned" : "Spent",
   253→      Category: t.category.replace(/_/g, " ").toLowerCase().replace(/\b\w/g, (c) => c.toUpperCase()),
   254→      Description: t.description,
   255→      Amount: t.type === "CREDIT" ? `+${t.amount}` : `-${t.amount}`,
   256→      "Coins After": coinsAfterById.get(t.id) ?? "",
   257→    }));
   258→    const csv = toCSV(rows);
   259→    const date = new Date().toISOString().slice(0, 10);
   260→    const suffix = hasActiveFilters ? "-filtered" : "";
   261→    downloadCSV(`kwikseller-wallet${suffix}-${date}`, csv);
   262→    kwikToast.success("CSV downloaded", `${txs.length} transaction${txs.length === 1 ? "" : "s"} exported.`);
   263→  }
   264→
   265→  if (isLoading) return <PageLoading label="Loading wallet…" />;
   266→
   267→  if (!wallet) {
   268→    return (
   269→      <EmptyState
   270→        variant="default"
   271→        icon={<WalletIcon className="h-12 w-12" />}
   272→        title="Wallet unavailable"
   273→        description="We couldn't load your KwikCoins wallet right now."
   274→      />
   275→    );
   276→  }
   277→
   278→  const CurrentTierIcon = TIER_ICONS[wallet.tier] ?? Coins;
   279→
   280→  return (
   281→    <div className="mx-auto max-w-4xl">
   282→      <Link
   283→        href="/profile"
   284→        className="inline-flex items-center gap-2 text-sm font-semibold text-kwik-muted transition hover:text-foreground"
   285→      >
   286→        <ArrowLeft className="h-4 w-4" /> Profile
   287→      </Link>
   288→
   289→      <h1 className="mt-3 font-heading text-2xl font-bold text-foreground">KwikCoins Wallet</h1>
   290→      <p className="mt-1 text-sm text-kwik-muted">
   291→        Earn coins on every order and redeem them for discounts, ad credit, and rewards.
   292→      </p>
   293→
   294→      {/* Balance card */}
   295→      <motion.div
   296→        initial={{ opacity: 0, y: 12 }}
   297→        animate={{ opacity: 1, y: 0 }}
   298→        className="mt-6 overflow-hidden rounded-3xl bg-kwik-gradient p-6 text-white shadow-lg sm:p-8"
   299→      >
   300→        <div className="flex flex-wrap items-start justify-between gap-4">
   301→          <div>
   302→            <div className="flex items-center gap-2 text-white/80">
   303→              <Coins className="h-5 w-5" />
   304→              <span className="text-sm font-medium uppercase tracking-wider">Current balance</span>
   305→            </div>
   306→            <p className="mt-2 font-heading text-4xl font-bold sm:text-5xl">
   307→              {wallet.balance.toLocaleString()}{" "}
   308→              <span className="text-lg font-medium text-white/80">KwikCoins</span>
   309→            </p>
   310→            <p className="mt-2 text-sm text-white/70">
   311→              ≈ ₦{wallet.nairaEquivalent.toLocaleString()} value · {wallet.earningRate}× earning rate
   312→            </p>
   313→          </div>
   314→          <div className="flex items-center gap-2 rounded-full bg-white/15 px-3 py-1.5 backdrop-blur-sm">
   315→            <CurrentTierIcon className="h-4 w-4" />
   316→            <span className="text-sm font-semibold">{wallet.tier} Tier</span>
   317→          </div>
   318→        </div>
   319→
   320→        {/* Tier progress */}
   321→        <div className="mt-6">
   322→          <div className="flex items-center justify-between text-xs text-white/80">
   323→            <span>Progress to {wallet.nextTier}</span>
   324→            <span>{wallet.balance} / {wallet.nextTierThreshold}</span>
   325→          </div>
   326→          <div className="mt-1.5 h-2.5 w-full overflow-hidden rounded-full bg-white/20">
   327→            <motion.div
   328→              initial={{ width: 0 }}
   329→              animate={{ width: `${wallet.tierProgress}%` }}

   331→              className="h-full rounded-full bg-white"
   332→            />
   333→          </div>

   359→            <Sparkles className="h-5 w-5" />

   362→        </div>
   363→      </motion.div>
   364→
   365→      {/* Expiring coins warning (cycle 7) */}
   366→      {wallet.expiringCoins && wallet.expiringCoins.amount > 0 && (

   371→      <div className="mt-6 grid grid-cols-2 gap-4 sm:grid-cols-3">
   372→        <div className="rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-4">
   373→          <div className="flex items-center gap-2 text-kwik-green">
   374→            <TrendingUp className="h-4 w-4" />
   375→            <span className="text-xs font-semibold uppercase tracking-wide">Lifetime earned</span>
   376→          </div>
   377→          <p className="mt-2 font-heading text-xl font-bold text-foreground">
   378→            {wallet.lifetimeEarned.toLocaleString()}
   379→          </p>
   380→          <p className="text-xs text-kwik-muted">KwikCoins</p>
   381→        </div>
   382→        <div className="rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-4">
   383→          <div className="flex items-center gap-2 text-kwik-red">
   384→            <TrendingDown className="h-4 w-4" />

   389→          </p>
   390→          <p className="text-xs text-kwik-muted">KwikCoins</p>

   392→        <div className="col-span-2 rounded-2xl border border-kwik-border-light bg-kwik-bg-surface p-4 sm:col-span-1">

   407→          <h2 className="font-heading text-lg font-bold text-foreground">Membership Tiers</h2>
   408→          <div className="mt-3 grid gap-3 sm:grid-cols-2 lg:grid-cols-4">

   418→                    "rounded-2xl border p-4",

   461→            <span className="rounded-full bg-kwik-orange-tint px-2 py-0.5 text-[11px] font-bold text-kwik-orange">
   462→              {txs.length}
   463→            </span>
   464→          </div>

   470→                disabled={txs.length === 0}
   471→                className="inline-flex h-8 items-center gap-1.5 rounded-full border border-kwik-border-light bg-kwik-bg-surface px-3 text-xs font-semibold text-foreground transition hover:border-kwik-orange hover:text-kwik-orange focus:outline-none focus-visible:ring-2 focus-visible:ring-kwik-orange/40 focus-visible:ring-offset-2 disabled:cursor-not-allowed disabled:opacity-50"
   472→                aria-label="Export transactions to CSV"
   473→                title="Export filtered transactions to CSV"
   474→              >

   477→              </button>
   478→              <span className="text-[10px] text-kwik-muted">with running balance</span>
   479→            </div>
   480→            <button
   481→              type="button"
   482→              onClick={() => setShowAdvanced((v) => !v)}
   483→              className={cn(
   484→                "inline-flex h-8 items-center gap-1.5 rounded-full px-3 text-xs font-medium transition",
   485→                showAdvanced

   487→                  : "border border-kwik-border-light text-kwik-muted hover:bg-kwik-bg-surface",
   488→              )}
   489→              aria-expanded={showAdvanced}

   491→              <SlidersHorizontal className="h-3.5 w-3.5" />

   495→        </div>

   497→        {/* Type filter chips — always visible */}
   498→        <div className="mt-3 flex flex-wrap items-center gap-2">
   499→          {(["ALL", "CREDIT", "DEBIT"] as const).map((f) => (
   500→            <button
