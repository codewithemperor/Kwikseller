    49→    49→/* ─── Product Description (renders HTML or plain text) ─── */
    50→    50→function ProductDescription({ description }: { description: string }) {
    51→    51→  const [expanded, setExpanded] = React.useState(false);
    52→    52→
    53→    53→  // Strip HTML for the compact preview
    54→    54→  const plainText = React.useMemo(() => {
    55→    55→    if (!hasHtmlMarkup(description)) return description;
    56→    56→    return description.replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim();
    57→    57→  }, [description]);
    58→    58→
    59→    59→  const fullContentClassName =
    60→    60→    "text-sm leading-6 text-kwik-gray dark:text-white/65 [&_h1]:mb-2 [&_h1]:text-lg [&_h1]:font-bold [&_h1]:text-kwik-dark [&_h1]:dark:text-white [&_h2]:mb-2 [&_h2]:text-base [&_h2]:font-semibold [&_h2]:text-kwik-dark [&_h2]:dark:text-white [&_h3]:mb-1.5 [&_h3]:text-sm [&_h3]:font-semibold [&_h3]:text-kwik-dark [&_h3]:dark:text-white [&_p]:mb-2 [&_ul]:mb-2 [&_ul]:list-disc [&_ul]:pl-5 [&_ol]:mb-2 [&_ol]:list-decimal [&_ol]:pl-5 [&_strong]:font-semibold [&_strong]:text-kwik-dark [&_strong]:dark:text-white";
    61→    61→
    62→    62→  return (
    63→    63→    <div>
    64→    64→      {expanded ? (
    65→    65→        hasHtmlMarkup(description) ? (
    66→    66→          <div className={fullContentClassName} dangerouslySetInnerHTML={{ __html: description }} />
    67→    67→        ) : (
    68→    68→          <p className={fullContentClassName}>{description}</p>
    69→    69→        )
    70→    70→      ) : (
    71→    71→        <p className="text-sm leading-6 text-kwik-gray dark:text-white/65 line-clamp-3">
    72→    72→          {plainText}
    73→    73→        </p>
    74→    74→      )}
    75→    75→      {plainText.length > 120 && (
    76→    76→        <button
    77→    77→          type="button"
    78→    78→          onClick={() => setExpanded(!expanded)}
    79→    79→          className="mt-1 text-xs font-semibold text-kwik-orange hover:underline"
    80→    80→        >
    81→    81→          {expanded ? "Show less" : "Read more"}
    82→    82→        </button>
    83→    83→      )}
    84→    84→    </div>
    85→    85→  );
    86→    86→}
