   291→   291→/* ─── Create Product Wizard Modal ─── */
   292→   292→
   293→   293→function CreateProductModal({
   294→   294→  isOpen,
   295→   295→  onClose,
   296→   296→  onCreated,
   297→   297→}: {
   298→   298→  isOpen: boolean;
   299→   299→  onClose: () => void;
   300→   300→  onCreated: () => void;
   301→   301→}) {
   302→   302→  const [form, setForm] = React.useState(blankForm);
   303→   303→  const [step, setStep] = React.useState(0);
   304→   304→  const [isSaving, setIsSaving] = React.useState(false);
   305→   305→  const [isUploading, setIsUploading] = React.useState(false);
   306→   306→
   307→   307→  const resetAndClose = () => {
   308→   308→    setForm(blankForm);
   309→   309→    setStep(0);
   310→   310→    onClose();
   311→   311→  };
   312→   312→
   313→   313→  const validateStep = (nextStep: number) => {
   314→   314→    if (step === 0 && (!form.name.trim() || !form.productType)) {
   315→   315→      kwikToast.error("Add product name and type first");
   316→   316→      return;
   317→   317→    }
   318→   318→    if (step === 1 && Number(form.price) <= 0) {
   319→   319→      kwikToast.error("Add a valid product price");
   320→   320→      return;
   321→   321→    }
   322→   322→    setStep(nextStep);
   323→   323→  };
   324→   324→
   325→   325→  const createProduct = async () => {
   326→   326→    if (!form.name.trim()) { kwikToast.error("Product name is required"); setStep(0); return; }
   327→   327→    if (Number(form.price) <= 0) { kwikToast.error("Price must be greater than zero"); setStep(1); return; }
   328→   328→
   329→   329→    setIsSaving(true);
   330→   330→    try {
   331→   331→      await vendorCommerceApi.createProduct({
   332→   332→        name: form.name,
   333→   333→        description: form.description || undefined,
   334→   334→        price: Number(form.price),
   335→   335→        comparePrice: form.comparePrice ? Number(form.comparePrice) : undefined,
   336→   336→        sku: form.sku || undefined,
   337→   337→        productType: form.productType,
   338→   338→        requiresShipping: form.productType === "PHYSICAL",
   339→   339→        trackInventory: form.productType === "PHYSICAL",
   340→   340→        initialStock: form.productType === "PHYSICAL" ? Number(form.initialStock) : undefined,
   341→   341→        lowStock: form.productType === "PHYSICAL" ? Number(form.lowStock) : undefined,
   342→   342→        images: form.images,
   343→   343→        poolEnabled: form.productType === "PHYSICAL" ? form.poolEnabled : false,
   344→   344→        poolBasePrice: form.poolEnabled ? Number(form.poolBasePrice || form.price) : undefined,
   345→   345→        poolMinSalePrice: form.poolEnabled ? Number(form.poolMinSalePrice || form.poolBasePrice || form.price) : undefined,
   346→   346→        poolMaxSelectableQuantity: form.poolEnabled && form.poolMaxSelectableQuantity ? Number(form.poolMaxSelectableQuantity) : undefined,
   347→   347→      });
   348→   348→      kwikToast.success("Product created");
   349→   349→      resetAndClose();
   350→   350→      onCreated();
   351→   351→    } catch (error) {
   352→   352→      kwikToast.error(error instanceof Error ? error.message : "Product creation failed");
   353→   353→    } finally {
   354→   354→      setIsSaving(false);
   355→   355→    }
   356→   356→  };
   357→   357→
   358→   358→  const uploadImages = async (files: FileList | null) => {
   359→   359→    if (!files?.length) return;
   360→   360→    setIsUploading(true);
   361→   361→    try {
   362→   362→      const uploads = await Promise.all(Array.from(files).slice(0, 5).map((f) => uploadApi.productImage(f)));
   363→   363→      const urls = uploads.map(uploadedUrl).filter(Boolean);
   364→   364→      setForm((c) => ({ ...c, images: [...c.images, ...urls].slice(0, 5) }));
   365→   365→      kwikToast.success(`${urls.length} image${urls.length === 1 ? "" : "s"} uploaded`);
   366→   366→    } catch (error) {
   367→   367→      kwikToast.error(error instanceof Error ? error.message : "Image upload failed");
   368→   368→    } finally {
   369→   369→      setIsUploading(false);
   370→   370→    }
   371→   371→  };
   372→   372→
   373→   373→  return (
   374→   374→    <AppModal isOpen={isOpen} onClose={resetAndClose} title="Add Product" description="Complete the required steps, then confirm." className="sm:max-w-3xl">
   375→   375→      <div className="space-y-6">
   376→   376→        {/* Step indicator */}
   377→   377→        <div className="grid grid-cols-4 gap-2">
   378→   378→          {wizardSteps.map((label, i) => {
   379→   379→            const done = i < step;
   380→   380→            const active = i === step;
   381→   381→            return (
   382→   382→              <button key={label} type="button" onClick={() => i <= step && setStep(i)} className="flex flex-col items-center gap-2 text-center">
   383→   383→                <span className={`flex h-10 w-10 items-center justify-center rounded-full border text-sm font-semibold ${done ? "border-gray-900 bg-gray-900 text-white" : active ? "border-gray-900 bg-white text-gray-900" : "border-gray-300 bg-white text-gray-400"}`}>
   384→   384→                  {done ? <Check className="h-4 w-4" /> : i + 1}
   385→   385→                </span>
   386→   386→                <span className="hidden text-xs font-medium text-gray-500 sm:block">{label}</span>
   387→   387→              </button>
   388→   388→            );
   389→   389→          })}
   390→   390→        </div>
   391→   391→
   392→   392→        {/* Step 0 — Basic Info */}
   393→   393→        {step === 0 && (
   394→   394→          <div className="space-y-4">
   395→   395→            <FieldInput required label="Product name" value={form.name} onChange={(e) => setForm((v) => ({ ...v, name: e.target.value }))} />
   396→   396→            <FieldSelect label="Product type" value={form.productType} onChange={(e) => setForm((v) => ({ ...v, productType: e.target.value as ProductType }))}>
   397→   397→              <option value="PHYSICAL">Physical</option>
   398→   398→              <option value="DIGITAL">Digital</option>
   399→   399→            </FieldSelect>
   400→   400→            <FieldTextarea label="Description" value={form.description} onChange={(e) => setForm((v) => ({ ...v, description: e.target.value }))} />
   401→   401→          </div>
   402→   402→        )}
   403→   403→
   404→   404→        {/* Step 1 — Pricing & Stock */}
   405→   405→        {step === 1 && (
   406→   406→          <div className="space-y-4">
   407→   407→            <div className="grid gap-3 sm:grid-cols-2">
   408→   408→              <FieldInput required type="number" min={0} label="Price" value={form.price} onChange={(e) => setForm((v) => ({ ...v, price: Number(e.target.value) }))} />
   409→   409→              <FieldInput type="number" min={0} label="Compare price" value={form.comparePrice} onChange={(e) => setForm((v) => ({ ...v, comparePrice: Number(e.target.value) }))} />
   410→   410→            </div>
   411→   411→            {form.productType === "PHYSICAL" ? (
   412→   412→              <div className="grid gap-3 sm:grid-cols-2">
   413→   413→                <FieldInput type="number" min={0} label="Initial stock" value={form.initialStock} onChange={(e) => setForm((v) => ({ ...v, initialStock: Number(e.target.value) }))} />
   414→   414→                <FieldInput type="number" min={0} label="Low stock alert" value={form.lowStock} onChange={(e) => setForm((v) => ({ ...v, lowStock: Number(e.target.value) }))} />
   415→   415→              </div>
   416→   416→            ) : (
   417→   417→              <p className="text-sm text-gray-500">Digital products default to unlimited delivery.</p>
   418→   418→            )}
   419→   419→          </div>
   420→   420→        )}
   421→   421→
   422→   422→        {/* Step 2 — Details */}
   423→   423→        {step === 2 && (
   424→   424→          <div className="space-y-4">
   425→   425→            <FieldInput label="SKU" value={form.sku} onChange={(e) => setForm((v) => ({ ...v, sku: e.target.value }))} />
   426→   426→            <label className="block">
   427→   427→              <span className="text-xs font-semibold text-gray-500">Product images</span>
   428→   428→              <div className="mt-2 flex min-h-44 cursor-pointer flex-col items-center justify-center rounded-lg border border-dashed border-gray-300 bg-gray-50 p-6 text-center">
   429→   429→                <Upload className="h-8 w-8 text-gray-400" />
   430→   430→                <p className="mt-2 text-sm font-medium text-gray-700">
   431→   431→                  <span className="text-blue-600 underline">Click to upload</span> or drag and drop
   432→   432→                </p>
   433→   433→                <p className="mt-1 text-xs text-gray-400">{`${form.images.length}/5 images added`}</p>
   434→   434→                <input type="file" accept="image/*" multiple onChange={(e) => uploadImages(e.target.files)} className="sr-only" />
   435→   435→              </div>
   436→   436→            </label>
   437→   437→            {form.images.length > 0 && (

   515→   515→/* ─── Pagination ─── */
   516→   516→
   517→   517→function Pagination({
   518→   518→  total,
   519→   519→  page,
   520→   520→  onPageChange,
   521→   521→}: {
   522→   522→  total: number;
   523→   523→  page: number;
   524→   524→  onPageChange: (p: number) => void;
   525→   525→}) {
   526→   526→  const totalPages = Math.max(1, Math.ceil(total / ITEMS_PER_PAGE));
   527→   527→  const start = (page - 1) * ITEMS_PER_PAGE + 1;
   528→   528→  const end = Math.min(page * ITEMS_PER_PAGE, total);
   529→   529→
   530→   530→  const pages: (number | "...")[] = [];
   531→   531→  if (totalPages <= 7) {
   532→   532→    for (let i = 1; i <= totalPages; i++) pages.push(i);
   533→   533→  } else {
   534→   534→    pages.push(1);
   535→   535→    if (page > 3) pages.push("...");
   536→   536→    for (let i = Math.max(2, page - 1); i <= Math.min(totalPages - 1, page + 1); i++) pages.push(i);
   537→   537→    if (page < totalPages - 2) pages.push("...");
   538→   538→    pages.push(totalPages);
   539→   539→  }
   540→   540→
   541→   541→  return (
   542→   542→    <div className="flex flex-col items-center justify-between gap-4 sm:flex-row">
   543→   543→      <p className="text-sm text-gray-500">
   544→   544→        Showing {total > 0 ? start : 0}–{end} of {total} products
   545→   545→      </p>
   546→   546→      <div className="flex items-center gap-1">
   547→   547→        <button
   548→   548→          type="button"
   549→   549→          disabled={page <= 1}
   550→   550→          onClick={() => onPageChange(page - 1)}
   551→   551→          className="inline-flex h-8 items-center gap-1 rounded px-2 text-sm font-medium text-gray-600 transition hover:bg-gray-100 disabled:cursor-not-allowed disabled:opacity-40"
   552→   552→        >
   553→   553→          <ChevronLeft className="h-3.5 w-3.5" />
   554→   554→          Prev
   555→   555→        </button>
   556→   556→        {pages.map((p, i) =>
   557→   557→          p === "..." ? (
   558→   558→            <span key={`dots-${i}`} className="px-1 text-sm text-gray-400">...</span>
   559→   559→          ) : (
   560→   560→            <button
   561→   561→              key={p}
   562→   562→              type="button"
   563→   563→              onClick={() => onPageChange(p as number)}
   564→   564→              className={`inline-flex h-8 w-8 items-center justify-center rounded text-sm font-medium transition ${p === page ? "bg-gray-900 text-white" : "text-gray-600 hover:bg-gray-100"}`}
   565→   565→            >
   566→   566→              {p}
   567→   567→            </button>
   568→   568→          )
   569→   569→        )}
   570→   570→        <button
   571→   571→          type="button"
   572→   572→          disabled={page >= totalPages}
   573→   573→          onClick={() => onPageChange(page + 1)}
   574→   574→          className="inline-flex h-8 items-center gap-1 rounded px-2 text-sm font-medium text-gray-600 transition hover:bg-gray-100 disabled:cursor-not-allowed disabled:opacity-40"
   575→   575→        >
   576→   576→          Next
   577→   577→          <ChevronRight className="h-3.5 w-3.5" />
   578→   578→        </button>
   579→   579→      </div>
   580→   580→    </div>
   581→   581→  );
   582→   582→}
