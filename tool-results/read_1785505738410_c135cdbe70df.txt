     7→  Search,

     9→  X,
    10→  LayoutGrid,

    14→} from "lucide-react";
    15→import { GenericProductCard as ProductCard } from "@kwikseller/ui";
    16→import { useWishlistStore } from "@/stores";
    17→import { useCartStore } from "@/stores";
    18→import { kwikToast } from "@kwikseller/utils";
    19→import { cn } from "@/lib/utils";
    20→import {
    21→  useProducts,
    22→  useCategories,
    23→  useStores,
    24→  type Product,
    25→} from "@/lib/api-hooks";

    28→import type { MarketplaceProduct } from "@/data/marketplace-home";
    29→
    30→function formatNGN(n: number) {
    31→  return new Intl.NumberFormat("en-NG", {
    32→    style: "currency",
    33→    currency: "NGN",
    34→    maximumFractionDigits: 0,
    35→  }).format(n);
    36→}
    37→
    38→type SortOption = "popular" | "newest" | "price-asc" | "price-desc" | "rating";
    39→
    40→const sortOptions: { label: string; value: SortOption }[] = [
    41→  { label: "Most Popular", value: "popular" },
    42→  { label: "Newest First", value: "newest" },
    43→  { label: "Price: Low to High", value: "price-asc" },
    44→  { label: "Price: High to Low", value: "price-desc" },
    45→  { label: "Top Rated", value: "rating" },
    46→];
    47→
    48→const PAGE_SIZE = 12;
    49→
    50→function ProductsBrowseContent() {
    51→  const router = useRouter();
    52→  const searchParams = useSearchParams();
    53→
    54→  const initialQuery = searchParams.get("q") ?? "";
    55→  const initialCategory = searchParams.get("category") ?? "all";
    56→  const initialVendor = searchParams.get("vendor") ?? "all";
    57→  const initialBrandId = searchParams.get("brandId") ?? "";
    58→
    59→  const [query, setQuery] = useState(initialQuery);
    60→  const [category, setCategory] = useState<string>(initialCategory);
    61→  const [vendor, setVendor] = useState<string>(initialVendor);
    62→  const [brandId, setBrandId] = useState<string>(initialBrandId);
    63→  const [sort, setSort] = useState<SortOption>("popular");
    64→  const [view, setView] = useState<"grid" | "list">("grid");
    65→  const [showFilters, setShowFilters] = useState(false);
    66→  const [priceMax, setPriceMax] = useState<number>(50000);
    67→  const [onlyDiscounted, setOnlyDiscounted] = useState(false);
    68→  const [onlyInStock, setOnlyInStock] = useState(false);
    69→  const [page, setPage] = useState(1);
    70→  const [debouncedQuery, setDebouncedQuery] = useState(initialQuery);
    71→
    72→  // Debounce the search query so we don't fire an API call on every keystroke.
    73→  useEffect(() => {
    74→    const t = setTimeout(() => setDebouncedQuery(query), 300);
    75→    return () => clearTimeout(t);
    76→  }, [query]);
    77→
    78→  // Wrap each filter setter so changing any filter resets to page 1. Doing it
    79→  // inline (rather than in a useEffect) avoids the "setState in effect" lint
    80→  // rule and the cascading render it warns about.
    81→  const updateFilter = useCallback(
    82→    <T,>(setter: (v: T) => void) =>
    83→      (v: T) => {
    84→        setter(v);
    85→        setPage(1);
    86→      },
    87→    [],
    88→  );
    89→  const updateQuery = useCallback(
    90→    (v: string) => {
    91→      setQuery(v);
    92→      setPage(1);
    93→    },
    94→    [],
    95→  );
    96→
    97→  // Sync filters to URL (shallow) so the page is shareable/bookmarkable.
    98→  useEffect(() => {
    99→    const params = new URLSearchParams();
   100→    if (query) params.set("q", query);
   101→    if (category !== "all") params.set("category", category);
   102→    if (vendor !== "all") params.set("vendor", vendor);
   103→    if (brandId) params.set("brandId", brandId);
   104→    const qs = params.toString();
   105→    router.replace(qs ? `/products?${qs}` : "/products", { scroll: false });
   106→  }, [query, category, vendor, brandId, router]);
   107→
   108→  const wishlistItems = useWishlistStore((s) => s.items);
   109→  const toggleWishlist = useWishlistStore((s) => s.toggleWishlist);
   110→  const addToCart = useCartStore((s) => s.addItem);
   111→
   112→  // ── Filter option lists from the API ───────────────────────────────────
   113→  const categoriesQuery = useCategories();
   114→  const storesQuery = useStores();
   115→
   116→  const categoryOptions = useMemo(() => {
   117→    const cats = categoriesQuery.data ?? [];
   118→    return [
   119→      { label: "All Categories", value: "all" },
   120→      ...cats.map((c: { name: string; slug: string }) => ({
   121→        label: c.name,
   122→        value: c.slug,
   123→      })),
   124→    ];
   125→  }, [categoriesQuery.data]);
   126→
   127→  const vendorOptions = useMemo(() => {
   128→    const stores = (storesQuery.data ?? []) as Array<{
   129→      name: string;
   130→      slug: string;
   131→    }>;
   132→    return [
   133→      { label: "All Vendors", value: "all" },
   134→      ...stores.map((s) => ({ label: s.name, value: s.slug })),
   135→    ];
   136→  }, [storesQuery.data]);
   137→
   138→  // ── Build API params from local filter state ───────────────────────────
   139→  const sortByParam =
   140→    sort === "popular"
   141→      ? "totalSales"
   142→      : sort === "newest"
   143→        ? "createdAt"
   144→        : sort === "rating"
   145→          ? "rating"
   146→          : "price";
   147→  const sortOrderParam: "asc" | "desc" =
   148→    sort === "price-asc" ? "asc" : "desc";
   149→
   150→  const productsQuery = useProducts({
   151→    search: debouncedQuery.trim() || undefined,
   152→    categoryId: category !== "all" ? category : undefined,
   153→    storeId: vendor !== "all" ? vendor : undefined,
   154→    brandId: brandId || undefined,
   155→    sortBy: sortByParam,
   156→    sortOrder: sortOrderParam,
   157→    page,
   158→    limit: PAGE_SIZE,
   159→  });
   160→
   161→  const apiProducts: MarketplaceProduct[] = productsQuery.data?.products ?? [];
   162→  const meta = productsQuery.data?.meta;
   163→  const totalPages = meta?.totalPages ?? 1;
   164→  const totalCount = meta?.total ?? apiProducts.length;
   165→
   166→  // Apply client-side filters the API doesn't support (price ceiling,
   167→  // "on sale" toggle, "in stock" toggle) to the page's products.
   168→  const filtered = useMemo(() => {
   169→    let list = apiProducts.slice();
   170→    list = list.filter((p) => p.price <= priceMax);
   171→    if (onlyDiscounted) {
   172→      list = list.filter((p) => p.comparePrice && p.comparePrice > p.price);
   173→    }
   174→    if (onlyInStock) {
   175→      list = list.filter((p) => (p.stock ?? 0) > 0);
   176→    }
   177→    return list;
   178→  }, [apiProducts, priceMax, onlyDiscounted, onlyInStock]);
   179→
   180→  const activeFilterCount =
   181→    (category !== "all" ? 1 : 0) +
   182→    (vendor !== "all" ? 1 : 0) +
   183→    (brandId ? 1 : 0) +
   184→    (onlyDiscounted ? 1 : 0) +
   185→    (onlyInStock ? 1 : 0) +
   186→    (priceMax < 50000 ? 1 : 0);
   187→
   188→  function clearFilters() {
   189→    setCategory("all");
   190→    setVendor("all");
   191→    setBrandId("");
   192→    setPriceMax(50000);
   193→    setOnlyDiscounted(false);
   194→    setOnlyInStock(false);
   195→    setPage(1);
   196→  }
   197→
   198→  function handleAddToCart(product: MarketplaceProduct) {
   199→    addToCart({
   200→      productId: product.id,
   201→      name: product.name,
   202→      price: product.price,
   203→      comparePrice: product.comparePrice,
   204→      image: product.image,
   205→      store: product.store,
   206→      storeSlug: product.storeSlug,
   207→      storeName: product.store,
   208→      productType: product.productType,
   209→      productSource: product.productSource,
   210→      requiresShipping: product.requiresShipping,
   211→    });
   212→    kwikToast.success("Added to cart", `${product.name} is in your cart.`);
   213→  }
   214→
   215→  function handleWishlist(product: MarketplaceProduct) {
   216→    toggleWishlist({
   217→      productId: product.id,
   218→      name: product.name,
   219→      price: product.price,
   220→      image: product.image,
   221→      store: product.store,
   222→      storeSlug: product.storeSlug,
   223→    });
   224→  }
   225→
   226→  const isLoading = productsQuery.isLoading;
   227→
   228→  return (
   229→    <div className="bg-background min-h-screen">
   230→      {/* ── Page header ── */}
   231→      <section className="kwik-gradient relative overflow-hidden">
   232→        <div className="absolute inset-0 bg-[radial-gradient(circle_at_top_right,rgba(255,255,255,0.18),transparent_55%)]" />
   233→        <div className="container mx-auto max-w-7xl px-4 py-10 md:py-14 relative">
   234→          <motion.div
   235→            initial={{ opacity: 0, y: 12 }}
   236→            animate={{ opacity: 1, y: 0 }}
   237→            transition={{ duration: 0.4 }}
   238→          >
   239→            <p className="text-xs font-semibold uppercase tracking-[0.2em] text-white/75">
   240→              Browse the marketplace
   241→            </p>
   242→            <h1 className="mt-2 font-heading text-3xl font-bold text-white md:text-4xl">
   243→              All Products
   244→            </h1>
   245→            <p className="mt-2 max-w-2xl text-sm leading-6 text-white/85 md:text-base">
   246→              Discover thousands of products from verified vendors across Africa.
   247→              Every purchase is protected by KwisCrow escrow.
   248→            </p>
   249→          </motion.div>
   250→        </div>
   251→      </section>
   252→
   253→      {/* ── Search + sort bar ── */}
   254→      <section className="sticky top-0 z-30 border-b border-border bg-background/95 backdrop-blur-md">
   255→        <div className="container mx-auto max-w-7xl px-4 py-3">
   256→          <div className="flex flex-col gap-3 md:flex-row md:items-center">
   257→            {/* Search */}
   258→            <div className="relative flex-1">
   259→              <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-kwik-muted" />
   260→              <input
   261→                type="text"
   262→                value={query}
   263→                onChange={(e) => updateQuery(e.target.value)}
   264→                placeholder="Search products, vendors, categories..."
   265→                aria-label="Search products"
   266→                className="h-11 w-full rounded-xl border border-border bg-surface pl-10 pr-9 text-sm text-foreground placeholder:text-kwik-muted focus:border-kwik-orange focus:outline-none focus:ring-2 focus:ring-kwik-orange/20"
   267→              />
   268→              {query ? (
   269→                <button
   270→                  type="button"
   271→                  onClick={() => setQuery("")}
   272→                  aria-label="Clear search"
   273→                  className="absolute right-2 top-1/2 -translate-y-1/2 rounded-lg p-1.5 text-kwik-muted hover:bg-muted hover:text-foreground"
   274→                >
   275→                  <X className="h-4 w-4" />
   276→                </button>
   277→              ) : null}
   278→            </div>
   279→
   280→            {/* Sort */}
   281→            <div className="flex items-center gap-2">
   282→              <div className="relative">
   283→                <select
   284→                  value={sort}
   285→                  onChange={(e) => {
   286→                    setSort(e.target.value as SortOption);
   287→                    setPage(1);
   288→                  }}
   289→                  aria-label="Sort products"
   290→                  className="h-11 appearance-none rounded-xl border border-border bg-surface pl-9 pr-8 text-sm font-medium text-foreground focus:border-kwik-orange focus:outline-none focus:ring-2 focus:ring-kwik-orange/20"
   291→                >
   292→                  {sortOptions.map((opt) => (
   293→                    <option key={opt.value} value={opt.value}>
   294→                      {opt.label}
   295→                    </option>
   296→                  ))}
   297→                </select>
   298→                <SlidersHorizontal className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-kwik-muted" />
   299→              </div>
   300→
   301→              {/* Filter toggle (mobile) */}
   302→              <button
   303→                type="button"
   304→                onClick={() => setShowFilters((v) => !v)}
   305→                className="relative inline-flex h-11 items-center gap-2 rounded-xl border border-border bg-surface px-3 text-sm font-medium text-foreground hover:bg-muted md:hidden"
   306→                aria-label="Toggle filters"
   307→              >
   308→                <SlidersHorizontal className="h-4 w-4" />
   309→                Filters
   310→                {activeFilterCount > 0 ? (
   311→                  <span className="absolute -right-1 -top-1 flex h-5 min-w-5 items-center justify-center rounded-full bg-kwik-orange px-1 text-[10px] font-bold text-white">
   312→                    {activeFilterCount}
   313→                  </span>
   314→                ) : null}
   315→              </button>
   316→
   317→              {/* View toggle */}
   318→              <div className="hidden items-center rounded-xl border border-border bg-surface p-1 md:flex">
   319→                <button
   320→                  type="button"
   321→                  onClick={() => setView("grid")}
   322→                  aria-label="Grid view"
   323→                  aria-pressed={view === "grid"}
   324→                  className={cn(
   325→                    "rounded-lg p-2 transition",
   326→                    view === "grid"
   327→                      ? "bg-kwik-orange text-white"
   328→                      : "text-kwik-muted hover:text-foreground",
   329→                  )}
   330→                >
   331→                  <LayoutGrid className="h-4 w-4" />
   332→                </button>
   333→                <button
   334→                  type="button"
   335→                  onClick={() => setView("list")}
   336→                  aria-label="List view"
   337→                  aria-pressed={view === "list"}
   338→                  className={cn(
   339→                    "rounded-lg p-2 transition",
   340→                    view === "list"
   341→                      ? "bg-kwik-orange text-white"
   342→                      : "text-kwik-muted hover:text-foreground",
   343→                  )}
   344→                >
   345→                  <List className="h-4 w-4" />
   346→                </button>
   347→              </div>
   348→            </div>
   349→          </div>
   350→        </div>
   351→      </section>
   352→
   353→      {/* ── Body: sidebar filters + product grid ── */}
   354→      <section className="container mx-auto max-w-7xl px-4 py-6">
   355→        <div className="flex gap-6">
   356→          {/* Sidebar filters (desktop) */}
   357→          <aside className="hidden w-64 shrink-0 md:block">
   358→            <FilterPanel
   359→              category={category}
   360→              setCategory={updateFilter(setCategory)}
   361→              vendor={vendor}
   362→              setVendor={updateFilter(setVendor)}
   363→              priceMax={priceMax}
   364→              setPriceMax={updateFilter(setPriceMax)}
   365→              onlyDiscounted={onlyDiscounted}
   366→              setOnlyDiscounted={updateFilter(setOnlyDiscounted)}
   367→              onlyInStock={onlyInStock}
   368→              setOnlyInStock={updateFilter(setOnlyInStock)}
   369→              activeFilterCount={activeFilterCount}
   370→              onClear={clearFilters}
   371→              categoryOptions={categoryOptions}
   372→              vendorOptions={vendorOptions}
   373→            />
   374→          </aside>
   375→
   376→          {/* Mobile filter drawer */}
   377→          <AnimatePresence>
   378→            {showFilters ? (
   379→              <motion.div
   380→                initial={{ opacity: 0 }}
   381→                animate={{ opacity: 1 }}
   382→                exit={{ opacity: 0 }}
   383→                className="fixed inset-0 z-50 md:hidden"
   384→              >
   385→                <div
   386→                  className="absolute inset-0 bg-black/40 backdrop-blur-sm"
   387→                  onClick={() => setShowFilters(false)}
   388→                />
   389→                <motion.div
   390→                  initial={{ x: "-100%" }}
   391→                  animate={{ x: 0 }}
   392→                  exit={{ x: "-100%" }}
   393→                  transition={{ type: "spring", damping: 28, stiffness: 280 }}
   394→                  className="absolute left-0 top-0 h-full w-80 max-w-[85vw] overflow-y-auto bg-background p-4 shadow-2xl"
   395→                >
   396→                  <div className="mb-4 flex items-center justify-between">
   397→                    <h2 className="font-heading text-lg font-semibold text-foreground">
   398→                      Filters
   399→                    </h2>
   400→                    <button
   401→                      type="button"
   402→                      onClick={() => setShowFilters(false)}
   403→                      aria-label="Close filters"
   404→                      className="rounded-lg p-1.5 text-kwik-muted hover:bg-muted hover:text-foreground"
   405→                    >
   406→                      <X className="h-5 w-5" />
   407→                    </button>
   408→                  </div>
   409→                  <FilterPanel
   410→                    category={category}
   411→                    setCategory={updateFilter(setCategory)}
   412→                    vendor={vendor}
   413→                    setVendor={updateFilter(setVendor)}
   414→                    priceMax={priceMax}
   415→                    setPriceMax={updateFilter(setPriceMax)}
   416→                    onlyDiscounted={onlyDiscounted}
   417→                    setOnlyDiscounted={updateFilter(setOnlyDiscounted)}
   418→                    onlyInStock={onlyInStock}
   419→                    setOnlyInStock={updateFilter(setOnlyInStock)}
   420→                    activeFilterCount={activeFilterCount}
   421→                    onClear={clearFilters}
   422→                    categoryOptions={categoryOptions}
   423→                    vendorOptions={vendorOptions}
   424→                  />
   425→                </motion.div>
   426→              </motion.div>
   427→            ) : null}
   428→          </AnimatePresence>
   429→
   430→          {/* Results */}
   431→          <div className="min-w-0 flex-1">
   432→            {/* Result count + active chips */}
   433→            <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   434→              <p className="text-sm text-kwik-muted">
   435→                Showing{" "}
   436→                <span className="font-semibold text-foreground">
   437→                  {filtered.length}
   438→                </span>{" "}
   439→                {filtered.length !== totalCount ? (
   440→                  <>
   441→                    of{" "}
   442→                    <span className="font-semibold text-foreground">
   443→                      {totalCount}
   444→                    </span>{" "}
   445→                  </>
   446→                ) : null}
   447→                products
   448→                {totalPages > 1 ? (
   449→                  <>
   450→                    {" "}
   451→                    · Page{" "}
   452→                    <span className="font-semibold text-foreground">{page}</span>{" "}
   453→                    of {totalPages}
   454→                  </>
   455→                ) : null}
   456→              </p>
   457→              {activeFilterCount > 0 ? (
   458→                <button
   459→                  type="button"
   460→                  onClick={clearFilters}
   461→                  className="inline-flex items-center gap-1 text-xs font-medium text-kwik-orange hover:opacity-80"
   462→                >
   463→                  <X className="h-3.5 w-3.5" /> Clear all filters
   464→                </button>
   465→              ) : null}
   466→            </div>
   467→
   468→            {/* Active filter chips */}
   469→            {activeFilterCount > 0 ? (
   470→              <div className="mb-4 flex flex-wrap gap-2">
   471→                {category !== "all" ? (
   472→                  <FilterChip
   473→                    label={
   474→                      categoryOptions.find((c) => c.value === category)?.label ??
   475→                      category
   476→                    }
   477→                    onRemove={() => {
   478→                      setCategory("all");
   479→                      setPage(1);
   480→                    }}
   481→                  />
   482→                ) : null}
   483→                {vendor !== "all" ? (
   484→                  <FilterChip
   485→                    label={
   486→                      vendorOptions.find((v) => v.value === vendor)?.label ??
   487→                      vendor
   488→                    }
   489→                    onRemove={() => {
   490→                      setVendor("all");
   491→                      setPage(1);
   492→                    }}
   493→                  />
   494→                ) : null}
   495→                {brandId ? (
   496→                  <FilterChip
   497→                    label="Brand filter"
   498→                    onRemove={() => {
   499→                      setBrandId("");
   500→                      setPage(1);
   501→                    }}
   502→                  />
   503→                ) : null}
   504→                {onlyDiscounted ? (
   505→                  <FilterChip label="On sale" onRemove={() => {
   506→                    setOnlyDiscounted(false);
   507→                    setPage(1);
   508→                  }} />
   509→                ) : null}
   510→                {onlyInStock ? (
   511→                  <FilterChip label="In stock" onRemove={() => {
   512→                    setOnlyInStock(false);
   513→                    setPage(1);
   514→                  }} />
   515→                ) : null}
   516→                {priceMax < 50000 ? (
   517→                  <FilterChip label={`Under ${formatNGN(priceMax)}`} onRemove={() => {
   518→                    setPriceMax(50000);
   519→                    setPage(1);
   520→                  }} />
   521→                ) : null}
   522→              </div>
   523→            ) : null}
   524→
   525→            {/* Loading skeleton */}
   526→            {isLoading ? (
   527→              <ProductGridSkeleton count={8} columns={view === "list" ? 2 : 4} />
   528→            ) : filtered.length === 0 ? (
   529→              <EmptyState
   530→                variant="search"
   531→                title="No products found"
   532→                description="Try adjusting your search or filters to find what you're looking for."
   533→                action={
   534→                  <button
   535→                    type="button"
   536→                    onClick={() => {
   537→                      setQuery("");
   538→                      clearFilters();
   539→                    }}
   540→                    className="inline-flex h-10 items-center justify-center rounded-xl bg-kwik-orange px-5 text-sm font-semibold text-white hover:bg-kwik-orange-hover"
   541→                  >
   542→                    Reset all
   543→                  </button>
   544→                }
   545→              />
   546→            ) : view === "grid" ? (
   547→              <div className="grid grid-cols-2 gap-4 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
   548→                {filtered.map((product, idx) => (
   549→                  <motion.div
   550→                    key={product.id}
   551→                    initial={{ opacity: 0, y: 16 }}
   552→                    animate={{ opacity: 1, y: 0 }}
   553→                    transition={{ duration: 0.3, delay: Math.min(idx * 0.03, 0.3) }}
   554→                  >
   555→                    <ProductCard
   556→                      image={product.image}
   557→                      imageAlt={product.name}
   558→                      name={product.name}
   559→                      price={product.price}
   560→                      comparePrice={product.comparePrice}
   561→                      rating={product.rating}
   562→                      reviewCount={product.reviewCount}
   563→                      storeName={product.store}
   564→                      category={product.category}
   565→                      tag={product.tag}
   566→                      isNew={product.isNew}
   567→                      href={`/products/${product.id}`}
   568→                      onAddToCart={() => handleAddToCart(product)}
   569→                      onWishlist={() => handleWishlist(product)}
   570→                      isWishlisted={wishlistItems.some((w) => w.productId === product.id)}
   571→                      variant="default"
   572→                    />
   573→                  </motion.div>
   574→                ))}
   575→              </div>
   576→            ) : (
   577→              <div className="flex flex-col gap-4">
   578→                {filtered.map((product, idx) => (
   579→                  <motion.div
   580→                    key={product.id}
   581→                    initial={{ opacity: 0, x: -8 }}
   582→                    animate={{ opacity: 1, x: 0 }}
   583→                    transition={{ duration: 0.25, delay: Math.min(idx * 0.02, 0.2) }}
   584→                  >
   585→                    <ProductCard
   586→                      image={product.image}
   587→                      imageAlt={product.name}
   588→                      name={product.name}
   589→                      price={product.price}
   590→                      comparePrice={product.comparePrice}
   591→                      rating={product.rating}
   592→                      reviewCount={product.reviewCount}
   593→                      storeName={product.store}
   594→                      category={product.category}
   595→                      tag={product.tag}
   596→                      isNew={product.isNew}
   597→                      href={`/products/${product.id}`}
   598→                      onAddToCart={() => handleAddToCart(product)}
   599→                      onWishlist={() => handleWishlist(product)}
   600→                      isWishlisted={wishlistItems.some((w) => w.productId === product.id)}
   601→                      variant="horizontal"
   602→                    />
   603→                  </motion.div>
   604→                ))}
   605→              </div>
   606→            )}
   607→
   608→            {/* Pagination */}
   609→            {totalPages > 1 && !isLoading ? (
   610→              <div className="mt-8 flex items-center justify-center gap-2">
   611→                <button
   612→                  type="button"
   613→                  onClick={() => setPage((p) => Math.max(1, p - 1))}
   614→                  disabled={page <= 1}
   615→                  className="inline-flex h-9 items-center gap-1 rounded-lg border border-border bg-surface px-3 text-sm font-medium text-foreground transition hover:bg-muted disabled:cursor-not-allowed disabled:opacity-40"
   616→                  aria-label="Previous page"
   617→                >
   618→                  <ChevronLeft className="h-4 w-4" />
   619→                  Prev
   620→                </button>
   621→                {Array.from({ length: totalPages }).map((_, i) => {
   622→                  const pageNum = i + 1;
   623→                  const isCurrent = pageNum === page;
   624→                  return (
   625→                    <button
   626→                      key={pageNum}
   627→                      type="button"
   628→                      onClick={() => setPage(pageNum)}
   629→                      className={cn(
   630→                        "inline-flex h-9 min-w-9 items-center justify-center rounded-lg border px-3 text-sm font-medium transition",
   631→                        isCurrent
   632→                          ? "border-kwik-orange bg-kwik-orange text-white"
   633→                          : "border-border bg-surface text-foreground hover:bg-muted",
   634→                      )}
   635→                      aria-current={isCurrent ? "page" : undefined}
   636→                    >
   637→                      {pageNum}
   638→                    </button>
   639→                  );
   640→                })}
   641→                <button
   642→                  type="button"
   643→                  onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   644→                  disabled={page >= totalPages}
   645→                  className="inline-flex h-9 items-center gap-1 rounded-lg border border-border bg-surface px-3 text-sm font-medium text-foreground transition hover:bg-muted disabled:cursor-not-allowed disabled:opacity-40"
   646→                  aria-label="Next page"
   647→                >
   648→                  Next
   649→                  <ChevronRight className="h-4 w-4" />
   650→                </button>
   651→              </div>
   652→            ) : null}
   653→          </div>
   654→        </div>
   655→      </section>
   656→    </div>
   657→  );
   658→}
