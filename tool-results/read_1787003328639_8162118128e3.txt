   111→   111→function statusToStepIndex(status: string): number {
   112→   112→  const idx = STEPS.findIndex((s) => s.key === status);
   113→   113→  if (status === "OUT_FOR_DELIVERY") return 3;
   114→   114→  if (status === "CANCELLED" || status === "REJECTED") return -1;
   115→   115→  return idx === -1 ? 0 : idx;
