   181→   181→/* ─── Inline Search Bar (shown in header on /search page) ──── */
   182→   182→
   183→   183→function InlineSearchBar({
   184→   184→  query,
   185→   185→  onSearch,
   186→   186→  onBack,
   187→   187→  showFilters,
   188→   188→  onToggleFilters,
   189→   189→}: {
   190→   190→  query: string;
   191→   191→  onSearch: (q: string) => void;
   192→   192→  onBack: () => void;
   193→   193→  showFilters: boolean;
   194→   194→  onToggleFilters: () => void;
   195→   195→}) {
   196→   196→  const [inputValue, setInputValue] = useState(query);
   197→   197→
   198→   198→  useEffect(() => {
   199→   199→    setInputValue(query);
   200→   200→  }, [query]);
   201→   201→
   202→   202→  const handleSubmit = (e: React.FormEvent) => {
   203→   203→    e.preventDefault();
   204→   204→    const trimmed = inputValue.trim();
   205→   205→    if (trimmed) onSearch(trimmed);
   206→   206→  };
   207→   207→
   208→   208→  return (
   209→   209→    <form onSubmit={handleSubmit} className="flex items-center gap-2 flex-1 min-w-0">
   210→   210→      <button
   211→   211→        type="button"
   212→   212→        onClick={onBack}
   213→   213→        className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl hover:bg-kwik-bg-light transition-colors"
   214→   214→        aria-label="Go back"
   215→   215→      >
   216→   216→        <ArrowLeft className="h-4 w-4 text-kwik-dark-medium" />
   217→   217→      </button>
   218→   218→
   219→   219→      <div className="relative flex-1 min-w-0">
   220→   220→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-kwik-muted" />
   221→   221→        <input
   222→   222→          type="text"
   223→   223→          value={inputValue}
   224→   224→          onChange={(e) => setInputValue(e.target.value)}
   225→   225→          placeholder="Search products, stores, categories..."
   226→   226→          className="w-full h-9 rounded-xl border border-kwik-border bg-kwik-bg-surface pl-9 pr-9 text-sm text-kwik-dark placeholder:text-kwik-muted outline-none focus:border-kwik-orange focus:ring-1 focus:ring-kwik-orange transition-colors"
   227→   227→          autoFocus
   228→   228→        />
   229→   229→        {inputValue && (
   230→   230→          <button
   231→   231→            type="button"
   232→   232→            onClick={() => setInputValue("")}
   233→   233→            className="absolute right-3 top-1/2 -translate-y-1/2 text-kwik-muted hover:text-kwik-dark-medium"
   234→   234→          >
   235→   235→            <X className="h-3.5 w-3.5" />
   236→   236→          </button>
   237→   237→        )}
   238→   238→      </div>
   239→   239→
   240→   240→      <button
   241→   241→        type="button"
   242→   242→        onClick={onToggleFilters}
   243→   243→        className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-xl transition-colors ${
   244→   244→          showFilters ? "bg-kwik-orange-tint text-kwik-orange" : "hover:bg-kwik-bg-light text-kwik-dark-medium"
   245→   245→        }`}
   246→   246→        aria-label="Toggle filters"
   247→   247→      >
   248→   248→        <SlidersHorizontal className="h-4 w-4" />
   249→   249→      </button>
   250→   250→    </form>
   251→   251→  );
   252→   252→}

   326→   326→  useEffect(() => {
   327→   327→    setIsClientMounted(true);

   355→   355→

   358→   358→

   362→   362→      const anchor = target?.closest?.("a[href]") as HTMLAnchorElement | null;
   363→   363→      if (!anchor || event.defaultPrevented) return;
   364→   364→      if (anchor.target && anchor.target !== "_self") return;
   365→   365→
   366→   366→      const nextUrl = new URL(anchor.href, window.location.href);
   367→   367→      if (nextUrl.origin !== window.location.origin) return;
   368→   368→      if (nextUrl.pathname === window.location.pathname && nextUrl.search === window.location.search) return;
   369→   369→
   370→   370→      startNavigationLoading();
   371→   371→    };
   372→   372→
   373→   373→    document.addEventListener("click", handleDocumentClick, true);
   374→   374→    return () => document.removeEventListener("click", handleDocumentClick, true);

   406→   406→
   407→   407→  const closeDrawer = useCallback(() => setIsDrawerOpen(false), []);
   408→   408→
   409→   409→  const handleSearchSubmit = useCallback(

   430→   430→      if (!showFilters) {
   431→   431→        params.set("filters", "true");
   432→   432→      } else {
   433→   433→        params.delete("filters");

   439→   439→
   440→   440→  return (

   450→   450→        <OfflineBanner />
   451→   451→        {!isSearchPage && !isVendorStorefrontRoute && (

   453→   453→            isOpen={isSearchOpen}
   454→   454→            onClose={() => setIsSearchOpen(false)}

   457→   457→
   458→   458→        {!hideTopNav && (

   460→   460→            <header

   462→   462→                isScrolled
