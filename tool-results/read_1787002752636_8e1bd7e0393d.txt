    34→function formatNGN(n: number) {
    35→  return new Intl.NumberFormat("en-NG", {
    36→    style: "currency",
    37→    currency: "NGN",
    38→    maximumFractionDigits: 0,
    39→  }).format(n);

    42→/**
    43→ * Generates a reasonably-unique idempotency key per checkout attempt.
    44→ * The backend uses it to dedupe a ParentCheckout (and its per-vendor
    45→ * Orders) so a network retry returns the original result instead of
    46→ * double-creating orders. We regenerate it whenever the user starts a
    47→ * new checkout submission (not on every render).
    48→ */
    49→function newIdempotencyKey(): string {
    50→  if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
    51→    return `chk_${crypto.randomUUID()}`;
    52→  }
    53→  return `chk_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    54→}

    56→// ─── Delivery options (NEW spec: Pickup + Standard Delivery ONLY) ──────────
    57→//
    58→// The old STANDARD / EXPRESS / PICKUP trio plus state-based price grid has
    59→// been removed. The backend is authoritative for every price/fee — at
    60→// checkout time we only tell it which delivery method the buyer chose.
    61→//   • PICKUP          → quoteStatus starts as AGREED (no fee to quote),
    62→//                       buyer can initialize payment immediately.
    63→//   • STANDARD_DELIVERY → quoteStatus starts as PENDING_VENDOR_QUOTE,
    64→//                       vendor must quote → customer accepts/requests
    65→//                       reduction → agreement → then payment.
    66→
    67→type DeliveryMethod = "PICKUP" | "STANDARD_DELIVERY";

   324→            name: i.name,
   325→            unitPrice: i.price,

   329→          vendorId: group.storeName,
   330→          vendorName: group.storeName,
   331→          deliveryAddress:
   332→            deliveryMethod === "STANDARD_DELIVERY"
   333→              ? `${address.fullName}, ${address.addressLine}, ${address.city}, ${address.state}. Phone: ${address.phone}${address.landmark ? `. Landmark: ${address.landmark}` : ""}`
   334→              : "PICKUP — buyer collects from vendor's store",
   335→        });
   336→      }
   337→
   338→      clearCart();
   339→
   340→      const orderCount = apiOrders.length || storeCount;
   341→      const isPickup = deliveryMethod === "PICKUP";
   342→      kwikToast.success(
   343→        "Order placed!",
   344→        isPickup
   345→          ? `${orderCount} order${orderCount > 1 ? "s" : ""} placed. You can initialize payment now.`
   346→          : `${orderCount} order${orderCount > 1 ? "s" : ""} sent to the vendor${orderCount > 1 ? "s" : ""} for a delivery quote. You'll be notified once they respond.`,
   347→      );
   348→

   350→      // page and send the buyer straight to the payment gateway.
   351→      if (isPickup) {
   352→        const firstOrderId = apiOrders[0]?.id;
   353→        if (firstOrderId) {
   354→          try {
   355→            const payRes = await api.post<{ authorizationUrl?: string }>(
   356→              `orders/${firstOrderId}/initialize-payment`,
   357→            );
   358→            const authorizationUrl = payRes.data?.authorizationUrl;
   359→            if (authorizationUrl) {
   360→              kwikToast.info(
   361→                "Redirecting to payment",
   362→                "Complete your payment to confirm the order.",
   363→              );
   364→              window.location.href = authorizationUrl;
   365→              return;
   366→            }
   367→          } catch (err) {
   368→            const message =
   369→              err instanceof Error ? err.message : "Failed to initialize payment.";
   370→            kwikToast.error(
   371→              "Payment initialization failed",
   372→              `${message} You can retry from the order page.`,
   373→            );
   374→          }
   375→          router.push(`/orders/${firstOrderId}`);
   376→        } else {
   377→          router.push("/orders");
   378→        }

   380→      }

   382→      // STANDARD_DELIVERY: redirect to the first API order's detail page if

   385→        router.push(`/orders/${apiOrders[0].id}`);
   386→      } else {
   387→        router.push("/orders");
   388→      }
   389→    } catch (err) {
   390→      const message =
   391→        err instanceof Error
   392→          ? err.message
   393→          : "Something went wrong. Please try again.";
   394→      kwikToast.error("Checkout failed", message);
   395→    }
   396→  }

   402→        <div className="flex flex-col items-center gap-3 text-center">

   452→      {/* ── Hero header ── */}

   461→            <p className="text-[11px] font-bold uppercase tracking-[0.18em] text-white/80">
   462→              Secure checkout

   474→      </section>
   475→
   476→      <section className="container mx-auto max-w-7xl px-4 py-8">
   477→        <div className="grid gap-8 lg:grid-cols-[1fr_400px]">
   478→          {/* ── Left column: Products → Delivery Option ── */}
   479→          <div className="space-y-6">
   480→            {/* ── 1. PRODUCTS ── */}
   481→            <section
   482→              aria-labelledby="checkout-products-heading"
   483→              className="rounded-2xl border border-border bg-card p-5 md:p-6"
