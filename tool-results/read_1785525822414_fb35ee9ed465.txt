    30→/**
    31→ * The api-client wraps responses in ApiResponse<T> = { success, data, ... }.
    32→ * The backend response interceptor also wraps. This helper handles every
    33→ * shape we've seen (data array nested 1 or 2 levels deep).
    34→ */
    35→function unwrapOrders(value: unknown): Order[] {
    36→  const payload = value as { data?: unknown } | undefined;
    37→  const nested = payload?.data as { data?: unknown; orders?: unknown } | undefined;
    38→  const data = nested?.data ?? nested?.orders ?? payload?.data ?? value;
    39→  if (Array.isArray(data)) return data as Order[];
    40→  const objectData = data as { orders?: unknown; data?: unknown } | undefined;
    41→  if (Array.isArray(objectData?.orders)) return objectData.orders as Order[];
    42→  if (Array.isArray(objectData?.data)) return objectData.data as Order[];
    43→  return [];
    44→}

    46→function filterByTab(orders: Order[], tab: TabKey): Order[] {
    47→  if (tab === "all") return orders;
    48→  const allowed =
    49→    tab === "active"
    50→      ? ACTIVE_STATUSES
    51→      : tab === "completed"
    52→        ? COMPLETED_STATUSES
    53→        : CANCELLED_STATUSES;
    54→  return orders.filter((order) => allowed.includes(order.status));
    55→}

    77→function getDeliveryAddress(order: Order): string | undefined {
    78→  const delivery = order.delivery as { deliveryAddress?: string } | undefined;
    79→  if (delivery?.deliveryAddress) return delivery.deliveryAddress;
    80→  const address = order.address;
    81→  if (address) {
    82→    return [address.line1, address.city, address.state]
    83→      .filter(Boolean)
    84→      .join(", ");
    85→  }
    86→  if (order.deliveryLocalGovernment && order.deliveryState) {
    87→    return `${order.deliveryLocalGovernment}, ${order.deliveryState}`;
    88→  }
    89→  return undefined;

   107→/**
   108→ * Determine which of the 4 macro workflow phases an order is in.
   109→ * Used for the mini progress bar on order cards.
   110→ */
   111→function getWorkflowPhase(status: string): { phase: number; label: string } {
   112→  const s = status as keyof typeof OrderStatus;
   113→  const quoteStatuses = [OrderStatus.PENDING_QUOTE, OrderStatus.QUOTED, OrderStatus.TO_PAY];
   114→  const payStatuses = [OrderStatus.PAID];
   115→  const shipStatuses = [OrderStatus.PROCESSING, OrderStatus.SHIPPED, OrderStatus.OUT_FOR_DELIVERY, OrderStatus.DELIVERED];
   116→  const receiveStatuses = [OrderStatus.RECEIVED, OrderStatus.COMPLETED];
   117→
   118→  if (quoteStatuses.includes(s as any)) return { phase: 1, label: "Quotation" };
   119→  if (payStatuses.includes(s as any)) return { phase: 2, label: "Payment" };
   120→  if (shipStatuses.includes(s as any)) return { phase: 3, label: "Shipping" };
   121→  if (receiveStatuses.includes(s as any)) return { phase: 4, label: "Received" };
   122→  return { phase: 0, label: "—" };
   123→}

   125→/**
   126→ * Mini 4-step progress bar for order cards (matches the order detail page's
   127→ * OrderProgressBar but compact).
   128→ */
   129→function MiniProgressBar({ status }: { status: string }) {
   130→  const { phase } = getWorkflowPhase(status);
   131→  const isAbnormal = status === "DISPUTED" || status === "CANCELLED" || status === "RETURNED";
   132→  const steps = ["Quote", "Pay", "Ship", "Receive"];
   133→  const fillWidth = isAbnormal ? 100 : (phase / 4) * 100;
   134→
   135→  return (
   136→    <div className="flex items-center gap-1.5">
   137→      {steps.map((step, idx) => {
   138→        const reached = !isAbnormal && phase > idx;
   139→        const isCurrent = !isAbnormal && phase === idx + 1;
   140→        return (
   141→          <div key={step} className="flex items-center gap-1">
   142→            <div
   143→              className={cn(
   144→                "flex h-5 w-5 items-center justify-center rounded-full text-[9px] font-bold transition",
   145→                reached
   146→                  ? "bg-primary-500 text-white"
   147→                  : isCurrent
   148→                    ? "bg-primary-100 text-primary-700 ring-2 ring-primary-500/30"
   149→                    : "bg-gray-100 text-gray-400",
   150→                isAbnormal && "bg-warning text-white",
   151→              )}
   152→            >
   153→              {reached ? "✓" : idx + 1}
   154→            </div>
   155→            {idx < steps.length - 1 && (
   156→              <div
   157→                className={cn(
   158→                  "h-0.5 w-4 rounded-full",
   159→                  reached ? "bg-primary-500" : "bg-gray-200",
   160→                )}
   161→              />
   162→            )}
   163→          </div>
   164→        );
   165→      })}
   166→    </div>
   167→  );
   168→}

   266→/**
   267→ * Convert an ApiOrder (created via POST /checkout in dummy mode, or by the

   269→ * placed via the checkout flow appear in the buyer's order list even when

   272→function apiOrderToOrder(o: ApiOrder): Order {
   273→  return {
   274→    id: o.id,
   275→    checkoutReference: o.orderNumber,
   276→    status: o.status,
   277→    paymentStatus: o.paymentStatus,
   278→    totalAmount: o.total,
   279→    createdAt: o.createdAt,
   280→    updatedAt: o.updatedAt,
   281→    store: {
   282→      id: o.storeId,
   283→      name: o.storeName,
   284→    },
   285→    items: o.items.map((item) => ({
   286→      id: item.id,
   287→      quantity: item.quantity,
   288→      product: {
   289→        id: item.productId,
   290→        name: item.product.name,
   291→        price: item.unitPrice,
   292→        images: item.product.image ? [{ url: item.product.image, isMain: true }] : [],
   293→      },
   294→    })),
   295→    delivery: { deliveryAddress: `${o.deliveryAddress.fullName}, ${o.deliveryAddress.addressLine1}, ${o.deliveryAddress.city}, ${o.deliveryAddress.state}` },
   296→  } as Order;

   299→/**
   300→ * Convert an OrderWorkflowState (from the Zustand mock store) into the Order
   301→ * shape this page expects, so the orders list is useful even when the live API
   302→ * is unreachable / the buyer is not authenticated.
   303→ */
   304→function mockOrderToOrder(state: OrderWorkflowState): Order {
   305→  const itemsTotal = state.items.reduce(
   306→    (sum, i) => sum + i.unitPrice * i.quantity,
   307→    0,
   308→  );
   309→  const deliveryFee = state.escrow?.amount
   310→    ? Math.max(0, state.escrow.amount - itemsTotal)
   311→    : state.quotation?.deliveryFee ?? 0;
   312→  const discount = state.quotation?.discount ?? 0;
   313→  const total = itemsTotal + deliveryFee - discount;
   314→
   315→  const paymentStatus =
   316→    state.status === "PAID" ||
   317→    state.status === "PROCESSING" ||
   318→    state.status === "SHIPPED" ||
   319→    state.status === "OUT_FOR_DELIVERY" ||
   320→    state.status === "DELIVERED" ||
   321→    state.status === "RECEIVED" ||
   322→    state.status === "COMPLETED"
   323→      ? "PAID"
   324→      : state.status === "CANCELLED" || state.status === "RETURNED"
   325→        ? "REFUNDED"
   326→        : "PENDING";
   327→
   328→  return {
   329→    id: state.id,
   330→    checkoutReference: state.ref,
   331→    status: state.status,
   332→    paymentStatus,
   333→    totalAmount: total,
   334→    createdAt: state.createdAt,
   335→    updatedAt: state.updatedAt,
   336→    store: {
   337→      id: state.vendor.id,
   338→      name: state.vendor.name,
   339→      logoUrl: state.vendor.logo ?? undefined,
   340→    },
   341→    items: state.items.map((item) => ({
   342→      id: item.productId,
   343→      quantity: item.quantity,
   344→      product: {
   345→        id: item.productId,
   346→        name: item.name,
   347→        price: item.unitPrice,
   348→        images: item.image ? [{ url: item.image, isMain: true }] : [],
   349→      },
   350→    })),
   351→    delivery: { deliveryAddress: state.deliveryAddress },
   352→  } as Order;
   353→}

   396→/**
   397→ * Build a CSV row list from an array of orders. Used by the "Export" buttons
   398→ * on the orders page so buyers can download their full order history.
   399→ */
   400→function ordersToCSVRows(orders: Order[]) {
   401→  return orders.map((o) => {
   402→    const delivery = o.delivery as { deliveryAddress?: string; carrier?: string } | undefined;
   403→    const payment = o.payment as { method?: string; gateway?: string } | undefined;
   404→    return {
   405→      "Order #": getOrderRef(o),
   406→      Status: o.status,
   407→      "Payment status": o.paymentStatus,
   408→      "Vendor / store": getStoreName(o) ?? "",
   409→      "Items": getItemCount(o),
   410→      "Item names": getItemNames(o).join("; "),
   411→      "Subtotal (NGN)": Number(o.subtotal ?? 0),
   412→      "Discount (NGN)": Number(o.discount ?? 0),
   413→      "Shipping fee (NGN)": Number(o.shippingFee ?? 0),
   414→      "Total (NGN)": Number(o.totalAmount ?? 0),
   415→      "Payment method": payment?.method ?? payment?.gateway ?? "",
   416→      "Delivery address": getDeliveryAddress(o) ?? delivery?.deliveryAddress ?? "",
   417→      "Carrier": delivery?.carrier ?? "",
   418→      "Placed at": formatDate(o.createdAt),
   419→      "Updated at": formatDate(o.updatedAt),
   420→    };
   421→  });
   422→}

   424→function BuyerOrdersPageInner() {
   425→  const router = useRouter();
   426→  const { isAuthenticated, isLoading: isAuthLoading } = useAuth();
   427→  const [activeTab, setActiveTab] = useState<TabKey>("all");
   428→
   429→  // Mock orders from the Zustand order-workflow store (always available).
   430→  const mockOrders = useOrderWorkflowStore((s) => s.orders);
   431→
   432→  // API orders — fetched via the shared `useMyOrders` hook which works in
   433→  // dummy mode without auth. These are orders created via POST /checkout
   434→  // (the real checkout → vendor flow). Polled every 5s while any are PENDING.
   435→  const { data: dummyApiOrders = [], isLoading: dummyLoading } = useMyOrders();
   436→
   437→  const { data: liveApiOrders = [], isLoading } = useQuery<Order[]>({
   438→    queryKey: ["buyer-orders"],
   439→    enabled: !isAuthLoading && isAuthenticated,
   440→    queryFn: async () => {
   441→      try {
   442→        const response = await ordersApi.list({ limit: 50 });
   443→        return unwrapOrders(response);
   444→      } catch (err) {
   445→        const message =
   446→          err instanceof Error ? err.message : "Failed to load orders.";
   447→        kwikToast.error("Couldn't load your orders", message);
   448→        throw err;
   449→      }
   450→    },
   451→    staleTime: 30 * 1000,
   452→  });
   453→
   454→  // Use live API orders when authenticated + available; otherwise merge
   455→  // dummy-API orders (created via checkout) with the mock workflow orders
   456→  // so the page is always useful (demo / no-backend).
   457→  const orders = useMemo<Order[]>(() => {
   458→    if (isAuthenticated && liveApiOrders.length > 0) return liveApiOrders;
   459→    const apiMapped = dummyApiOrders.map(apiOrderToOrder);
   460→    const mockMapped = mockOrders.map(mockOrderToOrder);
   461→    // De-duplicate by id (an order may exist in both stores).
   462→    const seen = new Set(apiMapped.map((o) => o.id));
   463→    return [...apiMapped, ...mockMapped.filter((o) => !seen.has(o.id))];
   464→  }, [isAuthenticated, liveApiOrders, dummyApiOrders, mockOrders]);
   465→
   466→  const isOrdersLoading = isLoading || dummyLoading;
   467→
   468→  const filteredOrders = useMemo(
   469→    () => filterByTab(orders, activeTab),
   470→    [orders, activeTab],
   471→  );
   472→
   473→  const counts = useMemo(
   474→    () => ({
   475→      all: orders.length,
   476→      active: filterByTab(orders, "active").length,
   477→      completed: filterByTab(orders, "completed").length,
   478→      cancelled: filterByTab(orders, "cancelled").length,
   479→    }),
   480→    [orders],
   481→  );
   482→
   483→  /**
   484→   * Export orders to CSV. `scope` controls whether we export the current
   485→   * filtered tab view or the entire order history regardless of filter.
   486→   */
   487→  function handleExportCSV(scope: "filtered" | "all") {
   488→    const rows = scope === "filtered" ? filteredOrders : orders;
   489→    if (!rows.length) {
   490→      kwikToast.info("Nothing to export", "Place an order first to download your history.");
   491→      return;
   492→    }
   493→    const csv = toCSV(ordersToCSVRows(rows));
   494→    const date = new Date().toISOString().slice(0, 10);
   495→    const suffix = scope === "filtered" && activeTab !== "all" ? `-${activeTab}` : "";
   496→    downloadCSV(`kwikseller-orders${suffix}-${date}`, csv);
   497→    kwikToast.success(
   498→      "Export ready",
   499→      `${rows.length} order${rows.length === 1 ? "" : "s"} downloaded as CSV.`,
   500→    );
   501→  }
   502→

   505→    return (
   506→      <div className="mx-auto max-w-4xl px-4 py-8">
   507→        <OrderListSkeleton />
   508→      </div>
   509→    );
   510→  }
   511→
   512→  const isDemoMode = !isAuthenticated || orders.length === 0;
   513→
   514→  return (
   515→    <div className="min-h-screen bg-background">
   516→      {/* Hero header (matches order detail page design) */}
   517→      <section className="kwik-gradient relative overflow-hidden">
   518→        <div className="absolute inset-0 bg-[radial-gradient(circle_at_top_right,rgba(255,255,255,0.18),transparent_55%)]" />
   519→        <div className="container mx-auto max-w-4xl px-4 py-8 relative">
   520→          {/* Back link */}
   521→          <Link
   522→            href="/"
   523→            className="mb-4 inline-flex items-center gap-1.5 text-sm font-medium text-white/85 transition-colors hover:text-white"
   524→          >
   525→            <ArrowLeft className="h-4 w-4" />
   526→            Back to marketplace
   527→          </Link>
   528→
   529→          <motion.div
   530→            initial={{ opacity: 0, y: 8 }}
   531→            animate={{ opacity: 1, y: 0 }}
   532→            transition={{ duration: 0.3 }}
   533→          >

   536→            </p>
   537→            <h1 className="mt-1 font-heading text-3xl font-bold text-white sm:text-4xl">
   538→              My Orders
   539→            </h1>
   540→            <p className="mt-2 max-w-2xl text-sm leading-6 text-white/85">
   541→              Track, confirm, and manage all your marketplace purchases. Every
   542→              order is protected by {KwisCrow.NAME} escrow.
   543→            </p>
   544→          </motion.div>
   545→        </div>
   546→      </section>
   547→
   548→      <motion.div
   549→        initial={{ opacity: 0, y: 10 }}
   550→        animate={{ opacity: 1, y: 0 }}
   551→        transition={{ duration: 0.3, delay: 0.1 }}
   552→        className="mx-auto max-w-4xl px-4 py-6"
   553→      >
   554→
   555→      {/* Demo-mode banner */}
   556→      {isDemoMode ? (
   557→        <div className="mb-6 flex items-start gap-3 rounded-xl border border-kwik-orange/20 bg-kwik-orange/5 p-4">
   558→          <Sparkles className="mt-0.5 h-5 w-5 shrink-0 text-kwik-orange" />
   559→          <div className="text-sm leading-5 text-kwik-dark">
   560→            <span className="font-semibold">Live + demo orders.</span>{" "}
   561→            Orders you place via checkout appear here in real time (vendor
   562→            quotation → escrow → shipping). Seeded demo orders show the full
   563→            workflow. Sign in to sync with your real account.
   564→          </div>
   565→        </div>
   566→      ) : null}
   567→
   568→      {/* Stats summary bar + export controls */}
   569→      {orders.length > 0 ? (
   570→        <div className="mb-6 flex flex-col gap-3 sm:flex-row sm:items-stretch">
   571→          <div className="grid flex-1 grid-cols-2 gap-3 rounded-2xl border border-border bg-surface p-4 sm:grid-cols-4">
   572→            <div className="flex items-center gap-2.5">
   573→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-primary-50 text-primary-600">

   575→              </div>
   576→              <div>
   577→                <p className="font-heading text-lg font-bold text-foreground">{orders.length}</p>
   578→                <p className="text-[11px] text-gray-500">Total orders</p>
   579→              </div>
   580→          </div>
   581→          <div className="flex items-center gap-2.5">
   582→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-secondary-50 text-secondary-600">
   583→              <Clock className="h-4 w-4" />
   584→            </div>
   585→            <div>
   586→              <p className="font-heading text-lg font-bold text-foreground">{counts.active}</p>
   587→              <p className="text-[11px] text-gray-500">Active</p>
   588→            </div>
   589→          </div>
   590→          <div className="flex items-center gap-2.5">
   591→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-success/10 text-success">
   592→              <CheckCircle2 className="h-4 w-4" />
   593→            </div>
   594→            <div>
   595→              <p className="font-heading text-lg font-bold text-foreground">{counts.completed}</p>
   596→              <p className="text-[11px] text-gray-500">Completed</p>
   597→            </div>
   598→          </div>
   599→          <div className="flex items-center gap-2.5">
   600→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-warning/10 text-warning">
   601→              <Wallet className="h-4 w-4" />
   602→            </div>

   605→                {formatCurrency(
   606→                  orders.reduce((sum, o) => sum + Number(o.totalAmount ?? 0), 0),
   607→                )}
   608→              </p>
   609→              <p className="text-[11px] text-gray-500">Total value</p>
   610→            </div>
   611→          </div>
   612→          </div>
   613→

   615→          <div className="flex shrink-0 flex-col justify-center gap-2 rounded-2xl border border-border bg-surface p-4 sm:flex-row sm:items-center">

   617→            <button
   618→              type="button"
   619→              onClick={() => handleExportCSV("filtered")}
   620→              disabled={filteredOrders.length === 0}
   621→              className="inline-flex h-9 items-center justify-center gap-1.5 rounded-lg border border-kwik-border-light bg-background px-3 text-xs font-semibold text-kwik-dark transition-colors hover:border-kwik-orange hover:text-kwik-orange disabled:cursor-not-allowed disabled:opacity-50"
   622→              aria-label="Export filtered orders to CSV"
   623→            >
   624→              <Download className="h-3.5 w-3.5" />
   625→              {activeTab === "all" ? "All orders" : `${tabLabel(activeTab)} only`}
   626→            </button>
   627→            <button
   628→              type="button"
   629→              onClick={() => handleExportCSV("all")}
   630→              disabled={orders.length === 0}
   631→              className="inline-flex h-9 items-center justify-center gap-1.5 rounded-lg bg-kwik-orange px-3 text-xs font-semibold text-white transition-colors hover:bg-kwik-orange-hover disabled:cursor-not-allowed disabled:opacity-50"
   632→              aria-label="Export all orders to CSV"
   633→            >
   634→              <FileSpreadsheet className="h-3.5 w-3.5" />
   635→              Full history
   636→            </button>
   637→          </div>
   638→        </div>
   639→      ) : null}
   640→
   641→      {/* Filter tabs */}
   642→      <div
   643→        role="tablist"
   644→        aria-label="Filter orders by status"
   645→        className="mb-6 flex flex-wrap gap-2"
   646→      >
   647→        {TABS.map((tab) => {
   648→          const isActive = activeTab === tab.key;
   649→          return (
   650→            <button
   651→              key={tab.key}
   652→              role="tab"
   653→              aria-selected={isActive}
   654→              type="button"
   655→              onClick={() => setActiveTab(tab.key)}
   656→              className={
   657→                "inline-flex items-center gap-2 rounded-full border px-4 py-2 text-sm font-semibold transition-colors " +

   659→                  ? "border-accent bg-accent text-accent-foreground"
   660→                  : "border-kwik-border bg-surface text-muted-foreground hover:border-accent/40 hover:text-foreground")

   662→            >
   663→              {tab.label}
   664→              <span
   665→                className={
   666→                  "inline-flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[11px] tabular-nums " +
   667→                  (isActive
   668→                    ? "bg-accent-foreground/15 text-accent-foreground"
   669→                    : "bg-default-100 text-muted-foreground")
   670→                }
   671→              >
   672→                {counts[tab.key]}
   673→              </span>
   674→            </button>
   675→          );
   676→        })}
   677→      </div>
   678→
   679→      {/* Content */}
   680→      {isOrdersLoading ? (
   681→        <OrderListSkeleton />
   682→      ) : filteredOrders.length === 0 ? (
   683→        <EmptyState
   684→          variant="orders"
   685→          title={
   686→            activeTab === "all"
   687→              ? "You haven't placed any orders yet"
   688→              : activeTab === "active"
   689→                ? "No active orders"
   690→                : activeTab === "completed"
   691→                  ? "No completed orders yet"
   692→                  : "No cancelled orders"
   693→          }
   694→          description={
   695→            activeTab === "all"
   696→              ? "When you make a purchase, your orders will appear here so you can track delivery, confirm receipt, or open a dispute."
   697→              : "Try switching tabs to see orders in other states."
   698→          }
   699→          action={{
   700→            label: "Start shopping",
   701→            onClick: () => router.push("/"),
   702→          }}
   703→          className="rounded-2xl border border-kwik-border bg-surface"
   704→        />
   705→      ) : (
   706→        <div className="space-y-4">
   707→          {filteredOrders.map((order) => (
   708→            <BuyerOrderCard
   709→              key={order.id}
   710→              order={order}
   711→              onClick={() => router.push(`/orders/${order.id}`)}
   712→            />
   713→          ))}
   714→        </div>
   715→      )}
   716→
   717→      {/* Reorder hint footer (only shown when there are orders) */}
   718→      {filteredOrders.length > 0 && (
   719→        <p className="mt-8 flex items-center justify-center gap-1.5 text-center text-xs text-muted-foreground">
   720→          <Package className="h-3.5 w-3.5" />
   721→          Tap an order to view full details, track delivery, or open a dispute.
   722→        </p>
   723→      )}
   724→      </motion.div>
   725→    </div>
   726→  );
   727→}
