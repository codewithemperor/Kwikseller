    36→function formatNGN(n: number) {
    37→  return new Intl.NumberFormat("en-NG", {
    38→    style: "currency",
    39→    currency: "NGN",
    40→    maximumFractionDigits: 0,
    41→  }).format(n);
    42→}

    60→export default function CheckoutPage() {
    61→  const router = useRouter();
    62→  const items = useCartStore((s) => s.items);
    63→  const updateQuantity = useCartStore((s) => s.updateQuantity);
    64→  const removeItem = useCartStore((s) => s.removeItem);
    65→  const clearCart = useCartStore((s) => s.clearCart);
    66→  const placeOrder = useOrderWorkflowStore((s) => s.placeOrder);
    67→  const checkout = useCheckout();
    68→
    69→  const [address, setAddress] = useState<AddressForm>({
    70→    fullName: "",
    71→    phone: "",
    72→    addressLine: "",
    73→    city: "",
    74→    state: "Lagos",
    75→    landmark: "",
    76→  });
    77→  const [paymentProvider, setPaymentProvider] =
    78→    useState<PaymentProviderKey>(DEFAULT_PAYMENT_PROVIDER);
    79→  const [placing, setPlacing] = useState(false);
    80→  const [errors, setErrors] = useState<Partial<Record<keyof AddressForm, string>>>({});
    81→
    82→  // ── Coupon state (TODO: coupon application in checkout) ──
    83→  const [couponCode, setCouponCode] = useState("");
    84→  const [appliedCoupon, setAppliedCoupon] = useState<{
    85→    code: string;
    86→    discountType: "PERCENT" | "AMOUNT";
    87→    discountValue: number;
    88→    message: string;
    89→  } | null>(null);
    90→  const [couponLoading, setCouponLoading] = useState(false);
    91→
    92→  async function applyCoupon() {
    93→    const code = couponCode.trim().toUpperCase();
    94→    if (!code) {
    95→      kwikToast.error("Enter a coupon code", "Type a code then tap Apply.");
    96→      return;
    97→    }
    98→    setCouponLoading(true);
    99→    try {
   100→      const res = await api.post<{ code: string; valid: boolean; discountType: "PERCENT" | "AMOUNT"; discountValue: number; message: string }>(
   101→        "cart/coupon",
   102→        { code },
   103→      );
   104→      const data = res.data;
   105→      if (!data?.valid) {
   106→        setAppliedCoupon(null);
   107→        kwikToast.error("Invalid coupon", data?.message || "This code is not valid or has expired.");
   108→        return;
   109→      }
   110→      setAppliedCoupon(data);
   111→      kwikToast.success("Coupon applied", data.message || `${data.discountValue}${data.discountType === "PERCENT" ? "%" : " NGN"} off your order.`);
   112→    } catch (e) {
   113→      setAppliedCoupon(null);
   114→      kwikToast.error("Invalid coupon", e instanceof Error ? e.message : "This code is not valid or has expired.");
   115→    } finally {
   116→      setCouponLoading(false);
   117→    }
   118→  }
   119→
   120→  function removeCoupon() {
   121→    setAppliedCoupon(null);
   122→    setCouponCode("");
   123→    kwikToast.info("Coupon removed", "The discount has been cleared.");
   124→  }
   125→
   126→  // Group items by store (since the 1688 workflow sends each vendor a separate order).
   127→  const itemsByStore = useMemo(() => {
   128→    const groups: Record<string, { storeName: string; items: typeof items; subtotal: number }> =
   129→      {};
   130→    for (const item of items) {
   131→      const key = item.storeSlug ?? item.store ?? "unknown";
   132→      if (!groups[key]) {
   133→        groups[key] = {
   134→          storeName: item.store ?? item.storeName ?? "Vendor",
   135→          items: [],
   136→          subtotal: 0,
   137→        };
   138→      }
   139→      groups[key].items.push(item);
   140→      groups[key].subtotal += item.price * item.quantity;
   141→    }
   142→    return groups;
   143→  }, [items]);
   144→
   145→  const subtotal = useMemo(
   146→    () => items.reduce((sum, i) => sum + i.price * i.quantity, 0),
   147→    [items],
   148→  );
   149→  const storeCount = Object.keys(itemsByStore).length;
   150→  // Estimated platform fee (demo) — the real delivery fee is set by the vendor
   151→  // when they submit the quotation (see the 1688 workflow).
   152→  const estimatedProcessingFee = Math.round(subtotal * 0.015);
   153→  // Coupon discount applied at checkout (TODO: coupon application).
   154→  const couponDiscount = useMemo(() => {
   155→    if (!appliedCoupon) return 0;
   156→    return appliedCoupon.discountType === "PERCENT"
   157→      ? Math.round((subtotal * appliedCoupon.discountValue) / 100)
   158→      : Math.min(appliedCoupon.discountValue, subtotal);
   159→  }, [appliedCoupon, subtotal]);
   160→  const totalDueNow = Math.max(0, subtotal - couponDiscount + estimatedProcessingFee);
   161→
   162→  function validate(): boolean {
   163→    const next: Partial<Record<keyof AddressForm, string>> = {};
   164→    if (!address.fullName.trim()) next.fullName = "Full name is required";
   165→    if (!address.phone.trim()) next.phone = "Phone number is required";
   166→    else if (!/^[0-9+\s-]{7,}$/.test(address.phone.trim()))
   167→      next.phone = "Enter a valid phone number";
   168→    if (!address.addressLine.trim()) next.addressLine = "Street address is required";
   169→    if (!address.city.trim()) next.city = "City is required";
   170→    if (!address.state.trim()) next.state = "State is required";
   171→    setErrors(next);
   172→    return Object.keys(next).length === 0;
   173→  }
   174→
   175→  async function handlePlaceOrder() {
   176→    if (items.length === 0) {
   177→      kwikToast.error("Your cart is empty", "Add products before checking out.");
   178→      return;
   179→    }
   180→    if (!validate()) {
   181→      kwikToast.error("Missing delivery details", "Please complete the delivery address.");
   182→      return;
   183→    }
   184→
   185→    setPlacing(true);
   186→
   187→    try {
   188→      // POST to the backend so the vendor actually RECEIVES the order
   189→      // (TODO #6). The backend groups items by vendor store and creates
   190→      // one order per store (split checkout), each linked to the vendor.
   191→      const result = await checkout.mutateAsync({
   192→        items: items.map((i) => ({
   193→          productId: i.productId,
   194→          quantity: i.quantity,
   195→        })),
   196→        shippingAddress: {
   197→          fullName: address.fullName,
   198→          phone: address.phone,
   199→          addressLine1: address.addressLine,
   200→          addressLine2: address.landmark,
   201→          city: address.city,
   202→          state: address.state,
   203→          country: "Nigeria",
   204→        },
   205→        paymentMethod: PAYMENT_PROVIDERS[paymentProvider]?.id ?? "CARD",
   206→        deliveryType: "STANDARD",
   207→        couponCode: appliedCoupon?.code,
   208→      });
   209→
   210→      const apiOrders = result?.orders ?? [];
   211→
   212→      // Also mirror into the local workflow store so the rich quotation UI
   213→      // (timeline, escrow badge) keeps working for these orders.
   214→      for (const group of Object.values(itemsByStore)) {
   215→        placeOrder({
   216→          items: group.items.map((i) => ({
   217→            productId: i.productId,
   218→            name: i.name,
   219→            unitPrice: i.price,
   220→            quantity: i.quantity,
   221→            image: i.image,
   222→          })),
   223→          vendorId: group.storeName,
   224→          vendorName: group.storeName,
   225→          deliveryAddress: `${address.fullName}, ${address.addressLine}, ${address.city}, ${address.state}. Phone: ${address.phone}${address.landmark ? `. Landmark: ${address.landmark}` : ""}`,
   226→        });
   227→      }
   228→
   229→      clearCart();
   230→      const count = apiOrders.length || Object.keys(itemsByStore).length;
   231→      kwikToast.success(
   232→        "Order placed!",
   233→        `${count} order${count > 1 ? "s" : ""} sent to vendor${count > 1 ? "s" : ""} for quotation. The vendor will confirm delivery & discount shortly.`,
   234→      );
   235→
   236→      // Redirect to the first API order's detail page (status: PENDING →
   237→      // vendor quotes delivery/discount → CONFIRMED → buyer pays).
   238→      if (apiOrders.length > 0 && apiOrders[0]?.id) {
   239→        router.push(`/orders/${apiOrders[0].id}`);
   240→      } else {
   241→        router.push("/orders");
   242→      }
   243→    } catch (err) {
   244→      setPlacing(false);
   245→      kwikToast.error(
   246→        "Checkout failed",
   247→        err instanceof Error ? err.message : "Please try again.",
   248→      );
   249→    }
   250→  }
   251→
   252→  // ── Empty cart state ──
   253→  if (items.length === 0) {
   254→    return (
   255→      <div className="bg-background min-h-screen">
   256→        <div className="container mx-auto max-w-2xl px-4 py-20 text-center">
   257→          <div className="mx-auto flex h-20 w-20 items-center justify-center rounded-full bg-gray-100">
   258→            <ShoppingBag className="h-10 w-10 text-gray-400" />
   259→          </div>
   260→          <h1 className="mt-6 font-heading text-2xl font-bold text-foreground">
   261→            Your cart is empty
   262→          </h1>
   263→          <p className="mt-2 text-sm text-gray-500">
   264→            Add products to your cart before checking out.
   265→          </p>
   266→          <button
   267→            type="button"
   268→            onClick={() => router.push("/products")}
   269→            className="mt-6 inline-flex h-11 items-center justify-center rounded-xl bg-secondary-500 px-6 text-sm font-semibold text-white hover:bg-secondary-600"
   270→          >
   271→            Browse products
   272→          </button>
   273→        </div>
   274→      </div>
   275→    );
   276→  }
   277→
   278→  return (
   279→    <div className="bg-background min-h-screen">
   280→      {/* ── Hero header (matches order detail page design) ── */}
   281→      <section className="kwik-gradient relative overflow-hidden">
   282→        <div className="absolute inset-0 bg-[radial-gradient(circle_at_top_right,rgba(255,255,255,0.18),transparent_55%)]" />
   283→        <div className="container mx-auto max-w-7xl px-4 py-8 relative">
   284→          <motion.div
   285→            initial={{ opacity: 0, y: 12 }}
   286→            animate={{ opacity: 1, y: 0 }}
   287→            transition={{ duration: 0.35 }}
   288→          >
   289→            <p className="text-[11px] font-bold uppercase tracking-[0.18em] text-white/80">
   290→              1688-style checkout
   291→            </p>
   292→            <h1 className="mt-1 font-heading text-2xl font-bold text-white md:text-4xl">
   293→              Review &amp; place your order
   294→            </h1>
   295→            <p className="mt-2 max-w-2xl text-sm leading-6 text-white/85">
   296→              No payment now — the vendor will send a quotation with the delivery
   297→              fee and ETA. Your funds stay protected by {KwisCrow.NAME} escrow
   298→              until you confirm receipt.
   299→            </p>
   300→
   301→            {/* Workflow steps strip */}
   302→            <div className="mt-4 flex flex-wrap items-center gap-2">
   303→              {[
   304→                { icon: Package, label: "1. Place order", active: true },
   305→                { icon: PackageCheck, label: "2. Vendor quotes", active: false },
   306→                { icon: Wallet, label: "3. Pay (escrow)", active: false },
   307→                { icon: Truck, label: "4. Delivery", active: false },
   308→                { icon: CheckCircle2, label: "5. Confirm", active: false },
   309→              ].map((step, idx) => (
   310→                <div
   311→                  key={idx}
   312→                  className={`flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-semibold backdrop-blur ${
   313→                    step.active
   314→                      ? "bg-white/25 text-white"
   315→                      : "bg-white/10 text-white/70"
   316→                  }`}
   317→                >
   318→                  <step.icon className="h-3.5 w-3.5" />
   319→                  {step.label}
   320→                </div>
   321→              ))}
   322→            </div>
   323→          </motion.div>
   324→        </div>
   325→      </section>
   326→
   327→      <section className="container mx-auto max-w-7xl px-4 py-8">
   328→        <div className="grid gap-8 lg:grid-cols-[1fr_400px]">
   329→          {/* ── Left: address + items + payment ── */}
   330→          <div className="space-y-6">
   331→            {/* Delivery address */}
   332→            <div className="rounded-2xl border border-border bg-surface p-5 md:p-6">
   333→              <div className="mb-4 flex items-center gap-2">
   334→                <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-primary-50 text-primary-600">
   335→                  <MapPin className="h-5 w-5" />
   336→                </div>
   337→                <div>
   338→                  <h2 className="font-heading text-base font-semibold text-foreground">
   339→                    Delivery address
   340→                  </h2>
   341→                  <p className="text-xs text-gray-500">
   342→                    Where should the vendor ship your order?
   343→                  </p>
   344→                </div>
   345→              </div>
   346→
   347→              <div className="grid gap-4 sm:grid-cols-2">
   348→                <Field
   349→                  label="Full name"
   350→                  required
   351→                  error={errors.fullName}
   352→                  value={address.fullName}
   353→                  onChange={(v) => setAddress((a) => ({ ...a, fullName: v }))}
   354→                  placeholder="Jane Doe"
   355→                />
   356→                <Field
   357→                  label="Phone number"
   358→                  required
   359→                  error={errors.phone}
   360→                  value={address.phone}
   361→                  onChange={(v) => setAddress((a) => ({ ...a, phone: v }))}
   362→                  placeholder="+234 801 234 5678"
   363→                />
   364→                <div className="sm:col-span-2">
   365→                  <Field
   366→                    label="Street address"
   367→                    required
   368→                    error={errors.addressLine}
   369→                    value={address.addressLine}
   370→                    onChange={(v) => setAddress((a) => ({ ...a, addressLine: v }))}
   371→                    placeholder="House 12, Allen Avenue, Ikeja"
   372→                  />
   373→                </div>
   374→                <Field
   375→                  label="City"
   376→                  required
   377→                  error={errors.city}
   378→                  value={address.city}
   379→                  onChange={(v) => setAddress((a) => ({ ...a, city: v }))}
   380→                  placeholder="Ikeja"
   381→                />
   382→                <div>
   383→                  <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wide text-gray-500">
   384→                    State <span className="text-danger">*</span>
   385→                  </label>
   386→                  <select
   387→                    value={address.state}
   388→                    onChange={(e) => setAddress((a) => ({ ...a, state: e.target.value }))}

   390→                  >
   391→                    {NIGERIAN_STATES.map((s) => (
   392→                      <option key={s} value={s}>
   393→                        {s}
   394→                      </option>
   395→                    ))}
   396→                  </select>
   397→                </div>
   398→                <div className="sm:col-span-2">
   399→                  <Field
   400→                    label="Landmark (optional)"
   401→                    value={address.landmark ?? ""}
   402→                    onChange={(v) => setAddress((a) => ({ ...a, landmark: v }))}
   403→                    placeholder="Opposite the petrol station"
   404→                  />
   405→                </div>
   406→              </div>
   407→            </div>
   408→
   409→            {/* Cart items grouped by vendor */}
   410→            <div className="rounded-2xl border border-border bg-surface p-5 md:p-6">
   411→              <div className="mb-4 flex items-center gap-2">
   412→                <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-primary-50 text-primary-600">
   413→                  <ShoppingBag className="h-5 w-5" />
   414→                </div>
   415→                <div>
   416→                  <h2 className="font-heading text-base font-semibold text-foreground">
   417→                    Your items
   418→                  </h2>
   419→                  <p className="text-xs text-gray-500">
   420→                    {items.length} item{items.length > 1 ? "s" : ""} from{" "}
   421→                    {storeCount} vendor{storeCount > 1 ? "s" : ""} — each vendor
   422→                    quotes delivery separately.
   423→                  </p>
   424→                </div>
   425→              </div>
   426→
   427→              <div className="space-y-5">
   428→                {Object.entries(itemsByStore).map(([key, group]) => (
   429→                  <div key={key} className="rounded-xl border border-border bg-background p-4">
   430→                    <div className="mb-3 flex items-center gap-2 border-b border-border pb-2">
   431→                      <Truck className="h-4 w-4 text-primary-600" />
   432→                      <span className="text-sm font-semibold text-foreground">
   433→                        {group.storeName}
   434→                      </span>
   435→                      <span className="ml-auto text-xs text-gray-500">
   436→                        Subtotal: {formatNGN(group.subtotal)}
   437→                      </span>
   438→                    </div>

   456→                              {formatNGN(item.price)}

   470→                                  updateQuantity(item.productId, item.quantity - 1, item.storeSlug)
   471→                                }
   472→                                className="flex h-8 w-8 items-center justify-center text-gray-500 hover:text-foreground"
   473→                              >

   483→                                  updateQuantity(item.productId, item.quantity + 1, item.storeSlug)

   486→                              >

   510→                <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-primary-50 text-primary-600">
   511→                  <CreditCard className="h-5 w-5" />

   749→function Field({
   750→  label,
   751→  value,
   752→  onChange,
   753→  placeholder,
   754→  required,
   755→  error,
   756→}: {
   757→  label: string;
   758→  value: string;
   759→  onChange: (v: string) => void;
   760→  placeholder?: string;
   761→  required?: boolean;
   762→  error?: string;
   763→}) {
   764→  return (
   765→    <div>
   766→      <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wide text-gray-500">
   767→        {label} {required ? <span className="text-danger">*</span> : null}
   768→      </label>
   769→      <input
   770→        type="text"
   771→        value={value}
   772→        onChange={(e) => onChange(e.target.value)}
   773→        placeholder={placeholder}
   774→        className={cn(
   775→          "h-11 w-full rounded-xl border bg-background px-3 text-sm text-foreground placeholder:text-gray-400 focus:outline-none focus:ring-2",
   776→          error
   777→            ? "border-danger focus:ring-danger/20"
   778→            : "border-border focus:border-primary-500 focus:ring-primary-500/20",
   779→        )}
   780→      />
   781→      {error ? <p className="mt-1 text-xs text-danger">{error}</p> : null}
   782→    </div>
   783→  );
   784→}
