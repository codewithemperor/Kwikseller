    37→function formatCount(n: number): string {
    38→  if (n >= 1000) return `${(n / 1000).toFixed(1).replace(/\.0$/, "")}K`;
    39→  return String(n);
    40→}

    42→function timeAgo(ts: number): string {
    43→  const diff = Date.now() - ts;
    44→  const mins = Math.floor(diff / 60000);
    45→  if (mins < 1) return "just now";
    46→  if (mins < 60) return `${mins}m ago`;
    47→  const hours = Math.floor(mins / 60);
    48→  if (hours < 24) return `${hours}h ago`;
    49→  const days = Math.floor(hours / 24);
    50→  if (days < 7) return `${days}d ago`;
    51→  return new Intl.DateTimeFormat("en-NG", { dateStyle: "medium" }).format(new Date(ts));
    52→}

    60→function isFiltersEmpty(f: SearchFiltersState): boolean {
    61→  return (
    62→    f.minPrice === undefined &&
    63→    f.maxPrice === undefined &&
    64→    f.rating === undefined &&
    65→    !f.category &&
    66→    !f.brandId &&
    67→    !f.storeId &&
    68→    !f.state
    69→  );
    70→}
    71→
    72→function parseFiltersFromParams(
    73→  params: URLSearchParams,
    74→): { query: string; filters: SearchFiltersState; sort: SortValue } {
    75→  const query = (params.get("q") ?? "").trim();
    76→  const filters: SearchFiltersState = {
    77→    minPrice: params.get("minPrice") ? Number(params.get("minPrice")) : undefined,
    78→    maxPrice: params.get("maxPrice") ? Number(params.get("maxPrice")) : undefined,
    79→    rating: params.get("rating") ? Number(params.get("rating")) : undefined,
    80→    category: params.get("category") ?? undefined,
    81→    brandId: params.get("brandId") ?? undefined,
    82→    storeId: params.get("storeId") ?? undefined,
    83→    state: params.get("state") ?? undefined,
    84→  };
    85→  // Strip NaN / undefined
    86→  for (const k of Object.keys(filters) as (keyof SearchFiltersState)[]) {
    87→    const v = filters[k];
    88→    if (v === undefined || v === null || (typeof v === "number" && Number.isNaN(v))) {
    89→      delete filters[k];
    90→    }
    91→  }
    92→  const sort = (params.get("sort") as SortValue) || "relevance";
    93→  return { query, filters, sort };
    94→}
    95→
    96→function buildSearchParams(
    97→  query: string,
    98→  filters: SearchFiltersState,
    99→  sort: SortValue,
   100→): URLSearchParams {
   101→  const params = new URLSearchParams();
   102→  if (query) params.set("q", query);
   103→  if (filters.minPrice !== undefined && !Number.isNaN(filters.minPrice)) {
   104→    params.set("minPrice", String(filters.minPrice));
   105→  }
   106→  if (filters.maxPrice !== undefined && !Number.isNaN(filters.maxPrice)) {
   107→    params.set("maxPrice", String(filters.maxPrice));
   108→  }
   109→  if (filters.rating !== undefined && !Number.isNaN(filters.rating)) {
   110→    params.set("rating", String(filters.rating));
   111→  }
   112→  if (filters.category) params.set("category", filters.category);
   113→  if (filters.brandId) params.set("brandId", filters.brandId);
   114→  if (filters.storeId) params.set("storeId", filters.storeId);
   115→  if (filters.state) params.set("state", filters.state);
   116→  if (sort !== "relevance") params.set("sort", sort);
   117→  return params;
   118→}

   122→function SearchPageContent() {
   123→  const searchParams = useSearchParams();
   124→  const router = useRouter();
   125→
   126→  // Read state from URL — single source of truth.
   127→  const { query, filters: urlFilters, sort: urlSort } = useMemo(
   128→    () => parseFiltersFromParams(searchParams),
   129→    [searchParams],
   130→  );
   131→
   132→  const [drawerOpen, setDrawerOpen] = useState(false);
   133→  const [quickViewProduct, setQuickViewProduct] = useState<MarketplaceProduct | null>(null);
   134→
   135→  // Build the API filters object. react-query's queryKey includes this
   136→  // object, so any change automatically resets to page 1.
   137→  const apiFilters: ApiSearchFilters = useMemo(
   138→    () => ({
   139→      q: query,
   140→      ...urlFilters,
   141→      sort: urlSort,
   142→      limit: PAGE_SIZE,
   143→    }),
   144→    [query, urlFilters, urlSort],
   145→  );
   146→
   147→  const hasQuery = query.length > 0 || !isFiltersEmpty(urlFilters);
   148→  const {
   149→    products,
   150→    meta,
   151→    isLoading,
   152→    isError,
   153→    isFetching,
   154→    isFetchingNextPage,
   155→    hasNextPage,
   156→    refetch,
   157→    fetchNextPage,
   158→  } = useSearchInfinite(apiFilters, hasQuery);
   159→
   160→  // Update URL when filters/sort change.
   161→  const pushUrl = useCallback(
   162→    (next: { query?: string; filters?: SearchFiltersState; sort?: SortValue }) => {
   163→      const nextQuery = next.query ?? query;
   164→      const nextFilters = next.filters ?? urlFilters;
   165→      const nextSort = next.sort ?? urlSort;
   166→      const params = buildSearchParams(nextQuery, nextFilters, nextSort);
   167→      const search = params.toString();
   168→      router.push(search ? `/search?${search}` : "/search", { scroll: false });
   169→    },
   170→    [query, urlFilters, urlSort, router],
   171→  );
   172→
   173→  const handleFilterChange = useCallback(
   174→    (next: Partial<SearchFiltersState>) => {
   175→      const merged = { ...urlFilters, ...next };
   176→      // Remove undefined values.
   177→      for (const k of Object.keys(merged) as (keyof SearchFiltersState)[]) {
   178→        if (merged[k] === undefined || merged[k] === null || Number.isNaN(merged[k] as number)) {
   179→          delete merged[k];
   180→        }
   181→      }
   182→      pushUrl({ filters: merged });
   183→    },
   184→    [urlFilters, pushUrl],
   185→  );
   186→
   187→  const handleSortChange = useCallback(
   188→    (next: SortValue) => {
   189→      pushUrl({ sort: next });
   190→    },
   191→    [pushUrl],
   192→  );
   193→
   194→  const handleResetFilters = useCallback(() => {
   195→    pushUrl({ filters: {} });
   196→  }, [pushUrl]);
   197→
   198→  const handleLoadMore = useCallback(() => {
   199→    fetchNextPage();
   200→  }, [fetchNextPage]);
   201→
   202→  const handleQuickView = useCallback((p: MarketplaceProduct) => {
   203→    setQuickViewProduct(p);
   204→  }, []);
   205→
   206→  // Run a search from a recent/trending term.
   207→  const runSearch = useCallback(
   208→    (term: string) => {
   209→      const params = new URLSearchParams({ q: term });
   210→      router.push(`/search?${params.toString()}`);
   211→    },
   212→    [router],
   213→  );
   214→
   215→  // Active filter chips — derived from URL state.
   216→  const activeChips = useMemo(() => {
   217→    const chips: Array<{ key: string; label: string; onRemove: () => void }> = [];
   218→    if (urlFilters.minPrice !== undefined && urlFilters.minPrice > 0) {
   219→      chips.push({
   220→        key: "minPrice",
   221→        label: `Min ₦${urlFilters.minPrice.toLocaleString()}`,
   222→        onRemove: () => handleFilterChange({ minPrice: undefined }),
   223→      });
   224→    }
   225→    if (urlFilters.maxPrice !== undefined && urlFilters.maxPrice > 0) {
   226→      chips.push({
   227→        key: "maxPrice",
   228→        label: `Max ₦${urlFilters.maxPrice.toLocaleString()}`,
   229→        onRemove: () => handleFilterChange({ maxPrice: undefined }),
   230→      });
   231→    }
   232→    if (urlFilters.rating !== undefined) {
   233→      chips.push({
   234→        key: "rating",
   235→        label: `${urlFilters.rating}★ & above`,
   236→        onRemove: () => handleFilterChange({ rating: undefined }),
   237→      });
   238→    }
   239→    if (urlFilters.category) {
   240→      const cat = meta?.categories?.find((c) => c.slug === urlFilters.category || c.id === urlFilters.category);
   241→      chips.push({
   242→        key: "category",
   243→        label: cat?.name ?? urlFilters.category,
   244→        onRemove: () => handleFilterChange({ category: undefined }),
   245→      });
   246→    }
   247→    if (urlFilters.storeId) {
   248→      const store = meta?.stores?.find((s) => s.id === urlFilters.storeId || s.slug === urlFilters.storeId);
   249→      chips.push({
   250→        key: "storeId",
   251→        label: store?.name ?? urlFilters.storeId,
   252→        onRemove: () => handleFilterChange({ storeId: undefined }),
   253→      });
   254→    }
   255→    if (urlFilters.brandId) {
   256→      const brand = meta?.brands?.find((b) => b.id === urlFilters.brandId || b.slug === urlFilters.brandId);
   257→      chips.push({
   258→        key: "brandId",
   259→        label: brand?.name ?? urlFilters.brandId,
   260→        onRemove: () => handleFilterChange({ brandId: undefined }),
   261→      });
   262→    }
   263→    if (urlFilters.state) {
   264→      const st = meta?.states?.find((s) => s.name === urlFilters.state || s.code === urlFilters.state);
   265→      chips.push({
   266→        key: "state",
   267→        label: st?.name ?? urlFilters.state,
   268→        onRemove: () => handleFilterChange({ state: undefined }),
   269→      });
   270→    }
   271→    return chips;
   272→  }, [urlFilters, meta, handleFilterChange]);
   273→
   274→  const trendingQuery = useTrending(5);
   275→  const trendingSearchesQuery = useTrendingSearches(12);
   276→  const { items: recentSearches, addSearch, removeSearch, clearSearches } = useRecentSearches();
   277→
   278→  // Persist every successful query into recent searches (max once per query change).
   279→  useEffect(() => {
   280→    if (query && !isFetching && !isLoading) {
   281→      addSearch(query);
   282→    }
   283→  }, [query, isFetching, isLoading, addSearch]);
   284→
   285→  const popularSearches = useMemo(
   286→    () => (trendingQuery.data ?? []).slice(0, 5),
   287→    [trendingQuery.data],
   288→  );
   289→  const trendingSearches = useMemo(
   290→    () => trendingSearchesQuery.data ?? [],
   291→    [trendingSearchesQuery.data],
   292→  );
   293→
   294→  // Suggested alternative queries when no results are returned.
   295→  const noResultSuggestions = useMemo(() => {
   296→    if (!query || (meta?.total ?? 0) > 0) return [];
   297→    const q = query.toLowerCase();
   298→    return trendingSearches
   299→      .filter((t) => !t.query.toLowerCase().includes(q) && !q.includes(t.query.toLowerCase()))
   300→      .slice(0, 4);
   301→  }, [query, meta?.total, trendingSearches]);
   302→
   303→  const totalResults = meta?.total ?? 0;
   304→  const hasResults = products.length > 0;
   305→
   306→  // ─── Render ──────────────────────────────────────────────────────────
   307→
   308→  return (
   309→    <div className="min-h-screen bg-background">
   310→      {/* Sticky results toolbar */}
   311→      {hasQuery ? (
   312→        <div className="sticky top-[var(--header-height)] z-20 border-b border-kwik-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80 dark:border-white/10">
   313→          <div className="container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   314→            <div className="flex items-center gap-3 py-3">
   315→              {/* Result count */}
   316→              <div className="flex-1 min-w-0">
   317→                {isLoading ? (
   318→                  <span className="text-sm text-kwik-muted">Searching…</span>
   319→                ) : (
   320→                  <p className="text-sm text-kwik-gray-light dark:text-white/70">
   321→                    <span className="font-semibold text-kwik-dark dark:text-white">{totalResults}</span>{" "}
   322→                    result{totalResults !== 1 ? "s" : ""}
   323→                    {query ? (
   324→                      <>
   325→                        {" "}for <span className="font-semibold text-kwik-orange">&ldquo;{query}&rdquo;</span>
   326→                      </>
   327→                    ) : null}
   328→                  </p>
   329→                )}
   330→              </div>
   331→
   332→              {/* Mobile: Filter button */}
   333→              <button
   334→                type="button"
   335→                onClick={() => setDrawerOpen(true)}
   336→                className="flex h-9 items-center gap-1.5 rounded-lg border border-kwik-border bg-background px-3 text-sm font-medium text-kwik-dark transition hover:border-kwik-orange/50 hover:bg-kwik-bg-light lg:hidden dark:bg-white/5 dark:text-white dark:border-white/10 dark:hover:bg-white/10"
   337→              >
   338→                <SlidersHorizontal className="h-3.5 w-3.5" />
   339→                Filters
   340→                {activeChips.length > 0 ? (
   341→                  <span className="flex h-4 min-w-4 items-center justify-center rounded-full bg-kwik-orange px-1 text-[10px] font-bold text-white">
   342→                    {activeChips.length}
   343→                  </span>
   344→                ) : null}
   345→              </button>
   346→
   347→              <SortDropdown value={urlSort} onChange={handleSortChange} />
   348→            </div>
   349→
   350→            {/* Active filter chips */}
   351→            {activeChips.length > 0 ? (
   352→              <ActiveFilters
   353→                chips={activeChips}
   354→                onClearAll={handleResetFilters}
   355→                className="pb-3"
   356→              />
   357→            ) : null}
   358→          </div>
   359→        </div>
   360→      ) : null}
   361→
   362→      {/* Results area */}
   363→      <div className="container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 py-6">
   364→        {/* No query state — show recent + trending searches & popular products */}
   365→        {!hasQuery ? (
   366→          <NoQueryState
   367→            recentSearches={recentSearches}
   368→            trendingSearches={trendingSearches}
   369→            trendingSearchesLoading={trendingSearchesQuery.isLoading}
   370→            onPickSearch={runSearch}
   371→            onRemoveSearch={removeSearch}
   372→            onClearSearches={clearSearches}
   373→            popularSearches={popularSearches}
   374→            onQuickView={handleQuickView}
   375→          />
   376→        ) : (
   377→          <div className="flex gap-6">
   378→            {/* Desktop filter sidebar */}
   379→            <aside className="hidden lg:block w-64 shrink-0">
   380→              <div className="sticky top-[calc(var(--header-height)+1rem)] rounded-2xl border border-kwik-border bg-background p-4 dark:bg-white/[0.02] dark:border-white/10">
   381→                <SearchFiltersPanel
   382→                  state={urlFilters}
   383→                  meta={meta}
   384→                  onChange={handleFilterChange}
   385→                  onReset={handleResetFilters}
   386→                />
   387→              </div>
   388→            </aside>
   389→
   390→            {/* Product results */}
   391→            <div className="flex-1 min-w-0">
   392→              {/* Loading — first page */}
   393→              {isLoading && !hasResults ? (
   394→                <ProductGridSkeleton count={10} columns={5} />
   395→              ) : isError ? (
   396→                <SearchErrorState onRetry={() => refetch()} />
   397→              ) : !hasResults && !isFetching ? (
   398→                <div className="space-y-8">
   399→                  <NoResultsState
   400→                    query={query}
   401→                    onClearFilters={handleResetFilters}
   402→                    onBrowseAll={() => router.push("/products")}
   403→                  />
   404→                  {noResultSuggestions.length > 0 ? (
   405→                    <section className="rounded-2xl border border-kwik-border bg-kwik-bg-surface p-5 dark:bg-white/[0.02] dark:border-white/10">
   406→                      <div className="mb-3 flex items-center gap-2">
   407→                        <Sparkles className="h-4 w-4 text-kwik-orange" />
   408→                        <h3 className="text-sm font-semibold text-kwik-dark dark:text-white">
   409→                          Try one of these trending searches
   410→                        </h3>
   411→                      </div>
   412→                      <div className="flex flex-wrap gap-2">
   413→                        {noResultSuggestions.map((t) => (
   414→                          <button
   415→                            key={t.id}
   416→                            type="button"
   417→                            onClick={() => runSearch(t.query)}
   418→                            className="group inline-flex items-center gap-2 rounded-full border border-kwik-border bg-background px-4 py-2 text-xs font-medium text-kwik-dark transition-all hover:border-kwik-orange/50 hover:bg-kwik-orange-tint hover:text-kwik-orange dark:bg-white/5 dark:text-white/80 dark:border-white/10 dark:hover:bg-white/10"
   419→                          >
   420→                            <Flame className="h-3 w-3 text-kwik-orange" />
   421→                            {t.label}
   422→                            <span className="text-[10px] text-kwik-muted group-hover:text-kwik-orange/70">
   423→                              {formatCount(t.count)}
   424→                            </span>
   425→                          </button>
   426→                        ))}
   427→                      </div>
   428→                    </section>
   429→                  ) : null}
   430→                </div>
   431→              ) : (
   432→                <>

   435→                    {products.map((product) => (
   436→                      <MarketplaceProductCard
   437→                        key={product.id}
   438→                        product={product}
   439→                        onQuickView={() => handleQuickView(product)}
   440→                      />
   441→                    ))}
   442→                    {/* Loading skeletons when fetching next page */}
   443→                    {isFetchingNextPage
   444→                      ? Array.from({ length: 5 }).map((_, i) => (
   445→                          <div
   446→                            key={`skeleton-${i}`}
   447→                            className="aspect-[3/4] animate-pulse rounded-xl bg-kwik-bg-light dark:bg-white/5"
   448→                          />
   449→                        ))
   450→                      : null}
   451→                  </div>
   452→
   453→                  {/* Load More */}
   454→                  {hasNextPage ? (
   455→                    <div className="mt-8 flex justify-center">
   456→                      <button
   457→                        type="button"
   458→                        onClick={handleLoadMore}
   459→                        disabled={isFetchingNextPage}
   460→                        className="inline-flex h-11 items-center justify-center gap-2 rounded-xl border border-kwik-border bg-background px-6 text-sm font-semibold text-kwik-dark transition hover:border-kwik-orange/50 hover:bg-kwik-bg-light disabled:opacity-50 disabled:cursor-not-allowed dark:bg-white/5 dark:text-white dark:border-white/10 dark:hover:bg-white/10"
   461→                      >
   462→                        {isFetchingNextPage ? (
   463→                          <>
   464→                            <span className="h-4 w-4 animate-spin rounded-full border-2 border-kwik-orange border-t-transparent" />
   465→                            Loading…
   466→                          </>
   467→                        ) : (
   468→                          <>
   469→                            Load more
   470→                            <span className="text-xs text-kwik-muted">
   471→                              ({totalResults - products.length} left)
   472→                            </span>
   473→                          </>
   474→                        )}
   475→                      </button>
   476→                    </div>
   477→                  ) : hasResults ? (
   478→                    <p className="mt-8 text-center text-xs text-kwik-muted">
   479→                      You&rsquo;ve reached the end — showing all {products.length} of {totalResults} results
   480→                    </p>
   481→                  ) : null}
   482→                </>
   483→              )}
   484→            </div>
   485→          </div>
   486→        )}
   487→      </div>
   488→
   489→      {/* Mobile filter drawer */}
   490→      <SearchFilterDrawer
   491→        open={drawerOpen}
   492→        onClose={() => setDrawerOpen(false)}
   493→        state={urlFilters}
   494→        meta={meta}
   495→        onChange={(next) => handleFilterChange(next)}
   496→        onReset={handleResetFilters}
   497→      />
   498→
   499→      {/* Quick View Modal */}
   500→      <QuickViewModal
   501→        product={quickViewProduct}
   502→        isOpen={!!quickViewProduct}
   503→        onClose={() => setQuickViewProduct(null)}
   504→      />
   505→    </div>
   506→  );
   507→}

   514→  trendingSearchesLoading,

   529→}) {
   530→  const router = useRouter();
   531→  return (
   532→    <div className="space-y-10">

   535→          <SearchIcon className="h-10 w-10 text-kwik-orange" />
   536→        </div>
   537→        <h3 className="text-xl font-semibold text-kwik-dark dark:text-white">Search Kwikseller</h3>
   538→        <p className="mt-2 max-w-sm text-center text-sm text-kwik-gray-light dark:text-white/60">
   539→          Find products, stores, brands, and categories. Start typing to see live results.
   540→        </p>
   541→      </div>
   542→
   543→      <div className="grid gap-6 lg:grid-cols-2">
   544→        {/* Recent searches */}
   545→        <section className="rounded-2xl border border-kwik-border bg-kwik-bg-surface p-5 dark:bg-white/[0.02] dark:border-white/10">
   546→          <div className="mb-4 flex items-center justify-between gap-3">
   547→            <div className="flex items-center gap-2">
   548→              <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-kwik-orange-tint text-kwik-orange">
   549→                <History className="h-4 w-4" />
   550→              </div>
   551→              <div>
   552→                <h2 className="text-sm font-semibold text-kwik-dark dark:text-white">Recent searches</h2>
   553→                <p className="text-[11px] text-kwik-muted">Pick up where you left off</p>
   554→              </div>
   555→            </div>
   556→            {recentSearches.length > 0 ? (
   557→              <button
   558→                type="button"
   559→                onClick={onClearSearches}
   560→                className="inline-flex items-center gap-1 rounded-full px-2.5 py-1 text-[11px] font-medium text-kwik-muted transition hover:bg-kwik-red/5 hover:text-kwik-red"
   561→              >
   562→                <Trash2 className="h-3 w-3" /> Clear all
   563→              </button>
   564→            ) : null}
   565→          </div>
   566→
   567→          {recentSearches.length === 0 ? (
   568→            <div className="flex flex-col items-center justify-center rounded-xl border border-dashed border-kwik-border-light py-8 text-center dark:border-white/10">
   569→              <Clock className="h-7 w-7 text-kwik-muted/60" />
   570→              <p className="mt-2 text-xs font-medium text-kwik-muted">No recent searches yet</p>
   571→              <p className="text-[11px] text-kwik-muted/70">Your search history will appear here.</p>
   572→            </div>
   573→          ) : (
   574→            <ul className="space-y-1.5">
   575→              <AnimatePresence initial={false}>
   576→                {recentSearches.map((s) => (
   577→                  <motion.li
   578→                    key={`${s.query}-${s.timestamp}`}
   579→                    layout
   580→                    initial={{ opacity: 0, x: -8 }}
   581→                    animate={{ opacity: 1, x: 0 }}
   582→                    exit={{ opacity: 0, x: 8, height: 0 }}
   583→                    className="group flex items-center gap-2 rounded-xl border border-transparent bg-background px-3 py-2 transition-colors hover:border-kwik-border-light hover:bg-kwik-bg-surface dark:hover:border-white/10 dark:hover:bg-white/5"
   584→                  >
   585→                    <button
   586→                      type="button"
   587→                      onClick={() => onPickSearch(s.query)}
   588→                      className="flex flex-1 items-center gap-2.5 text-left"
   589→                    >
   590→                      <Clock className="h-3.5 w-3.5 text-kwik-muted transition group-hover:text-kwik-orange" />
   591→                      <span className="truncate text-sm font-medium text-kwik-dark dark:text-white">{s.query}</span>
   592→                      <span className="ml-auto shrink-0 text-[10px] text-kwik-muted">
   593→                        {timeAgo(s.timestamp)}
   594→                      </span>
   595→                    </button>
   596→                    <button
   597→                      type="button"
   598→                      aria-label={`Remove ${s.query}`}
   599→                      onClick={() => onRemoveSearch(s.query)}
   600→                      className="flex h-6 w-6 items-center justify-center rounded-full text-kwik-muted opacity-0 transition hover:bg-kwik-red/10 hover:text-kwik-red group-hover:opacity-100"
   601→                    >
   602→                      <X className="h-3.5 w-3.5" />
   603→                    </button>
   604→                  </motion.li>
   605→                ))}
   606→              </AnimatePresence>
   607→            </ul>
   608→          )}
   609→        </section>
   610→
   611→        {/* Trending searches */}
   612→        <section className="rounded-2xl border border-kwik-border bg-kwik-bg-surface p-5 dark:bg-white/[0.02] dark:border-white/10">
   613→          <div className="mb-4 flex items-center gap-2">
   614→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-kwik-orange-tint text-kwik-orange">
   615→              <Flame className="h-4 w-4" />
   616→            </div>
   617→            <div>
   618→              <h2 className="text-sm font-semibold text-kwik-dark dark:text-white">Trending searches</h2>
   619→              <p className="text-[11px] text-kwik-muted">What other buyers are looking for</p>
   620→            </div>
   621→          </div>
   622→
   623→          {trendingSearchesLoading ? (
   624→            <div className="flex flex-wrap gap-2">
   625→              {Array.from({ length: 8 }).map((_, i) => (
   626→                <div
   627→                  key={i}
   628→                  className="h-8 animate-pulse rounded-full bg-kwik-orange/10"
   629→                  style={{ width: `${60 + (i % 4) * 22}px` }}
   630→                />
   631→              ))}
   632→            </div>
   633→          ) : (
   634→            <div className="flex flex-wrap gap-2">
   635→              {trendingSearches.map((t, i) => (
   636→                <motion.button
   637→                  key={t.id}
   638→                  type="button"
   639→                  initial={{ opacity: 0, y: 6 }}
   640→                  animate={{ opacity: 1, y: 0 }}
   641→                  transition={{ delay: Math.min(i * 0.04, 0.3) }}
   642→                  whileTap={{ scale: 0.96 }}
   643→                  onClick={() => onPickSearch(t.query)}
   644→                  className="group inline-flex items-center gap-1.5 rounded-full border border-kwik-border bg-background px-3 py-1.5 text-xs font-medium text-kwik-dark shadow-sm transition-all hover:border-kwik-orange/50 hover:bg-kwik-orange-tint hover:text-kwik-orange dark:bg-white/5 dark:text-white/80 dark:border-white/10 dark:hover:bg-white/10"
   645→                >
   646→                  <span className="flex h-4 w-4 items-center justify-center rounded-full bg-kwik-orange-tint text-[9px] font-bold text-kwik-orange">
   647→                    {i + 1}
   648→                  </span>
   649→                  {t.label}
   650→                  <span className="text-[10px] text-kwik-muted group-hover:text-kwik-orange/70">
   651→                    {formatCount(t.count)}
   652→                  </span>
   653→                </motion.button>
   654→              ))}
   655→            </div>
   656→          )}
   657→        </section>
   658→      </div>
   659→
   660→      {/* Quick picks */}
   661→      <section>
   662→        <div className="mb-3 flex items-center gap-2">
   663→          <Sparkles className="h-4 w-4 text-kwik-orange" />
   664→          <h2 className="text-sm font-semibold text-kwik-dark dark:text-white">Quick picks</h2>
   665→        </div>
   666→        <div className="flex flex-wrap gap-2">
   667→          {["Electronics", "Fashion", "Phones", "Beauty", "Food", "Home"].map((term) => (
   668→            <button
   669→              key={term}
   670→              type="button"
   671→              onClick={() => {
   672→                const params = new URLSearchParams({ q: term });
   673→                router.push(`/search?${params.toString()}`);
   674→              }}
   675→              className="rounded-full border border-kwik-border bg-kwik-bg-surface px-4 py-2 text-xs font-medium text-kwik-gray transition-all hover:border-kwik-orange/50 hover:bg-kwik-orange-tint hover:text-kwik-orange dark:bg-white/5 dark:text-white/80 dark:border-white/10 dark:hover:bg-white/10"
   676→            >
   677→              {term}
   678→            </button>
   679→          ))}
   680→        </div>
   681→      </section>
   682→
   683→      {/* Popular products */}
   684→      {popularSearches.length > 0 ? (
   685→        <section>
   686→          <div className="mb-4 flex items-center gap-2">
   687→            <TrendingUp className="h-5 w-5 text-kwik-orange" />
   688→            <h2 className="text-base font-semibold text-kwik-dark dark:text-white">Popular right now</h2>
   689→          </div>
   690→          <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 2xl:grid-cols-6">
   691→            {popularSearches.map((product) => (
   692→              <MarketplaceProductCard
   693→                key={product.id}
   694→                product={product}
   695→                onQuickView={() => onQuickView(product)}
   696→              />
   697→            ))}
   698→          </div>
   699→        </section>
   700→      ) : null}
   701→    </div>
   702→  );
   703→}
