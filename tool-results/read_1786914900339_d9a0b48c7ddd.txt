    67→// ==================== Configuration ====================

    69→const getBaseURL = (): string => {
    70→  const configuredUrl = process.env.NEXT_PUBLIC_API_URL
    71→
    72→  if (!configuredUrl) {
    73→    // Client (browser): relative "/api/v1" so requests go through the
    74→    // gateway and hit the in-app route handlers (dummy) or the proxied
    75→    // rewrite (real API).
    76→    // Server (SSR/RSC): absolute localhost:3000 so server-side fetches
    77→    // resolve against the Next.js app's own route handlers. In production
    78→    // set NEXT_PUBLIC_API_URL to the real NestJS backend.
    79→    return typeof window !== 'undefined' ? '/api/v1' : 'http://localhost:3000/api/v1'
    80→  }
    81→
    82→  const withProtocol = /^https?:\/\//i.test(configuredUrl)
    83→    ? configuredUrl
    84→    : `https://${configuredUrl}`
    85→  const withoutTrailingSlash = withProtocol.replace(/\/+$/, '')
    86→
    87→  return withoutTrailingSlash.endsWith('/api/v1')
    88→    ? withoutTrailingSlash
    89→    : `${withoutTrailingSlash}/api/v1`
    90→}

    92→// Storage keys
    93→const STORAGE_KEYS = {
    94→  ACCESS_TOKEN: 'kwikseller_access_token',
    95→  REFRESH_TOKEN: 'kwikseller_refresh_token',
    96→  AUTH_STORE: 'kwikseller_auth',
    97→}
    98→
    99→// ==================== Token Refresh Mutex ====================
   100→// Prevents concurrent 401 responses from triggering multiple refresh calls.
   101→// Only one refresh request is in-flight at a time; other requests queue and
   102→// resolve once the refresh completes.
   103→
   104→let isRefreshing = false
   105→let refreshQueue: Array<{
   106→  resolve: (token: string) => void
   107→  reject: (error: unknown) => void
   108→}> = []
   109→
   110→function processRefreshQueue(error: unknown, token: string | null = null): void {
   111→  refreshQueue.forEach(({ resolve, reject }) => {
   112→    if (error) reject(error)
   113→    else if (token) resolve(token)
   114→  })
   115→  refreshQueue = []
   116→}

   118→// ==================== Token Management ====================
   119→
   120→export const tokenManager = {
   121→  getAccessToken: (): string | null => {
   122→    if (typeof window === 'undefined') return null
   123→    return localStorage.getItem(STORAGE_KEYS.ACCESS_TOKEN)
   124→  },
   125→
   126→  getRefreshToken: (): string | null => {
   127→    if (typeof window === 'undefined') return null
   128→    return localStorage.getItem(STORAGE_KEYS.REFRESH_TOKEN)
   129→  },
   130→
   131→  setTokens: (accessToken: string, refreshToken?: string): void => {
   132→    if (typeof window === 'undefined') return
   133→    localStorage.setItem(STORAGE_KEYS.ACCESS_TOKEN, accessToken)
   134→    if (refreshToken) {
   135→      localStorage.setItem(STORAGE_KEYS.REFRESH_TOKEN, refreshToken)
   136→    }
   137→  },
   138→
   139→  clearTokens: (): void => {
   140→    if (typeof window === 'undefined') return
   141→    localStorage.removeItem(STORAGE_KEYS.ACCESS_TOKEN)
   142→    localStorage.removeItem(STORAGE_KEYS.REFRESH_TOKEN)
   143→    localStorage.removeItem(STORAGE_KEYS.AUTH_STORE)
   144→  },
   145→
   146→  isAuthenticated: (): boolean => {
   147→    return !!tokenManager.getAccessToken()
   148→  },
   149→}
   150→
   151→// ==================== Create Axios Instance ====================
   152→
   153→const createApiClient = (): AxiosInstance => {
   154→  const instance = axios.create({
   155→    baseURL: getBaseURL(),
   156→    timeout: 30000,
   157→    headers: {
   158→      'Content-Type': 'application/json',
   159→    },
   160→  })
   161→
   162→  const redirectToLogin = () => {
   163→    if (typeof window === 'undefined') return
   164→    tokenManager.clearTokens()
   165→    const currentPath = window.location.pathname
   166→    const currentHost = window.location.hostname
   167→    const isAdminApp = currentHost.includes('admin') || currentPath.startsWith('/admin')
   168→    const isRiderApp = currentHost.includes('rider') || currentPath.startsWith('/rider')
   169→    const isVendorApp = currentHost.includes('vendor') || currentPath.startsWith('/dashboard')
   170→    let loginPath = '/login'
   171→    if (isAdminApp) loginPath = '/admin/login'
   172→    else if (isRiderApp) loginPath = '/rider/login'
   173→    // Vendor app login is at /login (same as default), but we preserve
   174→    // the dashboard redirect so the user lands back where they were
   175→    window.location.href = `${loginPath}?redirect=${encodeURIComponent(currentPath)}`
   176→  }
   177→
   178→  // Request interceptor - Add auth token
   179→  instance.interceptors.request.use(
   180→    (config: InternalAxiosRequestConfig) => {
   181→      // Skip auth if explicitly requested
   182→      if ((config as RequestConfig).skipAuth) {
   183→        return config
   184→      }
   185→
   186→      const token = tokenManager.getAccessToken()
   187→      if (token && config.headers) {
   188→        config.headers.Authorization = `Bearer ${token}`
   189→      }
   190→      return config
   191→    },
   192→    (error) => Promise.reject(error)
   193→  )
   194→
   195→  // Response interceptor - Handle errors and token refresh with mutex
   196→  instance.interceptors.response.use(
   197→    (response: AxiosResponse) => response,
   198→    async (error: AxiosError<ApiError>) => {
   199→      const originalRequest = error.config as InternalAxiosRequestConfig & {
   200→        _retry?: boolean
   201→      }
   202→      const isAuthEndpoint = originalRequest?.url?.includes('/auth/')
   203→
   204→      // Handle 401 Unauthorized - Try to refresh token (with mutex to prevent race conditions)
   205→      if (error.response?.status === 401 && !originalRequest._retry && !isAuthEndpoint) {
   206→        if (isRefreshing) {
   207→          // Another refresh is already in-flight — queue this request
   208→          return new Promise((resolve, reject) => {
   209→            refreshQueue.push({
   210→              resolve: (newToken: string) => {
   211→                if (originalRequest.headers) {
   212→                  originalRequest.headers.Authorization = `Bearer ${newToken}`
   213→                }
   214→                resolve(instance(originalRequest))
   215→              },
   216→              reject: (err: unknown) => reject(err),
   217→            })
   218→          })
   219→        }
   220→
   221→        originalRequest._retry = true
   222→        isRefreshing = true
   223→
   224→        try {
   225→          const refreshToken = tokenManager.getRefreshToken()
   226→          if (refreshToken) {
   227→            const response = await axios.post(`${getBaseURL()}/auth/refresh`, {
   228→              refreshToken,
   229→            })
   230→
   231→            const { accessToken, refreshToken: newRefreshToken } = response.data.data || response.data
   232→            tokenManager.setTokens(accessToken, newRefreshToken)
   233→
   234→            // Notify queued requests
   235→            processRefreshQueue(null, accessToken)
   236→
   237→            // Retry original request with new token
   238→            if (originalRequest.headers) {
   239→              originalRequest.headers.Authorization = `Bearer ${accessToken}`
   240→            }
   241→            return instance(originalRequest)
   242→          }
   243→          // No refresh token available
   244→          processRefreshQueue(new Error('No refresh token available'), null)
   245→          tokenManager.clearTokens()
   246→          redirectToLogin()
   247→        } catch (refreshError) {
   248→          // Refresh failed
   249→          processRefreshQueue(refreshError, null)
   250→          tokenManager.clearTokens()
   251→          redirectToLogin()
   252→        } finally {
   253→          isRefreshing = false
   254→        }
   255→      }
   256→
   257→      // Handle other errors
   258→      const apiError: ApiError = {
   259→        statusCode: error.response?.status || 500,
   260→        message: error.response?.data?.message || error.message || 'An error occurred',
   261→        error: error.response?.data?.error,
   262→        details: error.response?.data?.details,
   263→      }
   264→
   265→      return Promise.reject(apiError)
   266→    }
   267→  )
   268→
   269→  return instance
   270→}
   271→
   272→// Create singleton instance
   273→const apiClient = createApiClient()

   277→export const api = {
   278→  get: <T = unknown>(
   279→    url: string,
   280→    config?: RequestConfig
   281→  ): Promise<ApiResponse<T>> => apiClient.get(url, config).then((res) => res.data),
   282→
   283→  post: <T = unknown>(
   284→    url: string,
   285→    data?: unknown,
   286→    config?: RequestConfig
   287→  ): Promise<ApiResponse<T>> => apiClient.post(url, data, config).then((res) => res.data),
   288→
   289→  put: <T = unknown>(
   290→    url: string,
   291→    data?: unknown,
   292→    config?: RequestConfig
   293→  ): Promise<ApiResponse<T>> => apiClient.put(url, data, config).then((res) => res.data),
   294→
   295→  patch: <T = unknown>(
   296→    url: string,
   297→    data?: unknown,
   298→    config?: RequestConfig
   299→  ): Promise<ApiResponse<T>> => apiClient.patch(url, data, config).then((res) => res.data),
   300→
   301→  delete: <T = unknown>(
   302→    url: string,
   303→    config?: RequestConfig

   305→}
