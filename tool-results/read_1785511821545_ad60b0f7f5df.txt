    39→function formatCurrency(amount: number) {
    40→  return new Intl.NumberFormat("en-NG", {
    41→    style: "currency",
    42→    currency: "NGN",
    43→    maximumFractionDigits: 0,
    44→  }).format(amount);
    45→}
    46→
    47→function formatDate(value?: string | Date | null) {
    48→  if (!value) return "Not available";
    49→  return new Intl.DateTimeFormat("en-NG", {
    50→    dateStyle: "medium",
    51→    timeStyle: "short",
    52→  }).format(new Date(value));
    53→}
    54→
    55→function orderReference(order: Order) {
    56→  return order.checkoutReference || order.id.slice(-8).toUpperCase();
    57→}
    58→
    59→function deliveryAddress(order: Order) {
    60→  const delivery = order.delivery as
    61→    | { deliveryAddress?: string; status?: string }
    62→    | undefined;
    63→  if (delivery?.deliveryAddress) return delivery.deliveryAddress;
    64→  const address = order.address;
    65→  if (address) {
    66→    return [address.line1, address.line2, address.city, address.state, address.country]
    67→      .filter(Boolean)
    68→      .join(", ");
    69→  }
    70→  if (order.deliveryLocalGovernment && order.deliveryState) {
    71→    return `${order.deliveryLocalGovernment}, ${order.deliveryState}, Nigeria`;
    72→  }
    73→  return "No delivery address on this order";
    74→}

   415→/**
   416→ * Tiny helper that mounts an EscrowBadge in compact mode and programmatically

   419→function EscrowBadgeCompactTrigger({
   420→  escrow,
   421→  onDone,

   423→  escrow: NonNullable<OrderWorkflowState["escrow"]>;
   424→  onDone: () => void;
   425→}) {

   429→  // its dialog. The user can also close via the dialog itself.

   431→    <div className="sr-only">
   432→      <EscrowBadge escrow={escrow} compact />
   433→      <button type="button" onClick={onDone} aria-label="Close escrow info">
   434→        close
   435→      </button>
   436→    </div>
   437→  );
   438→}

   442→function unwrapOrder(value: unknown): Order | null {
   443→  const payload = value as { data?: unknown } | undefined;
   444→  const nested = payload?.data as { data?: unknown; order?: unknown } | undefined;
   445→  const candidate = nested?.data ?? nested?.order ?? payload?.data ?? value;
   446→  return candidate && typeof candidate === "object" && "id" in candidate
   447→    ? (candidate as Order)
   448→    : null;
   449→}

   733→// ─── Inner component (uses hooks; wrapped in ErrorBoundary by default export) ──
   734→
   735→function OrderDetailPageInner() {
   736→  const router = useRouter();
   737→  const params = useParams<{ id: string }>();
   738→  const orderId = Array.isArray(params.id) ? params.id[0] : params.id;
   739→  const { isAuthenticated, isLoading: isAuthLoading } = useAuth();
   740→
   741→  // Pull the mock order from the Zustand store — this is always available
   742→  // (even when the API is down or the user is not authenticated), so the
   743→  // workflow UI can render.
   744→  const mockOrder = useOrderWorkflowStore((s) =>
   745→    s.orders.find((o) => o.id === orderId),
   746→  );
   747→
   748→  // If the user is not authenticated (sandbox: no tokens), skip the live API
   749→  // call entirely and go straight to the mock order view. This keeps the
   750→  // verify URL working without forcing a login redirect that loops back.
   751→  const skipLiveApi = !isAuthLoading && !isAuthenticated;
   752→
   753→  const { data: liveOrder, isLoading, isError } = useQuery<Order | null>({
   754→    queryKey: ["buyer-order", orderId],
   755→    enabled: Boolean(orderId) && !isAuthLoading && isAuthenticated && !mockOrder,
   756→    queryFn: async () => {
   757→      try {
   758→        const response = await ordersApi.get(orderId);
   759→        return unwrapOrder(response);
   760→      } catch (error) {
   761→        const message =
   762→          error instanceof Error ? error.message : "Failed to load order.";
   763→        // Don't toast — we fall back to the mock order silently.
   764→        console.warn("[orders/[id]] live API failed:", message);
   765→        throw error;
   766→      }
   767→    },
   768→    retry: 1,
   769→    refetchOnWindowFocus: false,
   770→  });
   771→
   772→  // Fallback to the dummy/real API order (no auth required in dummy mode).
   773→  // This is what renders orders created via POST /checkout (TODO #6), where
   774→  // the vendor receives the order and quotes delivery + discount.
   775→  const {
   776→    data: apiOrder,
   777→    isLoading: apiLoading,
   778→  } = useOrder(mockOrder ? undefined : orderId);
   779→
   780→  // Boot: init the workflow store once (registers escrow listener, etc.)
   781→  const initStore = useOrderWorkflowStore((s) => s._init);
   782→  React.useEffect(() => {
   783→    const unregister = initStore();
   784→    return unregister;
   785→  }, [initStore]);
   786→
   787→  // Hydration guard: the Zustand store is persisted to localStorage. On the
   788→  // server (and the very first client render) it uses the seed data, which
   789→  // contains Date.now()-relative timestamps. Once the persisted store
   790→  // rehydrates on the client, those timestamps change → React hydration
   791→  // mismatch warnings. We avoid that by rendering the skeleton until the
   792→  // client has mounted.
   793→  const [hasMounted, setHasMounted] = React.useState(false);
   794→  React.useEffect(() => setHasMounted(true), []);
   795→
   796→  // Loading state — only show the skeleton if we're actually waiting on the
   797→  // live API AND there's no mock order to fall back to.
   798→  if (!hasMounted || (isAuthLoading || (isLoading && !skipLiveApi)) && !mockOrder) {
   799→    return <OrderDetailSkeleton />;
   800→  }
   801→
   802→  // Prefer the mock order (so the new workflow UI is always used for seeded
   803→  // orders), then fall back to the live order, then show not-found.
   804→  if (mockOrder) {
   805→    return <MockOrderWorkflow order={mockOrder} />;
   806→  }
   807→
   808→  if (liveOrder) {
   809→    return <LiveOrderDetail order={liveOrder} />;
   810→  }
   811→
   812→  // API order from POST /checkout (vendor-received order with quotation).
   813→  if (apiLoading) {
   814→    return <OrderDetailSkeleton />;
   815→  }
   816→  if (apiOrder) {
   817→    return <ApiOrderDetail order={apiOrder} />;
   818→  }
   819→
   820→  // Live API failed AND no mock order — show not-found.
   821→  if (isError || skipLiveApi) {
   822→    return <OrderNotFound id={orderId} />;
   823→  }
   824→
   825→  return <OrderDetailSkeleton />;
   826→}
