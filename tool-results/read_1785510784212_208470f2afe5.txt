   181→/* ─── Inline Search Bar (shown in header on /search page) ──── */
   182→
   183→function InlineSearchBar({
   184→  query,
   185→  onSearch,
   186→  onBack,
   187→  showFilters,
   188→  onToggleFilters,
   189→}: {
   190→  query: string;
   191→  onSearch: (q: string) => void;
   192→  onBack: () => void;
   193→  showFilters: boolean;
   194→  onToggleFilters: () => void;
   195→}) {
   196→  const [inputValue, setInputValue] = useState(query);
   197→
   198→  useEffect(() => {
   199→    setInputValue(query);
   200→  }, [query]);
   201→
   202→  const handleSubmit = (e: React.FormEvent) => {
   203→    e.preventDefault();
   204→    const trimmed = inputValue.trim();
   205→    if (trimmed) onSearch(trimmed);
   206→  };
   207→
   208→  return (
   209→    <form onSubmit={handleSubmit} className="flex items-center gap-2 flex-1 min-w-0">
   210→      <button
   211→        type="button"
   212→        onClick={onBack}
   213→        className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl hover:bg-kwik-bg-light transition-colors"
   214→        aria-label="Go back"
   215→      >
   216→        <ArrowLeft className="h-4 w-4 text-kwik-dark-medium" />
   217→      </button>
   218→
   219→      <div className="relative flex-1 min-w-0">
   220→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-kwik-muted" />
   221→        <input
   222→          type="text"
   223→          value={inputValue}
   224→          onChange={(e) => setInputValue(e.target.value)}
   225→          placeholder="Search products, stores, categories..."
   226→          className="w-full h-9 rounded-xl border border-kwik-border bg-kwik-bg-surface pl-9 pr-9 text-sm text-kwik-dark placeholder:text-kwik-muted outline-none focus:border-kwik-orange focus:ring-1 focus:ring-kwik-orange transition-colors"
   227→          autoFocus
   228→        />
   229→        {inputValue && (
   230→          <button
   231→            type="button"
   232→            onClick={() => setInputValue("")}
   233→            className="absolute right-3 top-1/2 -translate-y-1/2 text-kwik-muted hover:text-kwik-dark-medium"
   234→          >
   235→            <X className="h-3.5 w-3.5" />
   236→          </button>
   237→        )}
   238→      </div>
   239→
   240→      <button
   241→        type="button"
   242→        onClick={onToggleFilters}
   243→        className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-xl transition-colors ${
   244→          showFilters ? "bg-kwik-orange-tint text-kwik-orange" : "hover:bg-kwik-bg-light text-kwik-dark-medium"
   245→        }`}
   246→        aria-label="Toggle filters"
   247→      >
   248→        <SlidersHorizontal className="h-4 w-4" />
   249→      </button>
   250→    </form>
   251→  );
   252→}

   297→/* ─── Main Layout Component ─────────────────────────────────── */
   298→
   299→export function MarketplaceLayout({ children }: { children: React.ReactNode }) {
   300→  const router = useRouter();
   301→  const pathname = usePathname();
   302→  const searchParams = useSearchParams();
   303→  const { user, isAuthenticated, logout, isLoading: isAuthLoading } = useAuth();
   304→  const [isDrawerOpen, setIsDrawerOpen] = useState(false);
   305→  const [isSearchOpen, setIsSearchOpen] = useState(false);
   306→  const [isAutoSuggestOpen, setIsAutoSuggestOpen] = useState(false);
   307→  const desktopSearchBtnRef = useRef<HTMLButtonElement>(null);
   308→  const [isWishlistOpen, setIsWishlistOpen] = useState(false);
   309→  const [isPageLoading, setIsPageLoading] = useState(true);
   310→  const [isScrolled, setIsScrolled] = useState(false);
   311→  const [showFilters, setShowFilters] = useState(false);
   312→  const isPageLoadingRef = useRef(true);
   313→  const [isClientMounted, setIsClientMounted] = useState(false);
   314→
   315→  const isSearchPage = pathname === "/search";
   316→  const isAuthPage = pathname === "/login" || pathname === "/register" || pathname.startsWith("/forgot-password") || pathname.startsWith("/reset-password");
   317→  const isVendorStorefrontRoute = pathname.startsWith("/vendor/");
   318→  // Account pages (cart, orders list, wishlist, profile/*) render their own
   319→  // AccountLayout shell — hide the full marketplace header/footer/widgets.
   320→  // NOTE: /orders/[id] (detail page) is intentionally NOT included here —
   321→  // it keeps the full marketplace chrome.
   322→  const isAccountRoute =
   323→    pathname === "/cart" ||
   324→    pathname === "/orders" ||
   325→    pathname === "/wishlist" ||
   326→    pathname === "/vendor-orders" ||
   327→    pathname.startsWith("/profile");
   328→  const hideFullChrome = isVendorStorefrontRoute || isAccountRoute;
   329→  const hideTopNav = isAuthPage || hideFullChrome;
   330→  const searchQuery = searchParams.get("q") || "";
   331→  const startNavigationLoading = useCallback(() => {
   332→    isPageLoadingRef.current = true;
   333→    setIsPageLoading(true);
   334→  }, []);
   335→
   336→  useEffect(() => {
   337→    setIsClientMounted(true);
   338→  }, []);
   339→
   340→  useEffect(() => {
   341→    if (!isDrawerOpen) return undefined;
   342→    const previousOverflow = document.body.style.overflow;
   343→    document.body.style.overflow = "hidden";
   344→    return () => {
   345→      document.body.style.overflow = previousOverflow;
   346→    };
   347→  }, [isDrawerOpen]);
   348→
   349→  useEffect(() => {
   350→    const handleScroll = () => {
   351→      setIsScrolled(window.scrollY > 12);
   352→    };
   353→
   354→    window.addEventListener("scroll", handleScroll, { passive: true });
   355→    return () => window.removeEventListener("scroll", handleScroll);
   356→  }, []);
   357→
   358→  useEffect(() => {
   359→    isPageLoadingRef.current = true;
   360→    setIsPageLoading(true);
   361→    const timer = setTimeout(() => {
   362→      isPageLoadingRef.current = false;
   363→      setIsPageLoading(false);
   364→    }, 420);
   365→
   366→    return () => clearTimeout(timer);
   367→  }, [pathname]);
   368→
   369→  useEffect(() => {
   370→    const handleDocumentClick = (event: MouseEvent) => {
   371→      const target = event.target as Element | null;
   372→      const anchor = target?.closest?.("a[href]") as HTMLAnchorElement | null;
   373→      if (!anchor || event.defaultPrevented) return;
   374→      if (anchor.target && anchor.target !== "_self") return;
   375→
   376→      const nextUrl = new URL(anchor.href, window.location.href);
   377→      if (nextUrl.origin !== window.location.origin) return;
   378→      if (nextUrl.pathname === window.location.pathname && nextUrl.search === window.location.search) return;
   379→
   380→      startNavigationLoading();
   381→    };
   382→
   383→    document.addEventListener("click", handleDocumentClick, true);
   384→    return () => document.removeEventListener("click", handleDocumentClick, true);
   385→  }, [startNavigationLoading]);
   386→
   387→  // Reset filters when leaving search page
   388→  useEffect(() => {
   389→    if (!isSearchPage) setShowFilters(false);
   390→  }, [isSearchPage]);
   391→
   392→  const cartItemCount = useCartStore((s) =>
   393→    s.items.length,
   394→  );
   395→  const [prevCartCount, setPrevCartCount] = useState(cartItemCount);
   396→  const [badgeKey, setBadgeKey] = useState(0);
   397→
   398→  // Animate badge when cart count changes
   399→  useEffect(() => {
   400→    if (cartItemCount !== prevCartCount) {
   401→      setBadgeKey((k) => k + 1);
   402→      setPrevCartCount(cartItemCount);
   403→    }
   404→  }, [cartItemCount, prevCartCount]);
   405→  const setCartOpen = useCartStore((s) => s.setCartOpen);
   406→
   407→  const handleLogout = async () => {
   408→    try {
   409→      await logout();
   410→      kwikToast.success("Logged out successfully");
   411→      router.refresh();
   412→    } catch {
   413→      kwikToast.error("Failed to log out");
   414→    }
   415→  };
   416→
   417→  const closeDrawer = useCallback(() => setIsDrawerOpen(false), []);
   418→
   419→  const handleSearchSubmit = useCallback(
   420→    (q: string) => {
   421→      const params = new URLSearchParams();
   422→      params.set("q", q);
   423→      const category = searchParams.get("category");
   424→      if (category) params.set("category", category);
   425→      startNavigationLoading();
   426→      router.push(`/search?${params.toString()}`);
   427→    },
   428→    [router, searchParams, startNavigationLoading],
   429→  );
   430→
   431→  const handleSearchBack = useCallback(() => {
   432→    startNavigationLoading();
   433→    router.push("/");
   434→  }, [router, startNavigationLoading]);
   435→
   436→  const handleToggleFilters = useCallback(() => {
   437→    setShowFilters((prev) => !prev);
   438→    if (isSearchPage) {
   439→      const params = new URLSearchParams(searchParams.toString());
   440→      if (!showFilters) {
   441→        params.set("filters", "true");
   442→      } else {
   443→        params.delete("filters");
   444→      }
   445→      startNavigationLoading();
   446→      router.replace(`/search?${params.toString()}`, { scroll: false });
   447→    }
   448→  }, [isSearchPage, router, searchParams, showFilters, startNavigationLoading]);
   449→
   450→  return (
   451→    <MarketplaceShellProvider
   452→      value={{
   453→        openSearch: () => setIsSearchOpen(true),
   454→        showFilters,
   455→        setShowFilters,
   456→      }}
   457→    >
   458→      <div className="flex min-h-screen flex-col bg-background">
   459→        {!hideFullChrome && <PageLoader isLoading={isPageLoading} />}
   460→        <OfflineBanner />
   461→        {!isSearchPage && !hideFullChrome && (
   462→          <EnhancedSearchOverlay
   463→            isOpen={isSearchOpen}
   464→            onClose={() => setIsSearchOpen(false)}
   465→          />
   466→        )}
   467→
   468→        {!hideTopNav && (
   469→          <>
   470→            <header
   471→              className={`fixed inset-x-0 top-0 z-80 bg-background/95 backdrop-blur-md transition-shadow duration-300 ${
   472→                isScrolled
   473→                  ? "shadow-md border-b border-kwik-orange/10"
   474→                  : "border-b border-kwik-border"
   475→              }`}
   476→            >
   477→              <div className=" mx-auto px-0 md:px-4">
   478→                {/* Top row: logo + actions */}
   479→                <div className="flex py-2 md:h-16 items-center justify-between">
   480→                  <div className="flex min-w-0 items-center gap-1 md:gap-2">
   481→                    <Button
   482→                      isIconOnly
   483→                      variant="ghost"
   484→                      className="md:hidden"
   485→                      onPress={() => setIsDrawerOpen(true)}
   486→                      aria-label="Open menu"
   487→                    >
   488→                      <Menu className="h-5 w-5" />
   489→                    </Button>
   490→
   491→                    {/* Logo */}
   492→                    <button
   493→                      type="button"
   494→                      onClick={() => {
   495→                        startNavigationLoading();
   496→                        router.push("/");
   497→                      }}
   498→                      className="flex min-w-0 items-center gap-1.5 transition-opacity hover:opacity-80 md:gap-2"
   499→                    >
   500→                      <div className="y-2">
   501→                        <Image
   502→                          src="/icon.png"
   503→                          alt="KWIKSELLER"
   504→                          width={20}
   505→                          height={20}
   506→                          className="rounded-md md:rounded-lg md:h-8! md:w-8!"
   507→                        />
   508→                      </div>
   509→                      <span className="truncate text-lg font-bold text-kwik-dark md:text-xl">
   510→                        KWIKSELLER
   511→                      </span>
   512→                    </button>
   513→                  </div>
   514→
   515→                  <MegaNav />
   516→
   517→                  <div className="flex items-center gap-0 md:gap-2">
   518→                    {/* Desktop search - auto suggest on desktop */}
   519→                    {!isSearchPage && (
   520→                      <div className="relative hidden md:block">
   521→                        <button
   522→                          ref={desktopSearchBtnRef}
   523→                          type="button"
   524→                          onClick={() => setIsAutoSuggestOpen(true)}
   525→                          className="flex h-11 min-w-[190px] items-center gap-2 rounded-2xl border border-kwik-border bg-kwik-bg-surface px-4 text-sm text-kwik-gray-light transition-colors hover:border-kwik-orange hover:shadow-md hover:shadow-kwik-orange/5 lg:min-w-[260px] cursor-pointer"
   526→                        >
   527→                          <Search className="h-4 w-4 shrink-0 text-kwik-muted" />
   528→                          <span className="truncate">Search products, brands...</span>
   529→                        </button>
   530→                        <SearchAutoSuggest
   531→                          isOpen={isAutoSuggestOpen}
   532→                          onClose={() => setIsAutoSuggestOpen(false)}
   533→                          anchorRef={desktopSearchBtnRef as React.RefObject<HTMLElement>}
   534→                        />
   535→                      </div>
   536→                    )}
   537→
   538→                    <Button
   539→                      isIconOnly
   540→                      variant="ghost"
   541→                      size="sm"
   542→                      onPress={() => setCartOpen(true)}
   543→                      aria-label="Shopping cart"
   544→                      className="relative text-kwik-gray-light"
   545→                    >
   546→                      <ShoppingCart className="h-4 w-4" />
   547→                      {isClientMounted && cartItemCount > 0 && (
   548→                        <motion.span
   549→                          key={badgeKey}
   550→                          initial={{ scale: 0 }}
   551→                          animate={{ scale: 1 }}
   552→                          transition={{
   553→                            type: "spring",
   554→                            stiffness: 500,
   555→                            damping: 15,
   556→                          }}
   557→                          className="absolute -right-1 -top-1 flex h-5 w-5 items-center justify-center rounded-full bg-kwik-orange text-[10px] font-bold text-white shadow-sm shadow-kwik-orange/30"
   558→                        >
   559→                          {/* Pulse ring animation */}
   560→                          <motion.span
   561→                            className="absolute inset-0 rounded-full bg-kwik-orange"
