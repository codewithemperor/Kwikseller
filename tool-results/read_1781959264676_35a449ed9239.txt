    86→    86→const SUBSCRIPTION_KEY = "kwikseller_vendor_subscription";
    87→    87→const BILLING_KEY = "kwikseller_vendor_billing_history";
    88→    88→
    89→    89→function loadSubscription(): CurrentSubscription {
    90→    90→  if (typeof window === "undefined")
    91→    91→    return { planId: "STARTER", status: "NONE" };
    92→    92→  try {
    93→    93→    const raw = localStorage.getItem(SUBSCRIPTION_KEY);
    94→    94→    return raw ? JSON.parse(raw) : { planId: "STARTER", status: "NONE" };
    95→    95→  } catch {
    96→    96→    return { planId: "STARTER", status: "NONE" };
    97→    97→  }
    98→    98→}
    99→    99→
   100→   100→function saveSubscription(sub: CurrentSubscription) {
   101→   101→  if (typeof window === "undefined") return;
   102→   102→  localStorage.setItem(SUBSCRIPTION_KEY, JSON.stringify(sub));
   103→   103→}
   104→   104→
   105→   105→function loadBilling(): BillingEntry[] {
   106→   106→  if (typeof window === "undefined") return [];
   107→   107→  try {
   108→   108→    const raw = localStorage.getItem(BILLING_KEY);
   109→   109→    return raw ? JSON.parse(raw) : [];
   110→   110→  } catch {
   111→   111→    return [];
   112→   112→  }
   113→   113→}
   114→   114→
   115→   115→function addBillingEntry(plan: Plan, status: "Paid" | "Pending" | "Failed") {
   116→   116→  const entries = loadBilling();
   117→   117→  const entry: BillingEntry = {
   118→   118→    id: `inv_${Date.now()}`,
   119→   119→    date: new Date().toISOString(),
   120→   120→    plan: plan.name,
   121→   121→    amount: plan.price,
   122→   122→    status,
   123→   123→  };
   124→   124→  entries.unshift(entry);
   125→   125→  localStorage.setItem(BILLING_KEY, JSON.stringify(entries));
   126→   126→}

   130→   130→function getStatusBadge(status: SubscriptionStatus) {
   131→   131→  switch (status) {

   143→   143→      );
   144→   144→    default:

   146→   146→  }
