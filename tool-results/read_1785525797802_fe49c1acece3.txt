    33→// ─── Text highlight ──────────────────────────────────────────────
    34→
    35→function highlightMatch(text: string, query: string) {
    36→  if (!query.trim()) return text
    37→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    38→  const regex = new RegExp(`(${escaped})`, 'gi')
    39→  const parts = text.split(regex)
    40→  return parts.map((part, i) =>
    41→    regex.test(part) ? (
    42→      <span key={i} className="font-semibold text-kwik-orange">
    43→        {part}
    44→      </span>
    45→    ) : (
    46→      part
    47→    ),
    48→  )
    49→}
    50→
    51→// ─── Suggestion type icon ────────────────────────────────────────
    52→
    53→function SuggestionIcon({ label }: { label: string }) {
    54→  // Heuristic: pick an icon based on the suggestion source so the user
    55→  // can tell whether the suggestion is a product, brand, store, or category.
    56→  const lower = label.toLowerCase()
    57→  if (lower.includes('dress') || lower.includes('shirt') || lower.includes('sneaker') || lower.includes('fashion')) {
    58→    return <Shirt className="h-4 w-4 text-pink-500" />
    59→  }
    60→  if (lower.includes('phone') || lower.includes('iphone') || lower.includes('samsung') || lower.includes('earbud') || lower.includes('speaker')) {
    61→    return <Smartphone className="h-4 w-4 text-cyan-500" />
    62→  }
    63→  if (lower.includes('beauty') || lower.includes('serum') || lower.includes('cream') || lower.includes('makeup')) {
    64→    return <Gem className="h-4 w-4 text-purple-500" />
    65→  }
    66→  if (lower.includes('kitchen') || lower.includes('home') || lower.includes('decor') || lower.includes('bed')) {
    67→    return <Home className="h-4 w-4 text-amber-500" />
    68→  }
    69→  if (lower.includes('food') || lower.includes('coffee') || lower.includes('rice') || lower.includes('drink')) {
    70→    return <UtensilsCrossed className="h-4 w-4 text-emerald-500" />
    71→  }
    72→  return <Package className="h-4 w-4 text-kwik-orange" />
    73→}

   126→      (p) => p.name.toLowerCase().includes(q) || p.category.toLowerCase().includes(q),

   137→      const key = s.toLowerCase()

   139→        seen.add(key)
   140→        merged.push({ label: s, source: 'api' })
   141→      }
   142→    }
   143→    for (const p of localSuggestions) {
   144→      const key = p.name.toLowerCase()
   145→      if (!seen.has(key)) {
   146→        seen.add(key)
   147→        merged.push({ label: p.name, source: 'local' })
   148→      }
   149→    }
   150→    return merged.slice(0, 8)
   151→  }, [liveSuggestions, localSuggestions, trimmedQuery])
   152→
   153→  const handleSearch = useCallback(
   154→    (term: string) => {
   155→      const trimmed = term.trim()
   156→      if (!trimmed) return
   157→      addSearch(trimmed)
   158→      onClose()

   160→    },
   161→    [addSearch, onClose, router],
   162→  )
   163→
   164→  const handleRemoveHistory = useCallback((term: string, e: React.MouseEvent) => {
   165→    e.stopPropagation()
   166→    removeSearch(term)
   167→  }, [removeSearch])
   168→
   169→  const handleKeyDown = useCallback(
   170→    (e: React.KeyboardEvent) => {
   171→      if (e.key === 'Enter') {
   172→        handleSearch(query)
   173→      }
   174→    },
   175→    [query, handleSearch],
   176→  )
   177→
   178→  const handleCategoryClick = useCallback(
   179→    (categoryName: string) => {
   180→      const trimmed = categoryName.trim()
   181→      if (!trimmed) return
   182→      addSearch(trimmed)
   183→      onClose()
   184→      router.push(`/search?q=${encodeURIComponent(trimmed)}`)
   185→    },
   186→    [addSearch, onClose, router],
   187→  )
   188→
   189→  const hasQuery = trimmedQuery.length > 0
   190→  const showSuggestions = hasQuery && mergedSuggestions.length > 0
   191→  const showNoResults = hasQuery && !suggestionsQuery.isLoading && mergedSuggestions.length === 0
   192→  const showRelated = hasQuery && !PRODUCT_SUGGESTIONS.some((p) => {
   193→    const q = trimmedQuery.toLowerCase()

   195→  })
   196→
   197→  return (

   199→      {isOpen && (

   204→            animate={{ opacity: 1 }}
   205→            exit={{ opacity: 0 }}
   206→            transition={{ duration: 0.2 }}
   207→            className="fixed inset-0 z-[90] bg-black/60 backdrop-blur-md"
   208→            onClick={onClose}
   209→            aria-hidden="true"
   210→          />
   211→
   212→          {/* Main container */}
   213→          <motion.div
   214→            initial={{ opacity: 0, y: -30, scale: 0.95 }}
   215→            animate={{ opacity: 1, y: 0, scale: 1 }}
   216→            exit={{ opacity: 0, y: -30, scale: 0.95 }}

   224→                  <Search className="w-5 h-5 text-kwik-muted flex-shrink-0" />
   225→                  <input
   226→                    ref={inputRef}
   227→                    type="text"
   228→                    value={query}
   229→                    onChange={(e) => setQuery(e.target.value)}
   230→                    onKeyDown={handleKeyDown}
   231→                    placeholder="Search products, stores, categories..."
   232→                    className="flex-1 bg-transparent text-base text-kwik-dark outline-none placeholder:text-kwik-muted dark:text-white"

   234→                    role="searchbox"

   239→                  {hasQuery && (

   246→                      aria-label="Clear search"

   248→                      <X className="w-4 h-4" />

   252→                    ESC

   254→                </div>

   256→
   257→              {/* Content panel */}
   258→              <motion.div
   259→                initial={{ opacity: 0, y: -10 }}
   260→                animate={{ opacity: 1, y: 0 }}
   261→                transition={{ delay: 0.1, duration: 0.2 }}
   262→                className="mt-2 rounded-2xl bg-kwik-bg-surface/95 dark:bg-background/95 backdrop-blur-xl border border-kwik-border shadow-2xl overflow-hidden"
   263→              >
   264→                <div className="overflow-y-auto max-h-[60vh] sm:max-h-[55vh]">

   269→                        key="suggestions"
   270→                        initial={{ opacity: 0 }}
   271→                        animate={{ opacity: 1 }}
   272→                        exit={{ opacity: 0 }}
   273→                        transition={{ duration: 0.15 }}
   274→                        className="p-4"
   275→                      >
   276→                        <div className="flex items-center justify-between mb-3 px-1">
   277→                          <h3 className="text-xs font-semibold text-kwik-muted uppercase tracking-wider">
   278→                            {showRelated ? 'Similar picks' : 'Suggestions'}
   279→                          </h3>
   280→                          <span className="text-xs text-kwik-muted">{mergedSuggestions.length} results</span>
   281→                        </div>
   282→                        <div className="space-y-1">
   283→                          {mergedSuggestions.map((suggestion, index) => {
   284→                            const localMatch = PRODUCT_SUGGESTIONS.find(
   285→                              (p) => p.name.toLowerCase() === suggestion.label.toLowerCase(),
   286→                            )
   287→                            return (
   288→                              <motion.button
   289→                                key={`${suggestion.label}-${index}`}
   290→                                initial={{ opacity: 0, x: -10 }}
   291→                                animate={{ opacity: 1, x: 0 }}
   292→                                transition={{ delay: index * 0.03 }}
   293→                                onClick={() => handleSearch(suggestion.label)}
   294→                                className="w-full flex items-center gap-3 px-3 py-2.5 rounded-xl hover:bg-kwik-bg-light transition-colors text-left group"
   295→                              >
   296→                                {/* Icon + colored chip */}
   297→                                {localMatch ? (
   298→                                  <div
   299→                                    className={cn(
   300→                                      'w-10 h-10 rounded-lg flex items-center justify-center flex-shrink-0 text-white text-xs font-bold',
   301→                                      localMatch.color,
   302→                                    )}
   303→                                  >
   304→                                    {localMatch.initials}
   305→                                  </div>
   306→                                ) : (
   307→                                  <div className="w-10 h-10 rounded-lg flex items-center justify-center flex-shrink-0 bg-kwik-orange-tint text-kwik-orange">

   310→                                )}

   315→                                  <div className="flex items-center gap-1.5 text-xs text-kwik-gray-light dark:text-white/55">
   316→                                    {localMatch ? (
   317→                                      <>
   318→                                        <Tag className="h-3 w-3" />
   319→                                        {localMatch.category}
   320→                                      </>
   321→                                    ) : (
   322→                                      <>
   323→                                        <StoreIcon className="h-3 w-3" />
   324→                                        Marketplace
   325→                                      </>
   326→                                    )}
   327→                                  </div>
   328→                                </div>
   329→                                {localMatch && (
   330→                                  <span className="flex-shrink-0 text-sm font-semibold text-kwik-dark dark:text-white">
   331→                                    {localMatch.price}
   332→                                  </span>
   333→                                )}
   334→                              </motion.button>
   335→                            )
   336→                          })}
   337→                        </div>
   338→                      </motion.div>
   339→                    )}
   340→
   341→                    {/* ── No-results while typing ── */}
   342→                    {showNoResults && (
   343→                      <motion.div
   344→                        key="no-results"
   345→                        initial={{ opacity: 0 }}
   346→                        animate={{ opacity: 1 }}
   347→                        exit={{ opacity: 0 }}
   348→                        transition={{ duration: 0.15 }}
   349→                        className="p-8 text-center"
   350→                      >
   351→                        <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-kwik-orange-tint text-kwik-orange">
   352→                          <Search className="h-7 w-7" />

   354→                        <p className="mt-3 font-heading text-base font-bold text-foreground">
   355→                          No matches for &ldquo;{trimmedQuery}&rdquo;

   563→function formatRelative(timestamp: number): string {
   564→  const diff = Date.now() - timestamp
   565→  if (diff < 60_000) return 'just now'
   566→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}m ago`
   567→  if (diff < 86_400_000) return `${Math.floor(diff / 3_600_000)}h ago`
   568→  if (diff < 7 * 86_400_000) return `${Math.floor(diff / 86_400_000)}d ago`
   569→  return new Date(timestamp).toLocaleDateString('en-NG', { dateStyle: 'medium' })
   570→}
