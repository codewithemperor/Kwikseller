   124→   124→function getTimelineStageIndex(status: DeliveryStatus): number {
   125→   125→  const stageKeys = TIMELINE_STAGES.map((s) => s.key);
   126→   126→  const statusIdx = stageKeys.indexOf(status as (typeof TIMELINE_STAGES)[number]["key"]);
   127→   127→  if (statusIdx >= 0) return statusIdx;
   128→   128→  // Statuses before ACCEPTED map to -1 (all dots empty)
   129→   129→  if (STATUS_ORDER.indexOf(status) < STATUS_ORDER.indexOf("ACCEPTED")) return -1;
   130→   130→  // Statuses after DELIVERED map to last index (all dots filled)
   131→   131→  return TIMELINE_STAGES.length - 1;
   132→   132→}

   134→   134→function getRelativeTime(dateStr: string): string {
   135→   135→  const now = new Date();
   136→   136→  const date = new Date(dateStr);
   137→   137→  const diffMs = now.getTime() - date.getTime();
   138→   138→  const diffMins = Math.floor(diffMs / 60000);
   139→   139→  const diffHours = Math.floor(diffMs / 3600000);
   140→   140→  const diffDays = Math.floor(diffMs / 86400000);
   141→   141→
   142→   142→  if (diffMins < 1) return "Just now";
   143→   143→  if (diffMins < 60) return `${diffMins}m ago`;
   144→   144→  if (diffHours < 24) return `${diffHours}h ago`;
   145→   145→  if (diffDays < 7) return `${diffDays}d ago`;
   146→   146→  return formatDate(dateStr);
   147→   147→}

   392→   392→function DetailTimeline({ delivery }: { delivery: DeliveryItem }) {
   393→   393→  const entries: Array<{ key: string; label: string; time?: string; color: string }> = [];
   394→   394→
   395→   395→  const push = (key: string, label: string, time?: string, color = "text-muted-foreground") => {
   396→   396→    if (time) entries.push({ key, label, time, color });
   397→   397→  };
   398→   398→
   399→   399→  push("created", "Order created", delivery.createdAt, "text-muted-foreground");
   400→   400→  push("accepted", "Accepted by vendor", delivery.acceptedAt, "text-muted-foreground");
   401→   401→  push("preparing", "Preparing order", delivery.preparingAt, "text-amber-600");
   402→   402→  push("ready", "Ready for pickup", delivery.readyAt, "text-orange-600");
   403→   403→  push("picked_up", "Picked up by rider", delivery.pickedUpAt, "text-blue-600");
   404→   404→  push("in_transit", "In transit", delivery.inTransitAt, "text-blue-600");
   405→   405→  push("arrived", "Arrived at destination", delivery.arrivedAt, "text-purple-600");
   406→   406→  push("delivered", "Delivered", delivery.deliveredAt, "text-green-600");
   407→   407→  push("completed", "Completed", delivery.completedAt, "text-green-700");
   408→   408→  if (delivery.cancelledAt) push("cancelled", "Cancelled", delivery.cancelledAt, "text-red-500");
   409→   409→  if (delivery.returnedAt) push("returned", "Returned", delivery.returnedAt, "text-red-400");
   410→   410→
   411→   411→  if (entries.length === 0) return null;
   412→   412→
   413→   413→  return (
   414→   414→    <div className="relative pl-6">
   415→   415→      <div className="absolute left-[7px] top-2 bottom-2 w-px bg-default-100" />
   416→   416→      <div className="space-y-4">
   417→   417→        {entries.map((entry) => (
   418→   418→          <div key={entry.key} className="relative flex items-start gap-3">
   419→   419→            <div className="absolute -left-6 top-1.5 h-3.5 w-3.5 rounded-full border-2 border-kwik-border bg-surface" />
   420→   420→            <div className="min-w-0">
   421→   421→              <p className="text-sm text-foreground">{entry.label}</p>
   422→   422→              <p className="text-xs text-muted-foreground mt-0.5">{getRelativeTime(entry.time!)}</p>
   423→   423→            </div>
   424→   424→          </div>
   425→   425→        ))}
   426→   426→      </div>
   427→   427→    </div>
   428→   428→  );
   429→   429→}

   795→   795→            </div>

   842→   842→    </motion.div>
   843→   843→  );
   844→   844→}
