    41→/**
    42→ * Generates a reasonably-unique idempotency key per checkout attempt.
    43→ * The backend uses it to dedupe a ParentCheckout (and its per-vendor
    44→ * Orders) so a network retry returns the original result instead of
    45→ * double-creating orders. We regenerate it whenever the user starts a
    46→ * new checkout submission (not on every render).
    47→ */
    48→function newIdempotencyKey(): string {
    49→  if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
    50→    return `chk_${crypto.randomUUID()}`;
    51→  }
    52→  return `chk_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    53→}

   208→          ...prev,
   209→          fullName: prev.fullName || fullName,
   210→          phone: prev.phone || phone,

   356→            image: i.image,
   357→          })),
   358→          vendorId: group.storeName,
   359→          vendorName: group.storeName,
   360→          deliveryAddress:
   361→            deliveryMethod === "STANDARD_DELIVERY"
   362→              ? `${address.fullName}, ${address.addressLine}, ${address.city}, ${address.state}. Phone: ${address.phone}${address.landmark ? `. Landmark: ${address.landmark}` : ""}`
   363→              : "PICKUP — buyer collects from vendor's store",
   364→        });
   365→      }
   366→
   367→      clearCart();
   368→
   369→      const orderCount = apiOrders.length || storeCount;
   370→      const isPickup = deliveryMethod === "PICKUP";
   371→      kwikToast.success(
   372→        "Order placed!",
   373→        isPickup
   374→          ? `${orderCount} order${orderCount > 1 ? "s" : ""} placed. You can initialize payment now.`
   375→          : `${orderCount} order${orderCount > 1 ? "s" : ""} sent to the vendor${orderCount > 1 ? "s" : ""} for a delivery quote. You'll be notified once they respond.`,
   376→      );
   377→
   378→      // PICKUP: the quote auto-agrees at checkout, so skip the order detail
   379→      // page and send the buyer straight to the payment gateway.
   380→      if (isPickup) {
   381→        const firstOrderId = apiOrders[0]?.id;
   382→        if (firstOrderId) {
   383→          try {
   384→            const payRes = await api.post<{ authorizationUrl?: string }>(
   385→              `orders/${firstOrderId}/initialize-payment`,
   386→            );
   387→            const authorizationUrl = payRes.data?.authorizationUrl;
   388→            if (authorizationUrl) {
   389→              kwikToast.info(
   390→                "Redirecting to payment",
   391→                "Complete your payment to confirm the order.",
   392→              );
   393→              window.location.href = authorizationUrl;
   394→              return;
   395→            }
   396→          } catch (err) {
   397→            const message =
   398→              err instanceof Error ? err.message : "Failed to initialize payment.";
   399→            kwikToast.error(
   400→              "Payment initialization failed",
   401→              `${message} You can retry from the order page.`,
   402→            );
   403→          }
   404→          router.push(`/orders/${firstOrderId}`);
   405→        } else {
   406→          router.push("/orders");
   407→        }
   408→        return;
   409→      }
   410→
   411→      // STANDARD_DELIVERY: redirect to the first API order's detail page if
   412→      // available (quote negotiation happens there), otherwise the orders list.

   414→        router.push(`/orders/${apiOrders[0].id}`);
   415→      } else {
   416→        router.push("/orders");
   417→      }
   418→    } catch (err) {
   419→      const message =
   420→        err instanceof Error
   421→          ? err.message
   422→          : "Something went wrong. Please try again.";

   481→      {/* ── Hero header ── */}

   483→        <div className="absolute inset-0 bg-[radial-gradient(circle_at_top_right,rgba(255,255,255,0.18),transparent_55%)]" />
   484→        <div className="container relative mx-auto max-w-7xl px-4 py-8">

   491→              Secure checkout

   500→            </p>
   501→          </motion.div>
   502→        </div>
   503→      </section>
   504→
   505→      <section className="container mx-auto max-w-7xl px-4 py-8">

  1101→}
