   352→/* ─── Sections Manager ─────────────────────────────────────── */
   353→
   354→function SectionsManager({
   355→  sections,
   356→  onChange,
   357→}: {
   358→  sections: string[];
   359→  onChange: (sections: string[]) => void;
   360→}) {
   361→  const [expandedId, setExpandedId] = React.useState<string | null>(null);
   362→
   363→  const toggleVisibility = (id: string) => {
   364→    if (sections.includes(id)) {
   365→      onChange(sections.filter((s) => s !== id));
   366→    } else {
   367→      onChange([...sections, id]);
   368→    }
   369→  };
   370→
   371→  const moveSection = (index: number, direction: "up" | "down") => {
   372→    const next = [...sections];
   373→    const target = direction === "up" ? index - 1 : index + 1;
   374→    if (target < 0 || target >= next.length) return;
   375→    [next[index], next[target]] = [next[target], next[index]];
   376→    onChange(next);
   377→  };
   378→
   379→  return (
   380→    <div className="space-y-0">
   381→      {ALL_SECTIONS.map((section) => {
   382→        const isVisible = sections.includes(section.id);
   383→        const visibleIndex = sections.indexOf(section.id);
   384→        const IconComp = section.icon;
   385→        const isExpanded = expandedId === section.id;
   386→
   387→        return (
   388→          <div
   389→            key={section.id}
   390→            className="flex items-center gap-3 border-t border-gray-200 py-3 first:border-t-0"
   391→          >
   392→            {/* Drag handle (decorative) */}
   393→            <GripVertical className="h-4 w-4 shrink-0 text-gray-300" />
   394→
   395→            {/* Icon */}
   396→            <IconComp className="h-4 w-4 shrink-0 text-gray-400" strokeWidth={1.5} />
   397→
   398→            {/* Label + expand toggle */}
   399→            <div className="min-w-0 flex-1">
   400→              <div className="flex items-center gap-2">
   401→                <span className="text-sm font-medium text-foreground">{section.label}</span>
   402→                {isVisible && (
   403→                  <span className="text-[10px] font-mono text-gray-400">#{visibleIndex + 1}</span>
   404→                )}
   405→              </div>
   406→              {isExpanded && (
   407→                <p className="mt-1 text-xs text-gray-400">
   408→                  {isVisible
   409→                    ? "Visible in storefront. Use arrows to reorder."
   410→                    : "Hidden from storefront. Toggle to show."}
   411→                </p>
   412→              )}
   413→            </div>
   414→
   415→            {/* Reorder arrows (only when visible) */}
   416→            {isVisible && (
   417→              <div className="flex flex-col">
   418→                <button
   419→                  type="button"
   420→                  onClick={() => moveSection(visibleIndex, "up")}
   421→                  disabled={visibleIndex === 0}
   422→                  className="rounded p-0.5 text-gray-400 transition hover:bg-gray-100 hover:text-gray-600 disabled:opacity-20 disabled:hover:bg-transparent"
   423→                  aria-label="Move up"
   424→                >
   425→                  <ChevronUp className="h-3.5 w-3.5" />
   426→                </button>
   427→                <button
   428→                  type="button"
   429→                  onClick={() => moveSection(visibleIndex, "down")}
   430→                  disabled={visibleIndex === sections.length - 1}
   431→                  className="rounded p-0.5 text-gray-400 transition hover:bg-gray-100 hover:text-gray-600 disabled:opacity-20 disabled:hover:bg-transparent"
   432→                  aria-label="Move down"
   433→                >
   434→                  <ChevronDown className="h-3.5 w-3.5" />
   435→                </button>
   436→              </div>
   437→            )}
   438→
   439→            {/* Info toggle */}
   440→            <button
   441→              type="button"
   442→              onClick={() => setExpandedId(isExpanded ? null : section.id)}
   443→              className="rounded p-1 text-gray-400 transition hover:bg-gray-100 hover:text-gray-600"
   444→              aria-label="Toggle info"
   445→            >
   446→              <Info className="h-3.5 w-3.5" />
   447→            </button>
   448→
   449→            {/* Visibility toggle */}
   450→            <AppSwitch
   451→              isSelected={isVisible}
   452→              onChange={() => toggleVisibility(section.id)}
   453→              label=""
   454→            />
   455→          </div>
   456→        );
   457→      })}
   458→    </div>
   459→  );
   460→}

   462→/* ─── Color Presets Picker ──────────────────────────────────── */
   463→
   464→function ColorPresetsPicker({
   465→  label,
   466→  value,
   467→  presets,
   468→  onChange,
   469→}: {
   470→  label: string;
   471→  value: string;
   472→  presets: string[];
   473→  onChange: (value: string) => void;
   474→}) {
   475→  const [showCustom, setShowCustom] = React.useState(false);
   476→  const [customHex, setCustomHex] = React.useState(value);
   477→
   478→  const applyCustom = () => {
   479→    const trimmed = customHex.trim();
   480→    if (/^#[0-9a-fA-F]{6}$/.test(trimmed)) {
   481→      onChange(trimmed.toUpperCase());
   482→      setShowCustom(false);
   483→    } else {
   484→      kwikToast.error("Enter a valid hex color (e.g. #111827)");
   485→    }
   486→  };
   487→
   488→  return (
   489→    <div className="space-y-2">
   490→      <span className="text-xs font-semibold text-muted">{label}</span>
   491→      <div className="flex flex-wrap gap-2">
   492→        {presets.map((color) => (
   493→          <button
   494→            key={color}
   495→            type="button"
   496→            onClick={() => onChange(color)}
   497→            className="group relative h-9 w-9 rounded-sm border border-gray-200 transition hover:scale-105"
   498→            style={{ backgroundColor: color }}
   499→            aria-label={`Color ${color}`}
   500→          >
   501→            {value.toUpperCase() === color.toUpperCase() && (
   502→              <div className="absolute inset-0 flex items-center justify-center">
   503→                <div className="h-2.5 w-2.5 rounded-full border-2 border-white bg-gray-900" />
   504→              </div>
   505→            )}
   506→          </button>
   507→        ))}
   508→        <button
   509→          type="button"
   510→          onClick={() => {
   511→            setShowCustom(!showCustom);
   512→            setCustomHex(value);
   513→          }}
   514→          className="flex h-9 w-9 items-center justify-center rounded-sm border border-dashed border-gray-300 text-xs text-gray-400 transition hover:border-gray-500 hover:text-gray-600"
   515→          aria-label="Custom color"
   516→        >
   517→          #
   518→        </button>
   519→      </div>
   520→      {showCustom && (
   521→        <div className="flex items-center gap-2">
   522→          <input
   523→            type="text"
   524→            value={customHex}
   525→            onChange={(e) => setCustomHex(e.target.value)}
   526→            placeholder="#000000"
   527→            maxLength={7}
   528→            className="h-9 w-28 rounded-sm border border-gray-300 bg-white px-2 text-sm font-mono text-foreground outline-none focus:border-gray-500"
   529→          />
   530→          <AppButton size="sm" onClick={applyCustom}>
   531→            Apply
   532→          </AppButton>
   533→        </div>
   534→      )}
   535→      <div className="flex items-center gap-2">
   536→        <AppColorPicker label="" value={value} onChange={onChange} />
   537→        <span className="font-mono text-xs text-gray-400">{value.toUpperCase()}</span>
   538→      </div>
   539→    </div>
   540→  );
   541→}

   545→function LayoutSelector<T extends string>({
   546→  label,
   547→  options,
   548→  value,

   552→  options: readonly { value: T; label: string; icon?: string }[];
   553→  value: T;
   554→  onChange: (value: T) => void;
