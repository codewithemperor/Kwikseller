    49→/* ─── Product Description (renders HTML or plain text) ─── */
    50→function ProductDescription({ description }: { description: string }) {
    51→  const [expanded, setExpanded] = React.useState(false);
    52→
    53→  // Strip HTML for the compact preview
    54→  const plainText = React.useMemo(() => {
    55→    if (!hasHtmlMarkup(description)) return description;
    56→    return description.replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim();
    57→  }, [description]);
    58→
    59→  const fullContentClassName =
    60→    "text-sm leading-6 text-kwik-gray dark:text-white/65 [&_h1]:mb-2 [&_h1]:text-lg [&_h1]:font-bold [&_h1]:text-kwik-dark [&_h1]:dark:text-white [&_h2]:mb-2 [&_h2]:text-base [&_h2]:font-semibold [&_h2]:text-kwik-dark [&_h2]:dark:text-white [&_h3]:mb-1.5 [&_h3]:text-sm [&_h3]:font-semibold [&_h3]:text-kwik-dark [&_h3]:dark:text-white [&_p]:mb-2 [&_ul]:mb-2 [&_ul]:list-disc [&_ul]:pl-5 [&_ol]:mb-2 [&_ol]:list-decimal [&_ol]:pl-5 [&_strong]:font-semibold [&_strong]:text-kwik-dark [&_strong]:dark:text-white";
    61→
    62→  return (
    63→    <div>
    64→      {expanded ? (
    65→        hasHtmlMarkup(description) ? (
    66→          <div className={fullContentClassName} dangerouslySetInnerHTML={{ __html: description }} />
    67→        ) : (
    68→          <p className={fullContentClassName}>{description}</p>
    69→        )
    70→      ) : (
    71→        <p className="text-sm leading-6 text-kwik-gray dark:text-white/65 line-clamp-3">
    72→          {plainText}
    73→        </p>
    74→      )}
    75→      {plainText.length > 120 && (
    76→        <button
    77→          type="button"
    78→          onClick={() => setExpanded(!expanded)}
    79→          className="mt-1 text-xs font-semibold text-kwik-orange hover:underline"
    80→        >
    81→          {expanded ? "Show less" : "Read more"}
    82→        </button>
    83→      )}
    84→    </div>
    85→  );
    86→}

    89→function FadeIn({
    90→  children,
    91→  delay = 0,
    92→  className = "",
    93→}: {
    94→  children: React.ReactNode;
    95→  delay?: number;
    96→  className?: string;
    97→}) {
    98→  return (
    99→    <motion.div
   100→      initial={{ opacity: 0, y: 16 }}
   101→      animate={{ opacity: 1, y: 0 }}
   102→      transition={{ duration: 0.4, delay, ease: "easeOut" }}
   103→      className={className}
   104→    >

   106→    </motion.div>
   107→  );
   108→}

   280→        const at = a.createdAt ? new Date(a.createdAt).getTime() : 0;
   281→        const bt = b.createdAt ? new Date(b.createdAt).getTime() : 0;
   282→        return bt - at;

   285→      sorted.sort((a, b) => b.rating - a.rating);
